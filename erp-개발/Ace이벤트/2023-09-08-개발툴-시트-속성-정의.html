ActionCount int ActionCount  스크립트에서ActionType을통해Sheet에Action을할때, Row의증감(+/-) 단위입니다.
    기본값은1입니다.
ActionType string ActionType  
    1.     ActionType_SetNumber = 1,
    StartRow==-1 && EndRow == -1 인경우,
    선택된row가없는경우DataRowCnt까지자동채번,
    선택된row가있는경우선택된row만자동채번
    예)
    SS1.ActiveColumnName = ‘EventNo’
    SS1.StartRow=1
    SS1.EndRow = SS1.DataRowCnt
    SS1.NumberStartValue=1
    SS1.ActionType = ‘ActionType_SetNumber’
     
    2.     ActionType_DeleteRow = 2,
    Row 삭제
                 예) 3번Row에서10개를삭제하는경우
    SS1.StartRow = 3
    SS1.ActionCount=10
    SS1.ActionType = ‘ActionType_InsertRow’
     
    3.     ActionType_DeleteCol = 3,
    Col삭제
    4.     ActionType_InsertRow = 4,
    Row추가
    5.     ActionType_InsertCol = 5,
    Col추가

    예) SS1.ActiveCellBackColor = 1233 // argb(1,1,1,1)
