금액 계산 (환율 포함)

-- 금액 계산

local Row

for Row = 0, SS2.DataRowCnt - 1 do

SS2.ActiveRow = Row

local PurPrice = tonumber(CellText(SS2, Row, 'PurPrice')) -- 단가

local PurQty = tonumber(CellText(SS2, Row, 'PurQty')) -- 매입수량

if PurPrice == nil then PurPrice = 0 end

if PurQty == nil then PurQty = 0 end

local purAmt = PurPrice * PurQty -- 금액 = 단가 * 매입수량

local exRate = tonumber(CellText(SS2, Row, 'ExRate')) -- 환율

if exRate == nil then exRate = 0 end -- 환율이 nil이면 0으로 처리

SetText(SS2, Row, 'PurAmt', purAmt)

SetText(SS2, Row, 'PurDomAmt', purAmt * exRate) -- 원화금액(PurDomAmt) = 금액 * 환율

end

RunPgmMethod('Master_Amt') -- 'Master_Amt' 메서드 실행
