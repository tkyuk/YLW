function fnAccDrAccAfterInfo_opas(ObjectSS, IntSRow, IntERow, StrCCtrSeq, StrCostDeptSeq, StrDrAccSeq, StrColRemSeq, StrColUMCostType, StrColCostTypeName, StrColMCodeHelpConst, StrColMCodeHelpParam)

local ds, Qty, UnitSTDQty, lExRate, CustSeq, CurrSeq

local columnSchema

local addData

StrCol = ObjectSS.ActiveColumnName

columnSchema = '{WorkingTag, string}{IDX_NO, int}{Status, int}{DataSeq, int}{Selected, int}{TABLE_NAME, string}'

..'{CCtrSeq, int}'

..'{CostDeptSeq, int}'

..'{DrAccSeq, int}'

ds = CreateDataSet(ds,'DataBlock2', columnSchema)

for iRow = IntSRow, IntERow, 1 do

addData = 'A'..CNST_SP..tostring(iRow)..CNST_SP..'0'..CNST_SP..tostring(iRow)..CNST_SP..'1'..CNST_SP..'DataBlock2'..CNST_SP

..CellText(ObjectSS, iRow, StrCostDeptSeq)..CNST_SP

..CellText(ObjectSS, iRow, StrCCtrSeq)..CNST_SP

..CellText(ObjectSS, iRow, StrDrAccSeq)

AddDataRow(ds, 'DataBlock2', addData)

end

rtnDs = ExecuteService(15920058, 3, ds, rtnDs, 'SQLWF')

if (rtnDs == nil) then

else

for row=0, GetDataBlock(rtnDs, 'DataBlock2').Rows.Count-1, 1 do

SetText(ObjectSS, IntSRow + row, StrColRemSeq, GetDataSetValue(rtnDs, 'DataBlock2', 'RemSeq', row))

SetText(ObjectSS, IntSRow + row, StrColUMCostType, GetDataSetValue(rtnDs, 'DataBlock2', 'UMCostType', row))

SetText(ObjectSS, IntSRow + row, StrColCostTypeName, GetDataSetValue(rtnDs, 'DataBlock2', 'CostTypeName', row))

SetText(ObjectSS, IntSRow + row, StrColMCodeHelpConst, GetDataSetValue(rtnDs, 'DataBlock2', 'MCodeHelpConst', row))

SetText(ObjectSS, IntSRow + row, StrColMCodeHelpParam, GetDataSetValue(rtnDs, 'DataBlock2', 'MCodeHelpParam', row))

if CellText(ObjectSS, IntSRow + row, 'MCodeHelpConst') == '' or CellText(ObjectSS, IntSRow + row, 'MCodeHelpConst') == '0' then

ObjectSS.ActiveColumnName = 'RemVal'

ObjectSS.ActiveCellReadOnly = true

else

ObjectSS.ActiveColumnName = 'RemVal'

ObjectSS.ActiveCellReadOnly = false        

end         

end

end

ObjectSS.ActiveColumnName = StrCol

end
