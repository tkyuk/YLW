local cnt, status, msg

cnt = 0

status = 0

msg = '여신한도가 반영되었습니다.'

for row = 0, SS1.DataRowCnt - 1 do

if CellText(SS1, row, 'Sel') == '1' then

status = GetDataSetValue(m_RecvXml.Data, 'DataBlock2', 'Status', cnt)

if status ~= 0 and status ~= '0' and status ~= '999998' and status ~= '999999' then

msg = GetDataSetValue(m_RecvXml.Data, 'DataBlock2', 'Result', cnt)

GoTo('END')

end

cnt = tonumber(cnt) + 1

end

end

MessageBoxShow(msg, '')

Msg = '차수 삭제되었습니다.'

if        GetDataBlock(m_RecvXml.Data, 'DataBlock6') ~= nil and

GetDataBlock(m_RecvXml.Data, 'DataBlock6').Rows.Count > 0 and

GetDataSetValue(m_RecvXml.Data, 'DataBlock6', 'Status', 0) ~= 0 and

GetDataSetValue(m_RecvXml.Data, 'DataBlock6', 'Status', 0) ~= '0' and

GetDataSetValue(m_RecvXml.Data, 'DataBlock6', 'Status', 0) ~= nil then

MessageBoxShow(GetDataSetValue(m_RecvXml.Data, 'DataBlock6', 'Result', 0), 'Message')

GoTo('END')

return

else

MessageBoxShow(Msg ,'')

end
