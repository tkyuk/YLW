선택된 행에 대해서 특정값 없는 경우 종료

local IsContinuous, msg

local IsSelect = 0

if SS1.DataRowCnt == 0 then

GoTo('END')

return

end

for row = 0, SS1.DataRowCnt do

if CellText(SS1, row, 'SELECT') == '1' then

-- 24.04.23 SJPARK

SMStatusName = CellText(SS1, row, 'SMStatusName')

if SMStatusName == '' or SMStatusName == '0' then

MessageBoxShow('데이터 가져오기 처리된 건만 확정처리 가능합니다','')

GoTo('END')

return

end

IsSelect = IsSelect + 1

end

end

if IsSelect == 0 then

msg = GetMessage('1119', GetDictionary(2529), '', '', '', '', '', '', '', '', '')

MessageBoxShow(msg, '')

GoTo('END')

end

for row = 0, SS1.DataRowCnt do

if CellText(SS1, row, 'SELECT') == '1' then

IsContinuous = '1'

break

end

end

if IsContinuous ~= '1' then

GoTo('END')

end
