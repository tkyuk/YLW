코드도움 행만큼 금액계산

local ActCol, StrRow, EndRow, ActiveRowOld, row

ActCol = SS1.ActiveColumnName

EndRow = SS1.ActiveRow + SS1.CodeHelpSelectedRows - 1

if SS1.ActiveRow > EndRow then

EndRow = SS1.ActiveRow

end

ActiveRowOld = SS1.ActiveRow

for row = SS1.ActiveRow, EndRow, 1 do

SS1.ActiveRow = row

if tonumber(CellText(SS1, SS1.ActiveRow, 'UMSelfPayType')) == 0 then

SetText(SS1, SS1.ActiveRow, 'IsVATDiv' ,'0')                

end

if chkIsKEAD.Text == '1' then

-- 부가세 분할여부 체크 건인 경우

if tonumber(CellText(SS1, SS1.ActiveRow, 'IsVATDiv')) == 1 then

SetText(SS1, SS1.ActiveRow, 'SupplyAmt' ,tonumber(CellText(SS1, row, 'OrgAmt')))         -- 부가세공급가액 = 기관부담금

SetText(SS1, SS1.ActiveRow, 'SupplyAmt_PG' , tonumber(CellText(SS1, row, 'PersAmt'))) -- 부가세공급가액(PG) = 개인부담금

else

SetText(SS1, SS1.ActiveRow, 'SupplyAmt' ,tonumber(CellText(SS1, row, 'OrgAmt')) + tonumber(CellText(SS1, row, 'PersAmt')))         -- 부가세공급가액 = 기관부담금 + 개인부담금

SetText(SS1, SS1.ActiveRow, 'SupplyAmt_PG' ,'0') -- 부가세공급가액(PG) = 0

end

end

end

SS1.ActiveRow = ActiveRowOld
