색 변경 / 이전차수 비교

-- 가변부에서 이전차수 데이터 비교하여 변경시 색상 표시

-- 가변부

local row, col, dynCol, toCol, aCol, dynColArray, colName,fixedColCnt

--row = SS1.ActiveRow

-- 가변부 컬럼수로 나누기!!

fixedColCnt = 5 -- 가변부 컬럼수

dynCol = (SS2.MaxCols - SS2.SheetFixCols) / fixedColCnt

dynColArray = {

[1] = 'Qty',

[2] = 'Price',

[3] = 'Amt',

[4] = 'DelvDate',

[5] = 'Remark'

}

if dynCol > 1 then

for row = 0, SS2.DataRowCnt - 1 do

for col = 1, dynCol -1 do

for i = 1, fixedColCnt do

colName = dynColArray[i]

if CellText(SS2, row, colName..col) ~= CellText(SS2, row, colName..(col-1)) then

SS2.ActiveColumnName = colName..col

SS2.ActiveRow = row

SS2.ActiveCellBackColor = '#DFFFE5'

end

end

end

end

end
