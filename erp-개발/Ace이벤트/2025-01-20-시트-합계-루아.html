시트 합계 루아

-- * 부자재 계산

local QtyNumerator, QtyDenominator, Price, CalcPrice, MatCostSUM, MatPrice, MatHandling, LogiCost

-- 부자재 핸들링비용

MatHandling = fltMatHandling.Value

-- 제품단위당 물류비

LogiCost = fltLogiCost.Value

-- 반영단가 계산

for row = 0, SS1.DataRowCnt - 1 do

SS1.ActiveRow = row

if (CellText(SS1, row, 'IsCustSupply'))        == '1' then

SetText(SS1, row, 'CalcPrice', 0)

else

QtyNumerator = tonumber(CellText(SS1, row, 'QtyNumerator'))        

QtyDenominator = tonumber(CellText(SS1, row, 'QtyDenominator'))

Price = tonumber(CellText(SS1, row, 'Price'))

CalcPrice = math.floor(QtyNumerator/QtyDenominator * Price + 0.5)

SetText(SS1, row, 'CalcPrice', CalcPrice)

end

end

MatPrice = GetSumValue(SS1, 0, 'CalcPrice', SS1.DataRowCnt-1, 'CalcPrice')                

-- 부자재소계                        

fltMatPrice.Value = MatPrice

-- 부자재소계(제품정보쪽)

fltMatPriceM.Value = MatPrice        

-- 부자재비용합계

fltMatCostSUM.Value        = math.floor(MatPrice * ( 1 +MatHandling / 100) + 0.5)                                                                 

-- 부자재비용합계(제품정보쪽)

fltMatCostSUMM.Value        = math.floor(MatPrice * ( 1 +MatHandling / 100) + 0.5)                                                 

-- 제품단위당 물류비

fltLogiCostM.Value = LogiCost
