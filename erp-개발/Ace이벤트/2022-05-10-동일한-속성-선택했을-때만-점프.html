동일한 속성 선택했을 때만 점프

-- 창고가 다른 건이 선택되었을 때 점프 안되게 에러처리

local WHName, DiffWHName, chkWHName

WHName = GetDataSetValue(m_SendXml.Data, 'DataBlock1', 'WHName', 0)

for row = 0, GetDataBlock(m_SendXml.Data, 'DataBlock1').Rows.Count - 1 do

DiffWHName = GetDataSetValue(m_SendXml.Data, 'DataBlock1', 'WHName', row)

if WHName ~= DiffWHName then

chkWHName = 1

end

end

if chkWHName == 1 then

msg1 = GetMessage('1122', GetDictionary(783), '', '', '', '', '', '', '', '', '' )

msgType = MessageBox(msg1, '', 'MsgBoxTypeOK')

_base.IsJumpEventCancel = true

GoTo('END')

return

end
