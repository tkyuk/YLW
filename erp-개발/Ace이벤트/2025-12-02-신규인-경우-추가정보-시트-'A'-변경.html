신규인 경우 추가정보 시트 'A' 변경

-- 20180321 신규일때만 추가정보시트의 WorkingTag를 'A'로 바꿔줌 성지원 추가 (INIT에서 하도록 수정)

if (fltReceiptSeq.Text == '0' or fltReceiptSeq.Text == '' or fltReceiptSeq.Text == nil) then

local Row1, MaxRow1

Row1 = 0

MaxRow1 = SS1.DataRowCnt

while MaxRow1 > Row1

do

SS1.ActiveRow = Row1

SS1.WorkingTag = 'A'

Row1 = Row1 + 1        

end

end
