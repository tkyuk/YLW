루아 날짜 계산

-- 유효일자 = 제조일자 + 시효기간(일)

-- ValiDate = MakingDate + LimitTermDay

local ValiDate, MakingDate, LimitTermDay

MakingDateQty = CellText(SS, SS.ActiveRow, 'MakingDate')

LimitTermDay = tonumber(CellText(SS, SS.ActiveRow, 'LimitTermDay'))

if MakingDateQty == '' then

ValiDate = ''

else

ValiDate = DateAdd('Day', LimitTermDay, MakingDateQty)

end

SetText(SS, SS.ActiveRow, 'ValiDate', ValiDate)        

local strDate

strDate = os.date('%Y%m%d')

txtCostYM.Value = string.sub(DateAdd('Month', -1, strDate), 1, 4)..'01'
