local rowCount, ItemSeq, StockQty, LotNo

rowCount = GetDataBlock(m_RecvXml.Data, 'DataBlock1').Rows.Count

for i = 0,rowCount - 1 do

ItemSeq = GetDataSetValue(m_RecvXml.Data, 'DataBlock1', 'ItemSeq', i)

StockQty = GetDataSetValue(m_RecvXml.Data, 'DataBlock1', 'StockQty', i)

LotNo = GetDataSetValue(m_RecvXml.Data, 'DataBlock1', 'LotNo', i)

for row = 0, SS1.DataRowCnt - 1 do

if CellText(SS1, row, 'ItemSeq') == ItemSeq and CellText(SS1, row, 'LotNo') == LotNo then

SetText(SS1, row, 'StockQty', StockQty)

end

end

end

for row = 0, SS1.DataRowCnt - 1 do

if CellText(SS1, row, 'StockQty') == nil or CellText(SS1, row, 'StockQty') == '0' then

SetText(SS1, row, 'StockQty', '0')

end

end

for row = 0, SS1.DataRowCnt - 1 do

SetText(SS1, row, 'QutoPrice', GetDataSetValue(m_RecvXml.Data, 'DataBlock2', 'QutoPrice', row))

SetText(SS1, row, 'CurrName', GetDataSetValue(m_RecvXml.Data, 'DataBlock2', 'CurrName', row))

SetText(SS1, row, 'CurrSeq', GetDataSetValue(m_RecvXml.Data, 'DataBlock2', 'CurrSeq', row))

end
