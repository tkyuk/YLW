행별 시트 전체 일괄 색 적용

local EndRow, Row

if SS2.CodeHelpSelectedRows > 0 then

EndRow = SS2.ActiveRow + SS2.CodeHelpSelectedRows - 1

for Row = SS2.ActiveRow, EndRow do RunPgmMethod('SS2_IsStop2') end

else

RunPgmMethod('SS2_IsStop2')

end

SS2.CodeHelpSelectedRows = 0

-- SS2_IsStop2

local COLOR_DIS = -1118482 -- 비활성(gray)

local COLOR_WHITE = -1 -- 기본(white))

local COLOR_CODEHELP = '#eef8ff' -- 코드도움(skyblue) 구ace:-2100741

local COLOR_NOS = '#fff5f5' -- NOS 필수값(pink) 구ace:-2100741

-- IsStop 값 가져오기

local isStop = CellText(SS2, SS2.ActiveRow, 'IsStop')

if isStop == nil or isStop == '' then isStop = '0' end

-- 제외할 컬럼 Map

local expColMap = {

['IsStop'] = true,

['ProductPlanSeq'] = true,

['Serl'] = true,

['IsPO'] = true,

['ItemSeq'] = true,

['InputUnitSeq'] = true,

['SMInOutType'] = true,

['PurCustSeq'] = true,

['CurrSeq'] = true,

['PurUnitSeq'] = true,

['VATRate'] = true,

['TotPurAmt'] = true,

['TotPurDomAmt'] = true,

['WHSeq'] = true,

['MakerSeq'] = true,

['MakerName'] = true,

['StdUnitName'] = true,

['StdUnitSeq'] = true,

['StdUnitQty'] = true,

['StdConvQty'] = true,

['InputUnitTotQty'] = true,

['StdUnitTotQty'] = true,

['PurPriceOld'] = true,

['CurAmt'] = true,

['CurVAT'] = true,

['DomAmt'] = true,

['DomVAT'] = true,

['IsConsumableMat'] = true

}

-- IsStop일 경우 선택 해제

if isStop == '1' then SetText(SS2, SS2.ActiveRow, 'Sel', '0') end

-- ActiveRow 열 순회

local maxCol = SS2.MaxCols - 1

local colName

local targetColor

for col = 0, maxCol do

SS2.ActiveCol = col

colName = SS2.ActiveColumnName

-- 제외 목록에 없는 컬럼만 로직 수행

if expColMap[colName] == nil then

if isStop == '1' then

-- 중단 상태인 경우 무조건 DIS 색상

targetColor = COLOR_DIS

elseif SS2.ColumnCellType == 'enCodeHelp' then

-- 중단이 아니면서 코드도움인 경우

targetColor = COLOR_CODEHELP

else

if string.find(SS2.ColumnControlKey, 'NOS') then

targetColor = COLOR_NOS

else

targetColor = COLOR_WHITE

end

end

-- col 색상 적용

SS2.ActiveCellBackColor = targetColor

end

end
