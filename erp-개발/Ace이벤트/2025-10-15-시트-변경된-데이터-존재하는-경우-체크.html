시트 변경된 데이터 존재하는 경우 체크

-- 행삭제 하는 경우

-- 25.10.02 SJPARK

local msg, result, cnt, ReturnValue

for row = 0, SS1.DataRowCnt - 1 do

SS1.ActiveRow = row

if SS1.WorkingTag == 'A' or SS1.WorkingTag == 'U' then

cnt = 1

end

end

if cnt == 1 then         

ReturnValue = CommonMessageBox(1019,'MsgBoxTypeYesNo', '', '', '')         

if ReturnValue ~= 1 then

GoTo('END')

return

else

RunPgmMethod('Save')

GoTo('END')

return                

end        

--if result == 1 then

--        RunPgmMethod('Save')

--else

--        GoTo('END')

--end

end

-- 변경된 내역 존재 시 조회하는 경우

for i = 0, SS1.DataRowCnt - 1 do

SS1.ActiveRow = i

if SS1.WorkingTag == 'U' then

rtnType = MessageBox('변경된 내역이 존재합니다. 저장하시겠습니까?','', 'MsgBoxTypeYesNo')

if rtnType == 1 then

RunPgmMethod('Save')

GoTo('END')

return

end

end

end
