컬럼 전부 DIS 처리 및 툴바 DIS 처리

--==============================================================

-- 컬럼 전부 DIS 처리 및 툴바 DIS 처리

--==============================================================

local ErrorChk

for row = 0, SS1.DataRowCnt-1 do

if CellText(SS1, row, 'DelvOrderSeq') ~= '0' then                

SS1.ActiveRow = row                

for col = 0, SS1.MaxCols - 1 do

SS1.ActiveRow = row

SS1.ActiveCol = col

SS1.ActiveCellReadOnly = true

end

else

ErrorChk = '1'

end

end

if ErrorChk == '1' then

GetToolbar('Save').EventArgs.Disable = false

GetToolbar('Cut').EventArgs.Disable = false

else

GetToolbar('Save').EventArgs.Disable = true

GetToolbar('Cut').EventArgs.Disable = true

end
