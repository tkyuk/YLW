선택된 컬럼 기준 합계 구하기

-- TotPartsCost 부품비

-- TotLaborCost 공임비

local TotPartsCost, TotLaborCost

TotPartsCost= 0

TotLaborCost= 0

for row = 0, SS9.DataRowCnt - 1 do

if CellText(SS9,row, 'IsLaborCost') == '1' then

TotLaborCost = TotLaborCost + CellText(SS9, row, 'CurAmtTotal')

else

TotPartsCost = TotPartsCost + CellText(SS9, row, 'CurAmtTotal')

end

end

fltTotPartsCost.Value = TotPartsCost

fltTotLaborCost.Value = TotLaborCost
