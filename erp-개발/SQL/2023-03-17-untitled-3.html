-- 예상재고, 과부족 수량 구하기 전에 ItemSeq 집계

SELECT A.ItemSeq ,

SUM(ISNULL(A.StockQty ,0)) AS StockQty , -- 현재고

SUM(ISNULL(A.SafeQty ,0)) AS SafeQty , -- 안전재고

SUM(ISNULL(A.NeedQty ,0)) AS NeedQty , -- 필요량

SUM(ISNULL(A.POQty ,0)) AS POQty , -- 구매발주

SUM(ISNULL(A.QCWaitQty,0)) AS QCWaitQty, -- 검사대기

0 AS ExpectQty, -- 예상재고

0 AS StkQty , -- 과부족

0 AS POUnitSeq, -- 구매단위코드

0 AS PoBaseQty -- 발주산정량

--=================================================================

-- 예상재고 구하기 (StockQty - NeedQty + POReqQty + AppReqQty + POQty + QCWaitQty)

-- 과부족 구하기

/*

IF (StockQty - NeedQty + POReqQty + AppReqQty + POQty + QCWaitQty) \< 0                                                                                                                                                        

--> SafeQty + [(StockQty - NeedQty + POReqQty + AppReqQty + POQty + QCWaitQty)*-1]                                                                                                                                                

ELSE                                                                                                                                                        

IF SafeQty \<= (StockQty - NeedQty + POReqQty + AppReqQty + POQty + QCWaitQty)                                                                                                                                                

--> 0                                                                                                                                        

IF SafeQty > (StockQty - NeedQty + POReqQty + AppReqQty + POQty + QCWaitQty)                                                                                                                                                

--> SafeQty - (StockQty - NeedQty + POReqQty + AppReqQty + POQty + QCWaitQty)                                                                                                                                        

*/

--=================================================================

UPDATE #Result

SET ExpectQty = StockQty - NeedQty + POReqQty + AppReqQty + POQty + QCWaitQty,

StkQty = CASE WHEN (StockQty - NeedQty + POReqQty + AppReqQty + POQty + QCWaitQty) \< 0

THEN SafeQty + ((StockQty - NeedQty + POReqQty + AppReqQty + POQty + QCWaitQty)*-1)

ELSE CASE WHEN SafeQty \<= (StockQty - NeedQty + POReqQty + AppReqQty + POQty + QCWaitQty)

THEN 0

ELSE SafeQty - (StockQty - NeedQty + POReqQty + AppReqQty + POQty + QCWaitQty) END END

FROM #Result

-- 발주산정량 구하기 : 과부족 수량을 구매단위로 환산한 수량

ISNULL(A.QCWaitQty , 0) AS QCWaitQty , -- 검사대기

ISNULL(A.ExpectQty , 0) AS ExpectQty , -- 예상재고

ISNULL(A.StkQty , 0) AS StkQty , -- 과부족

ISNULL(G.UnitName ,'') AS POUnit , -- 구매단위

ISNULL(A.POUnitSeq , 0) AS POUnitSeq , -- 구매단위코드

ISNULL(A.PoBaseQty , 0) AS PoBaseQty -- 발주산정량

FROM #Result AS A

JOIN _TDAItem AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.ItemSeq = A.ItemSeq

JOIN _TDAUnit AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.UnitSeq = B.UnitSeq

JOIN _TDAItemPurchase AS D WITH(NOLOCK) ON D.CompanySeq = B.CompanySeq

AND D.ItemSeq = B.ItemSeq

LEFT OUTER JOIN _TDAItemClass AS E WITH(NOLOCK) ON E.CompanySeq = @CompanySeq

AND E.ItemSeq = A.ItemSeq

AND E.UMajorItemClass IN (2001, 2004)

LEFT OUTER JOIN _VDAItemClass AS F WITH(NOLOCK) ON F.CompanySeq = E.CompanySeq

AND F.ItemClassSSeq = E.UMItemClass

LEFT OUTER JOIN _TDAUnit AS G WITH(NOLOCK) ON G.CompanySeq = @CompanySeq

AND G.UnitSeq = A.POUnitSeq

WHERE @IsStkQty = '0' OR (@IsStkQty = '1' AND A.StkQty > 0) -- 과부족건만 조회 체크 시 과부족이 0보다 큰 건만 조회

RETURN
