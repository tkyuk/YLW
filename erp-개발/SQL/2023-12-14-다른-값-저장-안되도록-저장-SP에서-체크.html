다른 값 저장 안되도록 저장 SP에서 체크

UPDATE #TSLOrderItem

SET Result = '향정품은 다른품목과 함께 주문 할 수 없습니다.',

MessageType = 17009,

Status = 9999

WHERE (1=1)

AND WorkingTag IN ('A','U')

AND Status = 0

AND @PgmSeq IN (35420024,35420068) -- 주문등록_nkp, 주문등록(담당자)_nkp

AND (

-- ''과 1이 같이 있는 경우

EXISTS (

SELECT 1

FROM _TSLOrderItem AS B WITH(NOLOCK)

LEFT OUTER JOIN nkp_TDAItemPam AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.ItemSeq = B.ItemSeq

WHERE B.CompanySeq = @CompanySeq

AND B.OrderSeq = #TSLOrderItem.OrderSeq

AND C.ISPSY IN ('', '1')

GROUP BY B.OrderSeq

HAVING COUNT(DISTINCT C.ISPSY) = 2

)

OR

-- 0과 1이 같이 있는 경우

EXISTS (

SELECT 1

FROM _TSLOrderItem AS B WITH(NOLOCK)

LEFT OUTER JOIN nkp_TDAItemPam AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.ItemSeq = B.ItemSeq

WHERE B.CompanySeq = @CompanySeq

AND B.OrderSeq = #TSLOrderItem.OrderSeq

AND C.ISPSY IN ('0', '1')

GROUP BY B.OrderSeq

HAVING COUNT(DISTINCT C.ISPSY) = 2

)

)

0,1 같이 있는 경우 저장 X
