구매요청 -> 수주 원천

-- 구매요청 진행 테이블

CREATE TABLE #TEMP_TPUORDPOReqProg

(

IDX_NO INT IDENTITY,

POReqSeq INT,

CompleteCHECK INT,

SMCurrStatus INT,

IsStop NCHAR(1),

OrderSeq INT

)

INSERT INTO #TEMP_TPUORDPOReqProg(POReqSeq, CompleteCHECK, SMCurrStatus, IsStop)

SELECT A.POReqSeq, -1, 0, A.IsStop

FROM #TEMP_TPUORDPOReq AS A

GROUP BY A.POReqSeq, A.IsStop

-- 수주 역추적 --

-- 원천테이블

CREATE TABLE #TMP_SOURCETABLE

(

IDOrder INT,

TABLENAME NVARCHAR(100)

)

-- 원천 데이터 테이블

CREATE TABLE #TCOMSourceTracking

(

IDX_NO INT,

IDOrder INT,

Seq INT,

Serl INT,

SubSerl INT,

Qty DECIMAL(19,5),

STDQty DECIMAL(19,5),

Amt DECIMAL(19,5),

VAT DECIMAL(19,5)

)

INSERT #TMP_SOURCETABLE

SELECT 1, '_TSLOrderItem'

EXEC _SWCOMSourceTracking @CompanySeq, '_TPUORDPOReqItem', '#TEMP_TPUORDPOReqProg','POReqSeq', '',''

UPDATE #TEMP_TPUORDPOReqProg

SET OrderSeq = C.Seq

FROM #TEMP_TPUORDPOReqProg AS A

JOIN #TCOMSourceTracking AS C ON A.IDX_No = C.IDX_No

-- 수주 역추적끝--
