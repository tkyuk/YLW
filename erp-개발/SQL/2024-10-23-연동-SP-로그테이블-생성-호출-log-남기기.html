IF NOT EXISTS (SELECT * FROM Sysobjects where Name = 'SSCNT_TMESCutOutMoveReqSPLog' AND xtype = 'U' )

CREATE TABLE SSCNT_TMESCutOutMoveReqSPLog

(

LogSeq INT IDENTITY(1,1) NOT NULL,

MESKey NVARCHAR(1000) NOT NULL,

WorkingTag NCHAR(1) NOT NULL, -- 'A' : 처리, 'D' : 취소

ProcDate NCHAR(8) NOT NULL, -- 처리일자

ProcNo NVARCHAR(20) NOT NULL, -- 처리번호

EmpSeq INT NOT NULL, -- 처리자

Status INT NULL,

Results NVARCHAR(250) NULL,

LogDateTime DATETIME NOT NULL

,SYSTEMUSER NVARCHAR(100) NULL

,SERVERNAME NVARCHAR(100) NULL

)

INSERT SSCNT_TMESCutOutMoveReqSPLog

SELECT @MESKey

,@WorkingTag

,@ProcDate

,@ProcNo

,@EmpSeq

,0, '', GetDate()

,SYSTEM_USER

,@@SERVERNAME
