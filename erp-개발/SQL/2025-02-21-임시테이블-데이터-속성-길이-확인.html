임시테이블 데이터 속성 길이 확인

SELECT c.name AS ColumnName,

t.name AS DataType,

c.max_length AS MaxLength,

c.precision,

c.scale,

CASE

WHEN t.name IN ('int', 'bigint', 'smallint', 'tinyint', 'bit', 'datetime', 'date', 'smalldatetime', 'money', 'smallmoney')

THEN t.name

WHEN t.name IN ('char', 'varchar', 'nchar', 'nvarchar')

THEN t.name + '(' +

CASE WHEN t.name LIKE 'n%' THEN CONVERT(NVARCHAR(10), c.max_length / 2)

ELSE CONVERT(NVARCHAR(10), c.max_length)

END + ')'

WHEN t.name IN ('decimal', 'numeric')

THEN t.name + '(' + CONVERT(NVARCHAR(10), c.precision) + ',' + CONVERT(NVARCHAR(10), c.scale) + ')'

ELSE t.name

END AS FormattedDataType

FROM tempdb.sys.columns c

JOIN tempdb.sys.types t ON c.user_type_id = t.user_type_id

WHERE OBJECT_ID('tempdb..#Result') = c.object_id;
