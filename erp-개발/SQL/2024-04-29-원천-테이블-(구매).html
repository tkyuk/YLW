원천 테이블 (구매)

-- 원천 테이블

CREATE TABLE #TMP_SOURCETABLE(IDOrder INT, TABLENAME NVARCHAR(100))

-- 원천 공통 SP OUTPUT 테이블

CREATE TABLE #TCOMSourceTracking(IDX_NO INT, IDOrder INT, Seq INT, Serl INT, SubSerl INT,

Qty DECIMAL(19, 5), STDQty DECIMAL(19, 5), Amt DECIMAL(19, 5), VAT DECIMAL(19, 5))

-- 현재 데이터 테이블

CREATE TABLE #DelvInSource(IDX_NO INT IDENTITY, DelvInSeq INT, DelvInSerl INT, DelvInNo NVARCHAR(100), DelvInDate NCHAR(8), DelvInQty DECIMAL(19,5))

-- 원천 테이블명 INSERT

INSERT #TMP_SOURCETABLE

SELECT 1, '_TPUDelvItem' -- 구매납품

UNION

SELECT 2, '_TPUORDPOItem' -- 구매발주

UNION

SELECT 3, '_TPUORDApprovalReqItem' -- 구매품의

UNION

SELECT 4, '_TPUORDPOReqItem' -- 구매요청

-- 현재 데이터 테이블 INSERT

INSERT INTO #DelvInSource(DelvInSeq, DelvInSerl, DelvInNo, DelvInDate, DelvInQty)

SELECT A.DelvInSeq, B.DelvInSerl, A.DelvInNo, A.DelvInDate, B.Qty

FROM _TPUDelvIn AS A WITH(NOLOCK)

LEFT OUTER JOIN _TPUDelvInItem AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.DelvInSeq = B.DelvInSeq

WHERE A.CompanySeq = 1

--AND LEFT(A.DelvInDate,4) = '2024'

AND B.DelvInSerl IS NOT NULL

EXEC _SwCOMSourceTracking @CompanySeq = 1,

@TableName = '_TPUDelvInItem' ,

@TempTableName = '#DelvInSource' ,

@TempSeqColumnName = 'DelvInSeq' ,

@TempSerlColumnName = 'DelvInSerl' ,

@TempSubSerlColumnName = ''

-- 원천 sp 사용할 수 있는지 간이 확인

SELECT B.ProgTableKorName AS 'From'

,C.ProgTableKorName AS 'To'

FROM _TCOMSourceDailyTrace AS A

JOIN _TCOMProgTable AS B ON B.ProgTableSeq = A.FromTableSeq

JOIN _TCOMProgTable AS C ON C.ProgTableSeq = A.ToTableSeq

WHERE B.ProgTableKorName LIKE '%%' -- From

AND C.ProgTableKorName LIKE '%%' -- To

SELECT --구매입고번호(DelvInNo), 입고일, 입고수량, 납품번호, 납품일, 납품수량, 발주번호, 발주일, 발주수량, 품의번호, 품의일, 품의수량, 요청번호, 요청일, 요청수량

A.DelvInNo AS 구매입고번호,

A.DelvInDate AS 입고일,

A.DelvInQty AS 입고수량,

C.DelvNo AS 납품번호,

C.DelvDate AS 납품일,

B1.Qty AS 납품수량,

D.PONo AS 발주번호,

D.PODate AS 발주일,

B2.Qty AS 발주수량,

E.ApproReqNo AS 품의번호,

E.ApproReqDate AS 품의일,

B3.Qty AS 품의수량,

F.POReqNo AS 요청번호,

F.ReqDate AS 요청일,

B4.Qty AS 요청수량

FROM #DelvInSource AS A

LEFT OUTER JOIN #TCOMSourceTracking AS B1 ON B1.IDOrder = 1

AND A.IDX_NO = B1.IDX_NO

LEFT OUTER JOIN _TPUDelv AS C WITH(NOLOCK) ON C.CompanySeq = 1

AND B1.Seq = C.DelvSeq

LEFT OUTER JOIN _TPUDelvItem AS C1 WITH(NOLOCK) ON C1.CompanySeq = 1

AND B1.Seq = C1.DelvSeq

AND B1.Serl = C1.DelvSerl

LEFT OUTER JOIN #TCOMSourceTracking AS B2 ON B2.IDOrder = 2

AND A.IDX_NO = B2.IDX_NO

LEFT OUTER JOIN _TPUORDPO AS D WITH(NOLOCK) ON D.CompanySeq = 1

AND B2.Seq = D.POSeq

LEFT OUTER JOIN _TPUORDPOItem AS D1 WITH(NOLOCK) ON D1.CompanySeq = 1

AND B2.Seq = D1.POSeq

AND B2.Serl = D1.POSerl

LEFT OUTER JOIN #TCOMSourceTracking AS B3 ON B3.IDOrder = 3

AND A.IDX_NO = B3.IDX_NO

LEFT OUTER JOIN _TPUORDApprovalReq AS E WITH(NOLOCK) ON E.CompanySeq = 1

AND B3.Seq = E.ApproReqSeq

LEFT OUTER JOIN _TPUORDApprovalReqItem AS E1 WITH(NOLOCK) ON E1.CompanySeq = 1

AND B3.Seq = E1.ApproReqSeq

AND B3.Serl = E1.ApproReqSerl

LEFT OUTER JOIN #TCOMSourceTracking AS B4 ON B4.IDOrder = 4

AND A.IDX_NO = B4.IDX_NO

LEFT OUTER JOIN _TPUORDPOReq AS F WITH(NOLOCK) ON F.CompanySeq = 1

AND B4.Seq = F.POReqSeq

DROP TABLE #TMP_SOURCETABLE

DROP TABLE #TCOMSourceTracking

DROP TABLE #DelvInSource
