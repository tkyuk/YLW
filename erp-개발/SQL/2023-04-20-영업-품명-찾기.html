영업 품명 찾기

SELECT TOP 100 PERCENT C.ItemSeq

FROM _TDAItem AS C WITH(NOLOCK)

LEFT OUTER JOIN emk_TSLWasteInfo AS I ON C.ItemSeq = I.ItemSeq

WHERE @CompanySeq = C.CompanySeq

AND (I.GCustSeq = @CustSeq OR I.GCustSeq IS NULL)

AND C.AssetSeq In (SELECT AssetSeq FROM _TDAItemAssetCodeHelp WITH(NOLOCK) WHERE CompanySeq = @CompanySeq And ItemCodeHelpSeq = 2)

ORDER BY C.ItemName
