내수/수출 추가 (조회조건, 컬럼)

cmbSMExpKindName        CodeHelp        내수/수출        DataBlock1        SMExpKindName        SMExpKind        내수/수출        QRY

--Init에 추가

local columnSchema

local ds

local addData

columnSchema = '{SMExpKindName, string}{SMExpKind, int}'

ds = CreateDataSet(ds, 'DataBlock1', columnSchema)

addData = ''

addData = ''..'전체'..CNST_SP..'0'

AddDataRow(ds, 'DataBlock1', addData)

addData = ''..'내수'..CNST_SP..'1'

AddDataRow(ds, 'DataBlock1', addData)

addData = ''..'수출'..CNST_SP..'2'

AddDataRow(ds, 'DataBlock1', addData)

DataSetClear(m_SendXml.Data, '')

MergeDataSet(m_SendXml, ds)

SetAddCombo(cmbSMExpKindName, ds, 'DataBlock1', false)

cmbSMExpKindName.Value = 0

@SMExpKind          INT

ISNULL(B.SMExpKind , 0) AS SMExpKind , -- 내수/수출코드

ISNULL(M.MinorName ,'') AS SMExpKindName , -- 내수/수출

LEFT OUTER JOIN _TDASMinor AS M WITH(NOLOCK) ON M.CompanySeq = B.CompanySeq

AND M.MinorSeq = B.SMExpKind

AND ( ( @SMExpKind = 0 )

OR( @SMExpKind = 1 AND B.SMExpKind IN (8009001, 8009002, 8009003))

OR( @SMExpKind = 2 AND B.SMExpKind = 8009004)

) -- 달라질수도
