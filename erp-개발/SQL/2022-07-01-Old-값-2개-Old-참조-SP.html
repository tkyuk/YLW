CREATE PROCEDURE hsg_SWPDReWorkPriceCheck

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

AS

-- 변수선언

DECLARE @Count INT,

@Seq INT,

@MessageType INT,

@Status INT,

@Results NVARCHAR(250)

----===============--

---- 필수입력 체크 --

----===============--

---- 필수입력 Message 받아오기

EXEC dbo._SWCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

1038 , -- 필수입력 항목을 입력하지 않았습니다. (SELECT * FROM _TCAMessageLanguage WHERE Message like '%필수%')

@LanguageSeq ,

0,'' -- SELECT * FROM _TCADictionary WHERE Word like '%%'

---- 필수입력 Check

UPDATE A

SET Result = @Results,

MessageType = @MessageType,

Status = @Status

FROM #BIZ_OUT_DataBlock1 AS A

WHERE A.WorkingTag IN ('A','U')

AND A.Status = 0

-- guide : 이곳에 필수입력 체크 할 항목을 조건으로 넣으세요.

-- e.g. :

AND (ISNULL(A.BizUnit , 0) = 0 OR

ISNULL(A.ReworkYear , '') = '' OR

ISNULL(A.ItemSeq , 0) = 0 OR

ISNULL(A.ReWorkPrice, 0) = 0

)

----==================================================================================--

-------- 데이터 유무 체크 : UPDATE, DELETE 시 데이터 존재하지 않으면 에러처리 ----------

----==================================================================================--

EXEC dbo._SWCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

7 , -- 자료가 등록되어 있지 않습니다.(SELECT * FROM _TCAMessageLanguage WHERE MessageSeq = 7)

@LanguageSeq ,

0,'' -- SELECT * FROM _TCADictionary WHERE Word like '%%'

UPDATE A

SET Result = @Results,

MessageType = @MessageType,

Status = @Status

FROM #BIZ_OUT_DataBlock1 AS A

LEFT OUTER JOIN hsg_TSLReWorkItemPrice AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND A.ReworkYearOld = B.ReworkYear

AND A.BizUnitOld = B.BizUnit

AND A.ItemSeqOld = B.ItemSeq

WHERE A.WorkingTag IN ('U', 'D')

AND B.CompanySeq IS NULL

----====================================--

---- 중복여부체크(PalletSeq이 2개 이상일 경우)

----====================================--

---- 중복체크Message 받아오기

EXEC dbo._SWCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

6 , -- 중복된@1 @2가(이) 입력되었습니다.(SELECT * FROM _TCAMessageLanguage WHERE MessageSeq = 6)

@LanguageSeq ,

0,'' -- SELECT * FROM _TCADictionary WHERE Word like '%구매카드번호%'

--========================================--

-- 이미저장된시트에서중복되는것있는지확인 --

--========================================--

--Old값 초기화

UPDATE A

SET ReworkYearOld = 0, BizUnitOld = 0, ItemSeqOld = 0

FROM #BIZ_OUT_DataBlock1 AS A

WHERE A.WorkingTag = 'A'

AND A.Status = 0

--중복체크

UPDATE A

SET Result = @Results ,

MessageType = @MessageType ,

Status = @Status

FROM #BIZ_OUT_DataBlock1 AS A

JOIN hsg_TSLReWorkItemPrice AS B WITH(NOLOCK) ON (A.ReworkYearOld \<> B.ReworkYear OR A.BizUnitOld \<> B.BizUnit OR A.ItemSeqOld \<> B.ItemSeq) --내가 아닌것, 내자신이 아님

AND A.ReworkYear = B.ReworkYear

AND A.BizUnit = B.BizUnit

AND A.ItemSeq = B.ItemSeq

WHERE A.WorkingTag IN ('A', 'U')

AND A.Status = 0

AND B.CompanySeq = @CompanySeq

--========================================--

-- 같은저장시트에서중복되는것이있는지확인 --

--========================================--

UPDATE A

SET Result = @Results ,

MessageType = @MessageType ,

Status = @Status

FROM #BIZ_OUT_DataBlock1 AS A

JOIN #BIZ_OUT_DataBlock1 AS B ON A.IDX_NO \<> B.IDX_NO

AND A.ReworkYear = B.ReworkYear

AND A.BizUnit = B.BizUnit

AND A.ItemSeq = B.ItemSeq

-- 중복 컬럼이 추가 될수록 똑같이 조건 넣어주기

WHERE A.WorkingTag IN ( 'A','U' )

AND A.Status = 0

RETURN
