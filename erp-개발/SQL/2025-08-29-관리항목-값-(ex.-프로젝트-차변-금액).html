관리항목 값 (ex. 프로젝트 차변 금액)

SELECT B.RemValSeq AS SVCPJTSeq, A.AccSeq,SUM(A.DrAmt) AS Amt

FROM _TACSlipRow AS A WITH(NOLOCK)

JOIN _TACSlipRem AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.SlipSeq = A.SlipSeq

AND B.RemSeq = 1036

WHERE A.CompanySeq = @CompanySeq

AND A.AccSeq IN (SELECT TitleSeq2 FROM #Title WHERE IsAccSeq = 1)

AND B.RemValSeq IN (SELECT SVCPJTSeq FROM #TempFixedCol )

GROUP BY B.RemValSeq, A.AccSeq
