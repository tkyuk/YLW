DECLARE @TempDate NCHAR(8),

@WeeKDay NCHAR(8),

@WeeKDayTitle NCHAR(10)

SELECT @TempDate = '20221207'

SELECT @WeeKDay = DATEPART(WEEKDAY, @TempDate)

SELECT @WeeKDayTitle = CASE @WeeKDay WHEN '1' THEN '(일)'

WHEN '2' THEN '(월)'

WHEN '3' THEN '(화)'

WHEN '4' THEN '(수)'

WHEN '5' THEN '(목)'

WHEN '6' THEN '(금)'

WHEN '7' THEN '(토)'

END

SELECT @WeeKDay,@WeeKDayTitle

@CompanySeq INT = 1,
