SELECT A.WorkItemSeq

,A.WorkItemName

,A.WorkItemVer

,C.ElementID

,C.ElementName

,B.SqlScript

FROM _TWPKGWorkItem AS A

LEFT OUTER JOIN _TWPKGWorkItemAdditionalElement AS B ON B.WorkItemSeq = A.WorkItemSeq

AND B.WorkItemVer = A.WorkItemVer

LEFT OUTER JOIN _TWPKGElement AS C ON C.ElementSeq = B.ElementSeq

WHERE A.WorkItemName LIKE '%%' -- 작업단위명

AND C.ElementID LIKE '%FrmWPDQCVcheckLotList_biont%' -- 요소ID

SELECT E.WorkItemName, A.ElementTypeSeq, C.MinorName AS ElementType, A.ElementSeq, A.Description, A.DBType AS DBTypeSeq, A.SqlScript, A.ExecuteSeq

, D.MinorName AS WorkType, A.WorkTypeSeq, A.WorkTarget, A.AddElementSeq, B.UserName AS LastUserSeq, A.LastDateTime, A.Remark

FROM _TWPKGWorkItemAdditionalElement AS A WITH(NOLOCK)

LEFT OUTER JOIN _TCAUser AS B ON A.LastUserSeq = B.UserSeq

JOIN _TCAMinor AS C WITH(NOLOCK) ON A.ElementTypeSeq = C.MinorSeq

AND C.LanguageSeq = 1

JOIN _TCAMinor AS D WITH(NOLOCK) ON ISNULL(A.WorkTypeSeq, 333001) = D.MinorSeq

AND D.LanguageSeq = 1

JOIN _TWPKGWorkItem AS E ON E.WorkItemSeq = A.WorkItemSeq

AND E.WorkItemVer = A.WorkItemVer

WHERE SqlScript LIKE '%스크립트내용%'

ORDER BY A.AddElementSeq
