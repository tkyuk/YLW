행 끼리, 행 별로 계산 CROSS JOIN

SELECT '06' AS GbnCd,

'합계' AS GbnNm,

'매입비율' AS CustNm,

COALESCE(G3.M1 / NULLIF(G4.M1, 0), 0) AS M1,

COALESCE(G3.M2 / NULLIF(G4.M2, 0), 0) AS M2,

COALESCE(G3.M3 / NULLIF(G4.M3, 0), 0) AS M3,

COALESCE(G3.M4 / NULLIF(G4.M4, 0), 0) AS M4,

COALESCE(G3.M5 / NULLIF(G4.M5, 0), 0) AS M5,

COALESCE(G3.M6 / NULLIF(G4.M6, 0), 0) AS M6,

COALESCE(G3.M7 / NULLIF(G4.M7, 0), 0) AS M7,

COALESCE(G3.M8 / NULLIF(G4.M8, 0), 0) AS M8,

COALESCE(G3.M9 / NULLIF(G4.M9, 0), 0) AS M9,

COALESCE(G3.M10 / NULLIF(G4.M10, 0), 0) AS M10,

COALESCE(G3.M11 / NULLIF(G4.M11, 0), 0) AS M11,

COALESCE(G3.M12 / NULLIF(G4.M12, 0), 0) AS M12,

COALESCE(G3.TotMM / NULLIF(G4.TotMM, 0), 0) AS TotMM

FROM

(SELECT M1, M2, M3, M4, M5, M6, M7, M8, M9, M10, M11, M12, TotMM

FROM #ylwa19005_jsb_003

WHERE GbnCd = '03') AS G3

CROSS JOIN

(SELECT M1, M2, M3, M4, M5, M6, M7, M8, M9, M10, M11, M12, TotMM

FROM #ylwa19005_jsb_003

WHERE GbnCd = '05') AS G4;
