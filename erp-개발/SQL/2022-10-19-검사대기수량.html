검사대기수량

-- 검사대기수량 : 조회일 기준 3개월 전까지 구매납품, 외주납품, 작업실적건 중 검사구분이 미검사인 수량 조회(입고창고 관계없이 전체 조회)

CREATE TABLE #QCWaitQty_Temp

(

ItemSeq INT,

QCWaitQty DECIMAL(19,5)

)

-- 기간구하기

SELECT @FoProdDate3 = CONVERT(NCHAR(8), DATEADD(MONTH, -3, @FoProdDate), 112)

-- 구매납품

INSERT #QCWaitQty_Temp(ItemSeq, QCWaitQty)

SELECT B.ItemSeq, SUM(B.Qty) AS Qty

FROM _TPUDelv AS A WITH(NOLOCK)

JOIN _TPUDelvItem AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.DelvSeq = A.DelvSeq

JOIN #Fix AS C ON C.ItemSeq = B.ItemSeq

LEFT OUTER JOIN _TPDQCTestReport AS D WITH(NOLOCK) ON D.CompanySeq = B.CompanySeq

AND D.SourceSeq = B.DelvSeq

AND D.SourceSerl = B.DelvSerl

AND D.SourceType = 6097001 -- 구매검사

WHERE A.CompanySeq = @CompanySeq

AND A.BizUnit = @BizUnit

AND A.DelvDate BETWEEN @FoProdDate3 AND @FoProdDate

AND A.SMImpType IN (8008001, 8008002, 8008003)

AND ISNULL(D.SMTestResult, B.SMQcType) = 6035002 -- 미검사

GROUP BY B.ItemSeq

-- 외주납품

INSERT #QCWaitQty_Temp(ItemSeq, QCWaitQty)

SELECT B.ItemSeq, SUM(B.Qty) AS Qty

FROM _TPDOSPDelv AS A WITH(NOLOCK)

JOIN _TPDOSPDelvItem AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.OSPDelvSeq = A.OSPDelvSeq

JOIN #Fix AS C ON C.ItemSeq = B.ItemSeq

LEFT OUTER JOIN _TPDQCTestReport AS D WITH(NOLOCK) ON D.CompanySeq = B.CompanySeq

AND D.SourceSeq = B.OSPDelvSeq

AND D.SourceSerl = B.OSPDelvSerl

AND D.SourceType = 6097002 -- 외주검사

WHERE A.CompanySeq = @CompanySeq

AND A.BizUnit = @BizUnit

AND A.OSPDelvDate BETWEEN @FoProdDate3 AND @FoProdDate

AND ISNULL(D.SMTestResult, B.SMQcType) = 6035002 -- 미검사

GROUP BY B.ItemSeq

-- 작업실적

INSERT #QCWaitQty_Temp(ItemSeq, QCWaitQty)

SELECT B.ItemSeq, SUM(A.ProdQty) AS Qty

FROM _TPDSFCWorkReport AS A WITH(NOLOCK)

JOIN #Fix AS B ON B.ItemSeq = A.GoodItemSeq

LEFT OUTER JOIN _TPDQCTestReport AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.SourceSeq = A.WorkReportSeq

AND C.SourceSerl = 0

AND C.SourceType IN (6097003, 6097004) -- 최종검사, 공정검사

WHERE A.CompanySeq = @CompanySeq

AND A.BizUnit = @BizUnit

AND A.WorkDate BETWEEN @FoProdDate3 AND @FoProdDate

AND ISNULL(C.SMTestResult, 0) = 6035002 -- 미검사

GROUP BY B.ItemSeq

-- 창고 관계없이 품목별로 집계

SELECT ItemSeq, SUM(QCWaitQty) AS QCWaitQty

INTO #QCWaitQty

FROM #QCWaitQty_Temp

GROUP BY ItemSeq

-- 검사대기수량

UPDATE A

SET QCWaitQty = ISNULL(B.QCWaitQty, 0)

FROM #Fix AS A

LEFT OUTER JOIN #QCWaitQty AS B ON A.ItemSeq = B.ItemSeq
