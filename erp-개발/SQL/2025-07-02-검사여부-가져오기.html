검사여부 가져오기

-- 검사구분 재UPDATE

DECLARE @DelvDate NCHAR(8)

--WBS 품질에 맞게 검사품 가져올 수 있도록 설정

CREATE TABLE #QCItem

(

ItemSeq INT,

CustSeq INT,

IsQC NCHAR(1)

)

SELECT @DelvDate = DelvDate,

@BizUnit = BizUnit

FROM #BIZ_OUT_DataBlock2

INSERT INTO #QCItem (ItemSeq, CustSeq)

SELECT ItemSeq, CustSeq

FROM #BIZ_OUT_DataBlock2

EXEC _SWPDQAGetIsQC @CompanySeq = @CompanySeq,

@BizUnit = @BizUnit,

@Date = @DelvDate, --일자

@SMQCKind = 6018005 -- 구매검사. 20250519. jhkweon.

UPDATE A

SET SMQCType = CASE WHEN ISNULL(E.IsNotPurQC,'0') = '1' THEN 6035001

ELSE (CASE WHEN B.IsQC = '1' THEN 6035002 --미검사

ELSE 6035001 --무검사

END

)

END

FROM #BIZ_OUT_DataBlock2 AS A

JOIN #QCItem AS B ON A.ItemSeq = B.ItemSeq

LEFT OUTER JOIN _TPDQCTestReport AS C WITH(NOLOCK) ON A.DelvSeq = C.SourceSeq

AND A.DelvSerl = C.SourceSerl

AND C.CompanySeq = @CompanySeq

AND C.SourceType = 6097001

LEFT OUTER JOIN das_TPDQAProjectNotQCKind AS E WITH(NOLOCK) ON E.CompanySeq = @CompanySeq

AND A.PJTSeq = E.PJTSeq

WHERE A.Status = 0

AND ISNULL(C.QCSeq, 0) = 0

-- AND A.WorkingTag = 'U'

AND A.WorkingTag IN ('A', 'U')
