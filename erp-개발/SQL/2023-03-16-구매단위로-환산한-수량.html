구매단위로 환산한 수량

DECLARE @CompanySeq INT

DECLARE @UserSeq INT

SELECT @CompanySeq = 1

SELECT @UserSeq = 27292

DECLARE @GoodQtyDecLength INT

EXEC @GoodQtyDecLength = dbo._SWCOMEnvR @CompanySeq, 5, @UserSeq, @@PROCID

SELECT A.ItemName,A.UnitSeq AS 기준단위, B.STDUnitSeq AS 구매단위,A.ItemSeq,

ROUND((CASE WHEN ISNULL(C.ConvDen,0) = 0 THEN 0 ELSE ISNULL(200,0) * ISNULL(C.ConvNum,0) / ISNULL(C.ConvDen,0) END), @GoodQtyDecLength)

FROM _TDAItem AS A WITH(NOLOCK)

JOIN _TDAItemDefUnit AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.ItemSeq = A.ItemSeq

JOIN _TDAItemUnit AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.ItemSeq = A.ItemSeq

AND C.UnitSeq = B.STDUnitSeq

WHERE UMModuleSeq = 1116001 -- 구매

AND A.AssetSeq NOT IN (2,4)

AND A.ItemSeq = 9

--AND ISNULL(A.Qty,0) \<> 0
