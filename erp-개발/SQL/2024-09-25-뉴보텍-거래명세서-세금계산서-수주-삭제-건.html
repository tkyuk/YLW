CREATE TABLE #TInvoiceBillFIFO

(

IDX_NO INT,

IDX_Serl INT,

Seq INT,

Serl INT,

BillProgQty DECIMAL(19,5),

BillProgSTDQty DECIMAL(19,5),

BillProgAmt DECIMAL(19,5),

BillProgVAT DECIMAL(19,5)

)

CREATE INDEX IX_#TInvoiceBillFIFO ON #TInvoiceBillFIFO (IDX_NO, IDX_Serl)

DECLARE @IDX_NO INT,

@IDX_Serl INT

SELECT @IDX_NO = 0, @IDX_Serl = 0 -- 초기화

WHILE 1=1

BEGIN

SELECT TOP 1

@IDX_NO = A.IDX_NO,

@IDX_Serl = 0

FROM #TOrderBill AS A

WHERE A.IDX_NO > @IDX_NO

ORDER BY A.IDX_NO

IF @@ROWCOUNT = 0 BREAK

WHILE 1=1

BEGIN

SELECT TOP 1

@IDX_Serl = A.IDX_Serl

FROM #TOrderInvoicePre AS A

WHERE A.IDX_NO = @IDX_NO

AND A.IDX_Serl > @IDX_Serl

ORDER BY A.IDX_Serl

IF @@ROWCOUNT = 0 BREAK

INSERT INTO #TInvoiceBillFIFO

(

IDX_NO, IDX_Serl, Seq, Serl, BillProgQty, BillProgSTDQty, BillProgAmt, BillProgVAT

)

SELECT A.IDX_NO, A.IDX_Serl, A.Seq, A.Serl,

CASE WHEN A.PlusYnQty + B.PlusYnQty = 0 THEN 0 ELSE CASE WHEN ABS(A.Qty ) \< ABS(B.Qty ) THEN A.Qty ELSE B.Qty END END AS Qty ,

CASE WHEN A.PlusYnSTDQty + B.PlusYnSTDQty = 0 THEN 0 ELSE CASE WHEN ABS(A.STDQty) \< ABS(B.STDQty) THEN A.STDQty ELSE B.STDQty END END AS STDQty,

CASE WHEN A.PlusYnAmt + B.PlusYnAmt = 0 THEN 0 ELSE CASE WHEN ABS(A.Amt ) \< ABS(B.Amt ) THEN A.Amt ELSE B.Amt END END AS Amt ,

CASE WHEN A.PlusYnVAT + B.PlusYnVAT = 0 THEN 0 ELSE CASE WHEN ABS(A.VAT ) \< ABS(B.VAT ) THEN A.VAT ELSE B.VAT END END AS VAT

FROM #TOrderInvoicePre AS A

JOIN #TOrderBill AS B ON A.IDX_NO = B.IDX_NO

WHERE A.IDX_NO = @IDX_NO

AND A.IDX_Serl = @IDX_Serl

UPDATE B

SET Qty = B.Qty - A.Qty ,

STDQty = B.STDQty - A.STDQty,

Amt = B.Amt - A.Amt ,

VAT = B.VAT - A.VAT

FROM #TOrderInvoicePre AS A

JOIN #TOrderBill AS B ON A.IDX_NO = B.IDX_NO

WHERE A.IDX_NO = @IDX_NO

AND A.IDX_Serl = @IDX_Serl

IF EXISTS(SELECT TOP 1 1 FROM #TOrderBill WHERE IDX_NO = @IDX_NO AND PlusYnQty \<> CASE WHEN Qty \<= 0 THEN -1 ELSE 1 END)

BEGIN

BREAK

END

END

END

AND X.OrderSeq = O.OrderSeq

LEFT OUTER JOIN NVT_TSLOrderAGCust AS A WITH (NOLOCK) ON O.CompanySeq = A.CompanySeq

AND O.OrderSeq = A.OrderSeq

LEFT OUTER JOIN NVT_TSLInvoiceAGCust AS D WITH (NOLOCK) ON D.CompanySeq = @CompanySeq

AND X.InvoiceSeq = D.InvoiceSeq

LEFT OUTER JOIN _TDACust AS B WITH (NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.AGCustSeq = B.CustSeq

AND D.AGCustSeq = C.CustSeq
