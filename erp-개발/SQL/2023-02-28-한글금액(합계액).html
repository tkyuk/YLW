한글금액(합계액)

CREATE TABLE #HapAmt ( OrderSeq INT,HapAmt DECIMAL(19,5))

INSERT INTO #HapAmt

SELECT A.OrderSeq, SUM(A.CurAmt) + SUM(A.CurVAT)

FROM _TSLOrderItem AS A WITH (NOLOCK)

JOIN #BIZ_IN_DataBlock1 AS B ON A.OrderSeq = B.OrderSeq

WHERE A.CompanySeq = @CompanySeq

GROUP BY A.OrderSeq

JOIN #HapAmt                                AS X ON A.OrderSeq = X.OrderSeq

REPLACE(dbo._fnSLConvertHanAmt( ISNULL(X.HapAmt,0)),'영','') AS HapAmtKor, -- 합계액한글
