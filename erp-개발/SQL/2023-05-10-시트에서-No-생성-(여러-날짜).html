시트에서 No 생성 (여러 날짜)

CREATE TABLE #Tmp_Max

(

InNo NVARCHAR(100),

InDate NCHAR(8),

)

CREATE TABLE #Tmp_InNo

(

IDX_NO INT,

InNo NVARCHAR(100)

)

-- 임시테이블에 해당하는 날짜별 InNo 최대값 구하기

INSERT INTO #Tmp_Max

SELECT A.InNo,A.InDate

FROM emk_TSLWasteIn AS A

INNER JOIN ( SELECT CompanySeq, InDate, MAX(InNo) AS InNo

FROM emk_TSLWasteIn

GROUP BY CompanySeq, InDate

) AS B ON A.CompanySeq = B.CompanySeq

AND A.InDate = B.InDate

AND A.InNo = B.InNo

WHERE A.InDate IN (SELECT DISTINCT InDate FROM #BIZ_OUT_DataBlock1)

-- InNo 생성

INSERT INTO #Tmp_InNo

SELECT A.IDX_NO,

CASE WHEN ISNULL(B.InNo,'') =''

THEN A.InDate + RIGHT('0000' + CAST(ROW_NUMBER() OVER(PARTITION BY A.InDate ORDER BY A.IDX_NO) AS NVARCHAR), 4)

ELSE B.InNo + ROW_NUMBER() OVER(PARTITION BY A.InDate ORDER BY A.IDX_NO) END

FROM #BIZ_OUT_DataBlock1 AS A

LEFT OUTER JOIN #Tmp_Max AS B ON B.InDate = A.InDate

-- InNo Update

UPDATE #BIZ_OUT_DataBlock1

SET InNo = B.InNo

FROM #BIZ_OUT_DataBlock1 AS A

JOIN #Tmp_InNo AS B ON B.IDX_NO = A.IDX_NO
