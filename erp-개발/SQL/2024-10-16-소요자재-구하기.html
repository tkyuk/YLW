소요자재 구하기

-- 소요자재 수량 대비 투입수량을 보여주기 위해서 로직 추가

-- 소요자재 가져오기

CREATE TABLE #MatNeed_GoodItem

(

IDX_NO INT IDENTITY(1,1),

BizUnit INT,

ItemSeq INT, -- 제품코드

ProcRev NCHAR(2), -- 공정흐름차수

ProcSeq INT, -- 공정코드

AssyItemSeq INT, -- 공정품코드

UnitSeq INT, -- 단위코드 (공정품코드가 있으면 공정품단위코드, 없으면 제품단위코드)

Qty DECIMAL(19,5), -- 제품수량 (공정품코드가 있으면 공정품수량)

ProdPlanSeq INT, -- 생산계획내부번호 (생산의뢰에서 추가된 옵션자재를 가져오기위해)

WorkOrderSeq INT, -- 작업지시내부번호 (작업지시에서 추가자재로 등록된 자재를 가져오기위해)

WorkOrderSerl INT, -- 작업지시내부순번(작업지시에서추가자재로등록된자재를가져오기위해)

IsOut NCHAR(1) -- 로스율 적용에 사용 '1'이면 OutLossRate 적용

, WorkReportSeq INT

, OSPDelvInSeq INT

, OSPDelvInSerl INT

)

CREATE TABLE #MatNeed_MatItem_Result

(

IDX_NO INT, -- IDX

MatItemSeq INT, -- 자재코드

UnitSeq INT, -- 자재단위

NeedQty DECIMAL(19,5), -- 소요량

STDUnitSeq INT, -- 기준단위내부코드

STDNeedQty DECIMAL(19,5), -- 기준단위소요량

InputType INT, -- 자재투입구분 6042

SMDelvType INT, -- 조달구분 6032

SMAddType INT, -- 가감구분 6048

)

-- 생산건 담기

INSERT INTO #MatNeed_GoodItem

( BizUnit,ItemSeq,ProcRev,ProcSeq,AssyItemSeq

, UnitSeq

, Qty,ProdPlanSeq,WorkOrderSeq,WorkOrderSerl

, IsOut, WorkReportSeq, OSPDelvInSeq, OSPDelvInSerl

)

SELECT DISTINCT

A.BizUnit, B.GoodItemSeq, ISNULL(B.ProcRev, N'00'), B.ProcSeq, B.AssyItemSeq

, CASE ISNULL(B.ProdUnitSeq, 0) WHEN 0 THEN D.UnitSeq ELSE B.ProdUnitSeq END --2023.03.14 김남익 추가

, B.ProdQty, 0, B.WorkOrderSeq, B.WorkOrderSerl

, 0, A.WorkReportSeq, A.OSPDelvInSeq, A.OSPDelvInSerl

FROM #Input AS A

JOIN _TPDSFCWorkReport AS B WITH(NOLOCK) ON A.WorkReportSeq = B.WorkReportSeq

AND B.CompanySeq = @CompanySeq

JOIN _TDAItem AS D WITH(NOLOCK) ON B.GoodItemSeq = D.ItemSeq

AND D.CompanySeq = @CompanySeq

WHERE A.WorkReportSeq \<> 0

-- 소요자재 가져오기

EXEC dbo._SWPDMMGetItemNeedQty

@CompanySeq = @CompanySeq ,

@SMDelvType = '0' ,

@IsMRP = '0' , -- MRP여부

@IsOutsourcing = '1' , -- 외주여부

@IsMatAdd = '0' -- 추가자재만조회

CREATE INDEX IDX_Input ON #Input(IDX_No, MatItemSeq)

CREATE INDEX IDX_MatNeed_MatItem_Result ON #MatNeed_MatItem_Result(IDX_No, MatItemSeq)

-- 소요자재 가져오기 끝
