운전자금(유산스)_mcnk

차입명                        LCNo                        BL일자                        수입처                        금융기관                        수입비용전표번호                통화                        LC개설일

DROP TABLE #TMP

DROP TABLE #TMP2

DECLARE @BLDateFr NCHAR(8),

@BLDateTo NCHAR(8),

@LCNo NVARCHAR(100)

SELECT @BLDateFr = '20210701'

SELECT @BLDateTo = '20220730'

CREATE TABLE #TMP

(

BLSeq INT,

LCSeq INT,

BLNo NVARCHAR(100),

BLDate NCHAR(8)

)

INSERT INTO #TMP

SELECT B.BLSeq,

A.MngValSeq,

B.BLNo,

B.BLDate

FROM _TSLExpImpInfo AS A WITH(NOLOCK)

JOIN _TUIImpBL AS B WITH(NOLOCK) ON A.CompanySeq =B.CompanySeq AND A.KeySeq =B.BLSeq

WHERE MngSerl = 2025

AND MngValSeq \<>0 -- 수입BL등록의 수입정보 탭에 LCNo가 있는 데이터

AND BLDate >= @BLDateFr

AND BLDate \<= @BLDateTo

AND A.TableSeq = 43

SELECT

ISNULL(T.BLSeq , 0) AS BLSeq ,

ISNULL(T.LCSeq , 0) AS LCSeq ,

ISNULL(T.BLDate ,'') AS BLDate ,

ISNULL(A.CustSeq , 0) AS CustSeq , -- 수입처코드

ISNULL(G.CustName , 0) AS CustName , -- 수입처

'' AS ExpenseSlip , -- 수입비용전표번호

0 AS ExpenseSlipSeq , -- 수입비용전표코드

ISNULL(H.PaymentDate ,'') AS LCDate -- LC개설일

INTO #TMP2

FROM #TMP AS T

JOIN _TSLExpExpense AS A WITH(NOLOCK) ON A.CompanySeq =1 --@CompanySeq

AND A.SourceSeq = T.BLSeq

AND A.SMSourceType = 8215004

LEFT OUTER JOIN _TDACust AS G WITH(NOLOCK) ON G.CompanySeq = A.CompanySeq

AND G.CustSeq = A.CustSeq

JOIN _TSLImpPayment AS H WITH(NOLOCK) ON H.CompanySeq =1 --@CompanySeq

AND H.PaymentSeq = T.LCSeq

ORDER BY T.BLSeq, T.LCSeq

SELECT * FROM #TMP2 ORDER BY BLSeq, LCSeq
