행 별로 반복 / 행 수량 만큼 반복 출력

SELECT CONVERT(NVARCHAR(100),ISNULL(D.OrderNo, ''))+'-'+CONVERT(NVARCHAR(100),RIGHT('00' + CAST(ISNULL(B.OrderSerl, 0) AS NVARCHAR), 2))+ RIGHT('000' + CAST(CONVERT(NVARCHAR(100),G.number) AS NVARCHAR), 3) AS SN,

FROM #TPDSFCWorkOrder AS A

LEFT OUTER JOIN #SOInfo AS C ON A.IDX_NO = C.IDX_NO

LEFT OUTER JOIN _TSLOrderItem AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.OrderSeq = C.OrderSeq

AND B.Orderserl = C.OrderSerl

LEFT OUTER JOIN _TSLOrder AS D WITH(NOLOCK) ON D.CompanySeq = @CompanySeq

AND B.OrderSeq = D.OrderSeq

JOIN master..spt_values AS G ON 1=1

AND number BETWEEN 1 AND 1000

AND type = 'P'

AND A.OrderQty >= G.number
