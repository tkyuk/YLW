AS

SET NOCOUNT ON;

DECLARE @IDX_NO INT,

@Sql NVARCHAR(1000),

@UserName NVARCHAR(100),

@ObjectName NVARCHAR(100),

@PermissionName NVARCHAR(100)

CREATE TABLE #TSYSUserGrant

(

IDX_NO INT IDENTITY(1, 1),

UserName NVARCHAR(100),

ObjectName NVARCHAR(100),

PermissionName NVARCHAR(100)

)

INSERT INTO #TSYSUserGrant(UserName, ObjectName, PermissionName)

-- VIEW 테이블 권한 부여

SELECT 'mes' AS UserName, 'gnst_VDASMinor' AS ObjectName, 'SELECT' AS PermissionName

UNION SELECT 'mes' AS UserName, 'gnst_VDAUMinor' AS ObjectName, 'SELECT' AS PermissionName

UNION SELECT 'mes' AS UserName, 'gnst_VDAItemAssetView' AS ObjectName, 'SELECT' AS PermissionName

UNION SELECT 'mes' AS UserName, 'gnst_VDADept' AS ObjectName, 'SELECT' AS PermissionName

UNION SELECT 'mes' AS UserName, 'gnst_VDAWH' AS ObjectName, 'SELECT' AS PermissionName

UNION SELECT 'mes' AS UserName, 'gnst_VPDBaseProcessView' AS ObjectName, 'SELECT' AS PermissionName

UNION SELECT 'mes' AS UserName, 'gnst_VDAItemView' AS ObjectName, 'SELECT' AS PermissionName

UNION SELECT 'mes' AS UserName, 'gnst_TMESOrderIF' AS ObjectName, 'SELECT' AS PermissionName

SELECT @IDX_NO = 0

WHILE 1=1

BEGIN

SELECT TOP 1

@IDX_NO = A.IDX_NO,

@UserName = '',

@ObjectName = '',

@PermissionName = ''

FROM #TSYSUserGrant AS A

WHERE A.IDX_NO > @IDX_NO

ORDER BY A.IDX_NO ASC

IF @@ROWCOUNT = 0 BREAK

SELECT @UserName = A.UserName,

@ObjectName = A.ObjectName,

@PermissionName = A.PermissionName

FROM #TSYSUserGrant AS A

WHERE A.IDX_NO = @IDX_NO

IF EXISTS(SELECT TOP 1 1 FROM sys.database_principals AS B WITH(NOLOCK) WHERE B.Name = @UserName)

AND EXISTS(SELECT TOP 1 1 FROM sys.objects AS C WITH(NOLOCK) WHERE C.Name = @ObjectName)

BEGIN

IF NOT EXISTS(SELECT TOP 1 1

FROM sys.database_permissions AS A WITH(NOLOCK)

JOIN sys.database_principals AS B WITH(NOLOCK) ON B.Principal_ID = A.Grantee_Principal_ID

JOIN sys.objects AS C WITH(NOLOCK) ON C.Object_ID = A.Major_ID

WHERE A.Permission_Name = @PermissionName

AND B.Name = @UserName

AND C.Name = @ObjectName

)

BEGIN

SELECT @Sql = N'GRANT ' + @PermissionName + N' ON OBJECT::' + @ObjectName + N' TO ' + @UserName

EXEC sp_executeSql @Sql

PRINT @Sql

END

END

END

DROP TABLE #TSYSUserGrant

RETURN
