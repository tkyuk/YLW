FROM _TCOMSourceDailyTrace AS Z WITH(NOLOCK)

WHERE Z.ToTableSeq = 49 -- 구매발주

SELECT DISTINCT FromTableSeq,A.*

FROM _TCOMSourceDaily AS Z WITH(NOLOCK)

JOIN _TCOMProgTable AS A ON A.ProgTableSeq = FromTableSeq

WHERE Z.ToTableSeq = 49 -- 구매발주

SELECT EmpSeq AS ReqEmpSeq

FROM _TCOMSourceDailyTrace AS Z WITH(NOLOCK)

JOIN _TPUORDPOReq AS D WITH(NOLOCK) ON Z.FromSeq = D.POReqSeq

WHERE Z.FromTableSeq = 12 -- 구매요청

AND Z.ToTableSeq = 48 -- 구매발주
