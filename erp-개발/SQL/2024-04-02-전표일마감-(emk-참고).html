전표일마감 (emk 참고)

DECLARE @ClosingSeq INT,

@AccYM NCHAR(6),

@StartDate NCHAR(8),

@EndDate NCHAR(8)

SELECT @ClosingSeq = 88 -- 마감업무코드(전표일마감)

--SELECT @ClosingSeq = 1145 -- 마감업무코드(승인일마감) 240222 tgkim 전표일마감 -> 승인일마감으로 변경 (원에코에만 적용했으며 다른DB에는 마감업무코드 88로 적용되어있음)

SELECT @AccYM = ISNULL(A.AccYM, '')

FROM #BIZ_OUT_DataBlock1 AS A

-- MR기준월입력_emk 에 등록된 기준월이 있을 경우 해당 월의 일자기간을, 아닐 경우 회계연월의 1일 ~ 말일로 일마감 처리합니다.

SELECT @StartDate = A.MRFrDate,

@EndDate = A.MRToDate

FROM emk_TACMRYM AS A WITH(NOLOCK)

WHERE A.CompanySeq = @CompanySeq

AND A.MRYM = @AccYM

IF ISNULL(@StartDate, '') = ''

BEGIN

SELECT @StartDate = @AccYM + '01'

SELECT @EndDate = CONVERT(NCHAR(8), EOMONTH(LEFT(@AccYM, 4) + '-' + RIGHT(@AccYM, 2) + '-01'), 112)

END

-- [회계업무]전표월마감 테이블에 데이터가 있으면 UPDATE

UPDATE A

SET IsClose = ISNULL(B.IsCfm, '0')

FROM _TCOMClosingDate AS A

-- 이태욱(2202): DtlUnitSeq 제외하고 조인

JOIN #BIZ_OUT_DataBlock1 AS B ON A.ClosingDate BETWEEN @StartDate AND @EndDate

AND A.ClosingSeq = @ClosingSeq --마감업무코드

AND A.CompanySeq = @CompanySeq

WHILE @StartDate \<= @EndDate

BEGIN

IF NOT EXISTS(SELECT TOP 1 1

FROM _TCOMClosingDate AS A WITH(NOLOCK)

WHERE A.ClosingDate = @StartDate

AND A.ClosingSeq = @ClosingSeq --마감업무코드

AND A.CompanySeq = @CompanySeq)

BEGIN

-- [회계업무]전표월마감 테이블에 데이터가 없으면 INSERT

INSERT INTO _TCOMClosingDate (CompanySeq, ClosingSeq , ClosingDate, UnitSeq, DtlUnitSeq,

IsClose , LastUserSeq, LastDateTime)

SELECT @CompanySeq, @ClosingSeq, @StartDate, B.AccUnit, C.SlipUnit,

ISNULL(A.IsCfm, '0'), @UserSeq, GETDATE()

FROM #BIZ_OUT_DataBlock1 AS A

LEFT OUTER JOIN _TDAAccUnit AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

LEFT OUTER JOIN _TACSlipUnit AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

END

-- + 1일

SELECT @StartDate = CONVERT(NCHAR(8), DATEADD(DAY, 1, LEFT(@StartDate, 4) + '-' + SUBSTRING(@StartDate, 5, 2) + '-' + RIGHT(@StartDate, 2)), 112)

END
