현재고 : 조회하는 시점 기준 사업부내 모든 창고의 재고 수량

-- 임시 테이블 생성

CREATE TABLE #GetInOutItem (ItemSeq INT)

CREATE TABLE #GetInOutStock

(

BizUnit INT ,

WHSeq INT ,

MainWHSeq INT ,

ItemSeq INT ,

UnitSeq INT ,

PrevQty DECIMAL(19, 5),

InQty DECIMAL(19, 5),

OutQty DECIMAL(19, 5),

StockQty DECIMAL(19, 5),

STDPrevQty DECIMAL(19, 5),

STDInQty DECIMAL(19, 5),

STDOutQty DECIMAL(19, 5),

STDStockQty DECIMAL(19, 5),

StockAmt DECIMAL(19, 5)

)

-- 대상 품목 수집

INSERT INTO #GetInOutItem (ItemSeq)

SELECT A.ItemSeq

FROM #Fix AS A

JOIN _TDAItem AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.ItemSeq = A.ItemSeq

JOIN _TDAItemAsset AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.AssetSeq = B.AssetSeq

WHERE C.IsQty \<> '1'

-- 재고조회

EXEC _SWLGGetInOutStock @CompanySeq = @CompanySeq,

@BizUnit = @BizUnit ,

@DateFr = @FoProdDate,

@DateTo = @FoProdDate,

@WHSeq = 0 ,

@SMWHKind = 0 ,

@IsSubDisplay = '1' ,

@IsOnlyStock = '1' ,

@IsStockAmt = '0' ,

@StockAmtType = 0 ,

@UnitType = 1 ,

@ConvUnitSeq = 0 ,

@QryType = 'S' ,

@IsTrustCust = ''

-- 창고상관없이 품목별로 집계

SELECT ItemSeq, SUM(StockQty) AS StkQty

INTO #StkQty

FROM #GetInOutStock

GROUP BY ItemSeq

-- 현재고

UPDATE A

SET StkQty = ISNULL(B.StkQty, 0)

FROM #Fix AS A

LEFT OUTER JOIN #StkQty AS B ON A.ItemSeq = B.ItemSeq
