AND C.MajorSeq = 1110142

AND C.Serl = 15000001

AND C.ValueSeq = A.UMStatusDetail

AND ISNULL(A.UMStatusDetailOld,0) \<> 0

AND C.MajorSeq = 1110142

AND C.Serl = 15000001

AND C.ValueSeq = A.UMStatusDetail

JOIN _TDAUMinorValue AS D WITH(NOLOCK) ON C.CompanySeq = D.CompanySeq -- mcm_품목상태별이메일정보.받는사람이메일
