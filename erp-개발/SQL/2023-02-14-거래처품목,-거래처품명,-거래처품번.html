거래처품목, 거래처품명, 거래처품번

CASE WHEN ISNULL(N.CustItemName, '') = '' THEN N2.CustItemName

ELSE N.CustItemName END AS CustItemName,

CASE WHEN ISNULL(N.CustItemNo, '') = '' THEN N2.CustItemNo

ELSE N.CustItemNo END AS CustItemNo,

LEFT OUTER JOIN _TSLCustItem         AS N WITH(NOLOCK) ON N.CompanySeq = A.CompanySeq

AND N.ItemSeq         = A.ItemSeq

AND N.CustSeq         = A.CustSeq

AND N.UnitSeq         = A.UnitSeq

LEFT OUTER JOIN _TSLCustItem         AS N2 WITH(NOLOCK) ON N2.CompanySeq = A.CompanySeq

AND N2.ItemSeq         = A.ItemSeq

AND N2.CustSeq         = A.CustSeq

AND N2.UnitSeq         = 0
