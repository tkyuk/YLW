활동센터간배부방법등록 조회

SELECT CASE WHEN A.LogType = 'D' THEN '삭제'

WHEN A.LogType = 'U' THEN '수정'

WHEN A.LogType = 'X' THEN '일괄삭제' END AS '타입',

A.LogDateTime AS '실행일시',

E.UserName AS '실행자',

M.CostYM AS '원가년월' ,

A.AllocName AS '배부처리명',

B1.CCtrName AS '보내는 활동센터',

C1.CCtrName AS '받는 활동센터',

CASE WHEN D.CostAccSeq = 0 THEN '아래 계정을 제외한 계정에 대한 설정'

ELSE D1.CostAccName

END AS '관리회계 계정과목',

D2.DriverName AS '자원동인 배부기준'

FROM _TESMDProdAllocNameLog AS A WITH(NOLOCK)

JOIN _TESMDCostKey AS M WITH(NOLOCK) ON A.CompanySeq = M.CompanySeq

AND A.CostKeySeq = M.CostKeySeq

LEFT OUTER JOIN _TESMDProdCCtrSendLOG AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.AllocSeq = A.AllocSeq

LEFT OUTER JOIN _TCAUser AS E WITH(NOLOCK) ON E.CompanySeq = A.CompanySeq

AND E.UserSeq = A.LogUserSeq

LEFT OUTER JOIN _TDACCtr AS B1 WITH(NOLOCK) ON B1.CompanySeq = B.CompanySeq

AND B1.CCtrSeq = B.SendCCtrSeq

LEFT OUTER JOIN _TESMDProdCCtrRevLOG AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.AllocSeq = A.AllocSeq

LEFT OUTER JOIN _TDACCtr AS C1 WITH(NOLOCK) ON C1.CompanySeq = C.CompanySeq

AND C1.CCtrSeq = C.RevCCtrSeq

LEFT OUTER JOIN _TESMDProdCCtrAccLOG AS D WITH(NOLOCK) ON D.CompanySeq = A.CompanySeq

AND D.AllocSeq = A.AllocSeq

LEFT OUTER JOIN _TESMBAccount AS D1 WITH(NOLOCK) ON D1.CompanySeq = D.CompanySeq

AND D1.CostAccSeq = D.CostAccSeq

LEFT OUTER JOIN _TESMBDriver AS D2 WITH(NOLOCK) ON D2.CompanySeq = D.CompanySeq

AND D2.DriverSeq = D.DriverSeq

WHERE CONVERT(NCHAR(8),A.LogDateTime,112) BETWEEN '20241001' AND '20241130'

OR CONVERT(NCHAR(8),B.LogDateTime,112) BETWEEN '20241001' AND '20241130'

OR CONVERT(NCHAR(8),C.LogDateTime,112) BETWEEN '20241001' AND '20241130'

OR CONVERT(NCHAR(8),D.LogDateTime,112) BETWEEN '20241001' AND '20241130'
