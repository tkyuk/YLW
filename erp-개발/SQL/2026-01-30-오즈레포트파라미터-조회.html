오즈레포트파라미터 조회

-- SS2 Param 등록

SELECT E.Caption,

E.PgmId,

A.OZPSeq,

B.OZPID,

C.OZPSerl,

C.OZPSerlID,

C.DtlRemark,

C.OZPEx,

C.OZPSubSerl,

C.SubParamID,

A.OZParam,

A.IsUserChg,

A.IsEnv,

A.EnvSeq,

D.EnvName

FROM _TCAOZReport AS A WITH(NOLOCK)

LEFT OUTER JOIN _TCAOZParam AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.OZPSeq = B.OZPSeq

LEFT OUTER JOIN _TCAOZParamItem AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.OZPSeq = C.OZPSeq

AND A.OZPSerl = C.OZPSerl

LEFT OUTER JOIN _TCOMEnv AS D WITH(NOLOCK) ON A.CompanySeq = D.CompanySeq

AND A.EnvSeq = D.EnvSeq

LEFT OUTER JOIN _TCAPgm AS E WITH(NOLOCK) ON E.PgmSeq = A.ReportSeq

ORDER BY B.Sort, C.DtlSort

-- SS3 Variant 등록

SELECT E.Caption,

E.PgmId,

A.VariantSeq AS VariantSeq,

A.VariantName AS VariantName,

A.VariantName AS VariantNameOld,

A.Remark AS Remark,

B.VariantValue AS VariantValue,

B.IsUserChg AS IsUserChg,

B.IsEnv AS IsEnv,

B.EnvSeq AS EnvSeq,

C.EnvName AS EnvName,

B.ReportSeq AS PrintPgmSeq,

B.Sort AS Sort

FROM _TCAOZReport AS Z

LEFT OUTER JOIN _TCAOZVariantValue AS B WITH(NOLOCK) ON B.ReportSeq = Z.ReportSeq

LEFT OUTER JOIN _TCAOZVariant AS A WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.VariantSeq = A.VariantSeq

LEFT OUTER JOIN _TCOMEnv AS C WITH(NOLOCK) ON B.CompanySeq = C.CompanySeq

AND B.EnvSeq = C.EnvSeq

LEFT OUTER JOIN _TCAPgm AS E WITH(NOLOCK) ON E.PgmSeq = Z.ReportSeq

UNION ALL

SELECT E.Caption,

E.PgmId,

B.VariantSeq AS VariantSeq,

B.VariantName AS VariantName,

B.VariantName AS VariantNameOld,

B.Remark AS Remark,

B.VariantValue AS VariantValue,

B.IsUserChg AS IsUserChg,

B.IsEnv AS IsEnv,

B.EnvSeq AS EnvSeq,

C.EnvName AS EnvName,

B.ReportSeq AS PrintPgmSeq,

B.Sort AS Sort

FROM _TCAOZReport AS Z

LEFT OUTER JOIN _TCAOZVariantValue AS B WITH(NOLOCK) ON B.ReportSeq = Z.ReportSeq

LEFT OUTER JOIN _TCOMEnv AS C WITH(NOLOCK) ON B.CompanySeq = C.CompanySeq

AND B.EnvSeq = C.EnvSeq

LEFT OUTER JOIN _TCAPgm AS E WITH(NOLOCK) ON E.PgmSeq = Z.ReportSeq

ORDER BY B.Sort
