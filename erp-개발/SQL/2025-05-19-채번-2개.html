채번 2개

--프로젝트 번호가 없으면

IF EXISTS ( SELECT 1 FROM #BIZ_OUT_DataBlock2 WHERE ISNULL(PJTNo,'') = '' AND WorkingTag \<> 'D' )

BEGIN

SELECT @MasterPJTNo = B.PJTNo

,@MasterPJTSeq = A.MasterPJTSeq

FROM #BIZ_OUT_DataBlock2 AS A

JOIN _TPJTProject AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.PJTSeq = A.MasterPjtSeq

--==========================================================================

-- 단암 채번 시작

--==========================================================================

/*

채번 규칙

제품프로젝트번호 = 사업프로젝트번호 + 품목소분류 (추가정보정의 : das_소분류채번코드) + 제품프로젝트명

ex. 소분류 (일치) 제품프로젝트명 ( 일치) => A-001 / A-002 / A-003

소분류 (일치) 제품프로젝트명 (불일치) => A-001 / B-001 / C-001

*/

-- 해당 사업프로젝트에 저장되어있는 데이터 담기

INSERT INTO #Tmp_MaxPJT(PJTName, ClassSerl, EngSerl, MaxSerl)

SELECT A.PJTName

,SUBSTRING(A.PJTNo,LEN(@MasterPJTNo)+1,2) AS ClassSerl

,SUBSTRING(A.PJTNo,LEN(@MasterPJTNo)+3,1) AS EngSerl

,MAX(RIGHT(A.PJTNo,3)) AS MaxSerl

FROM _TPJTProject AS A WITH(NOLOCK)

JOIN das_TPJTProject AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.PJTSeq = A.PJTSeq

WHERE B.MasterPjtSeq = @MasterPJTSeq

AND TRY_CAST(RIGHT(PJTNo, 3) AS INT) \< 900

GROUP BY A.PJTName

,SUBSTRING(A.PJTNo,LEN(@MasterPJTNo)+1,2)

,SUBSTRING(A.PJTNo,LEN(@MasterPJTNo)+3,1)

-- MAX 값 추출

SELECT ClassSerl, MAX(EngSerl) AS Max_EngSerl

INTO #Tmp_MaxClass

FROM #Tmp_MaxPJT

GROUP BY ClassSerl

INSERT INTO #Tmp_Num(DataSeq, ClassSeq, PJTName, ClassSerl, ClassPJT_RowCnt,TmpNo, EngSerl, MaxSerl, Max_EngSerl)

SELECT A.DataSeq

,A.ClassSeq

,A.PJTName

,B.ValueText AS ClassSerl

,ROW_NUMBER() OVER(PARTITION BY A.ClassSeq, A.PJTName ORDER BY (SELECT IDX_NO)) AS ClassPJT_RowCnt

,@MasterPJTNo + CONVERT(NCHAR(2),B.ValueText)

,C.EngSerl

,ISNULL(C.MaxSerl,'000')

,D.Max_EngSerl

FROM #BIZ_OUT_DataBlock2 AS A

LEFT OUTER JOIN _TDAUMinorValue AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.MinorSeq = A.ClassSeq

AND B.Serl = 15000029

LEFT OUTER JOIN #Tmp_MaxPJT AS C ON C.PJTName = A.PJTName

AND C.ClassSerl = B.ValueText

LEFT OUTER JOIN #Tmp_MaxClass AS D ON D.ClassSerl = B.ValueText

WHERE A.WorkingTag IN ('A')

AND A.Status = 0

AND ISNULL(A.PJTNo,'') = ''

-- 소분류 추가정보정의 세팅 없는 경우 에러 처리

UPDATE #BIZ_OUT_DataBlock2

SET Result = '소분류 채번코드 값이 미입력되었습니다.',

MessageType = 17009,

Status = 9999

FROM #BIZ_OUT_DataBlock2 AS A

JOIN #Tmp_Num AS B ON B.ClassSeq = A.ClassSeq

WHERE A.WorkingTag IN ('A')

AND A.Status = 0

AND ISNULL(B.ClassSerl,'') = ''

-- 소분류 코드 채번하기 위한 임시테이블 생성

SELECT DataSeq

,NCHAR(ASCII(A.Max_EngSerl) + DENSE_RANK() OVER(PARTITION BY A.ClassSeq ORDER BY A.PJTName)) AS EngSerl

INTO #Tmp_Num_Max

FROM #Tmp_Num AS A

WHERE A.EngSerl IS NULL

-- 소분류 코드 업데이트

UPDATE A

SET EngSerl = B.EngSerl

FROM #Tmp_Num AS A

JOIN #Tmp_Num_Max AS B ON B.DataSeq = A.DataSeq

-- 순번 업데이트

UPDATE A

SET Serl = RIGHT('000' + CONVERT(NVARCHAR(20), A.ClassPJT_RowCnt + A.MaxSerl),3)

FROM #Tmp_Num AS A

--SELECT ClassSeq,PJTName,TmpNo + EngSerl + '-'+ Serl

-- FROM #Tmp_Num

UPDATE #BIZ_OUT_DataBlock2

SET PJTNo = B.TmpNo + B.EngSerl + '-'+ B.Serl

,ItemNo = B.TmpNo + B.EngSerl + '-'+ B.Serl

FROM #BIZ_OUT_DataBlock2 AS A

JOIN #Tmp_Num AS B ON B.DataSeq = A.DataSeq

WHERE WorkingTag = 'A'

AND Status = 0

AND ISNULL(A.PJTNo,'') = ''

--SET ROWCOUNT 0

--==========================================================================

-- 종료 : 단암 채번

--==========================================================================

END
