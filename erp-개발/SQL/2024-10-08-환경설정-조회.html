--==============================================================

-- 환경설정 조회

--==============================================================

SELECT Z.ConfigName,B.*

FROM _TCOMEnvConfig AS Z

JOIN _TCOMEnvConfigItem AS A WITH(NOLOCK) ON A.ConfigSeq = Z.ConfigSeq

JOIN _TCOMEnv AS B WITH(NOLOCK) ON A.EnvSeq = B.EnvSeq

WHERE B.EnvSeq LIKE '%8876%'

AND B.EnvName LIKE '%%'
