관리자 이메일 / SMTP 서버 관리자 ID

-- 공통버전 받아오기

DECLARE @IsAppVersion                INT

,@AdminMailUserId NVARCHAR(1000)

SELECT @IsAppVersion = CASE WHEN CONVERT(INT, REPLACE (A.AppVersion, '.', '') + REPLICATE('0', 9 - LEN(REPLACE(A.AppVersion, '.', '')))) >= 617100800 THEN 1

ELSE 0 END

FROM (SELECT CASE WHEN AppVersion = N'MASTER' THEN N'999900000'

WHEN ISNULL(AppVersion, '') = '' THEN N'000000000'

ELSE AppVersion END AS AppVersion

FROM OTCAServerConfig) AS A

IF @IsAppVersion = 0

BEGIN

SELECT @AdminMailUserId = A.AdminMailUserId

FROM _TCACompanyConfig AS A WITH(NOLOCK)

WHERE A.CompanySeq = @CompanySeq

END

ELSE

BEGIN

EXEC _SYMMETRICKeyOpen

SELECT dbo._FWCOMDecrypt(A.AdminMailUserId, N'_TCACompanyConfig', 'AdminMailUserId', @CompanySeq) AS AdminMailUserId

FROM _TCACompanyConfig AS A WITH(NOLOCK)

WHERE A.CompanySeq = @CompanySeq

END
