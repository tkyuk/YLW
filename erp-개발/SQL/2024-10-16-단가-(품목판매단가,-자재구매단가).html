단가 (품목판매단가, 자재구매단가)

-- 품목판매단가 구하기.

SELECT B.ItemSeq, B.UnitSeq, MAX(B.StdPrice) Price

INTO #GoodItemPrice

FROM (SELECT DISTINCT GoodItemSeq AS ItemSeq

FROM #Input) AS A JOIN _TSLItemUnitPrice AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND A.ItemSeq = B.ItemSeq

AND @STDDate BETWEEN B.SDate AND B.EDate

AND B.CurrSeq = @DomCurrSeq

GROUP BY B.ItemSeq, B.UnitSeq

-- 자재구매단가 구하기.

SELECT A.ItemSeq AS MatItemSeq, MAX(B.Price) AS MatPrice

INTO #MatItemPrice

FROM (SELECT DISTINCT MatItemSeq AS ItemSeq, MatUnitSeq AS UnitSeq

FROM #Input) AS A JOIN _TDAItemDefUnit AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

AND A.ItemSeq = C.ItemSeq

AND C.UMModuleSeq = 1116001 -- 구매단위 select * from _TDASMinor where MajorSeq = 1116

JOIN _TPUBASEBuyPriceItem AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.ItemSeq = A.ItemSeq

AND B.UnitSeq = C.STDUnitSeq -- 구매단위로 변경

AND B.CurrSeq = @DomCurrSeq

GROUP BY A.ItemSeq
