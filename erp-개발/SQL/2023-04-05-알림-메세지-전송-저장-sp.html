알림 메세지 전송 저장 sp

SELECT @DeptSeq = DeptSeq FROM _TCAUser WITH (NOLOCK) where UserSeq = @UserSeq

SELECT @ID = 0

WHILE (1=1)

BEGIN

SELECT TOP 1

@ID = DataSeq,

@Msg1 = ISNULL(B.CustName ,''),

@CustSeq = CONVERT(NVARCHAR(10), A.CustSeq),

@Msg2 = ISNULL(C.ItemName ,''),

@Msg3 = CASE WHEN A.DVDate = '' THEN '' ELSE

SUBSTRING(ISNULL(A.DVDate ,''),1,4) + '-' +

SUBSTRING(ISNULL(A.DVDate ,''),5,2) + '-' +

SUBSTRING(ISNULL(A.DVDate ,''),7,2) END,

@Msg4 = dbo._fnGWAmtCommma(@CompanySeq,2,1,A.Qty,0,0,2,0) ,

@Msg5 = Dummy2

FROM #BIZ_OUT_DataBlock1 A

LEFT OUTER JOIN _TDACust B WITH(NOLOCK) ON B.CompanySeq =@CompanySeq AND A.CustSeq = B.CustSeq

LEFT OUTER JOIN _TDAItem C WITH(NOLOCK) ON C.CompanySeq =@CompanySeq AND A.ItemSeq = C.ItemSeq

WHERE A.DataSeq > @ID

AND A.WorkingTag = 'A'

AND A.Status = 0

ORDER BY DataSeq

IF @@ROWCOUNT = 0 BREAK

SELECT @JumpData = N'\<ROOT>\<DataBlock1>\<CustSeq> ' + @CustSeq + ' \</CustSeq >\<CustName>' + @Msg1 +'\</CustName>\</DataBlock1>\</ROOT>'

EXEC _SWCAMessageSendProc

@WorkingTag, -- (필수) WorkingTag (공란)

@CompanySeq, -- (필수) 법인내부코드

@LanguageSeq, -- (필수) 언어내부코드

@UserSeq, -- (필수) 발신자내부코드

@DeptSeq, -- (필수) 발신자의 부서내부코드

1, -- (필수) 메시지템플릿내부코드

-- 프로그램 관련 정보는 알림 메시지에서 프로그램으로 Jump 처리가 필요한 경우 입력

0, -- 프로그램의 Key값 (ex. 거래처내부코드, 수주내부코드 등)

100520055, -- 화면내부코드

@PgmSeq, -- JUMP 화면내부코드

150003, -- JUMP 유형 (150001 : 신규, 150002 : 조회, 150003 : 저장, 150006 : 추가)

@JumpData, -- JUMP 데이터 XML

@Msg1, -- 메시지 내용에 포함할 내용1

@Msg2, -- 메시지 내용에 포함할 내용2

@Msg3, -- 메시지 내용에 포함할 내용3

@Msg4, -- 메시지 내용에 포함할 내용4

@Msg5 -- 메시지 내용에 포함할 내용5

END
