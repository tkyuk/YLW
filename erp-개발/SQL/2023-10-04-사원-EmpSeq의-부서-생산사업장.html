사원 EmpSeq의 부서 / 생산사업장

_fnAdmEmpOrdRet(@CompanySeq,'') => 사원 코드도움

UPDATE A

SET A.DeptSeq = ISNULL(B.DeptSeq,0),

A.FactUnit = ISNULL(C.FactUnit,1)

FROM #TPDMMOutReqM AS A

LEFT OUTER JOIN _fnAdmEmpOrd(1,'') AS B ON B.EmpSeq = A.EmpSeq

LEFT OUTER JOIN _TDADept AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

AND C.DeptSeq = B.DeptSeq
