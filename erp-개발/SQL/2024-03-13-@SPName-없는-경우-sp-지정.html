/************************************************************************************************

설 명 - 근로소득원천징수영수증

작 성 일 - 2015-01-13

작 성 자 - CREATED BY

수정내역 - 2018.01.04 분개 SP 수정(2016 -> 2017)

- 2019.11.26 이범확 : 2018 -> 2019

- 2020.12.19 구은혜 : 2019 -> 2020

- 2021.12.29 이설우 : 2020 -> 2021

*************************************************************************************************/

CREATE PROCEDURE cymc_SWPRAdjTotAbrIncomePrint

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

AS

DECLARE @YY NCHAR(4)

,@SPName NVARCHAR(200)

SELECT @YY = ISNULL(YY, '')

FROM #BIZ_IN_DataBlock1

SELECT @SPName = '_SWPRAdjTotAbrIncomePrint_' + @YY

-- 2023년 인 경우 사이트 SP 호출 (직인 공백) 24.03.13 SJPARK

IF @YY = '2023'

SELECT @SPName = 'cymc_SWPRAdjTotAbrIncomePrint_2023'

IF NOT EXISTS(SELECT 1 FROM SYSOBJECTS WHERE id = object_id(@SPName) AND sysstat & 0xf = 4)

BEGIN

SELECT @SPName = 'dbo._SWPRAdjTotAbrIncomePrint_2021'

END

EXEC @SPName @ServiceSeq ,

@WorkingTag ,

@CompanySeq ,

@LanguageSeq ,

@UserSeq ,

@PgmSeq ,

@IsTransaction

RETURN
