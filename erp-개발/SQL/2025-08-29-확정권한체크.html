확정권한체크

-- 20250811.확정 권한체크 표준 확정관리 세팅하여 사용(권한용)

DECLARE @EnvValue60 NVARCHAR(2000),

@ConfirmTableSeq INT,

@ConfirmSeq INT

EXEC dbo._SWCOMEnv @CompanySeq, 60, @UserSeq, @@PROCID, @EnvValue60 OUTPUT

SELECT @ConfirmTableSeq = ConfirmTableSeq

FROM _TCOMConfirmTable

WHERE CompanySeq = @CompanySeq

AND ConfirmTableName = 'das_TDAItemReq'

SELECT @ConfirmSeq = ConfirmSeq

FROM _TCOMConfirmDef

WHERE CompanySeq = @CompanySeq

AND ConfirmTableSeq = @ConfirmTableSeq

CREATE TABLE #TCOMConfirm

(

IDX_NO INT,

Result NVARCHAR(250),

MessageType INT,

Status INT

)

INSERT INTO #TCOMConfirm(IDX_NO, Result, MessageType, Status)

SELECT IDX_NO,

Result ,

MessageType,

Status

FROM #BIZ_OUT_DataBlock2

WHERE Status = 0

IF ISNULL(@EnvValue60,'') = '' OR ISNULL(@EnvValue60,'') = '0'

BEGIN

EXEC _SWCOMConfirmSecuCheck @CompanySeq, @ConfirmSeq, @UserSeq, @LanguageSeq

END

IF EXISTS(SELECT TOP 1 1 FROM #TCOMConfirm WHERE Status \<> 0)

BEGIN

UPDATE A

SET Result = B.Result,

MessageType = B.MessageType,

Status = B.Status

FROM #BIZ_OUT_DataBlock2 AS A

JOIN #TCOMConfirm AS B ON A.IDX_NO = B.IDX_NO

WHERE A.Status = 0

AND B.Status \<> 0

END
