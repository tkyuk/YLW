단가 가져오기

/*

PriceStep :

0 단가 적용 안됨

1 품목단가 (단위 일치)

2 품목단가 (단위 불일치 : 기준단위)

3 품목단가 (단위 불일치 : 등록단위 유일)

4 유통단가 (단위 일치)

5 거래처단가(단위 일치)

6 프로모션단가

7 옵션단가

*/

----------------------------------- 1. 품목단가

--------------- 1 품목단가(단위 일치)

UPDATE #BIZ_OUT_DataBlock1

SET ItemStdPrice = B.StdPrice,

CustStdPrice = B.StdPrice,

ItemMinPrice = B.MinPrice,

PriceStep = 1

FROM #BIZ_OUT_DataBlock1 A

JOIN _TSLItemUnitPrice B ON A.ItemSeq = B.ItemSeq

AND A.UnitSeq = B.UnitSeq

AND A.CurrSeq = B.CurrSeq

AND A.SalesDate BETWEEN B.SDate AND B.EDate

WHERE B.CompanySeq = @CompanySeq

AND A.OptionSeq = 0

--------------- 2 품목단가(단위 불일치 : 기준단위)

UPDATE #BIZ_OUT_DataBlock1

SET ItemStdPrice = Case A.CurrSeq when @StdDomCurrSeq

THEN Round(B.StdPrice * CASE AAA.ConvDen WHEN 0 THEN 0 ELSE AAA.ConvNum / AAA.ConvDen END, @GoodDomPriceDecLength)

ELSE Round(B.StdPrice * CASE AAA.ConvDen WHEN 0 THEN 0 ELSE AAA.ConvNum / AAA.ConvDen END, @GoodPriceDecLength) END,

CustStdPrice = Case A.CurrSeq when @StdDomCurrSeq

THEN Round(B.StdPrice * CASE AAA.ConvDen WHEN 0 THEN 0 ELSE AAA.ConvNum / AAA.ConvDen END, @GoodDomPriceDecLength)

ELSE Round(B.StdPrice * CASE AAA.ConvDen WHEN 0 THEN 0 ELSE AAA.ConvNum / AAA.ConvDen END, @GoodPriceDecLength) END,

ItemMinPrice = Case A.CurrSeq when @StdDomCurrSeq

THEN Round(B.MinPrice * CASE AAA.ConvDen WHEN 0 THEN 0 ELSE AAA.ConvNum / AAA.ConvDen END, @GoodDomPriceDecLength)

ELSE Round(B.MinPrice * CASE AAA.ConvDen WHEN 0 THEN 0 ELSE AAA.ConvNum / AAA.ConvDen END, @GoodPriceDecLength) END,

PriceStep = 2

FROM #BIZ_OUT_DataBlock1 A

JOIN _TDAItem AA On A.ItemSeq = AA.ItemSeq

JOIN _TDAItemUnit AAA On A.ItemSeq = AAA.ItemSeq

AND A.UnitSeq = AAA.UnitSeq

AND AAA.CompanySeq = @CompanySeq

JOIN _TSLItemUnitPrice B ON A.ItemSeq = B.ItemSeq

AND A.CurrSeq = B.CurrSeq

AND A.SalesDate Between B.SDate And B.EDate

AND AA.UnitSeq = B.UnitSeq

AND AA.CompanySeq = B.CompanySeq

WHERE AA.CompanySeq = @CompanySeq AND A.PriceStep = 0

AND A.OptionSeq = 0

--------------- 3 품목단가(단위 불일치 : 단가유일)

UPDATE #BIZ_OUT_DataBlock1

SET ItemStdPrice = Case A.CurrSeq when @StdDomCurrSeq

THEN Round(B.StdPrice * CASE AAA.ConvDen WHEN 0 THEN 0 ELSE (AAA.ConvNum / AAA.ConvDen) END

* CASE BBB.ConvNum WHEN 0 THEN 0 ELSE (BBB.ConvDen / BBB.ConvNum) END, @GoodDomPriceDecLength)

ELSE Round(B.StdPrice * CASE AAA.ConvDen WHEN 0 THEN 0 ELSE (AAA.ConvNum / AAA.ConvDen) END

* CASE BBB.ConvNum WHEN 0 THEN 0 ELSE (BBB.ConvDen / BBB.ConvNum) END, @GoodPriceDecLength) END,

CustStdPrice = Case A.CurrSeq when @StdDomCurrSeq

THEN Round(B.StdPrice * CASE AAA.ConvDen WHEN 0 THEN 0 ELSE (AAA.ConvNum / AAA.ConvDen) END

* CASE BBB.ConvNum WHEN 0 THEN 0 ELSE (BBB.ConvDen / BBB.ConvNum) END, @GoodDomPriceDecLength)

ELSE Round(B.StdPrice * CASE AAA.ConvDen WHEN 0 THEN 0 ELSE (AAA.ConvNum / AAA.ConvDen) END

* CASE BBB.ConvNum WHEN 0 THEN 0 ELSE (BBB.ConvDen / BBB.ConvNum) END, @GoodPriceDecLength) END,

ItemMinPrice = Case A.CurrSeq when @StdDomCurrSeq

THEN Round(B.MinPrice * CASE AAA.ConvDen WHEN 0 THEN 0 ELSE (AAA.ConvNum / AAA.ConvDen) END

* CASE BBB.ConvNum WHEN 0 THEN 0 ELSE (BBB.ConvDen / BBB.ConvNum) END, @GoodDomPriceDecLength)

ELSE Round(B.MinPrice * CASE AAA.ConvDen WHEN 0 THEN 0 ELSE (AAA.ConvNum / AAA.ConvDen) END

* CASE BBB.ConvNum WHEN 0 THEN 0 ELSE (BBB.ConvDen / BBB.ConvNum) END, @GoodPriceDecLength) END,

PriceStep = 3

FROM #BIZ_OUT_DataBlock1 A

JOIN (SELECT X.IDX_NO, Y.CompanySeq, Y.ItemSeq, Y.CurrSeq, X.SalesDate

FROM #BIZ_OUT_DataBlock1 X

JOIN _TSLItemUnitPrice Y WITH(NOLOCK) ON Y.CompanySeq = @CompanySeq

AND Y.ItemSeq = X.ItemSeq

AND Y.CurrSeq = X.CurrSeq

AND X.SalesDate BETWEEN Y.SDate And Y.EDate

AND X.PriceStep = 0

GROUP BY X.IDX_NO, Y.CompanySeq, Y.ItemSeq, Y.CurrSeq, X.SalesDate

HAVING COUNT(1) = 1) AA ON A.IDX_NO = AA.IDX_NO

JOIN _TSLItemUnitPrice B ON AA.ItemSeq = B.ItemSeq

AND AA.CurrSeq = B.CurrSeq

AND AA.SalesDate Between B.SDate And B.EDate

AND AA.CompanySeq = B.CompanySeq

JOIN _TDAItemUnit AAA On A.ItemSeq = AAA.ItemSeq

AND A.UnitSeq = AAA.UnitSeq

AND AAA.CompanySeq = @CompanySeq

JOIN _TDAItemUnit BBB On B.ItemSeq = BBB.ItemSeq

AND B.UnitSeq = BBB.UnitSeq

AND B.CompanySeq = BBB.CompanySeq

WHERE AA.CompanySeq = @CompanySeq AND A.PriceStep = 0

AND A.OptionSeq = 0

--------------- 5 유통단가 (단위 일치) :: 20170131 박성호

UPDATE #BIZ_OUT_DataBlock1

SET CustStdPrice = CASE C.SMPriceKind WHEN 8011001 THEN CASE A.PriceStep WHEN 0 THEN 0

ELSE CASE ISNULL(C.DiscountRate, 0) WHEN 0 THEN A.CustStdPrice

ELSE (CASE A.CurrSeq WHEN @StdDomCurrSeq THEN ROUND(A.ItemStdPrice * (100. - C.DiscountRate) / 100., @GoodDomPriceDecLength)

ELSE ROUND(A.ItemStdPrice * (100. - C.DiscountRate) / 100., @GoodPriceDecLength)

END)

END

END

WHEN 8011002 THEN CASE A.PriceStep WHEN 0 THEN 0

ELSE CASE ISNULL(C.DiscountPrice, 0) WHEN 0 THEN A.CustStdPrice

ELSE (CASE A.CurrSeq WHEN @StdDomCurrSeq THEN ROUND(A.ItemStdPrice - C.DiscountPrice, @GoodDomPriceDecLength)

ELSE ROUND(A.ItemStdPrice - C.DiscountPrice, @GoodPriceDecLength)

END)

END

END

ELSE CASE ISNULL(C.Price, 0) WHEN 0 THEN A.CustStdPrice

ELSE (CASE A.CurrSeq WHEN @StdDomCurrSeq THEN ROUND(C.Price, @GoodDomPriceDecLength)

ELSE ROUND(C.Price, @GoodPriceDecLength)

END)

END

END,

PriceStep = (CASE C.SMPriceKind WHEN 8011001 THEN CASE A.PriceStep WHEN 0 THEN 0 ELSE 5 END

WHEN 8011002 THEN CASE A.PriceStep WHEN 0 THEN 0 ELSE 5 END

ELSE 5

END)

FROM #BIZ_OUT_DataBlock1 AS A

JOIN _TDACustClass AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.CustSeq = A.CustSeq

AND B.UMajorCustClass = 8004

JOIN _TSLChannelPrice AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

AND C.ItemSeq = A.ItemSeq

AND C.UnitSeq = A.UnitSeq

AND C.CurrSeq = A.CurrSeq

AND C.ChannelSeq = B.UMCustClass

AND A.SalesDate BETWEEN C.SDATE AND C.EDATE

WHERE A.OptionSeq = 0

--------------- 5 거래처단가(단위 일치)

UPDATE #BIZ_OUT_DataBlock1

SET CustStdPrice = CASE C.SMPriceKind

WHEN 8011001 THEN

CASE A.PriceStep WHEN 0 THEN 0 ELSE

CASE A.CurrSeq WHEN @StdDomCurrSeq THEN ROUND(A.ItemStdPrice * (100. - C.DiscountRate) / 100., @GoodDomPriceDecLength)

ELSE ROUND(A.ItemStdPrice * (100. - C.DiscountRate) / 100., @GoodPriceDecLength) END

END

WHEN 8011002 THEN

CASE A.PriceStep WHEN 0 THEN 0 ELSE

CASE A.CurrSeq WHEN @StdDomCurrSeq THEN ROUND(A.ItemStdPrice - C.DiscountPrice, @GoodDomPriceDecLength)

ELSE ROUND(A.ItemStdPrice - C.DiscountPrice, @GoodPriceDecLength) END

END

ELSE

CASE A.CurrSeq WHEN @StdDomCurrSeq THEN ROUND(C.Price, @GoodDomPriceDecLength)

ELSE ROUND(C.Price, @GoodPriceDecLength) END

END,

PriceStep = CASE C.SMPriceKind

WHEN 8011001 THEN

CASE A.PriceStep WHEN 0 THEN 0 ELSE 4 END

WHEN 8011002 THEN

CASE A.PriceStep WHEN 0 THEN 0 ELSE 4 END

ELSE 4 END

FROM #BIZ_OUT_DataBlock1 A

JOIN _TSLCustItemPrice C WITH(NOLOCK) ON A.ItemSeq = C.ItemSeq

AND A.UnitSeq = C.UnitSeq

AND A.CurrSeq = C.CurrSeq

AND A.CustSeq = C.CustSeq

AND A.SalesDate BETWEEN C.SDATE AND C.EDATE

WHERE C.CompanySeq = @CompanySeq

AND A.OptionSeq = 0

-- 6 프로모션단가 (적용예외거래처 고려)

UPDATE #BIZ_OUT_DataBlock1

SET CustStdPrice = CASE A.CurrSeq WHEN @StdDomCurrSeq THEN ROUND(B.Price, @GoodDomPriceDecLength)

ELSE ROUND(B.Price, @GoodPriceDecLength)

END,

PriceStep = 5

FROM #BIZ_OUT_DataBlock1 AS A

JOIN (SELECT O.IDX_NO, MIN(P.Price) AS Price

FROM #BIZ_OUT_DataBlock1 O

JOIN _TDACustClass Q WITH(NOLOCK) ON O.CustSeq = Q.CustSeq

AND Q.UMajorCustClass = 8004

AND Q.CompanySeq = @CompanySeq

JOIN _TSLPromotionPrice P WITH(NOLOCK) ON P.ItemSeq = O.ItemSeq

AND P.UnitSeq = O.UnitSeq

AND P.CurrSeq = O.CurrSeq

AND O.SalesDate BETWEEN P.SDate AND P.EDate

AND P.CompanySeq = @CompanySeq

AND P.QtyCond \<= O.Qty

LEFT OUTER JOIN _TSLPromotionPriceExCust AS R WITH(NOLOCK) ON R.PromotionSeq = P.PromotionSeq

AND R.CompanySeq = P.CompanySeq

AND R.CustSeq = O.CustSeq

ANd R.SMApplyKind = 100015002 -- 적용예외

WHERE R.CustSeq IS NULL

AND (P.BizUnit = 0 OR P.BizUnit = O.BizUnit)

AND (P.UMCustClass = 0 OR P.UMCustClass = Q.UMCustClass)

AND NOT EXISTS (SELECT 1

FROM _TSLPromotionPriceExCust WITH(NOLOCK)

WHERE CompanySeq = P.CompanySeq

AND PromotionSeq = P.PromotionSeq

AND SMApplyKind = 100015001) -- 특별적용

GROUP BY O.IDX_NO) AS B ON A.IDX_NO = B.IDX_NO

WHERE A.OptionSeq = 0

-- 6 프로모션단가 (특별적용거래처 고려)

UPDATE #BIZ_OUT_DataBlock1

SET CustStdPrice = CASE A.CurrSeq WHEN @StdDomCurrSeq THEN ROUND(B.Price, @GoodDomPriceDecLength)

ELSE ROUND(B.Price, @GoodPriceDecLength)

END,

PriceStep = 5

FROM #BIZ_OUT_DataBlock1 AS A

JOIN (SELECT O.IDX_NO, MIN(P.Price) AS Price

FROM #BIZ_OUT_DataBlock1 O

JOIN _TDACustClass Q WITH(NOLOCK) ON O.CustSeq = Q.CustSeq

AND Q.UMajorCustClass = 8004

AND Q.CompanySeq = @CompanySeq

JOIN _TSLPromotionPrice P WITH(NOLOCK) ON P.ItemSeq = O.ItemSeq

AND P.UnitSeq = O.UnitSeq

AND P.CurrSeq = O.CurrSeq

AND O.SalesDate BETWEEN P.SDate AND P.EDate

AND P.CompanySeq = @CompanySeq

AND P.QtyCond \<= O.Qty

JOIN _TSLPromotionPriceExCust AS R WITH(NOLOCK) ON R.PromotionSeq = P.PromotionSeq

AND R.CompanySeq = P.CompanySeq

AND R.CustSeq = O.CustSeq

ANd R.SMApplyKind = 100015001 -- 특별적용

WHERE (P.BizUnit = 0 OR P.BizUnit = O.BizUnit)

AND (P.UMCustClass = 0 OR P.UMCustClass = Q.UMCustClass)

GROUP BY O.IDX_NO) AS B ON A.IDX_NO = B.IDX_NO

WHERE A.OptionSeq = 0

UPDATE #BIZ_OUT_DataBlock1

SET MaxDate = (SELECT MAX(X.InvoiceDate) --AS MaxDate

FROM _TSLInvoice x WITH(NOLOCK)

JOIN _TSLInvoiceItem y WITH(NOLOCK) ON x.companyseq = y.companyseq

AND x.invoiceseq = y.invoiceseq

WHERE x.CompanySeq = @CompanySeq

AND x.CustSeq = a.CustSeq and x.CurrSeq = a.CurrSeq

AND y.ItemSeq = a.ItemSeq and y.UnitSeq = a.UnitSeq

AND (y.CurAmt \<> 0 and y.Qty \<> 0)

GROUP BY x.CompanySeq, x.CustSeq, x.CurrSeq, y.ItemSeq, y.UnitSeq )

FROM #BIZ_OUT_DataBlock1 AS a

UPDATE #BIZ_OUT_DataBlock1

SET CustStdPrice = CASE WHEN A.MaxDate IS NULL THEN A.CustStdPrice

ELSE ISNULL( (SELECT CASE K.IsVat WHEN '0' THEN J.CurAmt / J.Qty ELSE (J.CurAmt + J.CurVat) / J.Qty END

FROM (SELECT X.CompanySeq, X.CustSeq, X.CurrSeq, Y.ItemSeq, Y.UnitSeq,

MAX(Y.InvoiceSeq) AS MaxInvoiceSeq,

--MAX(Y.InvoiceSeq + Y.InvoiceSerl) - MAX(Y.InvoiceSeq) AS MaxInvoiceSerl

(SELECT MAX(InvoiceSerl)

FROM _TSLInvoiceItem WITH(NOLOCK)

WHERE CompanySeq = @CompanySeq

AND InvoiceSeq = MAX(Y.InvoiceSeq)

AND ItemSeq = A.ItemSeq) AS MaxInvoiceSerl

FROM _TSLInvoice AS X WITH(NOLOCK)

JOIN _TSLInvoiceItem AS Y WITH(NOLOCK) ON X.CompanySeq = Y.CompanySeq AND X.InvoiceSeq = Y.InvoiceSeq

WHERE X.CompanySeq = @CompanySeq

AND X.InvoiceDate = A.MaxDate

AND X.CustSeq = A.CustSeq

AND X.CurrSeq = A.CurrSeq

AND Y.ItemSeq = A.ItemSeq

AND Y.UnitSeq = A.UnitSeq

AND (Y.CurAmt \<> 0 and Y.Qty \<> 0)

GROUP BY X.CompanySeq, X.CustSeq, X.CurrSeq, Y.ItemSeq, Y.UnitSeq

) AS I

JOIN _TSLInvoiceItem AS J WITH(NOLOCK) ON I.CompanySeq = J.CompanySeq AND I.MaxInvoiceSeq = J.InvoiceSeq AND I.MaxInvoiceSerl = J.InvoiceSerl

JOIN _TDAItemSales AS K WITH(NOLOCK) ON K.CompanySeq = J.CompanySeq AND J.ItemSeq = K.ItemSeq

), CustStdPrice)

END,

PriceStep = 9

FROM #BIZ_OUT_DataBlock1 AS A
