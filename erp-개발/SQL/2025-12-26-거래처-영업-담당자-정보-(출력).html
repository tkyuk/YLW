DECLARE @InvoiceSeq INT,

@Cnt INT,

@PONo NVARCHAR(4000)

CREATE TABLE #TempPONoResult(

InvoiceSeq INT,

PONo NVARCHAR(100)

)

SELECT @Cnt = 0, @PONo = ''

WHILE (1=1)

BEGIN

SELECT TOP 1

@InvoiceSeq = InvoiceSeq,

@PONo = PONo

FROM #TempPONo

WHERE IDX > @Cnt

IF @@ROWCOUNT = 0 BREAK

IF EXISTS (SELECT 1 FROM #TempPONoResult AS A WHERE A.InvoiceSeq = @InvoiceSeq AND ISNULL(A.PONo, '') \<> '' )

BEGIN

UPDATE #TempPONoResult

SET PONo = PONo + ', ' + @PONo

FROM #TempPONoResult AS A

WHERE A.InvoiceSeq = @InvoiceSeq

END

ELSE

BEGIN

INSERT INTO #TempPONoResult(InvoiceSeq, PONo)

SELECT @InvoiceSeq, @PONo

END

SELECT @Cnt = @Cnt + 1

END
