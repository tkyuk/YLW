CREATE PROC mcf_SWPUORDPOCustItemListJumpQuery

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction CHAR(1)

AS

CREATE TABLE #QCDelvQty_Temp(

POSeq INT,

POSerl INT,

QCDelvQty DECIMAL(19,5)

)

-- 구매납품

INSERT #QCDelvQty_Temp(X.POSeq, X.POSerl, QCDelvQty)

SELECT X.POSeq, X.POSerl, SUM(ISNULL(D.PassedQty,0)) AS Qty

FROM #BIZ_IN_DataBlock1 AS X

OUTER APPLY (SELECT ISNULL(Z.ToSeq, 0) AS ToSeq , ISNULL(Z.ToSerl,0) AS ToSerl

FROM _TCOMSourceDailyTrace AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = @CompanySeq

AND Z.FromSeq = X.POSeq

AND Z.FromSerl = X.POSerl

AND Z.FromTableSeq = 13 -- 구매발주

AND Z.ToTableSeq = 10 -- 구매납품

) AS Y

JOIN _TPUDelv AS A WITH(NOLOCK) ON A.CompanySeq = @CompanySeq

AND Y.ToSeq = A.DelvSeq

JOIN _TPUDelvItem AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.DelvSeq = Y.ToSeq

AND B.DelvSerl = Y.ToSerl

LEFT OUTER JOIN _TPDQCTestReport AS D WITH(NOLOCK) ON D.CompanySeq = B.CompanySeq

AND D.SourceSeq = B.DelvSeq

AND D.SourceSerl = B.DelvSerl

AND D.SourceType = 6097001 -- 구매검사

WHERE A.CompanySeq = @CompanySeq

AND A.SMImpType IN (8008001, 8008002, 8008003)

AND ISNULL(D.SMTestResult, B.SMQcType) IN (6035003, 6035001, 6035006) -- 무검사, 합격, 부분합격

AND X.UMDelvKind = 1000498001 -- 구매발주

GROUP BY X.POSeq, X.POSerl

CREATE TABLE #QCOSPQty_Temp(

POSeq INT,

POSerl INT,

QCOSPQty DECIMAL(19,5)

)

-- 외주납품

INSERT #QCOSPQty_Temp(X.POSeq, X.POSerl, QCOSPQty)

SELECT X.POSeq, X.POSerl , SUM(ISNULL(D.PassedQty,0)) AS Qty

FROM #BIZ_IN_DataBlock1 AS X

OUTER APPLY (SELECT ISNULL(Z.ToSeq, 0) AS ToSeq , ISNULL(Z.ToSerl,0) AS ToSerl

FROM _TCOMSourceDailyTrace AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = @CompanySeq

AND Z.FromSeq = X.POSeq

AND Z.FromSerl = X.POSerl

AND Z.FromTableSeq = 4 -- 외주발주

AND Z.ToTableSeq = 3 -- 외주납품

) AS Y

LEFT OUTER JOIN _TPDOSPDelv AS A WITH(NOLOCK) ON A.CompanySeq = @CompanySeq

AND A.OSPDelvSeq = Y.ToSeq

LEFT OUTER JOIN _TPDOSPDelvItem AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.OSPDelvSeq = Y.ToSeq

AND B.OSPDelvSeq = Y.ToSerl

LEFT OUTER JOIN _TPDQCTestReport AS D WITH(NOLOCK) ON D.CompanySeq = B.CompanySeq

AND D.SourceSeq = B.OSPDelvSeq

AND D.SourceSerl = B.OSPDelvSerl

AND D.SourceType = 6097002 -- 외주검사

WHERE A.CompanySeq = @CompanySeq

AND ISNULL(D.SMTestResult, B.SMQcType) IN (6035003, 6035001, 6035006) -- 무검사, 합격, 부분합격

AND X.UMDelvKind = 1000498002 -- 외주납품

GROUP BY X.POSeq, X.POSerl

SELECT H.UMDelvKind, H.POSeq, H.POSerl, SUM(ISNULL(H.Qty,0)) AS CustDelvQty

INTO #CustDelvQty

FROM #BIZ_IN_DataBlock1 AS A

LEFT OUTER JOIN mcf_TPUCustDelvItem AS H WITH(NOLOCK) ON @CompanySeq = H.CompanySeq

AND A.POSeq = H.POSeq

AND A.POSerl = H.POSerl

AND A.UMDelvKind = H.UMDelvKind

GROUP BY H.UMDelvKind, H.POSeq, H.POSerl

SELECT '0' AS IsDelvCfm , -- 납품확인

0 AS CustDelvSeq , -- 협력사납품내부코드

D.MinorName AS UMDelvKindName , -- 구분

A.UMDelvKind AS UMDelvKind , -- 구분내부코드

A.POSeq AS POSeq , -- 발주내부코드

A.POSerl AS POSerl , -- 발주내부순번

(SELECT MinorName FROM _TDASMinor WHERE CompanySeq = @CompanySeq AND MinorSeq = 1133002) AS SMIsCfmName , -- 납품확인여부

'' AS CustDelvDate , -- 납품일

C.ItemSeq AS ItemSeq , -- 품목내부코드

E.ItemName AS ItemName , -- 품명

E.ItemNo AS ItemNo , -- 품번

E.Spec AS Spec , -- 규격

F.UnitName AS UnitName , -- 구매단위

C.UnitSeq AS UnitSeq , -- 구매단위내부코드

ISNULL(J.ToQty,0) AS TotDelvQty , -- 총납품수량

ISNULL(K.QCDelvQty,0) AS TotQcQty , -- 총합격수량

ISNULL(C.Qty,0) - ISNULL(L.CustDelvQty,0) AS Qty , -- 금회납품수량

0 AS CfmQty , -- 납품확인수량

'' AS Remark , -- 비고

G.CustName AS CustName , -- 거래처

B.CustSeq AS CustSeq , -- 거래처내부코드

'' AS IsFile -- 첨부파일

INTO #TempResult

FROM #BIZ_IN_DataBlock1 AS A

JOIN _TPUORDPO AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND A.POSeq = B.POSeq

JOIN _TPUORDPOItem AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

AND A.POSeq = C.POSeq

AND A.POSerl = C.POSerl

LEFT OUTER JOIN _TDAUMinor AS D WITH(NOLOCK) ON D.CompanySeq = @CompanySeq

AND D.MinorSeq = A.UMDelvKind

LEFT OUTER JOIN _TDAItem AS E WITH(NOLOCK) ON C.CompanySeq = E.CompanySeq

AND C.ItemSeq = E.ItemSeq

LEFT OUTER JOIN _TDAUnit AS F WITH(NOLOCK) ON C.CompanySeq = F.CompanySeq

AND C.UnitSeq = F.UnitSeq

LEFT OUTER JOIN _TDACust AS G WITH(NOLOCK) ON B.CompanySeq = G.CompanySeq

AND B.CustSeq = G.CustSeq

-- LEFT OUTER JOIN mcf_TPUCustDelvItem AS H WITH(NOLOCK) ON C.CompanySeq = H.CompanySeq

-- AND C.POSeq = H.POSeq

-- AND C.POSerl = H.POSerl

-- AND H.UMDelvKind = 1000498001

LEFT OUTER JOIN _TDAWH AS I WITH(NOLOCK) ON C.WHSeq = I.WHSeq

AND C.CompanySeq = I.CompanySeq

OUTER APPLY (SELECT ISNULL(Z.FromSeq, 0) AS FromSeq , ISNULL(Z.FromSerl,0) AS FromSerl, SUM(ISNULL(Z.ToQty,0)) AS ToQty

FROM _TCOMSourceDailyTrace AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = B.CompanySeq

AND Z.FromSeq = C.POSeq

AND Z.FromSerl = C.POSerl

AND Z.FromTableSeq = 13 -- 구매발주

AND Z.ToTableSeq = 10 -- 구매납품

GROUP BY Z.FromSeq, Z.FromSerl

) AS J

LEFT OUTER JOIN #QCDelvQty_Temp AS K ON A.POSeq = K.POSeq

AND A.POSerl = K.POSerl

LEFT OUTER JOIN #CustDelvQty AS L ON A.POSeq = L.POSeq

AND A.POSerl = L.POSerl

AND A.UMDelvKind = L.UMDelvKind

WHERE A.UMDelvKind = 1000498001

INSERT INTO #TempResult

SELECT '0' AS IsDelvCfm , -- 납품확인

0 AS CustDelvSeq , -- 협력사납품내부코드

D.MinorName AS UMDelvKindName , -- 구분

A.UMDelvKind AS UMDelvKind , -- 구분내부코드

A.POSeq AS POSeq , -- 발주내부코드

A.POSerl AS POSerl , -- 발주내부순번

(SELECT MinorName FROM _TDASMinor WHERE CompanySeq = @CompanySeq AND MinorSeq = 1133002) AS SMIsCfmName , -- 납품확인여부

'' AS CustDelvDate , -- 납품일

C.ItemSeq AS ItemSeq , -- 품목내부코드

E.ItemName AS ItemName , -- 품명

E.ItemNo AS ItemNo , -- 품번

E.Spec AS Spec , -- 규격

F.UnitName AS UnitName , -- 구매단위

C.UnitSeq AS UnitSeq , -- 구매단위내부코드

ISNULL(J.ToQty,0) AS TotDelvQty , -- 총납품수량

ISNULL(K.QCOSPQty,0) AS TotQcQty , -- 총합격수량

ISNULL(C.Qty,0) - ISNULL(L.CustDelvQty,0) AS Qty , -- 금회납품수량

0 AS CfmQty , -- 납품확인수량

'' AS Remark , -- 비고

G.CustName AS CustName , -- 거래처

B.CustSeq AS CustSeq , -- 거래처내부코드

'' AS IsFile -- 첨부파일

FROM #BIZ_IN_DataBlock1 AS A

JOIN _TPDOSPPO AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND A.POSeq = B.OSPPOSeq

JOIN _TPDOSPPOItem AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

AND A.POSeq = C.OSPPOSeq

AND A.POSerl = C.OSPPOSerl

LEFT OUTER JOIN _TDAUMinor AS D WITH(NOLOCK) ON D.CompanySeq = @CompanySeq

AND D.MinorSeq = A.UMDelvKind

LEFT OUTER JOIN _TDAItem AS E WITH(NOLOCK) ON C.CompanySeq = E.CompanySeq

AND C.ItemSeq = E.ItemSeq

LEFT OUTER JOIN _TDAUnit AS F WITH(NOLOCK) ON C.CompanySeq = F.CompanySeq

AND C.UnitSeq = F.UnitSeq

LEFT OUTER JOIN _TDACust AS G WITH(NOLOCK) ON B.CompanySeq = G.CompanySeq

AND B.CustSeq = G.CustSeq

-- LEFT OUTER JOIN mcf_TPUCustDelvItem AS H WITH(NOLOCK) ON C.CompanySeq = H.CompanySeq

-- AND C.OSPPOSeq = H.POSeq

-- AND C.OSPPOSerl = H.POSerl

-- AND H.UMDelvKind = 1000498002

OUTER APPLY (SELECT ISNULL(Z.FromSeq, 0) AS FromSeq , ISNULL(Z.FromSerl,0) AS FromSerl, SUM(ISNULL(Z.ToQty,0)) AS ToQty

FROM _TCOMSourceDailyTrace AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = B.CompanySeq

AND Z.FromSeq = C.OSPPOSeq

AND Z.FromSerl = C.OSPPOSerl

AND Z.FromTableSeq = 4 -- 외주발주

AND Z.ToTableSeq = 3 -- 외주납품

GROUP BY Z.FromSeq, Z.FromSerl

) AS J

LEFT OUTER JOIN #QCOSPQty_Temp AS K ON A.POSeq = K.POSeq

AND A.POSerl = K.POSerl

LEFT OUTER JOIN #CustDelvQty AS L ON A.POSeq = L.POSeq

AND A.POSerl = L.POSerl

AND A.UMDelvKind = L.UMDelvKind

WHERE A.UMDelvKind = 1000498002

INSERT INTO #BIZ_OUT_DataBlock1(BizUnit, BizUnitName)

SELECT DISTINCT A.BizUnit, M.BizUnitName

FROM #BIZ_IN_DataBlock1 AS A

LEFT OUTER JOIN _TDABizUnit AS M WITH(NOLOCK) ON M.CompanySeq = @CompanySeq

AND A.BizUnit = M.BizUnit

TRUNCATE TABLE #BIZ_OUT_DataBlock2

INSERT INTO #BIZ_OUT_DataBlock2

(

IsDelvCfm ,

CustDelvSeq ,

UMDelvKindName ,

UMDelvKind ,

POSeq ,

POSerl ,

SMIsCfmName ,

CustDelvDate ,

ItemName ,

ItemNo ,

Spec ,

ItemSeq ,

UnitName ,

UnitSeq ,

TotDelvQty ,

TotQcQty ,

Qty ,

CfmQty ,

Remark ,

CustName ,

CustSeq ,

IsFile

)

SELECT IsDelvCfm ,

CustDelvSeq ,

UMDelvKindName ,

UMDelvKind ,

POSeq ,

SMIsCfmName ,

CONVERT(NCHAR(8),GETDATE(),112),

ItemName ,

ItemNo ,

Spec ,

ItemSeq ,

UnitName ,

UnitSeq ,

TotDelvQty ,

TotQcQty ,

Qty ,

CfmQty ,

Remark ,

CustName ,

CustSeq ,

FROM #TempResult

WHERE Qty > 0
