CREATE PROC dbo._SPJTSupplyProgListQuery

@xmlDocument NVARCHAR(MAX) ,

@xmlFlags INT = 0,

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0

AS

SET NOCOUNT ON

SET TRANSACTION ISOLATION LEVEL READ UNCOMMITTED

DECLARE @docHandle INT,

@MRPEmpSeq INT,

@MRPDeptSeq INT,

@PJTSeq INT,

@WBSSeq INT,

@ItemSeq INT,

@FrDate NCHAR(8),

@ToDate NCHAR(8),

@MRPNo NVARCHAR(12),

@CMOrPM INT,

@ProgStatus INT,

@QryDateType INT,

@QryFrDate NCHAR(8),

@QryToDate NCHAR(8),

@PurchaseCustSeq INT,

@ItemName NVARCHAR(400) ,

@ItemNo NVARCHAR(200) ,

@TopUnitName NVARCHAR(200) ,

@TopUnitNo NVARCHAR(200) ,

@PJTName NVARCHAR(100) ,

@PJTNo NVARCHAR(100) ,

@POReqNoFr NVARCHAR(30) ,

@POReqNoTo NVARCHAR(30) ,

@UMItemClassS NVARCHAR(200) ,

@ProdPlanNo NVARCHAR(50) ,

@WorkOrderNo NVARCHAR(50) ,

@PJTTypeSeq INT,

@ReqIsStop NCHAR(1),

@DelvInFrDate NCHAR(8),

@DelvInToDate NCHAR(8),

@BizUnits INT,

@IsPJTPur                        INT,

@UMSupplyType INT,

@POCustSeq INT

EXEC sp_xml_preparedocument @docHandle OUTPUT, @xmlDocument

SELECT @MRPNo = ISNULL(MRPNo , ''),

@PJTSeq = ISNULL(PJTSeq , 0),

@WBSSeq = ISNULL(WBSSeq , 0),

@ItemSeq = ISNULL(ItemSeq , 0),

@FrDate = ISNULL(MRPFrDate , ''),

@ToDate = ISNULL(MRPToDate , ''),

@CMOrPM = ISNULL(CMOrPM , ''),

@ProgStatus = ISNULL(ProgStatus , ''),

@QryDateType = ISNULL(QryDateType , ''),

@QryFrDate = ISNULL(QryFrDate , ''),

@QryToDate = ISNULL(QryToDate , ''),

@PurchaseCustSeq = ISNULL(PurchaseCustSeq, 0),

@ItemName = ISNULL(ItemName , ''),

@ItemNo = ISNULL(ItemNo , ''),

@TopUnitName = ISNULL(TopUnitName , ''),

@TopUnitNo = ISNULL(TopUnitNo , ''),

@PJTName = ISNULL(PJTName , ''),

@PJTNo = ISNULL(PJTNo , ''),

@MRPEmpSeq = ISNULL(MRPEmpSeq , 0),

@MRPDeptSeq = ISNULL(MRPDeptSeq , 0),

@POReqNoFr = ISNULL(POReqNoFr , ''),

@POReqNoTo = ISNULL(POReqNoTo , ''),

@UMItemClassS = ISNULL(UMItemClassS , ''),

@ProdPlanNo = ISNULL(ProdPlanNo , ''),

@WorkOrderNo = ISNULL(WorkOrderNo , ''),

@PJTTypeSeq = ISNULL(PJTTypeSeq , 0),

@ReqIsStop = ISNULL(ReqIsStop , '0'),

@DelvInFrDate = ISNULL(DelvInFrDate , ''),

@DelvInToDate = ISNULL(DelvInToDate , ''),

@BizUnits = ISNULL(BizUnit , 0),

@IsPJTPur                 = ISNULL(IsPJTPur , 0),

@UMSupplyType = ISNULL(UMSupplyType , 0),

@POCustSeq = ISNULL(POCustSeq , 0)

FROM OPENXML(@docHandle, N'/ROOT/DataBlock1', @xmlFlags)

WITH ( MRPNo NVARCHAR(12),

PJTSeq INT,

WBSSeq INT,

ItemSeq INT,

MRPFrDate NCHAR(8),

MRPToDate NCHAR(8),

CMOrPM INT,

ProgStatus INT,

QryDateType INT,

QryFrDate NCHAR(8),

QryToDate NCHAR(8),

PurchaseCustSeq INT,

ItemName NVARCHAR(400),

ItemNo NVARCHAR(200),

TopUnitName NVARCHAR(200),

TopUnitNo NVARCHAR(200),

PJTName NVARCHAR(100),

PJTNo NVARCHAR(100),

MRPEmpSeq INT,

MRPDeptSeq INT,

POReqNoFr NVARCHAR(30) ,

POReqNoTo NVARCHAR(30) ,

UMItemClassS NVARCHAR(200),

ProdPlanNo NVARCHAR(50) ,

WorkOrderNo NVARCHAR(50),

PJTTypeSeq INT,

ReqIsStop NCHAR(1),

DelvInFrDate NCHAR(8),

DelvInToDate NCHAR(8),

BizUnit INT,

IsPJTPur INT,

UMSupplyType INT,

POCustSeq INT )

EXEC sp_xml_removedocument @docHandle

IF ISNULL(@ToDate , '') = '' SELECT @ToDate = '99991231'

IF ISNULL(@DelvInToDate, '') = '' SELECT @DelvInToDate = '99991231'

CREATE TABLE #TMP_PROGRESSTABLE

(

IDOrder INT,

TABLENAME NVARCHAR(100))

( IDX_NO INT,

(

OutReqQty DECIMAL(19,5),

EtcOutQty DECIMAL(19,5),

ReturnReqQty DECIMAL(19,5),

ReqQty DECIMAL(19,5) NULL,

)

INSERT #TMP_PROGRESSTABLE

SELECT 1, '_TPUORDPOItem' -- 구매발주

UNION

INSERT INTO #Temp_POReq( POReqSeq, POReqSerl, UnitSeq, SourceSeq, SourceSerl, WorkOrderSeq, WorkOrderNo, ProdPlanSeq, ProdPlanNo, ReqQty, UpperUnitName, UpperUnitNo, TopUnitName, TopUnitNo )

SELECT A.POReqSeq,

B.POReqSerl,

B.UnitSeq,

B.SourceSeq,

B.SourceSerl,

H.WorkOrderSeq,

H.WorkOrderNo,

F.ProdPlanSeq,

G.ProdPlanNo,

B.Qty,

M2.ItemName AS UpperUnitName, -- 상위Unit품명 :: 20.02.21 #임희진 (최종 Select문에서 조회조건 적용되는 부분으로 구문 이동)

M2.ItemNo AS UpperUnitNo, -- 상위Unit품번 :: 20.02.21 #임희진 (최종 Select문에서 조회조건 적용되는 부분으로 구문 이동)

M4.ItemName AS TopUnitName, -- 최상위Unit품명 :: 20.02.21 #임희진 (최종 Select문에서 조회조건 적용되는 부분으로 구문 이동)

M4.ItemNo AS TopUnitNo -- 최상위Unit품번 :: 20.02.21 #임희진 (최종 Select문에서 조회조건 적용되는 부분으로 구문 이동)

FROM _TPUORDPOReq AS A WITH(NOLOCK)

JOIN _TPUORDPOReqItem AS B WITH(NOLOCK) ON A.CompanySeq= B.CompanySeq AND A.POReqSeq = B.POReqSeq

LEFT OUTER JOIN _TDAItem AS C WITH(NOLOCK) ON A.CompanySeq= C.CompanySeq AND B.ItemSeq = C.ItemSeq

LEFT OUTER JOIN _TPJTProject AS D WITH(NOLOCK) ON A.CompanySeq= D.CompanySeq AND B.PJTSeq = D.PJTSeq

LEFT OUTER JOIN _TPJTBOM AS M5 WITH(NOLOCK) ON B.CompanySeq= M5.CompanySEq

AND B.PJTSeq = M5.PJTSeq

AND B.BOMSerl = M5.BOMSerl

LEFT OUTER JOIN _TPJTBOM AS M1 WITH(NOLOCK) ON B.CompanySeq= M1.CompanySeq

AND B.PJTSeq = M1.PJTSeq

AND M1.BOMSerl \<> -1

AND M5.UpperBOMSerl = M1.BOMSerl

AND ISNULL(M1.BeforeBOMSerl,0) = 0 -- 상위 BOM

LEFT OUTER JOIN _TDAItem AS M2 WITH(NOLOCK) ON B.CompanySEq = M2.CompanySeq

AND M1.ItemSeq = M2.ItemSeq

LEFT OUTER JOIN _TPJTBOM AS M3 WITH(NOLOCK) ON B.CompanySeq = M3.CompanySeq

AND B.PJTSeq = M3.PJTSeq

AND M3.BOMSerl \<> -1

AND ISNULL(M3.BeforeBOMSerl,0) = 0

AND SUBSTRING(M1.TreeCode,1,6) = M3.TreeCode -- 최상위

AND ISNUMERIC(REPLACE(M3.BOMLevel,'.','/')) = 1 -- 가장 최상위 조건 추가 :: 20.02.21 #임희진

LEFT OUTER JOIN _TDAItem AS M4 WITH(NOLOCK) ON B.CompanySeq = M4.CompanySeq

AND M3.ItemSeq = M4.ItemSeq

LEFT OUTER JOIN _TDAItemClass AS E WITH(NOLOCK) ON B.CompanySeq = E.CompanySeq

AND E.UMajorItemClass IN (2001, 2004)

AND B.ItemSeq = E.ItemSeq

LEFT OUTER JOIN _TPDMRPDailyitem AS F WITH(NOLOCK) ON B.CompanySeq = F.CompanySeq

AND B.UMPOReqType = 6501014

AND B.SourceSeq = F.MRPDailySeq

AND B.SourceSerl = F.Serl

LEFT OUTER JOIN _TPDMPSDailyProdPlan AS G WITH(NOLOCK) ON F.CompanySeq = G.CompanySeq

AND F.ProdPlanSeq = G.ProdPlanSeq

LEFT OUTER JOIN _TPDSFCWorkOrder AS H WITH(NOLOCK) ON F.CompanySeq = H.CompanySeq

AND F.ProdPlanSeq = H.ProdPlanSeq

WHERE A.CompanySeq = @CompanySeq

AND (@MRPNo = '' OR A.POReqNo like @MRPNo + '%')

AND (@PJTSeq = 0 OR B.PJTSeq = @PJTSeq)

AND (@WBSSeq = 0 OR B.WBSSeq = @WBSSeq)

AND (@ItemSeq = 0 OR B.ItemSeq = @ItemSeq)

AND (@FrDate = '' OR A.ReqDate >= @FrDate)

AND (@ToDate = '' OR A.ReqDate \<= @ToDate)

AND (@CMOrPM = 0 OR A.UMPOReqType = @CMOrPM)

AND (@ItemName = '' OR C.ItemName LIKE @ItemName + '%')

AND (@ItemNo = '' OR C.ItemNo LIKE @ItemNo + '%')

AND (@PurchaseCustSeq = 0 OR B.CustSeq = @PurchaseCustSeq)

AND (@PJTName = '' OR D.PJTName LIKE @PJTName + '%')

AND (@PJTTypeSeq = 0 OR D.PJTTypeSeq = @PJTTypeSeq)

AND (@PJTNo = '' OR D.PJTNo LIKE @PJTNo + '%')

AND (@TopUnitName = '' OR M4.ItemName LIKE @TopUnitName + '%')

AND (@TopUnitNo = '' OR M4.ItemNo LIKE @TopUnitNo + '%')

AND (@MRPEmpSeq = 0 OR A.EmpSeq = @MRPEmpSeq)

AND (@MRPDeptSeq = 0 OR A.DeptSeq = @MRPDeptSeq)

AND (@ProdPlanNo = '' OR G.ProdPlanNo LIKE @ProdPlanNo + '%')

AND (@WorkOrderNo = '' OR H.WorkOrderNo LIKE @WorkOrderNo + '%')

AND ( (@POReqNoFr = '' AND @POReqNoTo = '') OR (@POReqNoFr = '' AND A.POReqNo \<= @POReqNoTo)

OR (@POReqNoTo = '' AND A.POReqNo >= @POReqNoFr)

OR (A.POReqNo BETWEEN @POReqNoFr AND @POReqNoTo))

AND (@UMItemClassS = 0 OR E.UMItemClass = @UMItemClassS)

AND (@ReqIsStop = '0' OR B.IsStop \<> @ReqIsStop)

EXEC _SCOMProgressTracking @CompanySeq, '_TPUORDPOReqItem', '#Temp_POReq', 'POReqSeq', 'POReqSerl', ''

--프로젝트반품 따로 빼놓고..

EXEC _SLGGetInOutStock

@CompanySeq = @CompanySeq, -- 법인코드

@BizUnit = 0, -- 사업부문

@FactUnit = 0, -- 생산사업장
