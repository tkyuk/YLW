프로그램 pgm 컬럼 메소드 확인

SELECT A.Caption, B.ControlName, D.PgmMethodName

FROM _TCAPgm AS A

JOIN _TCAPgmControls AS B ON B.PgmSeq =A.PgmSeq AND B.ControlSeq = 66

JOIN _TCAPgmControlsEvent AS C ON C.PgmSeq =B.PgmSeq AND B.ControlSeq = C.ControlSeq

JOIN _TCAPgmMethod AS D ON D.PgmSeq =C.PgmSeq AND D.PgmMethodSeq= C.PgmMethodSeq

WHERE A.PgmSeq = 3569

SELECT LastDateTime,* FROM _TCAPgm WHERE PgmSeq = 3569
