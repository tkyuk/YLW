/** InvoiceSeq 찾기**/

CREATE TABLE #Tmp_Source(IDX_NO INT IDENTITY, FromSeq INT, FromSerl INT, Seq INT, Serl INT)

-- 진행

CREATE TABLE #TMP_SOURCETABLE

(

IDOrder INT,

TABLENAME NVARCHAR(100)

)

CREATE TABLE #TCOMSOURCETracking

( IDX_NO INT,

IDOrder INT,

Seq INT,

Serl INT,

SubSerl INT,

Qty DECIMAL(19, 5),

STDQty DECIMAL(19, 5),

Amt DECIMAL(19, 5) ,

VAT DECIMAL(19, 5) ,

--DOMAmt DECIMAL(19, 5) ,

--DOMVAT DECIMAL(19, 5)

)

INSERT #TMP_SOURCETABLE

SELECT 1,'_TSLInvoiceItem'

INSERT #Tmp_Source(FromSeq, FromSerl, Seq, Serl)

SELECT A.SalesSeq, A.SalesSerl, 0, 0

FROM #Temp_Sales AS X

JOIN _TSLSalesItem AS A WITH(NOLOCK) ON A.CompanySeq = @CompanySeq

AND X.SalesSeq = A.SalesSeq

EXEC _SWCOMSourceTracking @CompanySeq, '_TSLSalesItem', '#Tmp_Source', 'FromSeq', 'FromSerl' ,''

UPDATE #Tmp_Source

SET Seq = C.InvoiceSeq,

Serl = C.InvoiceSerl

FROM #Tmp_Source A JOIN #TCOMSOURCETracking AS B WITH(NOLOCK) ON A.IDX_NO = B.IDX_No

JOIN _TSLInvoiceItem AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

AND B.Seq = C.InvoiceSeq

AND B.Serl = C.InvoiceSerl
