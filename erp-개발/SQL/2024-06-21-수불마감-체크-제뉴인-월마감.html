수불마감 체크 제뉴인 월마감

UPDATE A

SET Result = '업무가 마감이 되었습니다.',

MessageType = @MessageType ,

Status = @Status

FROM #BIZ_OUT_DataBlock3 AS A

JOIN _TCOMClosingYM AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.ClosingSeq = 69 -- 수불

AND B.ClosingYM = LEFT(A.CfmDate,6)

AND B.DtlUnitSeq = 2 -- 제/상품

WHERE A.Status = 0

AND B.IsClose = '1'

-- 거래명세서 수불 월마감 체크

IF EXISTS (SELECT TOP 1 *

FROM _TCOMClosingYM AS A WITH(NOLOCK)

WHERE (1=1)

AND A.CompanySeq = @CompanySeq

AND A.ClosingSeq = 69 -- 69 : 마감업무명 수불업무월마감(마감업무정의 화면에서 확인 가능합니다.)

--AND DtlUnitSeq = 2 -- 1 : 자재, 2 : 제/상품([수불업무]월마감처리 화면에서 확인 가능합니다.)

AND A.ClosingYM = LEFT(@InvoiceDate,6)

AND A.IsClose = '1' )

BEGIN

UPDATE #TSLOrderTotal

SET Result = '수불이 마감되었습니다.',

MessageType = 17009,

Status = 1

FROM #TSLOrderTotal AS A

IF @@ERROR \<> 0 RETURN

SELECT * FROM #TSLOrderTotal

RETURN

END
