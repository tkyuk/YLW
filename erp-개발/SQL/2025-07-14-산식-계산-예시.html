CREATE PROCEDURE das_SWDFCPJTContractAmtListQuery

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

AS

DECLARE @MasterPJTName         NVARCHAR(100)

,@MasterPJTNo         NVARCHAR(100)

,@CustSeq INT

,@PJTName         NVARCHAR(100)

,@PJTNo         NVARCHAR(100)

,@UMItemGubun INT

,@UMDFCSheetSeq INT

,@UMProfitType INT

,@ContractDateFr        NCHAR(8)

,@ContractDateTo        NCHAR(8)

,@PJTSeq INT = 0

SELECT @MasterPJTName        = ISNULL(MasterPJTName ,'')        

,@MasterPJTNo        = ISNULL(MasterPJTNo ,'')

,@CustSeq = ISNULL(CustSeq , 0)

,@PJTName         = ISNULL(PJTName ,'')

,@PJTNo         = ISNULL(PJTNo ,'')

,@UMItemGubun = ISNULL(UMItemGubun , 0)

,@UMDFCSheetSeq = ISNULL(UMDFCSheetSeq , 0)

,@UMProfitType = ISNULL(UMProfitType , 0)

,@ContractDateFr = ISNULL(ContractDateFr,'')

,@ContractDateTo = ISNULL(ContractDateTo,'')

FROM #BIZ_IN_DataBlock1

--das_TDFCShtStruct

-- 정렬

CREATE TABLE #Tmp

( STSerl INT,

SortNum NVARCHAR(30),

UpperSortNum NVARCHAR(30) NULL,

QrySort INT,

UpperSTSerl INT ,

NodeCode NVARCHAR(6),

TreeCode NVARCHAR(100),

TmpSortNum NCHAR(30),

Pos INT

)

INSERT INTO #Tmp

(STSerl, SortNum, TmpSortnum)

SELECT STSerl, LTRIM(RTRIM(STLevel)), STLevel+'.'

FROM das_TDFCShtStruct

WHERE CompanySeq = 1

AND (@UMDFCSheetSeq = 0 OR UMDFCSheetSeq = @UMDFCSheetSeq)

UPDATE #Tmp

SET QrySort = CONVERT(INT, RIGHT(SortNum, CHARINDEX('.', REVERSE('.'+SortNum))-1)),

UpperSortNum = SUBSTRING(SortNum, 1, LEN(SortNum) - CHARINDEX('.', REVERSE(SortNum)))

UPDATE #Tmp

SET UpperSTSerl = CASE WHEN B.STSerl = A.STSerl THEN -1 ELSE B.STSerl END,

NodeCode = CASE WHEN B.STSerl = A.STSerl THEN REPLICATE('0',6- len(convert(varchar,A.QrySort)))+convert(varchar,A.QrySort)

ELSE REPLICATE('0',4- len(convert(varchar,A.QrySort)))+convert(varchar,A.QrySort) END --2017.10.11 박수영 ::4자리 지원하도록 수정

FROM #Tmp AS A

LEFT OUTER JOIN das_TDFCShtStruct AS B ON B.CompanySeq = @CompanySeq

AND A.STSerl = B.STSerl

AND A.UpperSortNum = B.STLevel

UPDATE #Tmp

SET TreeCode = NodeCode,

TmpSortnum = SortNum

WHERE UpperSTSerl = -1

DECLARE @Cnt INT

SELECT @Cnt = 0

WHILE(@Cnt \< 9)

BEGIN

UPDATE B

SET TreeCode = A.TreeCode + LTRIM(RTRIM(B.NodeCode))

FROM #Tmp AS A

JOIN #Tmp AS B ON A.STSerl = B.UpperSTSerl

WHERE A.TreeCode is not null

SELECT @Cnt = @Cnt+1

END

-- 상위없는것도 트리코드 부여

SELECT @Cnt = 0

WHILE(@Cnt \< 9)

BEGIN

UPDATE B

SET TreeCode = A.TreeCode + LTRIM(RTRIM(B.NodeCode))

FROM #Tmp AS A

JOIN #Tmp AS B ON A.SortNum = B.UpperSortNum

WHERE B.UpperSTSerl IS NULL

SELECT @Cnt = @Cnt+1

END

-- 컬럼 정보 생성

--================================================================================================================================

CREATE TABLE #TempTitle

(

ColIDX INT,

Title NVARCHAR(100),

TitleSeq INT

,ShtItemSeq INT

,IsSystemItem NCHAR(1)

)

INSERT #TempTitle(Title, TitleSeq, ColIDX, ShtItemSeq, IsSystemItem)

SELECT B.ShtItemName

,A.STSerl

,ROW_NUMBER() OVER(ORDER BY A1.TreeCode) - 1 AS ColIDX

,B.ShtItemSeq

,B.IsSystemItem

FROM das_TDFCShtStruct AS A WITH(NOLOCK)

JOIN das_TDFCSheetItem AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.ShtItemSeq = A.ShtItemSeq

JOIN #Tmp AS A1 WITH(NOLOCK) ON A.STSerl = A1.STSerl

WHERE A.CompanySeq = @CompanySeq

AND A.UMDFCSheetSeq = @UMDFCSheetSeq

ORDER BY A1.TreeCode

SELECT * FROM #TempTitle ORDER BY ColIDX

,MasterPJTName NVARCHAR(100)

,UMProfitType INT

,UMRiskRewardTypeName NVARCHAR(100)

INSERT INTO #TempFixedCol

(

MasterPJTName ,MasterPJTNo ,MasterPJTSeq ,CustSeq ,CustName ,ContractFrDate ,

ContractToDate ,PJTName ,PJTNo ,PJTSeq ,PJTTypeName ,PJTTypeSeq ,

UMItemGubunName ,UMItemGubun ,UMContractTypeName ,UMContractType ,UMProductionTypeName,UMProductionType ,

UMProfitTypeName ,UMProfitType ,UMRiskRewardTypeName,UMRiskRewardType ,SMImpTypeName ,SMImpType ,

RowIDX

)

-- 최종SELECT

SELECT E.PJTName AS MasterPJTName -- 사업프로젝트명

,E.PJTNo AS MasterPJTNo -- 사업프로젝트번호

,E.PJTSeq AS MasterPJTSeq -- 사업프로젝트코드

,A.CustSeq AS CustSeq -- 거래처

,C.CustName AS CustName -- 거래처

,A.ContractFrDate AS ContractFrDate -- 계약시작일

,A.ContractToDate AS ContractToDate -- 계약종료일

,A.PJTName AS PJTName -- 제품프로젝트명

,A.PJTNo AS PJTNo -- 제품프로젝트번호

,A.PJTSeq AS PJTSeq -- 제품프로젝트코드

,F.PJTTypeName AS PJTTypeName -- 제품프로젝트유형

,A.PJTTypeSeq AS PJTTypeSeq -- 제품프로젝트유형코드

,ISNULL(H.MinorName ,'') AS UMItemGubunName -- 제품구분

,ISNULL(G.UMItemGubun , 0) AS UMItemGubun -- 제품구분코드

,ISNULL(L.MinorName ,'') AS UMContractTypeName -- 계약구분

,ISNULL(G.UMContractType , 0) AS UMContractType -- 계약구분코드

,ISNULL(I.MinorName ,'') AS UMProductionTypeName -- 생산구분

,ISNULL(G.UMProductionType, 0) AS UMProductionType -- 생산구분코드

,ISNULL(M.MinorName ,'') AS UMProfitTypeName -- 이윤산정구분

,ISNULL(G.UMProfitType , 0) AS UMProfitType -- 이윤산정구분코드

,ISNULL(N.MinorName ,'') AS UMRiskRewardTypeName -- 위험보상구분

,ISNULL(G.UMRiskRewardType, 0) AS UMRiskRewardType -- 위험보상구분코드

,ISNULL(O.MinorName ,'') AS SMImpTypeName -- 내외자구분

,ISNULL(G.SMImpType , 0) AS SMImpType -- 내외자구분코드

,ROW_NUMBER() OVER(ORDER BY E.PJTNo, A.PJTNo) - 1 AS RowIDX

FROM _TPJTProject AS A

JOIN #TPJT_AUTH AS AUTH ON AUTH.PJTSeq = A.PJTSeq

JOIN _TPJTType AS F WITH (NOLOCK) ON F.CompanySeq = A.CompanySeq

AND F.PJTTypeSeq = A.PJTTypeSeq

LEFT OUTER JOIN _TDABizUnit AS B WITH (NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.BizUnit = A.BizUnit

LEFT OUTER JOIN _TDACust AS C WITH (NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.CustSeq = A.CustSeq

JOIN _TPJTProject AS E WITH (NOLOCK) ON E.CompanySeq = A.CompanySeq

AND E.PJTSeq = A.RelationPJTSeq

JOIN das_TPJTProject AS G WITH(NOLOCK) ON G.CompanySeq = A.CompanySeq

AND G.PJTSeq = A.PJTSeq

LEFT OUTER JOIN _TDAUMinor AS H WITH(NOLOCK) ON H.CompanySeq = G.CompanySeq

AND H.MinorSeq = G.UMItemGubun

LEFT OUTER JOIN _TDAUMinor AS I WITH(NOLOCK) ON I.CompanySeq = G.CompanySeq

AND I.MinorSeq = G.UMProductionType

LEFT OUTER JOIN das_TPJTProjectType AS J WITH(NOLOCK) ON J.CompanySeq = A.CompanySeq

AND J.PJTTypeSeq = A.PJTTypeSeq

LEFT OUTER JOIN _TDAUMinorValue AS K WITH(NOLOCK) ON K.CompanySeq = J.CompanySeq

AND K.MinorSeq = J.UMProjectKind

AND K.Serl = 15000001

LEFT OUTER JOIN _TDAUMinor AS L WITH(NOLOCK) ON L.CompanySeq = G.CompanySeq

AND L.MinorSeq = G.UMContractType

LEFT OUTER JOIN _TDAUMinor AS M WITH(NOLOCK) ON M.CompanySeq = G.CompanySeq

AND M.MinorSeq = G.UMProfitType

LEFT OUTER JOIN _TDAUMinor AS N WITH(NOLOCK) ON N.CompanySeq = G.CompanySeq

AND N.MinorSeq = G.UMRiskRewardType

LEFT OUTER JOIN _TDASMinor AS O WITH(NOLOCK) ON O.CompanySeq = G.CompanySeq

AND O.MinorSeq = G.SMImpType

WHERE A.CompanySeq = @CompanySeq

AND K.ValueText = '1' -- 제품프로젝트만 조회 25.03.20 SJPARK

AND (@CustSeq = 0 OR A.CustSeq = @CustSeq )

AND (@MasterPJTName = '' OR E.PJTName LIKE @MasterPJTName +'%')

AND (@MasterPJTNo = '' OR E.PJTNo LIKE @MasterPJTNo +'%')

AND (@PJTName = '' OR A.PJTName LIKE @PJTName +'%')

AND (@PJTNo = '' OR A.PJTNo LIKE @PJTNo +'%')

AND (@UMItemGubun = 0 OR G.UMItemGubun = @UMItemGubun )

AND (@UMProfitType = 0 OR G.UMProfitType = @UMProfitType )

ORDER BY E.PJTNo, A.PJTNo

SELECT * FROM #TempFixedCol ORDER BY RowIDX

CREATE TABLE #Formula

(

Seq INT,

PJTSeq INT,

Formula NVARCHAR(100),

ReplaceFormula NVARCHAR(4000),

Gubn NVARCHAR(100),

Amt DECIMAL(19, 5),

STSerl INT,

Sort INT,

)

CREATE TABLE #Formula_Result

(

Seq INT,

PJTSeq INT,

Formula NVARCHAR(100),

ReplaceFormula NVARCHAR(4000),

Gubn NVARCHAR(100),

Amt DECIMAL(19, 5),

STSerl INT,

Sort INT,

)

CREATE TABLE #Variable

(

PJTSeq INT,

Variable NVARCHAR(100),

Gubn NVARCHAR(100),

Amt DECIMAL(19, 5),

STSerl INT,

Sort INT,

)

DECLARE @Start INT ,

@End INT

SELECT @Start = MIN(A.RowIDX),

@End = MAX(A.RowIDX)

FROM #TempFixedCol AS A

WHILE (@Start \<= @End)

BEGIN

SELECT TOP 1

@PJTSeq = PJTSeq

FROM #TempFixedCol

WHERE RowIDX = @Start

INSERT INTO #Formula(Seq, PJTSeq, Formula, ReplaceFormula, Gubn, Amt, STSerl, Sort)

SELECT A.STSerl AS Seq

, Z.PJTSeq AS PJTSeq

, A.CalcString AS Formula

, A.CalcString AS ReplaceFormula

, '' AS Gubn

, 0 AS Amt

, A.STSerl AS STSerl

, A.STSerl AS Sort

FROM (SELECT DISTINCT PJTSeq FROM #TempFixedCol) AS Z

JOIN das_TDFCShtStruct AS A WITH(NOLOCK) ON 1=1

WHERE A.CompanySeq = @CompanySeq

AND A.UMDFCSheetSeq = @UMDFCSheetSeq

AND ISNULL(A.CalcString, '') \<> ''

AND Z.PJTSeq = @PJTSeq

INSERT INTO #Variable(PJTSeq, Variable, Gubn, Amt, STSerl, Sort)

SELECT A.PJTSeq AS PJTSeq

, C.CalcVariable AS Variable

, D.ShtItemName AS Gubn

, B.Amt AS Amt

, C.STSerl AS STSerl

, C.STVariSerl AS Sort

FROM #TempFixedCol AS A

JOIN das_TDFCPJTContractAmt AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.PJTSeq = A.PJTSeq

JOIN das_TDFCShtStructCalcVari AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.ShtItemSeq = B.ShtItemSeq

LEFT OUTER JOIN das_TDFCSheetItem AS D WITH(NOLOCK) ON D.CompanySeq = B.CompanySeq

AND D.ShtItemSeq = B.ShtItemSeq

WHERE A.PJTSeq = @PJTSeq

-- 재무상태표 등에서 사용하는 산식계산 표준 SP를 호출합니다.(예산)

EXEC dbo._SWCOMFormulaResult '#Formula' , -- 산식테이블

'STSerl' , -- 산식키컬럼

'Formula' , -- 산식컬럼

'ReplaceFormula' , -- 산식을숫자로변환할컬럼

'Amt' , -- 산식의결과값컬럼

'#Variable' , -- 변수테이블

'STSerl' , -- 변수일련번호컬럼

'Variable' , -- 변수컬럼

'Amt' -- 변수의값컬럼

INSERT INTO #Formula_Result(Seq, PJTSeq, Formula, ReplaceFormula, Gubn, Amt, STSerl, Sort)

SELECT Seq, PJTSeq, Formula, ReplaceFormula, Gubn, Amt, STSerl, Sort FROM #Formula

TRUNCATE TABLE #Formula

TRUNCATE TABLE #Variable

SELECT @Start = @Start +1

END

-- 산식 종료

--================================================================================================================================

-- 변동컬럼값 조회(관리항목)

--================================================================================================================================

SELECT B.RowIDX,

A.ColIDX,

IIF(A.IsSystemItem = '1', ISNULL(C.Amt, 0), ISNULL(D.Amt,0) ) AS Amt

FROM #TempTitle AS A

JOIN #TempFixedCol AS B ON 1=1

LEFT OUTER JOIN das_TDFCPJTContractAmt AS C WITH(NOLOCK) ON C.ShtItemSeq = A.ShtItemSeq

AND C.PJTSeq = B.PJTSeq

LEFT OUTER JOIN #Formula_Result AS D WITH(NOLOCK) ON D.Seq = A.TitleSeq

AND D.PJTSeq = B.PJTSeq

ORDER BY B.RowIDX, ColIDX

RETURN

begin tran

declare @CompanySeq int = 1

DECLARE @CalcSerl INT

SELECT @CalcSerl = 0

DECLARE @AccYM NCHAR(6),

@MRRev NCHAR(2),

@MM INT

SELECT @AccYM = '202212',

@MRRev = '02'

SELECT @MM = ISNULL(CONVERT(INT, RIGHT('00' + @AccYM, 2)), 0)

CREATE TABLE #FormulaSS2

(

Seq INT,

Formula NVARCHAR(100),

ReplaceFormula NVARCHAR(4000),

Gubn NVARCHAR(100),

BudgetAmt DECIMAL(19, 5),

CalcSerl INT,

Sort INT

)

CREATE TABLE #VariableSS2

(

Seq INT,

Variable NVARCHAR(100),

Gubn NVARCHAR(100),

BudgetAmt DECIMAL(19, 5),

CalcSerl INT,

Sort INT

)

CREATE TABLE #VariableSS2Temp

(

Seq INT,

Variable NVARCHAR(100),

Gubn NVARCHAR(100),

BudgetAmt DECIMAL(19, 5),

CalcSerl INT,

Sort INT

)

-- 변수(Variable)에 해당하는 항목의 금액을 집계합니다.

INSERT INTO #VariableSS2(Seq, Variable, Gubn, BudgetAmt, CalcSerl, Sort)

SELECT A.Seq,

A.Variable,

A.MRCheckItemName AS Gubn,

CASE WHEN @MM = 1 THEN a.Seq

WHEN @MM = 2 THEN a.Seq

WHEN @MM = 3 THEN a.Seq

WHEN @MM = 4 THEN a.Seq

WHEN @MM = 5 THEN a.Seq

WHEN @MM = 6 THEN a.Seq

WHEN @MM = 7 THEN a.Seq

WHEN @MM = 8 THEN a.Seq

WHEN @MM = 9 THEN a.Seq

WHEN @MM = 10 THEN a.Seq

WHEN @MM = 11 THEN a.Seq

WHEN @MM = 12 THEN a.Seq

ELSE 0

END AS BudgetAmt,

0 AS CalcSerl,

A.Sort AS Sort

FROM emk_TACMRDataCheckForm2 AS A WITH(NOLOCK)

LEFT OUTER JOIN emk_TACMR1 AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq -- 잔액O

AND A.MR1ItemSeq = B.Form1Seq

AND B.AccYM = @AccYM

AND B.MR1Rev = @MRRev

LEFT OUTER JOIN emk_TACMR2CF AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq -- 잔액O

AND A.MR2CFItemSeq = C.Form2Seq

AND C.AccYM = @AccYM

AND C.MR2Rev = @MRRev

WHERE A.CompanySeq = @CompanySeq

AND ISNULL(A.Variable, '') \<> ''

AND ISNULL(A.Formula, '') = ''

select '#VariableSS2', * from #VariableSS2

INSERT INTO #FormulaSS2(Seq, Formula, ReplaceFormula, Gubn, BudgetAmt, CalcSerl, Sort)

SELECT A.Seq,

A.Formula,

A.Formula AS ReplaceFormula,

A.MRCheckItemName AS Gubn,

0 AS BudgetAmt,

A.Seq AS CalcSerl,

A.Sort AS Sort

FROM emk_TACMRDataCheckForm2 AS A WITH(NOLOCK)

WHERE A.CompanySeq = @CompanySeq

AND ISNULL(A.Variable, '') = ''

AND ISNULL(A.Formula, '') \<> ''

ORDER BY A.Seq

select '#FormulaSS2',* from #FormulaSS2

WHILE 1=1

BEGIN

SELECT TOP 1

@CalcSerl = CalcSerl

FROM #FormulaSS2

WHERE CalcSerl > @CalcSerl

ORDER BY CalcSerl

INSERT INTO #VariableSS2Temp(Seq, Variable, Gubn, BudgetAmt, CalcSerl, Sort)

FROM #FormulaSS2 AS A

WHERE A.Formula LIKE '%' + ISNULL(B.Variable, '') + '%'

AND A.CalcSerl = @CalcSerl

END

select '#VariableSS2Temp',* from #VariableSS2Temp

-- 재무상태표 등에서 사용하는 산식계산 표준 SP를 호출합니다.(예산)

EXEC dbo._SWCOMFormulaResult '#FormulaSS2' , -- 산식테이블

'CalcSerl' , -- 산식키컬럼

'Formula' , -- 산식컬럼

'ReplaceFormula' , -- 산식을숫자로변환할컬럼

'BudgetAmt' , -- 산식의결과값컬럼

'#VariableSS2Temp' , -- 변수테이블

'CalcSerl' , -- 변수일련번호컬럼

'Variable' , -- 변수컬럼

'BudgetAmt' -- 변수의값컬럼

select BudgetAmt,* from #FormulaSS2

rollback tran
