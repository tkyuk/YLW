CREATE PROCEDURE biont_SWLGKumsTemplateMoveReq

@WorkingTag NVARCHAR(10) = '',

@CompanySeq INT = 1 ,

@LanguageSeq INT = 1 ,

@UserSeq INT = 0 ,

@PgmSeq INT = 0 ,

@MessageDiv CHAR(1)         = '',

@MessageType VARCHAR(50) = '',

@TemplateCode VARCHAR(30) = '',

@TemplateContent NVARCHAR(1000)='',

@Param1 NVARCHAR(4000) , -- ReqSeq

@Param2 NVARCHAR(4000) ,

@Param3 NVARCHAR(4000) ,

@Param4 NVARCHAR(4000) ,

@Param5 NVARCHAR(4000)

AS

DECLARE @ReplaceContent NVARCHAR(1000) = ''

, @RecvNames                        NVARCHAR(1000) = ''

, @RecvValues                NVARCHAR(1000) = ''

, @DisplayReportName NVARCHAR(1000) = ''

, @ReportName                NVARCHAR(200) = ''

, @IDXNO                                INT

, @ContactNumber                NVARCHAR(400) = ''

, @CNT                                INT = 1

, @SendEmail                        NVARCHAR(1000) = ''

, @Subject                        NVARCHAR(1000) = ''

/* [#{companyName}]

안녕하세요 고객님, ERP시스템에서 처리결과를 알려드립니다.

제목 : #{title}

발신자 : #{name}

\<#{itemTitle} 내용>

#{content} */

-- 템플릿내용 담기

SELECT @TemplateContent = (SELECT TemplateContent

FROM OTCAKumsTemplate WITH(NOLOCK)

WHERE CompanySeq = @CompanySeq

AND TemplateCode = @TemplateCode)

---- #{companyName}

-- SELECT @TemplateContent = REPLACE(@TemplateContent, '#{companyName}',

-- (SELECT CompanyName

-- FROM _TCACompany WITH(NOLOCK)

-- WHERE CompanySeq = @CompanySeq))

-- -- #{title} 이동요청

-- SELECT @TemplateContent = REPLACE(@TemplateContent, '#{title}',

-- (SELECT WORD

-- FROM _tcadictionary WITH(NOLOCK)

-- WHERE wordseq = 20279 and LanguageSeq = @LanguageSeq))

-- -- #{name}

-- SELECT @TemplateContent = REPLACE(@TemplateContent, '#{name}',

-- (SELECT UserName

-- FROM _TCAUser WITH(NOLOCK)

-- WHERE CompanySeq = @CompanySeq

-- AND UserSeq = @UserSeq))

-- -- #{itemTitle}

-- SELECT @TemplateContent = REPLACE(@TemplateContent, '#{itemTitle}',

-- (SELECT WORD

-- FROM _tcadictionary WITH(NOLOCK)

-- WHERE wordseq = 20279 and LanguageSeq = @LanguageSeq))

/***************************** #{content} 시작 **************************/

DECLARE @CompanyName NVARCHAR(100), @PgmName NVARCHAR(200), @UserDept NVARCHAR(100), @UserName NVARCHAR(100) , @CurrDate NCHAR(8), @Remark NVARCHAR(1000)

DECLARE @ReqNo NVARCHAR(20), @CustName NVARCHAR(100), @ReqDate NCHAR(10), @ItemName NVARCHAR(100), @WHName NVARCHAR(100), @WishDate NCHAR(10), @CommitWHName NVARCHAR(100)

DECLARE @InOutReqDetailKindName NVARCHAR(100)

SELECT @CurrDate = CONVERT(NCHAR(8), GETDATE(), 112)

SELECT @CompanyName = (SELECT CompanyName FROM _TCACompany WITH(NOLOCK) WHERE CompanySeq = @CompanySeq)

SELECT @PgmName = (SELECT Caption FROM _TCAPgm WITH(NOLOCK) WHERE PgmSeq = @PgmSeq)

SELECT @UserDept = (SELECT ISNULL(D.DeptName, ISNULL(H.DeptName, N''))

FROM _TCAUser AS A WITH(NOLOCK)

LEFT OUTER JOIN _TDAEmp AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.EmpSeq = B.EmpSeq

LEFT OUTER JOIN _THRAdmOrdEmp AS C WITH(NOLOCK) ON B.CompanySeq = C.CompanySeq

AND B.EmpSeq = C.EmpSeq -- 발령일내에 있는 사원.

AND C.OrdDate+RIGHT(N'00000'+CONVERT(NVARCHAR(5),C.IntSeq),5) = (SELECT MAX(OrdDate+RIGHT(N'00000'+CONVERT(NVARCHAR(5),IntSeq),5))

FROM _THRAdmOrdEmp

WHERE CompanySeq = @CompanySeq

AND EmpSeq = C.EmpSeq

AND IsOrdDateLast = N'1'

AND OrdDate \<= @CurrDate) -- 발령일자 기준 이전발령

LEFT OUTER JOIN _TDADept AS D WITH(NOLOCK) ON C.CompanySeq = D.CompanySeq

AND C.DeptSeq = D.DeptSeq

LEFT OUTER JOIN _TDADept AS H WITH(NOLOCK) ON B.CompanySeq = H.CompanySeq --제품군이 WBS인 경우는 _TDAEmpOutContract 테이블 사용하지 않음

AND B.WkDeptSeq = H.DeptSeq

WHERE A.CompanySeq = @CompanySeq AND A.UserSeq = @UserSeq)

SELECT @UserName = (SELECT UserName FROM _TCAUser WHERE UserSeq = @UserSeq AND CompanySeq = @CompanySeq )

SELECT @ReqNo = A.ReqNo ,

@ReqDate = LEFT(A.ReqDate, 4) + '-' + SUBSTRING(A.ReqDate, 5, 2) + '-' + RIGHT(A.ReqDate, 2),

@WHName = ISNULL(C.WHName,''),

@WishDate = ISNULL(A.CompleteWishDate,''),

@CommitWHName = ISNULL(D.WHName,''),

@Remark = ISNULL(A.Remark,''),

@InOutReqDetailKindName = ISNULL(E.MinorName, '')

FROM _TLGInOutReq AS A WITH(NOLOCK)

LEFT OUTER        JOIN _TDAWH AS C WITH(NOLOCK) ON A.OutWHSeq = C.WHSeq

AND A.CompanySeq = C.CompanySeq

LEFT OUTER        JOIN _TDAWH AS D WITH(NOLOCK) ON A.InWHSeq = D.WHSeq

AND A.CompanySeq = D.CompanySeq

LEFT OUTER JOIN _TLGInOutReqItem AS F WITH(NOLOCK) ON F.CompanySeq = A.CompanySeq

AND F.ReqSeq = A.ReqSeq

LEFT OUTER JOIN _TDAUMinor AS E WITH(NOLOCK) ON E.CompanySeq = F.CompanySeq

AND E.MinorSeq = F.InOutReqDetailKind

WHERE A.CompanySeq = @CompanySeq

AND A.ReqSeq = @Param1

SELECT @ItemName = C.ItemName

FROM _TLGInOutReq AS A WITH(NOLOCK)

JOIN _TLGInOutReqItem AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.ReqSeq = A.ReqSeq

JOIN _TDAItem AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.ItemSeq = B.ItemSeq

WHERE A.CompanySeq = @CompanySeq

AND A.ReqSeq = @Param1

GROUP BY C.ItemName, B.Qty

ORDER BY B.Qty

--SELECT @TemplateContent = REPLACE(@TemplateContent, '#{content}',

-- '요청번호 : ' + @ReqNo + NCHAR(13) +

-- '요청일자 : ' + @ReqDate + NCHAR(13) +

-- '입고창고 : ' + @CommitWHName + NCHAR(13) +

-- '출고창고 : ' + @WHName + NCHAR(13) +

-- '이동희망일 : ' + @WishDate + NCHAR(13) +

-- '대표품명 : ' + @ItemName )

SELECT @TemplateContent = REPLACE(@TemplateContent, '#{content}',

'[' + @CompanyName + ']' + NCHAR(13) +

'- 처리화면 : ' + @PgmName + ' / ' + @ReqNo + NCHAR(13) +

'- 발신자 : ' + @UserDept + ' / ' + @UserName + NCHAR(13) +

'- 이동구분 : ' + @InOutReqDetailKindName + NCHAR(13) + -- 이동구분 문구 추가

'- 입고창고 : ' + @CommitWHName + NCHAR(13) +

'- 출고창고 : ' + @WHName + NCHAR(13) +

'- 대표품명 : ' + @ItemName + NCHAR(13) +

'- 특이사항 : ' + @Remark)

/***************************** #{content} 끝 **************************/

-- 출력물명 이동요청서_201905220001.pdf

SELECT @ReportName = ISNULL(Word, N'이동요청서') FROM _TCADictionary WHERE WordSeq = 66968 AND LanguageSeq = @LanguageSeq -- 이동요청서

SELECT @DisplayReportName = @ReportName+'_'+@ReqNo

-- Mail 보내는 사람 세팅

SELECT @SendEmail = B.EMail FROM _TCAUser AS A JOIN _TDAEmpIn AS B ON A.CompanySeq = B.CompanySeq AND A.EmpSeq = B.EmpSeq WHERE A.CompanySeq = @CompanySeq AND A.UserSeq = @UserSeq         

-- 메일 제목

-- [영림원ERP]에서 영림원 제거

SELECT @Subject = '[ERP] 요청번호 '+@ReqNo+' 메일 발송 드립니다.'

IF @MessageDiv IN ('A','S')

BEGIN

-- 최종 Output (알림톡,SMS인 경우 템플렛내용대로 뿌려주기, 출력물명은 공백)

SELECT @TemplateCode AS TemplateCode,

@TemplateContent AS TemplateContent,

''                                 AS RecvNames,

''                                 AS RecvValues,

'' AS DisplayReportName

END

ELSE IF @MessageDiv = 'M'

BEGIN

-- 최종 Output (Mail인 경우 템플릿내용 비어둠, 출력물명 세팅, 보내는사람 메일 세팅, 메일 제목 세팅)

SELECT @TemplateCode AS TemplateCode,

REPLACE(REPLACE(@TemplateContent, CHAR(10), ''), CHAR(13), ' \<br> ') AS TemplateContent,

''                          AS RecvNames,

''                          AS RecvValues,

@DisplayReportName AS DisplayReportName,

@SendEmail                 AS SendEmail,

@Subject AS Subject

END

RETURN
