발령 테이블 (ex.입사인원 확인)

SELECT B.RecruitSeq, B.ResumeSeq

FROM _THRAdmOrdEmp AS A

JOIN ubase_THRAdmOrdEmp AS B ON A.CompanySeq = B.CompanySeq

AND a.EmpSeq = b.EmpSeq

AND A.IntSeq = B.IntSeq

JOIN _THRAdmOrd AS C ON A.CompanySeq = C.CompanySeq

AND A.OrdSeq = C.OrdSeq

AND C.SMOrdAppSeq = 3058002

JOIN #Temp_DdFixCol AS Z ON B.CompanySeq = @CompanySeq

AND Z.RecruitSeq = B.RecruitSeq
