생산의뢰 - 작업지시

BEGIN TRAN

DECLARE @CompanySeq INT = 1

--==================================================

-- 작업지시 담기

--==================================================

CREATE TABLE #WorkOrder

(

IDX_NO INT IDENTITY(1,1) ,

WorkOrderSeq INT , -- 작업지시내부코드

WorkOrderSerl INT , -- 작업지시내부순번

WorkType INT , -- 작업구분코드

OrderQty DECIMAL(19,5) , -- 지시수량

ProgressQty DECIMAL(19,5) , -- 진행수량

ProgStatus INT , -- 진행상태코드

IsWork NCHAR(1) , -- 재작업여부

Confirm INT NULL, -- 확정여부

ConfirmEmp NVARCHAR(100) NULL, -- 확정자

ConfirmDate NCHAR(8) NULL, -- 확정일

ConfirmReason NVARCHAR(500) NULL, -- 확정사유

IsStop NCHAR(1) NULL, -- 중단여부

StopEmp NVARCHAR(100) NULL, -- 중단자

StopDate NCHAR(8) NULL, -- 중단일

StopReason NVARCHAR(500) NULL, -- 중단사유

CustSeq INT NULL, -- 거래처코드

CompleteCHECK INT

)

INSERT INTO #WorkOrder

(

WorkOrderSeq, WorkOrderSerl , WorkType , OrderQty , ProgressQty,

ProgStatus , IsWork , CompleteCHECK

)

SELECT DISTINCT

top 10

O.WorkOrderSeq AS WorkOrderSeq -- 작업지시내부코드

, O.WorkOrderSerl AS WorkOrderSerl -- 작업지시내부순번

, O.WorkType AS WorkType -- 작업구분코드

, O.OrderQty AS OrderQty -- 지시수량

, 0 AS ProgressQty -- 진행수량

, 6036001 AS ProgStatus -- 진행상태코드

, N'0' AS IsWork

, -1

FROM _TPDSFCWorkOrder AS O WITH(NOLOCK)

JOIN _TPDBaseWorkCenter AS W WITH(NOLOCK) ON O.WorkCenterSeq = W.WorkCenterSeq

AND O.CompanySeq = W.CompanySeq

JOIN _TDAItem AS I WITH(NOLOCK) ON O.GoodItemSeq = I.ItemSeq

AND O.CompanySeq = I.CompanySeq

LEFT OUTER JOIN _TPDMPSDailyProdPlan AS M WITH(NOLOCK) ON O.ProdPlanSeq = M.ProdPlanSeq

AND O.CompanySeq = M.CompanySeq

LEFT OUTER JOIN _TDAItemClass AS A WITH(NOLOCK) ON I.CompanySeq = A.CompanySeq

AND I.ItemSeq = A.ItemSeq

AND A.UMajorItemClass = 2001 -- IN (2001,2004)

LEFT OUTER JOIN _VDAItemClass AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.UMItemClass = B.ItemClassSSeq

WHERE O.WorkOrderSeq = 10705

--==================================================

-- 원천 가져오기 (생산의뢰)

--==================================================

-- 원천조회 테이블 생성

CREATE TABLE #TMP_SOURCETABLE

(

IDOrder INT,

TABLENAME NVARCHAR(100)

)

CREATE TABLE #TCOMSourceTracking

(

IDX_NO INT,

IDOrder INT,

Seq INT,

Serl INT,

SubSerl INT,

Qty DECIMAL(19,5),

STDQty DECIMAL(19,5),

Amt DECIMAL(19,5),

VAT DECIMAL(19,5)

)

-- 원천 데이터 담기

CREATE TABLE #SourceData

(

IDX_NO INT,

ProdReqSeq INT, -- 생산의뢰내부코드

ProdReqNo NVARCHAR(100), -- 생산의뢰번호

CustSeq INT, -- 생산의뢰거래처코드

DeptSeq INT, -- 생산의뢰부서코드

EmpSeq INT, -- 생산의뢰자코드

DVDate NCHAR(8), -- 납기일

ProdReqQty DECIMAL(19,5), -- 의뢰수량

ProdReqSerl INT -- 생산의뢰내부순번

)

-- 원천 조회

INSERT INTO #TMP_SOURCETABLE

SELECT 1, N'_TPDMPSProdReqItem' -- 생산의뢰

-- 원천 데이터 가져오기

EXEC dbo._SWCOMSourceTracking @CompanySeq

, N'_TPDSFCWorkOrder'

, N'#WorkOrder'

, N'WorkOrderSeq'

, N'WorkOrderSerl'

, N''

-- 생산의뢰정보

INSERT INTO #SourceData

(

IDX_NO , ProdReqSeq, ProdReqNo, CustSeq, DeptSeq,

EmpSeq , DVDate , ProdReqSerl,ProdReqQty

)

SELECT DISTINCT

T.IDX_NO

, S.Seq

, R.ProdReqNo

, RI.CustSeq -- 2022.02.07 김기태 Item테이블 거래처로 수정

, R.DeptSeq

, R.EmpSeq

, RI.DelvDate

, RI.Serl

, RI.Qty

FROM #WorkOrder AS T

LEFT OUTER JOIN #TCOMSourceTracking AS S ON T.IDX_NO = S.IDX_NO

AND S.IDOrder = 1

LEFT OUTER JOIN _TPDMPSProdReqItem AS RI WITH(NOLOCK) ON S.Seq = RI.ProdReqSeq

AND S.Serl = RI.Serl

LEFT OUTER JOIN _TPDMPSProdReq AS R WITH(NOLOCK) ON S.Seq = R.ProdReqSeq

AND RI.CompanySeq = R.CompanySeq

WHERE RI.CompanySeq = @CompanySeq

SELECT * FROM #SourceData

ROLLBACK TRAN
