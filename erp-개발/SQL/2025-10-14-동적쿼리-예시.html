동적쿼리 예시

DECLARE @DBName NVARCHAR(100),

@BizSeq INT,

@SQLParam NVARCHAR(500),

-- 각 INSERT 문에 대한 "템플릿" 변수

@TPL_DirItemRate NVARCHAR(MAX),

@TPL_DirItemTgtAmt NVARCHAR(MAX),

@TPL_DirItemRlt NVARCHAR(MAX),

@TPL_CostSumRlt NVARCHAR(MAX),

@TPL_ItemCostRlt NVARCHAR(MAX),

@TPL_ProcCostTgtAmt NVARCHAR(MAX),

@TPL_ProcCostRate NVARCHAR(MAX),

@TPL_ProcCostRlt NVARCHAR(MAX),

-- 각 INSERT 문을 만들기 위한 "빌더" 변수

@SQL_DirItemRate NVARCHAR(MAX),

@SQL_DirItemTgtAmt NVARCHAR(MAX),

@SQL_DirItemRlt NVARCHAR(MAX),

@SQL_CostSumRlt NVARCHAR(MAX),

@SQL_ItemCostRlt NVARCHAR(MAX),

@SQL_ProcCostTgtAmt NVARCHAR(MAX),

@SQL_ProcCostRate NVARCHAR(MAX),

@SQL_ProcCostRlt NVARCHAR(MAX),

-- 임시 조각을 담을 변수

@CurrentFragment NVARCHAR(MAX);

-- 파라미터 준비

SET @SQLParam = N'@pCompanySeq INT, @pCostKeySeq INT';

-- 1. 쿼리 템플릿 정의 (##DBName## 이라는 Placeholder 사용)

SET @TPL_DirItemRate = N'

SELECT A.CompanySeq, 1, A.CostKeySeq, A.SendCCtrSeq, A.ItemSeq, A.WorkOrderSeq, A.ProcSeq, A.AssySeq, D.CostAccSeq, A.DriverSeq, A.DriverValue, A.DriverRate

FROM ##DBName##.dbo._TESMCProdDirItemRate AS A

LEFT JOIN ##DBName##.dbo._TESMBAccount C ON C.CompanySeq = A.CompanySeq AND C.CostAccSeq = A.CostAccSeq

LEFT JOIN dbo._TESMBAccount D ON D.CompanySeq = C.CompanySeq AND D.AccSeq = C.AccSeq

WHERE A.CostKeySeq = @pCostKeySeq AND A.CompanySeq = @pCompanySeq';

SET @TPL_DirItemTgtAmt = N'

SELECT A.CompanySeq, 1, A.CostKeySeq, A.CCtrSeq, D.CostAccSeq, A.Amt

FROM ##DBName##.dbo._TESMCProdDirItemTgtAmt AS A

LEFT JOIN ##DBName##.dbo._TESMBAccount C ON C.CompanySeq = A.CompanySeq AND C.CostAccSeq = A.CostAccSeq

LEFT JOIN dbo._TESMBAccount D ON D.CompanySeq = C.CompanySeq AND D.AccSeq = C.AccSeq

WHERE A.CostKeySeq = @pCostKeySeq AND A.CompanySeq = @pCompanySeq';

SET @TPL_DirItemRlt = N'

SELECT A.CompanySeq, 1, A.CostKeySeq, A.SendCCtrSeq, A.ItemSeq, A.WorkOrderSeq, A.ProcSeq, A.AssySeq, D.CostAccSeq, A.DriverSeq, A.Amt

FROM ##DBName##.dbo._TESMCProdDirItemRlt AS A

LEFT JOIN ##DBName##.dbo._TESMBAccount C ON C.CompanySeq = A.CompanySeq AND C.CostAccSeq = A.CostAccSeq

LEFT JOIN dbo._TESMBAccount D ON D.CompanySeq = C.CompanySeq AND D.AccSeq = C.AccSeq

WHERE A.CostKeySeq = @pCostKeySeq AND A.CompanySeq = @pCompanySeq';

SET @TPL_CostSumRlt = N'

SELECT A.CompanySeq, 1, A.CostKeySeq, A.TransKind, A.CCtrSeq, A.AllocSeq, D.CostAccSeq, A.Amt, A.RevAmt, A.SendAmt

FROM ##DBName##.dbo._TESMCProdCostSumRlt AS A

JOIN dbo._TDASMinor B WITH(NOLOCK) ON A.TransKind = B.MinorSeq AND B.MajorSeq = 5551 AND A.CompanySeq = B.CompanySeq

LEFT JOIN ##DBName##.dbo._TESMBAccount C ON C.CompanySeq = A.CompanySeq AND C.CostAccSeq = A.CostAccSeq

LEFT JOIN dbo._TESMBAccount D ON D.CompanySeq = C.CompanySeq AND D.AccSeq = C.AccSeq

WHERE A.CostKeySeq = @pCostKeySeq AND A.CompanySeq = @pCompanySeq AND CONVERT(INT, B.MinorValue) >= 6';

SET @TPL_ItemCostRlt = N'

SELECT A.CompanySeq, 1, A.CostKeySeq, A.ItemSeq, A.WorkOrderSeq, A.ProcSeq, A.AssySeq, D.CostAccSeq, A.TransKind, A.RevAmt, A.Amt

FROM ##DBName##.dbo._TESMCProdItemCostRlt AS A

LEFT JOIN ##DBName##.dbo._TESMBAccount C ON C.CompanySeq = A.CompanySeq AND C.CostAccSeq = A.CostAccSeq

LEFT JOIN dbo._TESMBAccount D ON D.CompanySeq = C.CompanySeq AND D.AccSeq = C.AccSeq

WHERE A.CostKeySeq = @pCostKeySeq AND A.CompanySeq = @pCompanySeq';

SET @TPL_ProcCostTgtAmt = N'

SELECT A.CompanySeq, 1, A.CostKeySeq, A.ItemSeq, D.CostAccSeq, A.Amt

FROM ##DBName##.dbo._TESMCProdProcCostTgtAmt AS A

LEFT JOIN ##DBName##.dbo._TESMBAccount C ON C.CompanySeq = A.CompanySeq AND C.CostAccSeq = A.CostAccSeq

LEFT JOIN dbo._TESMBAccount D ON D.CompanySeq = C.CompanySeq AND D.AccSeq = C.AccSeq

WHERE A.CostKeySeq = @pCostKeySeq AND A.CompanySeq = @pCompanySeq';

SET @TPL_ProcCostRate = N'

SELECT A.CompanySeq, 1, A.CostKeySeq, A.ItemSeq, A.WorkOrderSeq, A.ProcSeq, A.AssySeq, D.CostAccSeq, A.DriverSeq, A.DriverValue, A.DriverRate

FROM ##DBName##.dbo._TESMCProdProcCostRate AS A

LEFT JOIN ##DBName##.dbo._TESMBAccount C ON C.CompanySeq = A.CompanySeq AND C.CostAccSeq = A.CostAccSeq

LEFT JOIN dbo._TESMBAccount D ON D.CompanySeq = C.CompanySeq AND D.AccSeq = C.AccSeq

WHERE A.CostKeySeq = @pCostKeySeq AND A.CompanySeq = @pCompanySeq';

SET @TPL_ProcCostRlt = N'

SELECT A.CompanySeq, 1, A.CostKeySeq, A.ItemSeq, A.WorkOrderSeq, A.ProcSeq, A.AssySeq, D.CostAccSeq, A.Amt

FROM ##DBName##.dbo._TESMCProdProcCostRlt AS A

LEFT JOIN ##DBName##.dbo._TESMBAccount C ON C.CompanySeq = A.CompanySeq AND C.CostAccSeq = A.CostAccSeq

LEFT JOIN dbo._TESMBAccount D ON D.CompanySeq = C.CompanySeq AND D.AccSeq = C.AccSeq

WHERE A.CostKeySeq = @pCostKeySeq AND A.CompanySeq = @pCompanySeq';

-- 2. 빌더 변수 초기화 및 단일 루프 실행

SELECT @BizSeq = 0,

@SQL_DirItemRate = N'', @SQL_DirItemTgtAmt = N'', @SQL_DirItemRlt = N'',

@SQL_CostSumRlt = N'', @SQL_ItemCostRlt = N'', @SQL_ProcCostTgtAmt = N'',

@SQL_ProcCostRate = N'', @SQL_ProcCostRlt = N'';

WHILE (1=1)

BEGIN

SELECT TOP 1 @BizSeq = BizSeq, @DBName = BizDNS

FROM sys_TMDMBiz AS A

WHERE A.CompanySeq = @pCompanySeq AND A.IsMainBiz \<> '1' AND A.BizSeq > @BizSeq

ORDER BY A.BizSeq;

IF @@ROWCOUNT = 0 BREAK;

-- 템플릿의 ##DBName## 부분을 실제 DB 이름으로 교체하여 조각(Fragment) 생성 후 UNION ALL로 연결

SET @CurrentFragment = REPLACE(@TPL_DirItemRate, '##DBName##', QUOTENAME(@DBName));

IF LEN(@SQL_DirItemRate) > 0 SET @SQL_DirItemRate = @SQL_DirItemRate + N' UNION ALL ';

SET @SQL_DirItemRate = @SQL_DirItemRate + @CurrentFragment;

SET @CurrentFragment = REPLACE(@TPL_DirItemTgtAmt, '##DBName##', QUOTENAME(@DBName));

IF LEN(@SQL_DirItemTgtAmt) > 0 SET @SQL_DirItemTgtAmt = @SQL_DirItemTgtAmt + N' UNION ALL ';

SET @SQL_DirItemTgtAmt = @SQL_DirItemTgtAmt + @CurrentFragment;

SET @CurrentFragment = REPLACE(@TPL_DirItemRlt, '##DBName##', QUOTENAME(@DBName));

IF LEN(@SQL_DirItemRlt) > 0 SET @SQL_DirItemRlt = @SQL_DirItemRlt + N' UNION ALL ';

SET @SQL_DirItemRlt = @SQL_DirItemRlt + @CurrentFragment;

SET @CurrentFragment = REPLACE(@TPL_CostSumRlt, '##DBName##', QUOTENAME(@DBName));

IF LEN(@SQL_CostSumRlt) > 0 SET @SQL_CostSumRlt = @SQL_CostSumRlt + N' UNION ALL ';

SET @SQL_CostSumRlt = @SQL_CostSumRlt + @CurrentFragment;

SET @CurrentFragment = REPLACE(@TPL_ItemCostRlt, '##DBName##', QUOTENAME(@DBName));

IF LEN(@SQL_ItemCostRlt) > 0 SET @SQL_ItemCostRlt = @SQL_ItemCostRlt + N' UNION ALL ';

SET @SQL_ItemCostRlt = @SQL_ItemCostRlt + @CurrentFragment;

SET @CurrentFragment = REPLACE(@TPL_ProcCostTgtAmt, '##DBName##', QUOTENAME(@DBName));

IF LEN(@SQL_ProcCostTgtAmt) > 0 SET @SQL_ProcCostTgtAmt = @SQL_ProcCostTgtAmt + N' UNION ALL ';

SET @SQL_ProcCostTgtAmt = @SQL_ProcCostTgtAmt + @CurrentFragment;

SET @CurrentFragment = REPLACE(@TPL_ProcCostRate, '##DBName##', QUOTENAME(@DBName));

IF LEN(@SQL_ProcCostRate) > 0 SET @SQL_ProcCostRate = @SQL_ProcCostRate + N' UNION ALL ';

SET @SQL_ProcCostRate = @SQL_ProcCostRate + @CurrentFragment;

SET @CurrentFragment = REPLACE(@TPL_ProcCostRlt, '##DBName##', QUOTENAME(@DBName));

IF LEN(@SQL_ProcCostRlt) > 0 SET @SQL_ProcCostRlt = @SQL_ProcCostRlt + N' UNION ALL ';

SET @SQL_ProcCostRlt = @SQL_ProcCostRlt + @CurrentFragment;

END;

-- 3. 루프 종료 후, 생성된 SQL들을 각각 실행

DECLARE @FinalSQL NVARCHAR(MAX);

IF LEN(@SQL_DirItemRate) > 0

BEGIN

SET @FinalSQL = N'

INSERT INTO _TESMCProdDirItemRate (

CompanySeq, CostUnit, CostKeySeq, SendCCtrSeq, ItemSeq, WorkOrderSeq, ProcSeq, AssySeq,

CostAccSeq, DriverSeq, DriverValue, DriverRate

)

SELECT

Z.CompanySeq, Z.CostUnit, Z.CostKeySeq, Z.SendCCtrSeq, Z.ItemSeq, Z.WorkOrderSeq, Z.ProcSeq, Z.AssySeq,

Z.CostAccSeq, Z.DriverSeq, SUM(Z.DriverValue), AVG(Z.DriverRate)

FROM (' + @SQL_DirItemRate + N') AS Z

GROUP BY

Z.CompanySeq, Z.CostUnit, Z.CostKeySeq, Z.SendCCtrSeq, Z.ItemSeq, Z.WorkOrderSeq, Z.ProcSeq, Z.AssySeq,

Z.CostAccSeq, Z.DriverSeq;';

EXEC sp_executesql @FinalSQL, @SQLParam, @pCompanySeq = @CompanySeq, @pCostKeySeq = @CostKeySeq;

END

IF LEN(@SQL_DirItemTgtAmt) > 0

BEGIN

SET @FinalSQL = N'

INSERT INTO _TESMCProdDirItemTgtAmt (

CompanySeq, CostUnit, CostKeySeq, CCtrSeq, CostAccSeq, Amt

)

SELECT Z.CompanySeq, Z.CostUnit, Z.CostKeySeq, Z.CCtrSeq, Z.CostAccSeq, SUM(Z.Amt)

FROM (' + @SQL_DirItemTgtAmt + N') AS Z

GROUP BY Z.CompanySeq, Z.CostUnit, Z.CostKeySeq, Z.CCtrSeq, Z.CostAccSeq;';

EXEC sp_executesql @FinalSQL, @SQLParam, @pCompanySeq = @CompanySeq, @pCostKeySeq = @CostKeySeq;

END

IF LEN(@SQL_DirItemRlt) > 0

BEGIN

SET @FinalSQL = N'

INSERT INTO _TESMCProdDirItemRlt (

CompanySeq, CostUnit, CostKeySeq, SendCCtrSeq, ItemSeq, WorkOrderSeq, ProcSeq, AssySeq,

CostAccSeq, DriverSeq, Amt

)

SELECT

Z.CompanySeq, Z.CostUnit, Z.CostKeySeq, Z.SendCCtrSeq, Z.ItemSeq, Z.WorkOrderSeq, Z.ProcSeq, Z.AssySeq,

Z.CostAccSeq, Z.DriverSeq, SUM(Z.Amt)

FROM (' + @SQL_DirItemRlt + N') AS Z

GROUP BY

Z.CompanySeq, Z.CostUnit, Z.CostKeySeq, Z.SendCCtrSeq, Z.ItemSeq, Z.WorkOrderSeq, Z.ProcSeq, Z.AssySeq,

Z.CostAccSeq, Z.DriverSeq;';

EXEC sp_executesql @FinalSQL, @SQLParam, @pCompanySeq = @CompanySeq, @pCostKeySeq = @CostKeySeq;

END

IF LEN(@SQL_CostSumRlt) > 0

BEGIN

SET @FinalSQL = N'

INSERT INTO _TESMCProdCostSumRlt (

CompanySeq, CostUnit, CostKeySeq, TransKind, CCtrSeq, AllocSeq, CostAccSeq, Amt, RevAmt, SendAmt

)

SELECT

Z.CompanySeq, Z.CostUnit, Z.CostKeySeq, Z.TransKind, Z.CCtrSeq, Z.AllocSeq, Z.CostAccSeq,

SUM(Z.Amt), SUM(Z.RevAmt), SUM(Z.SendAmt)

FROM (' + @SQL_CostSumRlt + N') AS Z

GROUP BY

Z.CompanySeq, Z.CostUnit, Z.CostKeySeq, Z.TransKind, Z.CCtrSeq, Z.AllocSeq, Z.CostAccSeq;';

EXEC sp_executesql @FinalSQL, @SQLParam, @pCompanySeq = @CompanySeq, @pCostKeySeq = @CostKeySeq;

END

IF LEN(@SQL_ItemCostRlt) > 0

BEGIN

SET @FinalSQL = N'

INSERT INTO _TESMCProdItemCostRlt (

CompanySeq, CostUnit, CostKeySeq, ItemSeq, WorkOrderSeq, ProcSeq, AssySeq,

CostAccSeq, TransKind, RevAmt, Amt

)

SELECT

Z.CompanySeq, Z.CostUnit, Z.CostKeySeq, Z.ItemSeq, Z.WorkOrderSeq, Z.ProcSeq, Z.AssySeq,

Z.CostAccSeq, Z.TransKind, SUM(Z.RevAmt), SUM(Z.Amt)

FROM (' + @SQL_ItemCostRlt + N') AS Z

GROUP BY

Z.CompanySeq, Z.CostUnit, Z.CostKeySeq, Z.ItemSeq, Z.WorkOrderSeq, Z.ProcSeq, Z.AssySeq,

Z.CostAccSeq, Z.TransKind;';

EXEC sp_executesql @FinalSQL, @SQLParam, @pCompanySeq = @CompanySeq, @pCostKeySeq = @CostKeySeq;

END

IF LEN(@SQL_ProcCostTgtAmt) > 0

BEGIN

SET @FinalSQL = N'

INSERT INTO _TESMCProdProcCostTgtAmt (

CompanySeq, CostUnit, CostKeySeq, ItemSeq, CostAccSeq, Amt

)

SELECT Z.CompanySeq, Z.CostUnit, Z.CostKeySeq, Z.ItemSeq, Z.CostAccSeq, SUM(Z.Amt)

FROM (' + @SQL_ProcCostTgtAmt + N') AS Z

GROUP BY Z.CompanySeq, Z.CostUnit, Z.CostKeySeq, Z.ItemSeq, Z.CostAccSeq;';

EXEC sp_executesql @FinalSQL, @SQLParam, @pCompanySeq = @CompanySeq, @pCostKeySeq = @CostKeySeq;

END

IF LEN(@SQL_ProcCostRate) > 0

BEGIN

SET @FinalSQL = N'

INSERT INTO _TESMCProdProcCostRate (

CompanySeq, CostUnit, CostKeySeq, ItemSeq, WorkOrderSeq, ProcSeq, AssySeq,

CostAccSeq, DriverSeq, DriverValue, DriverRate

)

SELECT

Z.CompanySeq, Z.CostUnit, Z.CostKeySeq, Z.ItemSeq, Z.WorkOrderSeq, Z.ProcSeq, Z.AssySeq,

Z.CostAccSeq, Z.DriverSeq, SUM(Z.DriverValue), AVG(Z.DriverRate)

FROM (' + @SQL_ProcCostRate + N') AS Z

GROUP BY

Z.CompanySeq, Z.CostUnit, Z.CostKeySeq, Z.ItemSeq, Z.WorkOrderSeq, Z.ProcSeq, Z.AssySeq,

Z.CostAccSeq, Z.DriverSeq;';

EXEC sp_executesql @FinalSQL, @SQLParam, @pCompanySeq = @CompanySeq, @pCostKeySeq = @CostKeySeq;

END

IF LEN(@SQL_ProcCostRlt) > 0

BEGIN

SET @FinalSQL = N'

INSERT INTO _TESMCProdProcCostRlt (

CompanySeq, CostUnit, CostKeySeq, ItemSeq, WorkOrderSeq, ProcSeq, AssySeq, CostAccSeq, Amt

)

SELECT

Z.CompanySeq, Z.CostUnit, Z.CostKeySeq, Z.ItemSeq, Z.WorkOrderSeq, Z.ProcSeq, Z.AssySeq, Z.CostAccSeq, SUM(Z.Amt)

FROM (' + @SQL_ProcCostRlt + N') AS Z

GROUP BY

Z.CompanySeq, Z.CostUnit, Z.CostKeySeq, Z.ItemSeq, Z.WorkOrderSeq, Z.ProcSeq, Z.AssySeq, Z.CostAccSeq;';

EXEC sp_executesql @FinalSQL, @SQLParam, @pCompanySeq = @CompanySeq, @pCostKeySeq = @CostKeySeq;

END;
