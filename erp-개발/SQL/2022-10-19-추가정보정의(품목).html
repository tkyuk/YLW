추가정보정의(품목)

CASE WHEN B.Price IS NOT NULL

THEN CASE WHEN CONVERT(DECIMAL(19,5),ISNULL(Q.MngValText,0)) = 0 THEN B.Price ELSE B.Price / Q.MngValText END

ELSE (ROUND(CASE WHEN ISNULL(B.Qty,0) = 0 THEN 0 ELSE (CASE WHEN B.IsInclusedVAT = '1' THEN ((ISNULL(B.CurAmt,0) + ISNULL(B.CurVat,0)) / ISNULL(B.Qty,0))

ELSE (ISNULL(B.CurAmt,0) / ISNULL(B.Qty,0)) END) END, @CurPriceLen)) END) AS Price, --판매단가

LEFT OUTER JOIN _TDAItemUserDefine AS Q WITH(NOLOCK) ON Q.CompanySeq = B.CompanySeq

AND Q.ItemSeq = B.ItemSeq

AND Q.MngSerl = 1000108

LEFT OUTER JOIN _TDAItemUserDefine AS F WITH (NOLOCK) ON F.CompanySeq = B.CompanySeq

AND F.ItemSeq = B.ItemSeq

AND F.MngSerl = 1000061 -- 제형

LEFT OUTER JOIN _TDAUMinor AS G WITH (NOLOCK) ON F.CompanySeq = G.CompanySeq

AND F.MngValSeq = G.MinorSeq
