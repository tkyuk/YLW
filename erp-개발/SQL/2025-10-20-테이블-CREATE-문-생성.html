테이블 CREATE 문 생성

-- 1건 생성

SET NOCOUNT ON

DECLARE @TableName NVARCHAR(128)

DECLARE @SQL NVARCHAR(MAX)

DECLARE @CRLF CHAR(2) = NCHAR(13) + NCHAR(10) -- 줄바꿈 문자 (\r\n)

-- 1. 조회하려는 테이블 이름을 여기에 설정하세요.

SET @TableName = 'PE_DelvPJTSalesAmt' -- 예시 테이블명

-- 2. 컬럼 정보를 임시 테이블에 저장 (저희 팀이 선호하는 임시테이블 방식)

IF OBJECT_ID('tempdb..#Columns') IS NOT NULL DROP TABLE #Columns

SELECT

C.name AS ColumnName

, T.name AS DataType

, C.max_length AS MaxLength

, C.precision AS Precision

, C.scale AS Scale

, C.is_nullable AS IsNullable

, C.is_identity AS IsIdentity

, C.column_id

INTO #Columns

FROM

sys.columns AS C WITH(NOLOCK)

JOIN

sys.tables AS TBL WITH(NOLOCK) ON TBL.object_id = C.object_id

JOIN

sys.types AS T WITH(NOLOCK) ON T.user_type_id = C.user_type_id

WHERE

TBL.name = @TableName

ORDER BY

C.column_id

-- 3. CREATE TABLE 구문 문자열 동적 구성

SET @SQL = 'CREATE TABLE [dbo].[' + @TableName + '](' + @CRLF

-- 모든 컬럼 정보를 순회하며 구문 생성

SELECT @SQL = @SQL + '        [' + ColumnName + '] [' + DataType + ']'

-- 데이터 타입에 따라 길이/정밀도 추가

+ CASE

-- NCHAR(6) 등 고정 길이 문자열 (NCHAR, CHAR)

WHEN DataType IN ('nchar', 'char') THEN '(' + CAST(MaxLength / CASE WHEN DataType = 'nchar' THEN 2 ELSE 1 END AS NVARCHAR) + ')'

-- VARCHAR, NVARCHAR, VARBINARY 등 가변 길이 문자열

WHEN DataType IN ('varchar', 'nvarchar', 'varbinary') THEN '(' + CASE WHEN MaxLength = -1 THEN 'MAX' ELSE CAST(MaxLength / CASE WHEN DataType = 'nvarchar' THEN 2 ELSE 1 END AS NVARCHAR) END + ')'

-- DECIMAL, NUMERIC 등 정밀도/스케일이 필요한 형식

WHEN DataType IN ('decimal', 'numeric') THEN '(' + CAST(Precision AS NVARCHAR) + ', ' + CAST(Scale AS NVARCHAR) + ')'

ELSE ''

END + ' '

-- Null 허용 여부 추가 (NULL 또는 NOT NULL)

+ CASE WHEN IsNullable = 1 THEN 'NULL' ELSE 'NOT NULL' END

-- 마지막 컬럼이 아니면 쉼표(,) 추가

+ CASE WHEN column_id = (SELECT MAX(column_id) FROM #Columns WITH(NOLOCK)) THEN '' ELSE ',' END

+ @CRLF

FROM

#Columns WITH(NOLOCK)

ORDER BY

column_id

-- 괄호 닫고 ON [PRIMARY] 추가 (기본 설정으로 마무리)

SET @SQL = @SQL + ') ON [PRIMARY]'

-- 4. 최종 결과 출력

PRINT @SQL

-- 5. 임시 테이블 정리

IF OBJECT_ID('tempdb..#Columns') IS NOT NULL DROP TABLE #Columns

-- 여러 건 생성

-- 테이블명 목록 정의 (요청하신 테이블 리스트)

DECLARE @TableList NVARCHAR(MAX) = 'danam_PBMProject,danam_PBMProduct,danam_PBMProductDetail,PE_TDAEmp,PE_DelvSalesAmt,PE_DelvPJTSalesAmt,PE_PlanSalesAmt,PE_SalesDirCost,PE_DevelopDirCost,PE_YearSalesProfit,PE_Pay,PE_RetEst,PE_DepreAmt,PE_ETCCost,PE_EmpYearAmt,OPBIC_SalesPlanAmt,OPBIC_PayBasicAmt,OPBIC_YearAmt,OPBIC_VacAmt,OPBIC_OTAmt,OPBIC_CarAmt,OPBIC_PayAmt,OPBIC_RetEst,OPBIC_DepreAmt,OPBIC_EmpCostMM'

DECLARE @TableName NVARCHAR(128)

DECLARE @SQL NVARCHAR(MAX)

DECLARE @CRLF CHAR(2) = NCHAR(13) + NCHAR(10) -- 줄바꿈 문자

DECLARE @CurrentID INT

DECLARE @MaxID INT

SET NOCOUNT ON

-- 1. 테이블 목록 파싱 및 임시 테이블에 저장

IF OBJECT_ID('tempdb..#TableNameList') IS NOT NULL DROP TABLE #TableNameList

CREATE TABLE #TableNameList (

ID INT IDENTITY(1,1) PRIMARY KEY,

TableName NVARCHAR(128)

)

-- 문자열 분할 (일반적으로 저희 팀이 사용하는 문자열 분할 로직이 있다면 그것을 사용하시되, 여기서는 T-SQL의 기본 XML 파싱 또는 CTE/임시 테이블 활용 로직을 사용합니다.)

-- 간단하게 STRING_SPLIT (SQL Server 2016 이상) 대신, 이전 버전 호환을 위해 WHLLE 루프 기반의 커스텀 파싱 로직을 사용합니다.

-- **주의: 아래 파싱 로직은 저희 팀의 표준 파싱 함수가 있다면 그걸로 대체하는 것이 좋습니다.**

SET @TableList = REPLACE(@TableList, ' ', '') + ',' -- 공백 제거 및 마지막 쉼표 보장

WHILE LEN(@TableList) > 0

BEGIN

SET @TableName = LEFT(@TableList, CHARINDEX(',', @TableList) - 1)

IF @TableName \<> ''

INSERT INTO #TableNameList (TableName) VALUES (@TableName)

SET @TableList = STUFF(@TableList, 1, CHARINDEX(',', @TableList), '')

END

-- ID 초기화

SELECT @CurrentID = MIN(ID), @MaxID = MAX(ID) FROM #TableNameList WITH(NOLOCK)

-- 2. 테이블 목록을 순회하며 CREATE TABLE 구문 생성 및 출력

WHILE @CurrentID IS NOT NULL AND @CurrentID \<= @MaxID

BEGIN

-- 현재 테이블 이름 가져오기

SELECT @TableName = TableName FROM #TableNameList WITH(NOLOCK) WHERE ID = @CurrentID

-- 해당 테이블이 데이터베이스에 존재하는지 확인 (NOLOCK 사용)

IF OBJECT_ID('dbo.' + @TableName) IS NOT NULL

BEGIN

-- 임시 테이블 정리

IF OBJECT_ID('tempdb..#Columns') IS NOT NULL DROP TABLE #Columns

-- 컬럼 정보 추출 및 임시 테이블에 저장

SELECT

C.name AS ColumnName

, T.name AS DataType

, C.max_length AS MaxLength

, C.precision AS Precision

, C.scale AS Scale

, C.is_nullable AS IsNullable

, C.is_identity AS IsIdentity

, C.column_id

INTO #Columns

FROM

sys.columns AS C WITH(NOLOCK)

JOIN

sys.tables AS TBL WITH(NOLOCK) ON TBL.object_id = C.object_id

JOIN

sys.types AS T WITH(NOLOCK) ON T.user_type_id = C.user_type_id

WHERE

TBL.name = @TableName

ORDER BY

C.column_id

-- CREATE TABLE 구문 초기화

SET @SQL = 'CREATE TABLE [dbo].[' + @TableName + '](' + @CRLF

-- 컬럼 순회하며 구문 생성

SELECT @SQL = @SQL + '        [' + ColumnName + '] [' + DataType + ']'

-- 데이터 타입에 따라 길이/정밀도 추가

+ CASE

WHEN DataType IN ('nchar', 'char') THEN '(' + CAST(MaxLength / CASE WHEN DataType = 'nchar' THEN 2 ELSE 1 END AS NVARCHAR) + ')'

WHEN DataType IN ('varchar', 'nvarchar', 'varbinary') THEN '(' + CASE WHEN MaxLength = -1 THEN 'MAX' ELSE CAST(MaxLength / CASE WHEN DataType = 'nvarchar' THEN 2 ELSE 1 END AS NVARCHAR) END + ')'

WHEN DataType IN ('decimal', 'numeric') THEN '(' + CAST(Precision AS NVARCHAR) + ', ' + CAST(Scale AS NVARCHAR) + ')'

ELSE ''

END + ' '

-- IDENTITY 속성 추가

+ CASE WHEN IsIdentity = 1 THEN 'IDENTITY(1,1) ' ELSE '' END

-- Null 허용 여부 추가

+ CASE WHEN IsNullable = 1 THEN 'NULL' ELSE 'NOT NULL' END

-- 마지막 컬럼이 아니면 쉼표(,) 추가

+ CASE WHEN column_id = (SELECT MAX(column_id) FROM #Columns WITH(NOLOCK)) THEN '' ELSE ',' END

+ @CRLF

FROM

#Columns WITH(NOLOCK)

ORDER BY

column_id

-- 괄호 닫고 ON [PRIMARY] 추가

SET @SQL = @SQL + ') ON [PRIMARY]' + @CRLF

-- 최종 CREATE TABLE 구문 출력 및 구분선 추가

PRINT @SQL

PRINT '------------------------------------------------------------------------------------------------'

-- 임시 테이블 정리

IF OBJECT_ID('tempdb..#Columns') IS NOT NULL DROP TABLE #Columns

END

ELSE

BEGIN

-- 테이블이 존재하지 않는 경우 알림

PRINT 'WARNING: Table [dbo].[' + @TableName + '] does not exist in the database.'

PRINT '------------------------------------------------------------------------------------------------'

END

-- 다음 테이블로 이동

SET @CurrentID = @CurrentID + 1

END

-- 최종 임시 테이블 정리

IF OBJECT_ID('tempdb..#TableNameList') IS NOT NULL DROP TABLE #TableNameList
