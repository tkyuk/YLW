전자결재 기본 데이터 테이블

SELECT CONVERT(NVARCHAR(100),@CompanySeq)                AS CompanySeq,

CONVERT(NVARCHAR(100),@LanguageSeq)                AS LanguageSeq,

CONVERT(NVARCHAR(100),@WorkKind)                AS WorkKind,

CONVERT(NVARCHAR(100),@DocWorkKind)                AS DocWorkKind,

CONVERT(NVARCHAR(100),@PgmSeq)                        AS PgmSeq,

CONVERT(NVARCHAR(100),@Value4)                        AS TblKey,

ISNULL(A.EnvDocSeq, 0)                                        AS EnvDocSeq,

--ISNULL(A.DocKindName, '')                                AS DocKindName, --품의서,... 양식문서 자체 명칭

CASE WHEN ISNULL(A.DocTypeSeq, 0) = 338002 THEN

ISNULL(A.LinkDocWorkKindName, '') ELSE ISNULL(A.DocKindName, '') END AS WorkKindName,

ISNULL(A.DocTitle        , '')                                        AS DocTitle,

ISNULL(A.ReportID, '')                                        AS ReportID,

ISNULL(A.ReportSP, '')                                        AS ReportSP,

ISNULL(A.DocIDTemplate, '')                                AS DocIDTemplate,

CASE WHEN ISNULL(A.OrgType, 0) = 0 THEN 1 ELSE A.OrgType END AS OrgType,

IsNull(A.OrgTypeName, '')                                AS OrgTypeName,

REPLACE(REPLACE(REPLACE(CONVERT(VARCHAR, GetDate(), 120),' ',''),'-',''),':','') AS CurrDateTime,

ISNULL(IsUseRecentLine, 0)                                AS IsUseRecentLine

FROM OTCAERPApprovalEnvDoc AS A WITH(NOLOCK)

LEFT OUTER JOIN _TCOMEnvGroupWare AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq AND A.LinkDocWorkKindKey = B.WorkKind

LEFT OUTER JOIN _TCOMEnvGroupWarePgm AS C WITH(NOLOCK) ON B.CompanySeq = C.CompanySeq AND B.WorkKind = C.WorkKind
