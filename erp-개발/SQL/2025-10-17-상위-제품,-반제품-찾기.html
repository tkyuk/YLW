상위 제품, 반제품 찾기

--==================================================

-- 시작 : 상위 제품, 반제품 찾기

--==================================================

DECLARE @MaxLevel INT = 20

CREATE TABLE #DiscontItemMat(

SubItemSeq INT -- 자재

)

CREATE TABLE #ExclusiveMat( -- output

GoodItemSeq INT, -- 최상위 제품

BOMRev NCHAR(2),

ItemSeq INT -- 자재

)

INSERT INTO #DiscontItemMat

SELECT ItemSeq

FROM #TDAItem

DELETE #DiscontItemMat

FROM #DiscontItemMat AS A

JOIN _TDAItem AS I WITH(NOLOCK) ON I.CompanySeq = @CompanySeq

AND I.ItemSeq = A.SubItemSeq

JOIN _TDAItemAsset AS B WITH(NOLOCK) ON B.CompanySeq = I.CompanySeq

AND B.AssetSeq = I.AssetSeq

JOIN _TDASMinor AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.MinorSeq = B.SMAssetGrp

AND C.MinorValue \<> '1' -- 자재가 아닌 건은 제거

-- 전용자재 구하기(중간에 있는 제품/반제품 제외)

;WITH CTE_ReverseTPDBOM(ParentItemSeq, ParentItemBomRev, SubItemSeq, SubItemBomRev, Lvl)

AS

(

SELECT B.ItemSeq AS ParentItemSeq

,B.ItemBomRev AS ParentItemBomRev

,B.SubItemSeq AS SubItemSeq

,B.SubItemBomRev AS SubItemBomRev

,1 AS Lvl

FROM #DiscontItemMat AS A

JOIN _TPDBOM AS B WITH (NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.SubItemSeq = A.SubItemSeq

JOIN _TDAItem AS C WITH (NOLOCK) ON C.CompanySeq = @CompanySeq

AND C.ItemSeq = A.SubItemSeq

WHERE B.CompanySeq = @CompanySeq

UNION ALL

SELECT B.ItemSeq AS ParentItemSeq

,B.ItemBomRev AS ParentItemBomRev

,B.SubItemSeq AS SubItemSeq

,B.SubItemBomRev AS SubItemBomRev

,A.lvl + 1 AS lvl

FROM CTE_ReverseTPDBOM AS A

JOIN _TPDBOM AS B WITH (NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.SubItemSeq = A.ParentItemSeq

AND B.SubItemBomRev = A.ParentItemBomRev

JOIN _TDAItem AS C WITH (NOLOCK) ON C.CompanySeq = @CompanySeq

AND C.ItemSeq = A.ParentItemSeq

WHERE A.Lvl \< @MaxLevel -- 재귀호출 방지

AND B.CompanySeq = @CompanySeq

),

CTE_GoodItemSeq AS ( -- 최상위 제품 찾기, 입력된 자재도 함께 담기

SELECT DISTINCT

R.SubItemSeq AS SubItemSeq,

R.SubItemBomRev AS SubItemBomRev,

R.ParentItemSeq AS GoodItemSeq

FROM CTE_ReverseTPDBOM AS R

JOIN _TDAItem AS I WITH(NOLOCK) ON I.CompanySeq = @CompanySeq

AND I.ItemSeq = R.SubItemSeq

JOIN _TDAItemAsset AS IA WITH(NOLOCK) ON IA.CompanySeq = I.CompanySeq

AND IA.AssetSeq = I.AssetSeq

JOIN _TDASMinor AS F WITH(NOLOCK) ON F.CompanySeq = IA.CompanySeq

AND F.MinorSeq = IA.SMAssetGrp

JOIN _TDAItem AS I2 WITH(NOLOCK) ON I2.CompanySeq = @CompanySeq

AND I2.ItemSeq = R.ParentItemSeq

JOIN _TDAItemAsset AS IA2 WITH(NOLOCK) ON IA2.CompanySeq = I2.CompanySeq

AND IA2.AssetSeq = I2.AssetSeq

WHERE NOT EXISTS (

SELECT 1

FROM _TPDBOM AS B WITH(NOLOCK)

WHERE B.CompanySeq = @CompanySeq

AND B.SubItemSeq = R.ParentItemSeq) -- 최상위 품목: 자식으로 쓰인적 없는 품목

AND F.MinorValue = '1' -- 자재만 담기

AND IA2.SMAssetGrp IN (6008002, 6008004) -- 최상위 품목이 제품, 반제품일 때만

)

INSERT INTO #ExclusiveMat(GoodItemSeq, ItemSeq, BOMRev)

SELECT A.GoodItemSeq

,A.SubItemSeq

,A.SubItemBomRev

FROM CTE_GoodItemSeq AS A

JOIN _TDAItem AS I WITH(NOLOCK) ON I.CompanySeq = @CompanySeq

AND I.ItemSeq = A.GoodItemSeq

WHERE I.SMStatus IN (2001002, 2001003)

SELECT B.ITEMNO, C.ITEMNO,A.*

FROM #ExclusiveMat AS A

JOIN _TDAITEM AS B ON B.ITEMSEQ =A.ITEMSEQ

JOIN _TDAITEM AS C ON C.ITEMSEQ =A.GOODITEMSEQ

WHERE B.ITEMNO LIKE '%HCA102KA%'

ORDER BY 1,2

--==================================================

-- 종료 : 상위 제품, 반제품 찾기

--==================================================

CREATE PROCEDURE gnst_SWDAExclusiveMat

-- gnst_SWDAExclusiveMat : 제품을 통해 단산품목전용자재만 반환하는 SP 입니다.

-- gnst_SWDAExclusiveMat2: 자재를 통해 단산품목을 먼저 찾고 전용자재인 것만 반환하는 SP 입니다.

@CompanySeq INT,

@MaxLevel INT = 20

AS

/*

CREATE TABLE #DiscontItem( -- input

GoodItemSeq INT,

BOMRev NCHAR(2)

)

CREATE TABLE #ExclusiveMat( -- output

GoodItemSeq INT,

BOMRev NCHAR(2),

ItemSeq INT

)

*/

--==================================================

-- 단산품목 소요자재 중 자재인 것만 가져오기

--==================================================

;WITH CTE_TPDBOM(ItemSeq, ItemBomRev, SubItemSeq, SubItemBomRev,Lvl, GoodItemSeq)

AS

(

SELECT B.ItemSeq AS ItemSeq

,B.ItemBomRev AS ItemBomRev

,B.SubItemSeq AS SubItemSeq

,B.SubItemBomRev AS SubItemBomRev

,1 AS Lvl

,A.GoodItemSeq AS GoodItemSeq

FROM #DiscontItem AS A

JOIN _TPDBOM AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.ItemSeq = A.GoodItemSeq

AND B.ItemBomRev = A.BOMRev

JOIN _TDAItem AS C WITH(NOLOCK) ON B.CompanySeq = C.CompanySeq

AND B.SubItemSeq = C.ItemSeq

WHERE B.CompanySeq = @CompanySeq

UNION ALL

SELECT B.ItemSeq AS ItemSeq

,B.ItemBomRev AS ItemBomRev

,B.SubItemSeq AS SubItemSeq

,B.SubItemBomRev AS SubItemBomRev

,A.Lvl + 1 AS Lvl

,A.GoodItemSeq AS GoodItemSeq

FROM CTE_TPDBOM AS A

JOIN _TPDBOM AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.ItemSeq = A.SubItemSeq

AND B.ItemBomRev = A.SubItemBomRev

JOIN _TDAItem AS C WITH(NOLOCK) ON B.CompanySeq = C.CompanySeq

AND B.SubItemSeq = C.ItemSeq

WHERE B.CompanySeq = @CompanySeq

AND A.Lvl \< @MaxLevel -- 재귀호출 방지

)

SELECT DISTINCT A.GoodItemSeq,A.SubItemSeq

INTO #DiscontItemMat

FROM CTE_TPDBOM AS A

JOIN _TDAItem AS I WITH(NOLOCK) ON I.CompanySeq = @CompanySeq

AND A.SubItemSeq = I.ItemSeq

JOIN _TDAItemAsset AS B WITH(NOLOCK) ON B.CompanySeq = I.CompanySeq

AND B.AssetSeq = I.AssetSeq

JOIN _TDASMinor AS C WITH(NOLOCK) ON B.CompanySeq = C.CompanySeq

AND B.SMAssetGrp = C.MinorSeq

AND C.MinorValue = '1' -- 자재

WHERE I.CompanySeq = @CompanySeq

DELETE FROM #DiscontItemMat

WHERE SubItemSeq IN (

SELECT SubItemSeq

FROM #DiscontItemMat

GROUP BY SubItemSeq

HAVING COUNT(DISTINCT GoodItemSeq) >= 2

)

--==================================================

-- 전용자재 구하기(중간에 있는 제품/반제품 제외)

--==================================================

;WITH CTE_ReverseTPDBOM(ParentItemSeq, ParentItemBomRev, SubItemSeq, SubItemBomRev, Lvl, STDMatItemSeq,GoodItemSeq)

AS

(

SELECT B.ItemSeq AS ParentItemSeq

,B.ItemBomRev AS ParentItemBomRev

,B.SubItemSeq AS SubItemSeq

,B.SubItemBomRev AS SubItemBomRev

,1 AS Lvl

,A.SubItemSeq AS STDMatItemSeq

,A.GoodItemSeq AS GoodItemSeq

FROM #DiscontItemMat AS A

JOIN _TPDBOM AS B WITH (NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.SubItemSeq = A.SubItemSeq

JOIN _TDAItem AS C WITH (NOLOCK) ON @CompanySeq = C.CompanySeq

AND A.SubItemSeq = C.ItemSeq

WHERE B.CompanySeq = @CompanySeq

UNION ALL

SELECT B.ItemSeq AS ParentItemSeq

,B.ItemBomRev AS ParentItemBomRev

,B.SubItemSeq AS SubItemSeq

,B.SubItemBomRev AS SubItemBomRev

,A.lvl + 1 AS lvl

,A.STDMatItemSeq AS STDMatItemSeq

,A.GoodItemSeq AS GoodItemSeq

FROM CTE_ReverseTPDBOM AS A

JOIN _TPDBOM AS B WITH (NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.SubItemSeq = A.ParentItemSeq

AND B.SubItemBomRev = A.ParentItemBomRev

JOIN _TDAItem AS C WITH (NOLOCK) ON @CompanySeq = C.CompanySeq

AND A.ParentItemSeq = C.ItemSeq

WHERE A.Lvl \< @MaxLevel -- 재귀호출 방지

AND B.CompanySeq = @CompanySeq

)

SELECT DISTINCT A.ParentItemSeq, A.SubItemSeq,A.SubItemBomRev , A.GoodItemSeq

INTO #TempExclusiveMat -- 단산품목 소요자재 역전개 테이블

FROM CTE_ReverseTPDBOM AS A

JOIN _TDAItem AS B WITH(NOLOCK) ON A.GoodItemSeq = B.ItemSeq

JOIN _TDAItemAsset AS C WITH(NOLOCK) ON B.AssetSeq = C.AssetSeq

AND C.SMAssetGrp IN(6008002 ,6008004) -- 제품,반제품

JOIN _TDAItem AS D WITH(NOLOCK) ON A.SubItemSeq = D.ItemSeq

JOIN _TDAItemAsset AS E WITH(NOLOCK) ON D.AssetSeq = E.AssetSeq

JOIN _TDASMinor AS F WITH(NOLOCK) ON E.SMAssetGrp = F.MinorSeq

AND F.MinorValue = '1' -- 자재만

WHERE B.CompanySeq = @CompanySeq

DELETE FROM #TempExclusiveMat

WHERE SubItemSeq IN (

SELECT SubItemSeq

FROM #TempExclusiveMat

GROUP BY SubItemSeq

HAVING COUNT(DISTINCT ParentItemSeq) >= 2

)

INSERT INTO #ExclusiveMat(GoodItemSeq,ItemSeq, BOMRev)

SELECT ParentItemSeq,SubItemSeq,SubItemBomRev

FROM #TempExclusiveMat AS A

JOIN #DiscontItem AS B ON A.ParentItemSeq = B.GoodItemSeq

RETURN
