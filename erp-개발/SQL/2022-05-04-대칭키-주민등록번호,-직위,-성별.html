제뉴인 : dbo._FCOMDecrypt(A.PersonId, '_TDACust', 'PersonId', A.CompanySeq) AS PersonId

참고sp : mcnk_SXHRInfEmpList

EXEC _SYMMETRICKeyOpen

mcnv_SXPRPayResultAmtCurrUploadQuery

주민등록번호

ISNULL(dbo._FWCOMDecrypt(B.ResidID, '_TDAEmp', 'ResidID', @CompanySeq),'') AS ResidId

직위

D.MinorName AS UMJpName

LEFT JOIN _TDADept AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.DeptSeq = C.DeptSeq

LEFT JOIN _TDAUMinor AS D WITH(NOLOCK) ON A.CompanySeq = D.CompanySeq

AND A.UMJpSeq = D.MinorSeq

성별

F.MinorName AS SMSexName

LEFT JOIN _TDAEmpIn AS E WITH(NOLOCK) ON A.CompanySeq = E.CompanySeq

AND A.EmpSeq = E.EmpSeq

LEFT JOIN _TDASMinor AS F WITH(NOLOCK) ON E.CompanySeq = F.CompanySeq

AND E.SMSexSeq = F.MinorSeq
