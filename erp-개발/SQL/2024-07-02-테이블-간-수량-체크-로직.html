CREATE TABLE #Temp_error

(

IDX_NO INT,

POSeq INT,

POSerl INT

)

INSERT #Temp_error(IDX_NO, POSeq, POSerl)

-- 구매발주

SELECT A.IDX_NO, A.POSeq, A.POSerl

FROM #BIZ_OUT_DataBlock2 AS A

JOIN _TPUORDPOItem AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.POSeq = A.POSeq

AND B.POSerl = A.POSerl

AND A.UMDelvKind = 1000498001

LEFT OUTER JOIN mcf_TPUCustDelvItem AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

AND C.POSeq = A.POSeq

AND C.POSerl = A.POSerl

AND C.UMDelvKind = A.UMDelvKind

WHERE A.WorkingTag = 'A' AND A.Status = 0

GROUP BY A.IDX_NO, A.POSeq, A.POSerl

HAVING MAX(ISNULL(B.Qty, 0)) - (MAX(ISNULL(A.Qty, 0)) + SUM(ISNULL(C.Qty, 0))) \< 0

UNION

-- 외주발주

SELECT A.IDX_NO, A.POSeq, A.POSerl

FROM #BIZ_OUT_DataBlock2 AS A

JOIN _TPDOSPPOItem AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.OSPPOSeq = A.POSeq

AND B.OSPPOSerl = A.POSerl

AND A.UMDelvKind = 1000498002

LEFT OUTER JOIN mcf_TPUCustDelvItem AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

AND C.POSeq = A.POSeq

AND C.POSerl = A.POSerl

AND C.UMDelvKind = A.UMDelvKind

WHERE A.WorkingTag = 'A' AND A.Status = 0

GROUP BY A.IDX_NO, A.POSeq, A.POSerl

HAVING MAX(ISNULL(B.Qty, 0)) - (MAX(ISNULL(A.Qty, 0)) + SUM(ISNULL(C.Qty, 0))) \< 0

UPDATE A

SET Status = 99999,

Result = '발주수량 초과하였습니다.'

FROM #BIZ_OUT_DataBlock2 AS A

JOIN #Temp_error AS B ON B.IDX_NO = A.IDX_NO

WHERE A.Status = 0
