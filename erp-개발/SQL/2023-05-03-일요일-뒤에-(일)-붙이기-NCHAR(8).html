일요일 뒤에 (일) 붙이기 NCHAR(8)

CREATE TABLE #TempSero

(

DD NCHAR(8)

)

SELECT CASE WHEN SUBSTRING(DD, 7, 2) = '32'

THEN '총계' ELSE SUBSTRING(DD, 5, 2) + '/'+SUBSTRING(DD, 7, 2)

+ CASE DATEPART(W, DD) WHEN 1

THEN '(일)'

ELSE '' END END

FROM #TempSero ORDER BY DD
