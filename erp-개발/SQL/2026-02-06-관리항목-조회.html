관리항목 조회

SELECT z.AccName,

B.RemName, B.RemSeq, B.CodeHelpSeq, B.CodeHelpParams,

CASE B.SMInputType

WHEN 4016001 THEN 'enText'

WHEN 4016002 THEN 'enCodeHelp'

WHEN 4016003 THEN 'enFloat'

WHEN 4016004 THEN 'enFloat'

WHEN 4016005 THEN 'enDate'

WHEN 4016006 THEN 'enText'

WHEN 4016007 THEN 'enFloat'

WHEN 4016008 THEN 'enFloat'

ELSE 'enText'

END AS CellType -- 입력형태

, B.SMInputType AS SMInputType_Qry -- 입력형태(코드값)

FROM _TDAAccount as z

join _TDAAccountSub AS A WITH (NOLOCK) on a.AccSeq = z.AccSeq

INNER JOIN _VDAAccountRem B WITH (NOLOCK)

ON A.CompanySeq = B.CompanySeq

AND A.RemSeq = B.RemSeq

AND B.LanguageSeq = 1

WHERE A.CompanySeq = 1

--AND A.AccSeq = @Param1

ORDER BY 1
