연결된 서버 추가 및 삭제, 조회

--생성

sp_addlinkedserver @Server='ylw',@srvproduct='SQLSERVER',@Provider='SQLOLEDB',@Datasrc='해당DB',@Catalog=''

EXEC sp_addlinkedserver

@server = 'ylw', -- 연결된 서버의 논리적 이름 (사용자가 정함)

@srvproduct = 'SQLSERVER',-- 서버 제품명 (SQL Server일 경우 'SQLSERVER', Oracle 등은 생략 가능)

@provider = 'SQLOLEDB', -- OLE DB 공급자 (SQL Server 기본 공급자는 'SQLOLEDB' 또는 'MSOLEDBSQL')

@datasrc = '해당DB', -- 실제 데이터 소스(서버 이름 또는 IP)

@catalog = ''; -- 기본 데이터베이스 (보통 생략 가능, 기본값은 빈 문자열)

sp_addlinkedsrvlogin 'ylw','False','sa','해당DB 로그인 ID','해당DB PW'

--삭제

sp_droplinkedsrvlogin @RmtSrvname ='ylw', @locallogin = 'sa'

sp_dropserver @server ='ylw'

﻿
