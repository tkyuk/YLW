-- 메시지 그룹 사용자

SELECT A.UserSeq                                AS UserSeq,

ISNULL(C.UserName, '')        AS UserName,

A.UserSeq                                AS UserSeqOld

FROM OTCANoticeGroupMember AS A WITH(NOLOCK) JOIN OTCANoticeGroup AS B WITH(NOLOCK)

ON B.CompanySeq        = A.CompanySeq

AND B.GroupSeq                = A.GroupSeq

LEFT OUTER JOIN _TCAUser AS C WITH(NOLOCK)

ON C.CompanySeq        = A.CompanySeq

AND C.UserSeq                = A.UserSeq

WHERE A.CompanySeq                                 = @CompanySeq

AND A.GroupSeq                                 = @GroupSeq

AND ISNULL(B.IsNotiGroup, '') = '1'

END
