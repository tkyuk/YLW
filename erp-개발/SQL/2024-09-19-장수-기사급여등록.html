장수 - 기사급여등록

생성 SP : SSADriverPayProc_jsb

SELECT PayYM ,

SUM(DelvAmt ) AS DelvAmt

,SUM(MVAmt ) AS MVAmt

,SUM(DelvAmt ) + SUM(MVAmt )

FROM TSADriverPayMM_jsb

WHERE PayYM LIKE '2024%'

GROUP BY PayYM

ORDER BY 1 DESC

SELECT * FROM TSADriverPayProc_jsb ORDER BY 1 DESC , 2 DESC
