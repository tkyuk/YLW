UPDATE A

SET CCtrSeq = ISNULL(E.CCtrSeq, 0),

LotNo = CASE WHEN C.IsLotMng = '1' THEN ISNULL(E.PJTNo, '') ELSE '' END,

VATRate = CASE WHEN A.VATRate IS NULL THEN ISNULL(G.VATRate, 10) ELSE A.VATRate END,

STDUnitQty = CASE WHEN A.UnitSeq = A.STDUnitSeq THEN A.Qty

ELSE dbo._FWPDGetDecPointQty((CASE WHEN ISNULL(D.ConvDen, 0) = 0 THEN 1

ELSE D.ConvNum / D.ConvDen

END) * A.Qty,

@QtyDec, F.SMDecPointSeq)

END

FROM #TDelvItem AS A

JOIN _TDAItemSales AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq AND B.ItemSeq = A.ItemSeq

JOIN _TDAItemStock AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq AND C.ItemSeq = A.ItemSeq

LEFT OUTER JOIN _TDAItemUnit AS D WITH(NOLOCK) ON D.CompanySeq = @CompanySeq

AND D.ItemSeq = A.ItemSeq

AND D.UnitSeq = A.UnitSeq

LEFT OUTER JOIN _TPJTProject AS E WITH(NOLOCK) ON E.CompanySeq = @CompanySeq

AND E.PJTSeq = A.PJTSeq

LEFT OUTER JOIN _TDAUnit AS F WITH(NOLOCK) ON F.CompanySeq = @CompanySeq

AND F.UnitSeq = A.STDUnitSeq

OUTER APPLY(SELECT TOP 1 Z.VatRate

FROM _TDAVatRate AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = B.CompanySeq

AND Z.SMVatType = ISNULL(B.SMVatType, 8028001)

AND Z.SDate \<= A.DelvDate

ORDER BY Z.SDate DESC

) AS G

WHERE A.WorkingTag IN ('A', 'U')

AND A.Status = 0
