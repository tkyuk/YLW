CREATE PROCEDURE _SXPRWkCalendarHolidayQuery

@ServiceSeq INT = 0 ,

@WorkingTag NVARCHAR(10) = '',

@CompanySeq INT = 1 ,

@LanguageSeq INT = 1 ,

@UserSeq INT = 0 ,

@PgmSeq INT = 0 ,

@IsTransaction BIT = 0

AS

-- 사용할 변수 선언

DECLARE @BaseYY NCHAR(4), -- 기준연도

@BegYm NCHAR(6), -- 시작월

@EndYm NCHAR(6), -- 종료월

@SMHolidayTypeSeq NCHAR(8), -- 휴일구분코드

@DayTypeSeq INT , -- 일구분코드

@HoliBaseSeq INT

SELECT @BaseYY = ISNULL(BaseYY , ''),

@BegYm = ISNULL(BegYm , ''),

@EndYm = ISNULL(EndYm , ''),

@SMHolidayTypeSeq = ISNULL(SMHolidayTypeSeq, 0),

@DayTypeSeq = ISNULL(DayTypeSeq , 0),

@HoliBaseSeq = ISNULL(HoliBaseSeq , 0)

FROM #BIZ_IN_DataBlock1

-- 요일을 언어별로 가져오기 위해서

IF @LanguageSeq = 1

BEGIN

SET LANGUAGE 'Korean'

END

ELSE IF @LanguageSeq = 3

BEGIN

SET LANGUAGE 'Japanese'

END

ELSE IF @LanguageSeq = 4

BEGIN

SET LANGUAGE 'Simplified Chinese'

END

ELSE IF @LanguageSeq = 5

BEGIN

SET LANGUAGE 'Traditional Chinese'

END

ELSE

BEGIN

SET LANGUAGE 'English'

END

-- 시트에 값을 출력 하는 부분

SELECT ISNULL(A.HoliBaseSeq , 0) AS HoliBaseSeq , -- 휴일기준코드

ISNULL(A.SolarDate , '') AS SolarDate , -- 일자

--DATENAME(WEEKDAY, A.SolarDate) AS WeekName , -- 요일

CASE WHEN @LanguageSeq IN (1, 3, 4, 5) THEN

LEFT(DATENAME(WEEKDAY, A.SolarDate), 1)

ELSE

LEFT(DATENAME(WEEKDAY, A.SolarDate), 3) END AS WeekName ,

ISNULL(A.HolidayName , '') AS HolidayName , -- 휴일

ISNULL(B.MinorName , '') AS SMHolidayTypeName, -- 휴일구분

ISNULL(A.SMHolidayTypeSeq, 0) AS SMHolidayTypeSeq , -- 휴일구분코드

ISNULL(C.DayTypeName , '') AS DayTypeName , -- 일구분

ISNULL(A.DayTypeSeq , 0) AS DayTypeSeq , -- 일구분코드

ISNULL(A.Remark , '') AS Remark , -- 비고

ISNULL(A.SolarDate , '') AS SolarDateOld -- 이전일자

FROM _TXPRWkCalendarHoliday AS A WITH(NOLOCK) LEFT OUTER JOIN _TXBMSMinor AS B ON A.CompanySeq = B.CompanySeq

AND A.SMHolidayTypeSeq = B.MinorSeq

LEFT OUTER JOIN _TXPRWkDayType AS C ON A.CompanySeq = C.Companyseq

AND A.DayTypeSeq = C.DayTypeSeq

WHERE A.CompanySeq = @CompanySeq

AND A.HoliBaseSeq = @HoliBaseSeq

AND LEFT(SolarDate, 4) = @BaseYY

AND (LEFT(A.SolarDate, 6) BETWEEN @BegYm AND @EndYm OR @BegYm = '')

AND (LEFT(A.SolarDate, 6) BETWEEN @BegYm AND @EndYm OR @EndYm = '')

AND (A.DayTypeSeq = @DayTypeSeq OR @DayTypeSeq = 0)

AND (A.SMHolidayTypeSeq = @SMHolidayTypeSeq OR @SMHolidayTypeSeq = 0)

RETURN
