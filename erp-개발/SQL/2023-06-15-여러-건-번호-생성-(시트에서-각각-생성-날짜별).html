여러 건 번호 생성 (시트에서 각각 생성 / 날짜별)

-- 여러건 번호 생성

-- INSERT 번호부여(맨 마지막 처리)

SELECT @Count = COUNT(1) FROM #BIZ_OUT_DataBlock1 WHERE WorkingTag = 'A' AND Status = 0

IF @Count > 0

BEGIN

-- 키값생성코드부분 시작

EXEC @Seq = dbo._SWCOMCreateSeq @CompanySeq, 'emk_TTMSOrder', 'OrderSeq', @Count

-- Temp Table 에 생성된 키값 UPDATE

UPDATE #BIZ_OUT_DataBlock1

SET OrderSeq = @Seq + DataSeq

WHERE WorkingTag = 'A'

AND Status = 0

SELECT @Date = ISNULL(MAX(ReqDate),REPLACE(CONVERT(CHAR(10),GETDATE(),121),'-',''))

FROM #BIZ_OUT_DataBlock1 WHERE WorkingTag = 'A' AND Status = 0

DECLARE @Start NCHAR(8),

@iDate NCHAR(8),

@End NCHAR(8)

DECLARE @StartSeq INT,

@EndSeq INT

SELECT @StartSeq = MIN(A.DataSeq)

FROM #BIZ_OUT_DataBlock1 AS A

SELECT @EndSeq = MAX(A.DataSeq)

FROM #BIZ_OUT_DataBlock1 AS A

SELECT @iDate = @StartSeq

--반복처리

WHILE (@StartSeq \<= @EndSeq)

BEGIN

SELECT @Date = A.ReqDate

FROM #BIZ_OUT_DataBlock1 AS A

WHERE A.DataSeq = @StartSeq

-- 번호생성코드부분 시작

EXEC dbo._SWCOMCreateNo 'SL', 'emk_TTMSOrder', @CompanySeq, 0, @Date, @MaxNo OUTPUT

-- Temp Talbe 에 생성된 키값 UPDATE

UPDATE #BIZ_OUT_DataBlock1

SET OrderNo = @MaxNo

FROM #BIZ_OUT_DataBlock1 AS A

WHERE A.WorkingTag = 'A'

AND A.Status = 0

AND A.DataSeq = @StartSeq

SELECT @StartSeq = @StartSeq + 1

END

END
