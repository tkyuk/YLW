진행 수량 확인

SELECT A.LASTDATETIME,B.ProgTableKorName,C.ProgTableKorName,FromSeq,ToSeq,FromQty,ToQty,*

FROM _TCOMSourceDaily AS A

JOIN _TCOMProgTable AS B ON B.ProgTableSeq = A.FromTableSeq

JOIN _TCOMProgTable AS C ON C.ProgTableSeq = A.ToTableSeq

ORDER BY 1 DESC

SELECT A.LASTDATETIME,B.ProgTableKorName,C.ProgTableKorName,FromSeq,ToSeq,FromQty,ToQty,*

FROM _TCOMSourceDailyTrace AS A

JOIN _TCOMProgTable AS B ON B.ProgTableSeq = A.FromTableSeq

JOIN _TCOMProgTable AS C ON C.ProgTableSeq = A.ToTableSeq
