사이트 sp 일괄 생성

-- =============================================

-- 설명: 기존 SP를 새 SP로 일괄 생성하고,

-- 생성된 SP의 전체 스크립트를 SELECT로 조회하는 스크립트

-- 작성자: 우희철

-- 마음대로 사용하셔도 되지만 사용에 대한 책임은 지지 않습니다

-- =============================================

SET NOCOUNT ON;

-- 변수 선언

DECLARE @SourceSP NVARCHAR(255), @CreateSP NVARCHAR(255);

DECLARE @SQL NVARCHAR(MAX);

DECLARE @SiteInit NVARCHAR(100)

-- 1. SP 목록을 담을 임시 테이블 생성 및 데이터 삽입

-- 이곳에 변환할 SP 목록을 추가하거나 수정하시면 됩니다.

IF OBJECT_ID('tempdb..#SPList_First') IS NOT NULL DROP TABLE #SPList_First;

CREATE TABLE #SPList_First (

SourceSql NVARCHAR(255),

)

IF OBJECT_ID('tempdb..#SPList') IS NOT NULL DROP TABLE #SPList;

CREATE TABLE #SPList (

SourceSql NVARCHAR(255),

CreateSql NVARCHAR(255)

);

SELECT @SiteInit = 'svs'

INSERT INTO #SPList_First

SELECT RIGHT(SqlScriptID,LEN(SqlScriptID)-LEN(@SiteInit))

FROM SYSDEVWKCommon.DBO._TCASQLScripts

WHERE SqlScriptSeq >= 149710308 -- 스크립트 시작 Seq

AND SqlScriptSeq \<= 149710310 -- 스크립트 종료 Seq

AND LastUserSeq = 51399

INSERT INTO #SPList (SourceSql, CreateSql)

SELECT SourceSql, @SiteInit+SourceSql

FROM #SPList_First

--SELECT RIGHT(SqlScriptID,LEN(SqlScriptID)-3) AS SourceSql ,SqlScriptID AS CreateSql

-- FROM sysdevwkcommon.._TCASqlScripts

-- WHERE SqlScriptSeq BETWEEN A AND B

SELECT * FROM #SPList

-- 2. 커서(Cursor)를 사용하여 테이블을 순회

DECLARE sp_cursor CURSOR FOR

SELECT SourceSql, CreateSql FROM #SPList;

OPEN sp_cursor;

FETCH NEXT FROM sp_cursor INTO @SourceSP, @CreateSP;

WHILE @@FETCH_STATUS = 0

BEGIN

-- 3. sp_helptext를 사용하여 기존 SP의 정의를 임시 테이블에 저장

IF OBJECT_ID('tempdb..#SPText') IS NOT NULL DROP TABLE #SPText;

CREATE TABLE #SPText (Text NVARCHAR(MAX));

BEGIN TRY

INSERT INTO #SPText

EXEC sp_helptext @SourceSP;

END TRY

BEGIN CATCH

PRINT @SourceSP + N' 프로시저를 찾을 수 없습니다. 건너뜁니다.';

FETCH NEXT FROM sp_cursor INTO @SourceSP, @CreateSP;

CONTINUE;

END CATCH

-- 4. SP 정의를 하나의 문자열로 합치고, CREATE PROCEDURE 구문 변경

SET @SQL = (SELECT Text FROM #SPText FOR XML PATH(''), TYPE).value('.', 'NVARCHAR(MAX)');

SET @SQL = STUFF(@SQL,

CHARINDEX(@SourceSP, @SQL, CHARINDEX('PROCEDURE', @SQL)),

LEN(@SourceSP),

@CreateSP);

-- 5. 기존에 동일한 이름의 SP가 있다면 삭제

IF OBJECT_ID(@CreateSP, 'P') IS NOT NULL

BEGIN

EXEC('DROP PROCEDURE ' + @CreateSP);

END

-- 6. 변경된 SQL을 실행하여 새로운 SP 생성

BEGIN TRY

EXEC sp_executesql @SQL;

PRINT @CreateSP + N' 프로시저가 성공적으로 생성되었습니다.';

-- 7. 생성된 SP의 스크립트를 SELECT 문으로 결과 창에 출력

IF OBJECT_ID('tempdb..#SPResultText') IS NOT NULL DROP TABLE #SPResultText;

CREATE TABLE #SPResultText (Text NVARCHAR(MAX));

INSERT INTO #SPResultText

EXEC sp_helptext @CreateSP;

-- 결과 그리드에 출력 (이 내용을 복사해서 사용하시면 됩니다)

SELECT @CreateSP AS SPName,'DROP PROC ' + @CreateSP AS SP

UNION ALL

SELECT @CreateSP AS SPName, 'GO' AS SP

UNION ALL

SELECT @CreateSP AS SPName,(SELECT Text FROM #SPResultText FOR XML PATH(''), TYPE).value('.', 'NVARCHAR(MAX)') AS SP

DROP TABLE #SPResultText;

END TRY

BEGIN CATCH

PRINT @CreateSP + N' 프로시저 생성 중 오류가 발생했습니다. 원본 SP: ' + @SourceSP;

PRINT ERROR_MESSAGE();

END CATCH

FETCH NEXT FROM sp_cursor INTO @SourceSP, @CreateSP;

END

-- 커서 및 임시 테이블 정리

CLOSE sp_cursor;

DEALLOCATE sp_cursor;

SELECT

A.SourceSql,

A.CreateSql,

LEN(S.definition) AS SourceSqlLength,

LEN(C.definition) AS CreateSqlLength,

-- 두 프로시저의 길이 차이

ISNULL(LEN(S.definition), 0) - ISNULL(LEN(C.definition), 0) AS LengthDifference

FROM

#SPList AS A

LEFT OUTER JOIN

sys.sql_modules AS S WITH(NOLOCK) ON S.object_id = OBJECT_ID(A.SourceSql)

LEFT OUTER JOIN

sys.sql_modules AS C WITH(NOLOCK) ON C.object_id = OBJECT_ID(A.CreateSql);

DROP TABLE #SPList;

IF OBJECT_ID('tempdb..#SPText') IS NOT NULL DROP TABLE #SPText;

SET NOCOUNT OFF;
