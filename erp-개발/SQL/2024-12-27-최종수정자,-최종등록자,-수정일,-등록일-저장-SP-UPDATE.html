DROP PROCEDURE rot_SDAOilPriceSave

GO

CREATE PROCEDURE rot_SDAOilPriceSave

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

AS

DECLARE @TableColumns NVARCHAR(4000)

SELECT @TableColumns = dbo._FGetColumnsForLog('rot_TDAOilPrice')

-- 로그테이블 남기기(마지막 파라메터는 반드시 한줄로 보내기)

EXEC _SCOMLog @CompanySeq ,

@UserSeq ,

'rot_TDAOilPrice', -- 원테이블명

'#BIZ_OUT_DataBlock1', -- 템프테이블명

'StdYYMM, UMOilTypeSeq' , -- 키가 여러개일 경우는 , 로 연결한다.

@TableColumns,

'', @PgmSeq

-- 신규일 때는 X 수정시에 들어간다

-- DELETE

IF EXISTS (SELECT 1 FROM #BIZ_OUT_DataBlock1 WHERE WorkingTag = 'D' AND Status = 0 )

BEGIN

DELETE rot_TDAOilPrice

FROM #BIZ_OUT_DataBlock1 AS A

JOIN rot_TDAOilPrice AS B ON A.StdYYMMOld = B.StdYYMM

AND A.UMOilTypeSeqOld = B.UMOilTypeSeq

WHERE B.CompanySeq = @CompanySeq

AND A.WorkingTag = 'D'

AND A.Status = 0

IF @@ERROR \<> 0

BEGIN

RETURN

END

END

---- Update

IF EXISTS (SELECT 1 FROM #BIZ_OUT_DataBlock1 WHERE WorkingTag = 'U' AND Status = 0 )

BEGIN

UPDATE rot_TDAOilPrice

SET StdYYMM = A.StdYYMM ,

UMOilTypeSeq = A.UMOilTypeSeq ,

StdPrice = A.StdPrice ,

Remark = ISNULL(A.Remark ,''),

FuelRate = ISNULL(A.FuelRate, 0),

LastDateTime = GETDATE() ,

LastUserSeq = @UserSeq

FROM #BIZ_OUT_DataBlock1 AS A

JOIN rot_TDAOilPrice AS B ON A.StdYYMMOld = B.StdYYMM

AND A.UMOilTypeSeqOld = B.UMOilTypeSeq

WHERE B.CompanySeq = @CompanySeq

AND A.WorkingTag = 'U'

AND A.Status = 0

IF @@ERROR \<> 0

BEGIN

RETURN

END

END

-- INSERT

IF EXISTS (SELECT 1 FROM #BIZ_OUT_DataBlock1 WHERE WorkingTag = 'A' AND Status = 0 )

BEGIN

INSERT INTO rot_TDAOilPrice

(

CompanySeq ,

StdYYMM ,

UMOilTypeSeq ,

StdPrice ,

Remark ,

FuelRate ,

RegUserSeq ,

RegDateTime ,

LastUserSeq ,

LastDateTime

)

SELECT @CompanySeq ,

A.StdYYMM ,

A.UMOilTypeSeq ,

A.StdPrice ,

ISNULL(A.Remark ,''),

ISNULL(A.FuelRate, 0),

@UserSeq ,

GETDATE() ,

@UserSeq ,

GETDATE()

FROM #BIZ_OUT_DataBlock1 AS A

WHERE WorkingTag = 'A'

AND Status = 0

IF @@ERROR \<> 0

BEGIN

RETURN

END

END

UPDATE A

SET StdYYMMOld = StdYYMM,

UMOilTypeSeqOld = UMOilTypeSeq

FROM #BIZ_OUT_DataBlock1 AS A

WHERE A.WorkingTag IN ('A', 'U')

AND A.Status = 0

UPDATE A

SET RegUserName = ISNULL(C.UserName,N''),

RegDateTime = B.RegDateTime

FROM #BIZ_OUT_DataBlock1 AS A

JOIN rot_TDAOilPrice AS B WITH(NOLOCK) ON A.StdYYMM = B.StdYYMM

AND A.UMOilTypeSeq = B.UMOilTypeSeq

LEFT OUTER JOIN _TCAUser AS C WITH(NOLOCK) ON B.RegUserSeq = C.UserSeq

WHERE A.WorkingTag IN (N'A')

AND A.Status = 0

AND B.CompanySeq = @CompanySeq

UPDATE A

SET LastUserName = ISNULL(C.UserName,N''),

LastDateTime = B.LastDateTime

FROM #BIZ_OUT_DataBlock1 AS A

JOIN rot_TDAOilPrice AS B WITH(NOLOCK) ON A.StdYYMM = B.StdYYMM

AND A.UMOilTypeSeq = B.UMOilTypeSeq

LEFT OUTER JOIN _TCAUser AS C WITH(NOLOCK) ON B.LastUserSeq = C.UserSeq

WHERE A.WorkingTag IN (N'A',N'U')

AND A.Status = 0

AND B.CompanySeq = @CompanySeq

RETURN

UPDATE A

SET RegUserName = ISNULL(C.UserName,N''),

RegDateTime = B.RegDateTime

FROM #BIZ_OUT_DataBlock1 AS A

JOIN rot_TDAOilPrice AS B WITH(NOLOCK) ON A.StdYYMM = B.StdYYMM

AND A.UMOilTypeSeq = B.UMOilTypeSeq

LEFT OUTER JOIN _TCAUser AS C WITH(NOLOCK) ON B.RegUserSeq = C.UserSeq

WHERE A.WorkingTag IN (N'A')

AND A.Status = 0

AND B.CompanySeq = @CompanySeq

UPDATE A

SET LastUserName = ISNULL(C.UserName,N''),

LastDateTime = B.LastDateTime

FROM #BIZ_OUT_DataBlock1 AS A

JOIN rot_TDAOilPrice AS B WITH(NOLOCK) ON A.StdYYMM = B.StdYYMM

AND A.UMOilTypeSeq = B.UMOilTypeSeq

LEFT OUTER JOIN _TCAUser AS C WITH(NOLOCK) ON B.LastUserSeq = C.UserSeq

WHERE A.WorkingTag IN (N'A',N'U')

AND A.Status = 0

AND B.CompanySeq = @CompanySeq
