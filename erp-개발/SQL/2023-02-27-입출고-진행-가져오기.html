INSERT INTO #Temp_InOutItemProg(InOutType, InOutSeq, InOutSerl, CompleteCHECK, IsStop)

SELECT A.InOutType, A.InOutSeq, A.InOutSerl, -1, '0'

FROM _TLGInOutDailyItem                                AS A WITH(NOLOCK)

JOIN _TLGInOutDaily                                AS B WITH(NOLOCK) ON A.CompanySeq                = B.CompanySeq

AND A.InOutType                = B.InOutType

AND A.InOutSeq                        = B.InOutSeq

LEFT OUTER JOIN _TDAItem                AS C WITH(NOLOCK) ON A.CompanySeq                = C.CompanySeq

AND A.ItemSeq                        = C.ItemSeq

LEFT OUTER JOIN _TDAItemSales        AS D WITH(NOLOCK) ON D.CompanySeq                = C.CompanySeq

AND D.ItemSeq                        = C.ItemSeq

LEFT OUTER JOIN _TDAItemClass        AS E WITH(NOLOCK) ON C.CompanySeq                = E.CompanySeq

AND C.ItemSeq                        = E.ItemSeq

AND E.UMajorItemClass IN (2001,2004)

LEFT OUTER JOIN _VDAItemClass        AS F WITH(NOLOCK) ON F.CompanySeq                = C.CompanySeq

AND F.ItemClassSSeq        = E.UMItemClass

LEFT OUTER JOIN _TDADept AS G WITH(NOLOCK) ON B.CompanySeq = G.CompanySeq

AND B.DeptSeq = G.DeptSeq

WHERE (@InOutType = 0

OR A.InOutType = @InOutType

OR ( (A.InOutType IN (80, 81) AND @InOutType = 80)

OR (A.InOutType IN (82, 83) AND @InOutType = 82)

OR (A.InOutType IN (70, 71) AND @InOutType = 70)

OR (A.InOutType IN (72, 73) AND @InOutType = 72)

OR (A.InOutType IN (50, 51) AND @InOutType = 50)

OR (A.InOutType = 600 AND @InOutType = 30 AND D.IsSet = '1')                --기타출고입력(세트품) : 20170612 장경선

OR (A.InOutType = 610 AND @InOutType = 40 AND D.IsSet = '1'))         --기타입고입력(세트품) : 20170612 장경선

)

AND (@BizUnit = 0 OR B.BizUnit = @BizUnit)

AND (@InOutDateFr = '' OR B.InOutDate >= @InOutDateFr)

AND (@InOutDateTo = '' OR B.InOutDate \<= @InOutDateTo)

AND (@InOutNo = '' OR B.InOutNo LIKE @InOutNo + N'%')

-- AND (@DeptSeq = 0 OR B.DeptSeq = @DeptSeq) -- 처리부서 조회조건 코드도움에서 TEXT로 변경 221201 HCWOO

AND (@DeptName = '' OR G.DeptName LIKE @DeptName + N'%') -- 처리부서 조회조건 코드도움에서 TEXT로 변경 221201 HCWOO

AND (@EmpSeq = 0 OR B.EmpSeq = @EmpSeq)

AND (@IsCompleted = '' OR B.IsCompleted = @IsCompleted)

AND (@CompleteDeptSeq = 0 OR B.CompleteDeptSeq = @CompleteDeptSeq)

AND (@CompleteEmpSeq = 0 OR B.CompleteEmpSeq = @CompleteEmpSeq)

AND (@CustSeq = 0 OR B.CustSeq = @CustSeq)

AND (@OutWHSeq = 0 OR B.OutWHSeq = @OutWHSeq)

AND (@InWHSeq = 0 OR B.InWHSeq = @InWHSeq)

AND (@InOutDetailKind = 0 OR A.InOutDetailKind = @InOutDetailKind)

AND (@ItemName = '' OR C.ItemName LIKE @ItemName + N'%')

AND (@ItemNo = '' OR C.ItemNo LIKE @ItemNo + N'%')

AND (@Spec = '' OR C.Spec LIKE @Spec + N'%')

AND (@CCtrSeq = 0 OR A.CCtrSeq = @CCtrSeq)

AND (@UseDeptSeq = 0 OR B.UseDeptSeq = @UseDeptSeq)

AND (@CompleteDateFr = '' OR B.CompleteDate >= @CompleteDateFr)

AND (@CompleteDateTo = '' OR B.CompleteDate \<= @CompleteDateTo)

AND (@ReqBizUnit = 0 OR B.ReqBizUnit = @ReqBizUnit)

AND (@InOutKind = 0 OR A.InOutKind = @InOutKind)

AND (@MoveTransKind = 0 OR (@MoveTransKind = 8099001 AND B.InOutType IN (80,82))

OR (@MoveTransKind = 8099002 AND B.InOutType IN (81,83)))

AND (@BizUnit = 0 OR B.BizUnit = @BizUnit)

AND (@AssetSeq                = 0                OR C.AssetSeq                        = @AssetSeq)

AND (@ItemClassSSeq        = 0                OR E.UMItemClass                = @ItemClassSSeq)

AND (@ItemClassMSeq        = 0                OR F.ItemClassMSeq                = @ItemClassMSeq)

AND (@ItemClassLSeq        = 0                OR F.ItemClassLSeq                = @ItemClassLSeq)

AND (@InOutType2 = 0 OR (@InOutType2 = 8025001 AND A.InOutType IN (70, 72)) OR (@InOutType2 = 8025002 AND A.InOutType IN (71,73)))

AND A.CompanySeq = @CompanySeq

INSERT #TMP_PROGRESSTABLE

/** ReqNo 찾기 2021.12.17 김기태 **/

UPDATE #Temp_InOutItemProg

SET ReqSeq = ISNULL(C.ReqSeq, 0)

, ReqNo = ISNULL(C.ReqNo, '')

, ReqDeptSeq = ISNULL(C.DeptSeq, 0)        

, ReqEmpSeq = ISNULL(C.EmpSeq, 0)        

, Qty         = ISNULL(D.Qty, 0)        

FROM #Temp_InOutItemProg                        AS A

JOIN #TCOMSourceTracking                        AS B ON A.IDX_NO = B.IDX_NO

AND B.IDOrder = 2

JOIN _TLGInOutReq                                        AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq AND B.Seq = C.ReqSeq

LEFT OUTER JOIN _TLGInOutReqItem        AS D WITH(NOLOCK) ON D.CompanySeq = @CompanySeq AND D.ReqSeq = C.ReqSeq

--2022.08.10 InvoiceNo 찾기 :: 우승현

UPDATE #Temp_InOutItemProg

SET InvoiceNo = ISNULL(C.InvoiceNo, '')

FROM #Temp_InOutItemProg AS A

JOIN #TCOMSourceTracking AS B ON B.IDX_NO = A.IDX_NO

AND B.IDOrder = 3

JOIN _TSLInvoice AS C ON C.InvoiceSeq = B.Seq

--[기타출고품목조회] 화면 금액 컬럼 수정 :: 22.04.07 우승현

--[기타입고품목조회] InOutType조건 삭제 :: 22.06.17 방혁

UPDATE #Temp_InOutItemProg

SET Amt = ISNULL(C.Amt, 0)

FROM #Temp_InOutItemProg AS A

JOIN (

SELECT A.InOutType, A.InOutSeq, A.InOutSerl, SUM(B.Amt) AS AMT

FROM #Temp_InOutItemProg AS A

JOIN _TLGInOutStock AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.InOutType = A.InOutType

AND B.InOutSeq = A.InOutSeq

AND B.InOutSerl = A.InOutSerl

WHERE B.CompanySeq = @CompanySeq

--AND B.InOutType = 30

GROUP BY A.InOutType, A.InOutSeq, A.InOutSerl

) AS C ON C.InOutType = A.InOutType

AND C.InOutSeq = A.InOutSeq

AND C.InOutSerl = A.InOutSerl

--WHERE A.InOutType = 30
