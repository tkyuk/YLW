세트품

SELECT DISTINCT

A.ItemName AS SetItemName, -- 세트품명

A.ItemNo AS SetItemNo, -- 세트품번

A.Spec                 AS SetSpec, -- 세트규격

A.ItemSeq AS SetItemSeq -- 세트품코드

FROM _TDAItem AS A WITH(NOLOCK)

INNER JOIN _TDAItemSales AS B WITH(NOLOCK) ON A.ItemSeq = B.ItemSeq

AND B.IsSet = '1'

AND A.CompanySeq = B.CompanySeq

WHERE A.SMStatus NOT IN (2001001, 2001005)

AND A.SMStatus NOT IN (SELECT ItemStatus FROM _TDAItemStatusCodeHelp WITH(NOLOCK) WHERE ItemCodeHelpSeq = 2)

ORDER BY A.ItemName
