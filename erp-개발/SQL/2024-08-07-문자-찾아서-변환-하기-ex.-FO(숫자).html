문자 찾아서 변환 하기 ex. FO(숫자)

CASE -- FO 패턴 찾기 (소수점 포함)

WHEN GoodItemSpec LIKE '%FO([0-9]% %' THEN

'Fail Open (' +

SUBSTRING(

GoodItemSpec,

CHARINDEX('FO(', GoodItemSpec) + 3,

CHARINDEX(')', GoodItemSpec, CHARINDEX('FO(', GoodItemSpec)) - CHARINDEX('FO(', GoodItemSpec) - 3

) + ')'

-- FC 패턴 찾기 (소수점 포함)

WHEN GoodItemSpec LIKE '%FC([0-9]% %' THEN

'Fail Close (' +

SUBSTRING(

GoodItemSpec,

CHARINDEX('FC(', GoodItemSpec) + 3,

CHARINDEX(')', GoodItemSpec, CHARINDEX('FC(', GoodItemSpec)) - CHARINDEX('FC(', GoodItemSpec) - 3

) + ')'

ELSE NULL
