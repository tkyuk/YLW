사원 부서 테이블 (발령 / 내부 외부직원)

SELECT A.EmpSeq, C.DeptSeq

FROM #BIZ_OUT_DataBlock1 AS Z

JOIN _TDAEmpIn AS A WITH(NOLOCK) ON A.CompanySeq = @CompanySeq

AND A.EmpSeq = Z.EmpSeq

LEFT OUTER JOIN _THRAdmOrdEmp AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.EmpSeq = C.EmpSeq

AND CASE WHEN CONVERT(NVARCHAR(8), GETDATE(), 112) \<= A.EntDate THEN A.EntDate -- 입사발령이 미래시점으로 난 경우 입사일

WHEN CONVERT(NVARCHAR(8), GETDATE(), 112) >= A.RetireDate THEN A.RetireDate -- 퇴사자는 퇴사일 기준

ELSE CONVERT(NVARCHAR(8), GETDATE(), 112)

END BETWEEN C.OrdDate AND C.OrdEndDate

AND C.IsOrdDateLast = '1'

UNION

SELECT ISNULL(B.EmpSeq , 0)        AS EmpSeq , -- 사원, 사원코드 ,

(CASE WHEN ISNULL(C.ContractSeq, 0) = 0 THEN ISNULL(B.WkDeptSeq, 0) ELSE ISNULL(C.WkDeptSeq, 0 ) END) AS WkDeptSeq -- 부서코드

FROM #BIZ_OUT_DataBlock1 AS Z

JOIN _TDAEmp AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.EmpSeq = Z.EmpSeq

LEFT OUTER JOIN _THRLabDailyEmpContract AS C WITH(NOLOCK)

ON B.CompanySeq = C.CompanySeq

AND B.EmpSeq = C.EmpSeq

AND C.BegDate = (SELECT MAX(BegDate)

FROM _THRLabDailyEmpContract WITH(NOLOCK)

WHERE CompanySeq = C.CompanySeq

AND EmpSeq = C.EmpSeq

AND BegDate \<= CONVERT(NCHAR(8), GETDATE(), 112) )

LEFT OUTER JOIN _TDADept AS D WITH(NOLOCK) ON D.CompanySeq = B.CompanySeq

AND D.DeptSeq = B.WkDeptSeq
