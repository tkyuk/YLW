중복체크 로직

EXEC dbo._SWCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

6 , -- 중복된 @1 @2가(이) 입력되었습니다.

@LanguageSeq ,

25961,'' -- '대체'

,29408,'' -- 'serialNo'

UPDATE A

SET A.Result = @Results,

A.MessageType = @MessageType,

A.Status = @Status

FROM #BIZ_OUT_DataBlock2 AS A

WHERE A.Status = 0

AND A.WorkingTag IN ('A', 'U')

AND (

-- 기존 테이블 중복체크

EXISTS (

SELECT 1

FROM _TLGItemConvertDesc AS B WITH(NOLOCK)

WHERE B.CompanySeq = @CompanySeq

AND A.ConvertSeq = B.ConvertSeq

AND A.Dummy2 = B.Dummy2

AND (A.WorkingTag = 'A' OR (A.WorkingTag = 'U' AND A.ConvertSerl \<> B.ConvertSerl))

)

OR

-- 입력 시트 내 중복 체크

EXISTS (

SELECT 1

FROM #BIZ_OUT_DataBlock2 AS C

WHERE A.IDX_NO \<> C.IDX_NO -- 자기 자신 제외

AND A.ConvertSeq = C.ConvertSeq

AND A.Dummy2 = C.Dummy2

)

)

EXEC dbo._SWCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

6 , -- 중복된 @1 @2가(이) 입력되었습니다.(SELECT * FROM _TCAMessageLanguage WHERE Message LIKE '%중복%')

@LanguageSeq ,

25961,'' -- SELECT * FROM _TCADictionary WHERE Word like '대체' AND LanguageSeq = 1

,29408,'' -- SELECT * FROM _TCADictionary WHERE Word like '%serialNo%' AND LanguageSeq = 1

-- 신규입력시 기존데이터와 중복된 경우

UPDATE #BIZ_OUT_DataBlock1

SET Result = @Results,

MessageType = @MessageType,

Status = @Status

FROM #BIZ_OUT_DataBlock2 AS A

JOIN _TLGItemConvertDesc AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND A.ConvertSeq = B.ConvertSeq

AND A.Dummy2 = B.Dummy2

WHERE A.WorkingTag = 'A'

AND A.Status = 0

-- 수정시 기존데이터와 중복된 경우

UPDATE #BIZ_OUT_DataBlock1

SET Result = @Results,

MessageType = @MessageType,

Status = @Status

FROM #BIZ_OUT_DataBlock2 AS A

JOIN _TLGItemConvertDesc AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND A.Dummy2 = B.Dummy2

AND A.ConvertSeq = B.ConvertSeq

AND A.ConvertSerl \<> B.ConvertSerl

WHERE A.WorkingTag = 'U'

AND A.Status = 0

-- 시트 내에서 중복된경우

UPDATE A

SET Result = @Results,

MessageType = @MessageType,

Status = @Status

FROM #BIZ_OUT_DataBlock2 AS A

JOIN #BIZ_OUT_DataBlock2 AS B ON A.IDX_NO \<> B.IDX_NO

AND A.ConvertSeq = B.ConvertSeq

AND A.Dummy2 = B.Dummy2

WHERE A.WorkingTag IN ('A', 'U')

AND A.Status = 0
