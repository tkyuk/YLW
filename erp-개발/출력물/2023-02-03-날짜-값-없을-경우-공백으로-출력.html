// 1. 값이 null이 아니고, 공백을 제거했을 때도 빈 값이 아닌지 확인

if (date != null && date != "") {

// 데이터가 8자리(YYYYMMDD)인 경우에만 포맷팅 실행

if (date.length === 8) {

var year = date.substring(0, 4);

var month = date.substring(4, 6);

var day = date.substring(6, 8);

// 연도 4자리를 다 쓰고 싶다면 아래 줄 사용

// var year = date.substring(0, 4);

This.SetText(year + "." + month + "." + day);

} else {

// 8자리가 아니면 가져온 값 그대로 출력하거나 별도 처리

This.SetText(date);

}

} else {

// 2. 데이터가 공백이면 라벨도 공백으로 설정

This.SetText("");

}
