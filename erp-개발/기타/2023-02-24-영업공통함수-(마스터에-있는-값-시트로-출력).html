function fnSLGetStdQty_djb(ObjectSS, IntSRow, IntERow, StrColItemSeq, StrColUnitSeq, StrColQty, StrColSTDUnitName, StrColSTDUnitSeq, StrColSTDQty, StrColIsQtyChange, StrColUnitName)

local ds

local columnSchema

local addData

local ActColName

ActColName = ObjectSS.ActiveColumnName

columnSchema = '{WorkingTag, string}{IDX_NO, int}{Status, int}{DataSeq, int}{Selected, int}{TABLE_NAME, string}'

..'{ItemSeq, int}{UnitSeq, int}{Qty, double}'

..'{StdUnitSeq, int}{StdUnitName, string}{StdQty, double}{IsQtyChange, string}{ConvUnitSeq, int}{ConvQty, double}{UnitName, string}'

ds = CreateDataSet(ds,'DataBlock1', columnSchema)

for iRow = IntSRow, IntERow, 1 do

ItemSeq = CellText(ObjectSS, iRow, StrColItemSeq)

UnitSeq = CellText(ObjectSS, iRow, StrColUnitSeq)

Qty = CellText(ObjectSS, iRow, StrColQty)

UnitName = CellText(ObjectSS, iRow, StrColUnitName)

if ItemSeq == '' then

ItemSeq = 0

end

if UnitSeq == '' then

UnitSeq = 0

end

if Qty == '' then

Qty = 0

end

addData = 'A'..CNST_SP..tostring(iRow)..CNST_SP..'0'..CNST_SP..tostring(iRow)..CNST_SP..'1'..CNST_SP..'DataBlock1'..CNST_SP

..tostring(ItemSeq)..CNST_SP

..tostring(UnitSeq)..CNST_SP

..tostring(Qty)..CNST_SP

..'0'..CNST_SP

..''..CNST_SP

..'0'..CNST_SP

..'0'..CNST_SP

..'0'..CNST_SP

..'0'..CNST_SP

..tostring(UnitName)

AddDataRow(ds, 'DataBlock1', addData)

end

rtnDs = ExecuteService(501822, 1, ds, rtnDs, 'SQLWF')

if (rtnDs == nil) then

else

for row=0, GetDataBlock(rtnDs, 'DataBlock1').Rows.Count-1, 1 do

SetText(ObjectSS, IntSRow + row, StrColSTDUnitName, GetDataSetValue(rtnDs, 'DataBlock1', 'StdUnitName', row))

SetText(ObjectSS, IntSRow + row, StrColSTDUnitSeq, GetDataSetValue(rtnDs, 'DataBlock1', 'StdUnitSeq', row))

SetText(ObjectSS, IntSRow + row, StrColSTDQty, GetDataSetValue(rtnDs, 'DataBlock1', 'StdQty', row))

if StrColUnitSeq ~= '' and CellText(ObjectSS, IntSRow + row, StrColUnitSeq) == '0' then

SetText(ObjectSS, IntSRow + row, StrColUnitSeq, GetDataSetValue(rtnDs, 'DataBlock1', 'UnitSeq', row))

SetText(ObjectSS, IntSRow + row, StrColUnitName, GetDataSetValue(rtnDs, 'DataBlock1', 'UnitName', row))

end

if StrColIsQtyChange ~= '' then

ObjectSS.ActiveColumnName = StrColIsQtyChange

ObjectSS.ActiveRow = IntSRow + row

if GetDataSetValue(rtnDs, 'DataBlock1', 'IsQtyChange', row) == '1' then

ObjectSS.ActiveCellReadOnly = false

else

ObjectSS.ActiveCellReadOnly = true

end

end

end

end

ObjectSS.ActiveColumnName = ActColName

end
