--=====================================================================

-- Serial 수불반영 _TLGInOutSerialStock / _TLGInOutSerialSub

--=====================================================================

-- 셀바스 추가 25.11.17 SJPARK

IF ISNULL(OBJECT_ID('tempdb..#TLGInOutSerialStock'),1)=1

BEGIN

CREATE TABLE #TLGInOutSerialStock

(

WorkingTag NCHAR(1),

IDX_NO INT,

DataSeq INT,

Selected INT,

MessageType INT,

Status INT,

Result NVARCHAR(255),

InOutType INT,

InOutSeq INT,

InOutSerl INT,

InOutSubSerl INT,

InOutDataSerl INT,

InOutSerialSerl INT,

DataKind INT,

SerialNo NVARCHAR(100),

SerialNoOld NVARCHAR(100)

)

END

INSERT INTO #TLGInOutSerialStock

(

WorkingTag, IDX_NO, DataSeq, Selected, Status, MessageType, Result, InOutType,

InOutSeq, InOutSerl, InOutSubSerl, InOutDataSerl, InOutSerialSerl, DataKind, SerialNo, SerialNoOld

)

SELECT DISTINCT

Z.WorkingTag AS WorkingTag

,Z.IDX_NO AS IDX_NO

,Z.DataSeq AS DataSeq

,Z.Selected AS Selected

,Z.Status AS Status

,Z.MessageType AS MessageType

,Z.Result AS Result

,A.InOutType AS InOutType

,A.InOutSeq AS InOutSeq

,A.InOutSerl AS InOutSerl

,0 AS InOutSubSerl

,1 AS InOutDataSerl

,A.InOutSerialSerl AS InOutSerialSerl

,ISNULL(A.DataKind,0) AS DataKind

,A.SerialNo AS SerialNo

,A.SerialNo AS SerialNoOld

FROM #BIZ_OUT_DataBlock1 AS Z

JOIN _TLGInOutSerialSub AS A WITH(NOLOCK) ON A.CompanySeq = @CompanySeq

AND A.InOutType = Z.InOutType

AND A.InOutSeq = Z.InvoiceSeq

WHERE Z.Status = 0

IF EXISTS (SELECT TOP 1 1 FROM #TLGInOutSerialStock)

BEGIN

EXEC svs_SWLGInOutSerialStockProc @CompanySeq = @CompanySeq,

@UserSeq = @UserSeq,

@PgmSeq = @PgmSeq

IF EXISTS ( SELECT TOP 1 1 FROM #TLGInOutSerialStock WHERE Status \<> 0 )

BEGIN

UPDATE #BIZ_OUT_DataBlock1

SET MessageType = B.MessageType

,Status = B.Status

,Result = B.Result

FROM #BIZ_OUT_DataBlock1 AS A

JOIN #TLGInOutSerialStock AS B ON A.IDX_NO = B.IDX_NO

WHERE B.Status \<> 0

END

END
