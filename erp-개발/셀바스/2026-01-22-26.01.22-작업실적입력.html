BEGIN TRAN

INSERT INTO svs_TPDSFCWorkReportSerial

(

CompanySeq ,WorkReportSeq ,SerialNo ,SerialSeq ,ItemSeq ,FW1Ver ,FW2Ver ,FW3Ver ,CellNo ,

Remark ,OutDate ,WarrantySD ,WarrantyPeriod,WarrantyED ,CustomerName ,CustomerDept ,CustomerEmp ,UMRegion ,

Address ,CPNo ,TelNo ,FaxNo ,Email1 ,Email2 ,AddInfo1 ,AddInfo2 ,CustSeq ,

WorkDate ,PgmSeq ,RegUserSeq ,RegDateTime ,LastUserSeq ,LastDateTime

)

SELECT E.CompanySeq AS CompanySeq

,C.WorkReportSeq AS WorkReportSeq

,E.SerialNo AS SerialNo

,E.SerialSeq AS SerialSeq

,D.ItemSeq AS ItemSeq

,E.FW1Ver AS FW1Ver

,E.FW2Ver AS FW2Ver

,E.FW3Ver AS FW3Ver

,E.CellNo AS CellNo

,E.Remark AS Remark

,E.OutDate AS OutDate

,E.WarrantySD AS WarrantySD

,E.WarrantyPeriod AS WarrantyPeriod

,E.WarrantyED AS WarrantyED

,E.CustomerName AS CustomerName

,E.CustomerDept AS CustomerDept

,E.CustomerEmp AS CustomerEmp

,E.UMRegion AS UMRegion

,E.Address AS Address

,E.CPNo AS CPNo

,E.TelNo AS TelNo

,E.FaxNo AS FaxNo

,E.Email1 AS Email1

,E.Email2 AS Email2

,E.AddInfo1 AS AddInfo1

,E.AddInfo2 AS AddInfo2

,D.CustSeq AS CustSeq

,E.WorkDate AS WorkDate

,9999 AS PgmSeq

,1 AS RegUserSeq

,GETDATE() AS RegDateTime

,1 AS LastUserSeq

,GETDATE() AS LastDateTime

FROM _TLGInOutSerialSub AS A

JOIN _TPDSFCGoodIn AS B ON B.GoodInSeq = A.InOutSeq

JOIN _TPDSFCWorkReport AS C ON C.WorkReportSeq = B.WorkReportSeq

JOIN _TLGSerialMaster AS D ON D.SerialNo = A.SerialNo

JOIN svs_TLGSerialMaster AS E ON E.SerialNo = A.SerialNo

WHERE A.InOutType = 140

ROLLBACK TRAN
