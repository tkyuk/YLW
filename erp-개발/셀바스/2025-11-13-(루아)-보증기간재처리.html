(루아) 보증기간재처리

-- 보증만료일 = 보증개시일 + 보증기간(개월)

local WarrantySD, WarrantyPeriod, WarrantyED

-- 보증개시일

WarrantySD = CellText(SS3, SS3.ActiveRow, 'WarrantySD')

-- 보증기간

WarrantyPeriod = tonumber(CellText(SS3, SS3.ActiveRow, 'WarrantyPeriod'))

if WarrantyPeriod == nil or WarrantyPeriod == 0 then

WarrantyPeriod = 0

end

if WarrantySD == '' or WarrantySD == nil then

WarrantyED = ''

else

WarrantyED = DateAdd('Month', WarrantyPeriod, WarrantySD)

end

-- 계산된 보증 만료일을 'WarrantyED' 컬럼에 설정합니다.

SetText(SS3, SS3.ActiveRow, 'WarrantyED', WarrantyED)
