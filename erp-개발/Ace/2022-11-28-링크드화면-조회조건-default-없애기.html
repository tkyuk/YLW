링크드화면 - 조회조건 default 없애기

default 공백으로 넣으면 동작하지 않기 때문에

초기 - pgmseq에 따라서 값 넣어주기

if _base.PgmSeq == 97920008 then

txtDeptName.Text = _base.DeptName

txtEmpName.Text = _base.EmpName

txtDeptName.TextCd = _base.DeptSeq

txtEmpName.TextCd = _base.EmpSeq

txtReqDeptName.Text = 0

txtReqEmpName.Text = 0

txtReqDeptName.TextCd = ''

txtReqEmpName.TextCd = ''

else

txtDeptName.Text = 0

txtEmpName.Text = 0

txtDeptName.TextCd = ''

txtEmpName.TextCd = ''

txtReqDeptName.Text = _base.DeptName

txtReqEmpName.Text = _base.EmpName

txtReqDeptName.TextCd = _base.DeptSeq

txtReqEmpName.TextCd = _base.EmpSeq

end
