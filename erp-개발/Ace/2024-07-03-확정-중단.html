확정 중단

1. 확정테이블관리(2014)

_TCOMConfimTable = 확정을 사용하겠다

_TCOMConfirm => 확정 사용할 때 필수로 join해야하는 테이블

-- 확정 테이블 생성 (ConfirmTableSeq)

SELECT ConfirmTableName AS '확정테이블'

,ProgressTableName AS '체크진행테이블'

,*

FROM _TCOMConfirmTable WHERE ConfirmTableSeq = 15000042

-- 확정 ID (확정명) -> 확정테이블 연결 (ConfirmSeq -> ConfirmTableSeq)

SELECT * FROM _TCOMConfirmDef WHERE ConfirmSeq = 15000053

-- 확정 프로그램 생성 (ConfirmSeq \<-> PgmSeq)

SELECT * FROM _TCOMConfirmPgm WHERE ConfirmSeq = 15000053

-- 실제 확정 데이터 저장하는 테이블 (ConfirmSeq / CfmSeq(키 값)) (CfmCode = '1' 이면 확정 / 확정날짜, 확정한 사용자 조회가능)

SELECT * FROM _TCOMConfirm WHERE ConfirmSeq = 15000053

-- 중단

-- 중단설정프로그램 - 1. 진행 테이블 조회

SELECT * FROM _TCOMProgTable

-- 중단 사용할 프로그램 세팅 (_TCOMStopPGM)

SELECT C.Caption,

CASE WHEN B.IsReadOnly = '1' THEN '읽기전용' ELSE '' END,

B.ControlName,B.SeqColumn,B.SerlColumn,

A.*

FROM _TCOMProgTable AS A

JOIN _TCOMStopPGM AS B ON B.TableSeq = A.ProgTableSeq

JOIN _TCAPgm AS C ON C.PgmSeq = B.PgmSeq
