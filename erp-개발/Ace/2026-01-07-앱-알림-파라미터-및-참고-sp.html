CREATE PROC svs_SWFlexLGKumsTemplateEtcOutReq

@WorkingTag NVARCHAR(10)= ''

,@CompanySeq INT = 1

,@LanguageSeq INT = 1

,@UserSeq INT = 0

,@PgmSeq INT = 0

,@MessageDiv CHAR(1)

,@MessageType VARCHAR(50)

,@TemplateCode VARCHAR(30)

,@TemplateContent NVARCHAR(1000)

,@Param1 NVARCHAR(4000) -- ReqSeq

,@Param2 NVARCHAR(4000) -- InOutReqType

,@Param3 NVARCHAR(4000)

,@Param4 NVARCHAR(4000)

,@Param5 NVARCHAR(4000)

AS

-- WITH(NOLOCK) 효과

SET TRANSACTION ISOLATION LEVEL READ UNCOMMITTED

/*******************************************

-- 2-1. 제목

*******************************************/

DECLARE @Subjects NVARCHAR(1000) = '' -- (OUTPUT) 제목

,@RecvNames NVARCHAR(MAX) = '' -- (OUTPUT) 수신자

,@RecvValues NVARCHAR(MAX) = '' -- (OUTPUT) 수신자내부코드

,@WordNotice NVARCHAR(200) -- 단어 : 공지

,@WordAlarm NVARCHAR(200) -- 단어 : 알림

,@MbFlexAppNoti_FlexAppID NVARCHAR(50) -- OTCAKumsTemplatePgm 의 MbFlexAppNoti_FlexAppID

DECLARE @FlexAppID NVARCHAR(100)

SELECT @FlexAppID = ISNULL(A.MbFlexAppNoti_FlexAppID, N'')

FROM OTCAKumsTemplatePgm AS A WITH(NOLOCK)

WHERE A.CompanySeq = @CompanySeq

AND ISNULL(A.PgmSeq , 0) = 501110 -- 기타출고요청입력

/***************************** #{content} 시작 **************************/

DECLARE @CompanyName NVARCHAR(100), @PgmName NVARCHAR(200), @UserDept NVARCHAR(100), @UserName NVARCHAR(100) , @CurrDate NCHAR(8), @Remark NVARCHAR(1000)

DECLARE @ReqNo NVARCHAR(20), @ReqDate NCHAR(10), @ItemName NVARCHAR(100), @WHName NVARCHAR(100), @WishDate NCHAR(10)

SELECT @CurrDate = CONVERT(NCHAR(8), GETDATE(), 112)

SELECT @CompanyName = (SELECT CompanyName FROM _TCACompany WITH(NOLOCK) WHERE CompanySeq = @CompanySeq)

SELECT @PgmName = (SELECT Caption FROM _TCAPgm WITH(NOLOCK) WHERE PgmSeq = @PgmSeq)

SELECT @UserDept = (SELECT ISNULL(D.DeptName, ISNULL(H.DeptName, N''))

FROM _TCAUser AS A WITH(NOLOCK)

LEFT OUTER JOIN _TDAEmp AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.EmpSeq = B.EmpSeq

LEFT OUTER JOIN _THRAdmOrdEmp AS C WITH(NOLOCK) ON B.CompanySeq = C.CompanySeq

AND B.EmpSeq = C.EmpSeq -- 발령일내에 있는 사원.

AND C.OrdDate+RIGHT(N'00000'+CONVERT(NVARCHAR(5),C.IntSeq),5) = (SELECT MAX(OrdDate+RIGHT(N'00000'+CONVERT(NVARCHAR(5),IntSeq),5))

FROM _THRAdmOrdEmp

WHERE CompanySeq = @CompanySeq

AND EmpSeq = C.EmpSeq

AND IsOrdDateLast = N'1'

AND OrdDate \<= @CurrDate) -- 발령일자 기준 이전발령

LEFT OUTER JOIN _TDADept AS D WITH(NOLOCK) ON C.CompanySeq = D.CompanySeq

AND C.DeptSeq = D.DeptSeq

LEFT OUTER JOIN _TDADept AS H WITH(NOLOCK) ON B.CompanySeq = H.CompanySeq --제품군이 WBS인 경우는 _TDAEmpOutContract 테이블 사용하지 않음

AND B.WkDeptSeq = H.DeptSeq

WHERE A.CompanySeq = @CompanySeq AND A.UserSeq = @UserSeq)

SELECT @UserName = (SELECT UserName FROM _TCAUser WHERE UserSeq = @UserSeq AND CompanySeq = @CompanySeq )

SELECT @ReqNo = A.ReqNo ,

@ReqDate = LEFT(A.ReqDate, 4) + '-' + SUBSTRING(A.ReqDate, 5, 2) + '-' + RIGHT(A.ReqDate, 2),

@WHName = C.WHName,

@WishDate = A.CompleteWishDate,

@Remark = ISNULL(A.Remark,'')

FROM _TLGInOutReq AS A WITH(NOLOCK)

JOIN _TDAWH AS C WITH(NOLOCK) ON A.OutWHSeq = C.WHSeq

AND A.CompanySeq = C.CompanySeq

WHERE A.CompanySeq = @CompanySeq

AND A.ReqSeq = @Param1

SELECT @ItemName = C.ItemName

FROM _TLGInOutReq AS A WITH(NOLOCK)

JOIN _TLGInOutReqItem AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.ReqSeq = A.ReqSeq

JOIN _TDAItem AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.ItemSeq = B.ItemSeq

WHERE A.CompanySeq = @CompanySeq

AND A.ReqSeq = @Param1

GROUP BY C.ItemName, B.Qty

ORDER BY B.Qty

SELECT @TemplateContent = REPLACE(@TemplateContent, '#{content}',

'[' + @CompanyName + ']' + NCHAR(13) +

'- 처리화면 : ' + @PgmName + ' / ' + @ReqNo + NCHAR(13) +

'- 발신자 : ' + @UserDept + ' / ' + @UserName + NCHAR(13) +

'- 출고창고 : ' + @WHName + NCHAR(13) +

'- 대표품명 : ' + @ItemName + NCHAR(13) +

'- 특이사항 : ' + @Remark)

-- EXEC @WordAlarm = _FCOMGetWord @LanguageSeq, 1677 , N'알림'

SELECT @Subjects = '[ERP] 요청번호 '+@ReqNo+' 알림 드립니다.'

/**********************************************************************************

--3. 최종결과

**********************************************************************************/

SELECT @Subjects = ISNULL(@Subjects , N'')

, @TemplateContent = ISNULL(@TemplateContent , N'')

, @RecvNames = ISNULL(@RecvNames , N'')

, @RecvValues = ISNULL(@RecvValues , N'')

, @MbFlexAppNoti_FlexAppID = ISNULL(MbFlexAppNoti_FlexAppID, N'')

FROM OTCAKumsTemplatePgm

WHERE CompanySeq = @CompanySeq

AND PgmSeq = 527040

AND TemplateSeq = 2

-- 수신자 지정

--SELECT @RecvNames = '홍길동'

--SELECT @RecvValues = '146'

--SELECT * FROM _TCAUser

IF @Subjects \<> N'' AND @TemplateContent \<> N'' AND @RecvNames \<> N'' AND @RecvValues \<> N'' AND @MbFlexAppNoti_FlexAppID \<> N''

BEGIN

SELECT @Subjects AS Subjects -- 알림 제목 : 공지 알림

, @TemplateContent AS TemplateContent -- 알림 내용 : 새로운 공지가 등록되었습니다. '(#공지사항제목)' 지금 확인하세요.

, @RecvNames AS RecvNames -- 수신자 이름 : UserName

, @RecvValues AS RecvValues -- 수신자 SEQ : UserSeq

, N'' AS RecvDeptNames -- 수신자 부서 : kums알림 보낼때만 있으면 됨

, @MbFlexAppNoti_FlexAppID AS FlexAppID

END

SELECT @Subjects AS Subjects -- 알림 제목 : [증명서] #증명서구분 (승인/반려)

, @TemplateContent AS TemplateContent -- 알림 내용 : 신청하신 (#증명서구분)이(가) (#알림종류)되었습니다.

, @RecvNames AS RecvNames -- 수신자 이름 : UserName

, @RecvValues AS RecvValues -- 수신자 SEQ : UserSeq

, N'' AS RecvDeptNames -- 수신자 부서 : kums알림 보낼때만 있으면 됨

, @FlexAppID AS FlexAppID -- 수신FlexApp

RETURN

GO
