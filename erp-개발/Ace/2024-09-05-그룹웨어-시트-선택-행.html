AS

DECLARE @TmpSeq NVARCHAR(100),

@CustSeq INT,

@CustBankAccSerl INT,

@DocHandle INT

EXEC sp_xml_preparedocument @DocHandle OUTPUT, @XmlDocument;

--SELECT @TmpSeq = CustSeq

-- FROM OPENXML(@docHandle, N'/ROOT/DataBlock1', @xmlFlags)

FROM OPENXML(@docHandle, N'/ROOT/DataBlock1', @xmlFlags)

WITH (CustSeq INT)

--IF (ISNULL(@TmpSeq, '') \<> '')

--END

DECLARE @AppEmpName NVARCHAR(100)

SELECT @AppEmpName = A.UserName

FROM _TCAUser AS A

WHERE A.CompanySeq = @CompanySeq

AND A.UserSeq = @UserSeq

EXEC _SYMMETRICKeyOpen

DECLARE @EnvValue NVARCHAR(1000), -- 환경설정(사업자번호형식)

@EnvValue2 NVARCHAR(1000) -- 환경설정(주민등록번호형식)

-- (INPUT) 사용자정의코드 대분류

CREATE TABLE #TDACustBankAcc(

CompanySeq INT

, CustBankAccSerl INT

, CustSeq INT

, BankAccNo NVARCHAR(400)

)

INSERT INTO #TDACustBankAcc(CompanySeq, CustBankAccSerl, CustSeq, BankAccNo)

SELECT DISTINCT A.CompanySeq, A.CustBankAccSerl, A.CustSeq, dbo._FWCOMDecrypt(A.BankAccNo, N'_TDACustBankAcc', N'BankAccNo', @CompanySeq)

FROM _TDACustBankAcc AS A WITH(NOLOCK)

JOIN #Tmp_TDACustBankAcc AS B WITH(NOLOCK) ON B.CustSeq = A.CustSeq

LEFT OUTER JOIN _TDACustKind AS D WITH(NOLOCK) ON D.CompanySeq = A.CompanySeq

AND D.CustSeq = A.CustSeq -- 거래처종류

WHERE A.CompanySeq = @CompanySeq

--AND A.CustSeq = @CustSeq

--AND A.CustBankAccSerl = @CustBankAccSerl

-- 최종 SELECT

SELECT CONVERT(NCHAR(8),GETDATE(),112) AS AppDate -- 신청일

,@AppEmpName AS AppEmpName -- 신청자

,C.FullName AS FullName -- 상호 (거래처등록 상호명)

,C.CustName AS CustName -- 거래처명

,CASE WHEN ISNULL(C.BizNo,'') = '' THEN '0' ELSE '1' END AS IsBizNo -- 사업자여부 (사업자번호 없는 경우 X)

,ISNULL(B.MinorName, '') AS SMBizPersName -- 사업자종류 (법인/개인 )

,CASE WHEN LEFT( dbo._FDAGetCustKindNames(A.CompanySeq,A.CustSeq),1) = N''

THEN SUBSTRING(dbo._FDAGetCustKindNames(A.CompanySeq,A.CustSeq),2,Len(dbo._FDAGetCustKindNames(A.CompanySeq,A.CustSeq)))

ELSE dbo._FDAGetCustKindNames(A.CompanySeq,A.CustSeq) END AS UMCustKindName -- 거래처종류

,ISNULL(C.Owner, '') AS CustOwner -- 대표자명

,CASE WHEN ISNULL(C.BizNo, N'') = N'' AND ISNULL(C.PersonId, N'') \<> N''

THEN ISNULL(dbo._fnResidMask(dbo._FCOMMaskConv(@EnvValue2, dbo._FWCOMDecrypt(C.PersonId, N'_TDACust', N'PersonId', @CompanySeq))), N'')

ELSE ISNULL(dbo._FCOMMaskConv(@EnvValue, REPLACE(C.BizNo, '-', '')), '') END AS BizNo -- 사업자번호

,dbo._FWCOMDecrypt(C.PersonId, '_TDACust', 'PersonId', @CompanySeq) AS PersonId -- 주민번호

,ISNULL(C.BizType , '') AS BizType -- 업태

,ISNULL(C.BizKind , '') AS BizKind -- 업종

,ISNULL(L.Addr , '') AS BizAddr -- 주소

,ISNULL(G.EmpName , '') AS EmpName -- 영업담당자

,ISNULL(I.MinorName , '') AS UMReceiptCond -- 회수조건 (거래처 추가정보 - 회수조건)

,ISNULL(B.MinorName , '') AS UMBankHQName -- 금융기관

,ISNULL(A.Owner , '') AS Owner -- 예금주

,ISNULL(X.BankAccNo , '') AS BankAccNo -- 계좌번호

,ISNULL(A.IsDefault ,'0') AS IsDefault -- 주계좌여부(기본계좌여부)

,ISNULL(D.SubAccNoPurPose, '') AS SubAccNoPurPose -- 보조계좌등록목적 24.05.20 SJPARK

,ISNULL(A.Remark , '') AS Remark -- 비고

--,A.CustSeq, -- 거래처내부코드

-- C.CustNo, -- 거래처번호 컬럼 추가 by.sykim.

-- A.CustBankAccSerl AS CustBankAccSerl, -- 계좌순번

-- ISNULL(dbo._FCOMMaskConv(@EnvValue, REPLACE(C.BizNo, '-', '')), '') AS TaxNo, -- 사업자번호 -- 추후 삭제예정

-- A.UMBankHQ, -- 금융기관내부코드

-- A.IsCfm, -- 확정여부

-- A.IsNotUse, -- 미사용여부

-- CONVERT(NCHAR(8), A.LastDateTime, 112) AS LastDate, -- 수정일

-- A.CustSeq AS CustSeqOld, -- 거래처내부코드Old

-- A.CustBankAccSerl AS CustBankAccSerlOld, -- 계좌순번Old

-- X.BankAccNo AS BankAccNoOld,

-- ISNULL(A.CMSCode , N'') AS CMSCode , -- CMS코드

-- ISNULL(A.LastUserSeq, 0) AS LastUserSeq,

-- ISNULL(R.UserName , N'') AS LastUserName,

-- A.LastDateTime AS LastDateTime

FROM #TDACustBankAcc AS X

JOIN _TDACustBankAcc AS A WITH(NOLOCK) ON A.CompanySeq = X.CompanySeq AND A.CustBankAccSerl = X.CustBankAccSerl

JOIN _TDACust AS C WITH(NOLOCK) ON C.CompanySeq = X.CompanySeq AND C.CustSeq = X.CustSeq

LEFT OUTER JOIN _TCAUser AS R WITH(NOLOCK) ON R.UserSeq = A.LastUserSeq

LEFT OUTER JOIN #TDAUMinorLanguage AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.MinorSeq = A.UMBankHQ

AND B.MajorSeq = 4003 -- 금융기관

LEFT OUTER JOIN emk_TDACustBankAcc AS D WITH(NOLOCK) ON D.CompanySeq = A.CompanySeq AND D.CustBankAccSerl = A.CustBankAccSerl

LEFT OUTER JOIN _TDASMinor AS E WITH(NOLOCK) ON E.CompanySeq = C.CompanySeq

AND E.MinorSeq = C.SMBizPers

LEFT OUTER JOIN _VWCOMAddress AS L ON C.CompanySeq = L.CompanySeq

AND C.BizAddrSeq = L.AddressSeq

AND L.IsMin = '1'

LEFT OUTER JOIN _TSLCustSalesEmp AS F WITH(NOLOCK) ON A.CompanySeq = F.CompanySeq

AND A.CustSeq = F.CustSeq

AND CONVERT(NCHAR(8), A.LastDateTime, 112) BETWEEN F.SDate AND F.EDate

LEFT OUTER JOIN _TDAEmp AS G WITH(NOLOCK) ON G.CompanySeq = F.CompanySeq

AND G.EmpSeq = F.EmpSeq

LEFT OUTER JOIN _TDACustUserDefine AS H WITH(NOLOCK) ON H.CompanySeq = C.CompanySeq

AND H.CustSeq = C.CustSeq

AND H.MngSerl = 1000030

LEFT OUTER JOIN _TDAUMinor AS I WITH(NOLOCK) ON I.CompanySeq = H.CompanySeq

AND I.MinorSeq = H.MngValSeq

ORDER BY C.CustName, A.IsDefault DESC, A.IsNotUse

RETURN
