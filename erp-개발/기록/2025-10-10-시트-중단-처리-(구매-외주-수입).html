AND (j.StopSubSerl = 0 OR i.StopSubSerl = j.StopSubSerl)

GROUP BY j.StopSeq, j.StopSerl, j.StopSubSerl) B ON A.StopSeq = B.StopSeq

AND A.StopSerl = B.StopSerl

AND A.StopSubSerl = B.StopSubSerl

UPDATE #TCOMStop

SET Result = [Message]

FROM #TCOMStop A

JOIN _TCAMessageLanguage B ON A.Status = B.MessageSeq

WHERE LanguageSeq = @LanguageSeq AND STATUS \<> 0

UPDATE #BIZ_OUT_DATABLOCK1

SET Result = B.Result

, MessageType = B.MessageType

, Status = B.Status

FROM #BIZ_OUT_DATABLOCK1 AS A

JOIN #TCOMStop AS B ON A.IDX_NO = B.IDX_NO

WHERE B.Status \<> 0

UPDATE #BIZ_OUT_DATABLOCK1

SET StopDate = @StopDate

,StopEmp = (SELECT EmpName FROM _TDAEmp WITH(NOLOCK) WHERE CompanySeq = @CompanySeq AND EmpSeq = @EmpSeq)

FROM #BIZ_OUT_DATABLOCK1 AS A

JOIN #TCOMStop AS B ON A.IDX_NO = B.IDX_NO

WHERE A.Status = 0

AND B.Status = 0
