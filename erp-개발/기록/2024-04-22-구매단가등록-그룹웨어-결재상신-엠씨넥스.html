구매단가등록 그룹웨어 결재상신 엠씨넥스

SELECT ItemSeq,

DUMMY1 as 'PriceKey',

DUMMY2 as 'GroupKey',

DUMMY3 as '결재상신여부',

DUMMY4 as '결재완료여부',

DUMMY5 as '단가상태'

,C.UserName

,B.LastDateTime

FROM _TPUBASEBuyPriceItem AS A

LEFT OUTER JOIN _TCOMGroupWare AS B on a.Dummy1 = b.TblKey

and b.WorkKind = 'PurPrice'

LEFT OUTER JOIN _TCAUser AS C on C.CompanySeq = B.CompanySeq

and C.UserSeq = B.LastUserSeq

WHERE A.CustSeq = 1528

AND A.Dummy3 = '1'

ORDER BY 1

SELECT LastDateTime,DUMMY5,* FROM _TPUBASEBuyPriceItem WHERE ISNULL(DUMMY5,'') \<> '' ORDER BY 1

SELECT LastDateTime,DUMMY5,* FROM _TPUBASEBuyPriceItem WHERE ISNULL(DUMMY5,'') = '' ORDER BY 1 DESC

SELECT *

FROM _TCOMGroupWare AS B

WHERE b.WorkKind = 'PurPrice' AND B.Tblkey = '36474_1528_1_2_8197'

SELECT *

FROM _TCOMGroupWareLog AS B

WHERE b.WorkKind = 'PurPrice' AND B.Tblkey = '36474_1528_1_2_8197'
