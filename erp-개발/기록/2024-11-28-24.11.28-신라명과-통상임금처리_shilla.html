SELECT A.EmpSeq ,

C.EmpName ,

C.EmpID ,

C.DeptName ,

--SUM(D.Amt) AS Amt

D.Amt

FROM _TPRPayResult AS A WITH(NOLOCK)

LEFT OUTER JOIN _TPRBasPb AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.PbSeq = B.PbSeq

LEFT OUTER JOIN _VXHREmpInLast AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.EmpSeq = C.EmpSeq

LEFT OUTER JOIN _TPRPayCalc AS D WITH(NOLOCK) ON A.CompanySeq = D.CompanySeq -- 계산항목

AND A.PbYM = D.PBYm

AND A.SerialNo = D.SerialNo

AND A.EmpSeq = D.EmpSeq

LEFT OUTER JOIN _TXPRBasPayItemByPt AS E WITH(NOLOCK) ON D.CompanySeq = E.CompanySeq

AND D.ItemSeq = E.ItemSeq

WHERE A.CompanySeq = 1

-- AND B.SMPbType = 3001001 -- 231004 HCWOO 급여제외 -> 급여포함 -> 급여만

AND A.PuSeq = 1

AND A.PbYm = '202410'

--AND E.IsGen = '1'

AND A.EmpSeq = 119

SELECT A.EmpSeq ,

SUM(D.Amt) AS Amt

FROM _TPRPayResult AS A WITH(NOLOCK)

LEFT OUTER JOIN _TPRBasPb AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.PbSeq = B.PbSeq

LEFT OUTER JOIN _TPRPayCalc AS D WITH(NOLOCK) ON A.CompanySeq = D.CompanySeq -- 계산항목

AND A.PbYM = D.PBYm

AND A.SerialNo = D.SerialNo

AND A.EmpSeq = D.EmpSeq

LEFT OUTER JOIN _TXPRBasPayItemByPt AS E WITH(NOLOCK) ON D.CompanySeq = E.CompanySeq

AND D.ItemSeq = E.ItemSeq

WHERE A.CompanySeq = 1

-- AND B.SMPbType = 3001001 -- 231004 HCWOO 급여제외 -> 급여포함 -> 급여만

AND A.PuSeq = 1

AND A.PbYm = '202410'

--AND E.IsGen = '1'

AND A.EmpSeq = 119

GROUP BY A.EmpSeq
