CREATE PROC sys_SPgmLinkListQuery

@ServiceSeq INT

, @WorkingTag NVARCHAR(10)

, @CompanySeq INT

, @LanguageSeq INT

, @UserSeq INT

, @PgmSeq INT

, @IsTransaction BIT

AS

DECLARE @OriPgmSeq INT,

@OriServiceSeq INT,

@OriSqlScriptSeq INT

SELECT @OriPgmSeq = ISNULL(A.OriPgmSeq , 0),

@OriServiceSeq = ISNULL(A.OriServiceSeq , 0),

@OriSqlScriptSeq = ISNULL(A.OriSqlScriptSeq, 0)

FROM #BIZ_IN_DataBlock1 AS A

IF (@OriPgmSeq + @OriServiceSeq + @OriSqlScriptSeq) = 0

BEGIN

RETURN

END

CREATE TABLE #BIZ_OUT_DataBlock

(

Sort1 INT, Sort2 INT,

PgmID NVARCHAR(100), PgmName NVARCHAR(100), PgmSeq INT,

PgmMethodID NVARCHAR(100), PgmMethodSeq INT,

BizServiceID NVARCHAR(100), BizServiceName NVARCHAR(100), BizServiceSeq INT,

BizMethodID NVARCHAR(100), BizMethodName NVARCHAR(100), BizMethodSeq INT,

DatServiceID NVARCHAR(100), DatServiceName NVARCHAR(100), DatServiceSeq INT,

DatMethodID NVARCHAR(100), DatMethodName NVARCHAR(100), DatMethodSeq INT,

SqlScriptID NVARCHAR(100), SqlScriptName NVARCHAR(100), SqlScriptSeq INT

)

INSERT INTO #BIZ_OUT_DataBlock

(

Sort1, Sort2,

PgmID, PgmName, PgmSeq, PgmMethodID, PgmMethodSeq,

BizServiceID, BizServiceName, BizServiceSeq, BizMethodID, BizMethodName, BizMethodSeq,

DatServiceID, DatServiceName, DatServiceSeq, DatMethodID, DatMethodName, DatMethodSeq,

SqlScriptID, SqlScriptName, SqlScriptSeq

)

SELECT DISTINCT

C.ExecuteSeq, F.OrderSeq,

A.PgmID, A.Caption PgmName, A.PgmSeq, B.PgmMethodName AS PgmMethodID, B.PgmMethodSeq,

D.ServiceID BizServiceID, D.ServiceName BizServiceName, D.ServiceSeq BizServiceSeq, E.MethodID BizMethodID, E.MethodName BizMethodName, E.MethodSeq BizMethodSeq,

G.ServiceID DatServiceID, G.ServiceName DatServiceName, G.ServiceSeq DatServiceSeq, H.MethodID DatMethodID, H.MethodName DatMethodName, H.MethodSeq DatMethodSeq,

J.SqlScriptID SqlScriptID, J.SqlScriptName SqlScriptName, J.SqlScriptSeq SqlScriptSeq

FROM SYSDEVCommon.._TCAPgm AS A

JOIN SYSDEVCommon.._TCAPgmMethod AS B ON A.CompanySeq = B.CompanySeq AND A.PgmSeq = B.PgmSeq

JOIN SYSDEVCommon.._TCAPgmMethodItem AS C ON B.CompanySeq = C.CompanySeq AND B.PgmSeq = C.PgmSeq AND B.PgmMethodSeq = C.PgmMethodSeq AND C.ServiceSeq \<> 0 AND ISNULL(C.ExecuteYN, '') = '1'

JOIN SYSDEVCommon.._TCAService AS D ON C.CompanySeq = D.CompanySeq AND C.ServiceSeq = D.ServiceSeq AND D.ServiceLayer = 130002

JOIN SYSDEVCommon.._TCAServiceMethod AS E ON C.CompanySeq = E.CompanySeq AND C.ServiceSeq = E.ServiceSeq AND C.MethodSeq = E.MethodSeq AND ISNULL(E.WFType, 0) \<> 228002 -- SQL_WF

JOIN SYSDEVCommon.._TCAServiceMethodKWF AS F ON E.CompanySeq = F.CompanySeq AND E.ServiceSeq = F.ServiceSeq AND E.MethodSeq = F.MethodSeq AND F.ProcessServiceSeq \<> 0 AND ISNULL(F.NotUse, '') \<> '1'

JOIN SYSDEVCommon.._TCAService AS G ON F.CompanySeq = G.CompanySeq AND F.ProcessServiceSeq = G.ServiceSeq

JOIN SYSDEVCommon.._TCAServiceMethod AS H ON F.CompanySeq = H.CompanySeq AND F.ProcessServiceSeq = H.ServiceSeq AND F.ProcessMethodSeq = H.MethodSeq

JOIN SYSDEVCommon.._TCAServiceMethodScript AS I ON H.CompanySeq = I.CompanySeq AND H.ServiceSeq = I.ServiceSeq AND H.MethodSeq = I.MethodSeq AND ISNULL(I.SqlScriptSeq, 0) \<> 0

JOIN SYSDEVCommon.._TCASQLScripts AS J ON I.CompanySeq = J.CompanySeq AND I.SqlScriptSeq = J.SqlScriptSeq

WHERE A.CompanySeq = 0

AND (@OriPgmSeq = 0 OR A.PgmSeq = @OriPgmSeq)

AND (@OriServiceSeq = 0 OR (C.ServiceSeq = @OriServiceSeq) OR (G.ServiceSeq = @OriServiceSeq))

AND (@OriSqlScriptSeq = 0 OR J.SqlScriptSeq = @OriSqlScriptSeq)

INSERT INTO #BIZ_OUT_DataBlock

(

Sort1, Sort2,

PgmID, PgmName, PgmSeq, PgmMethodID, PgmMethodSeq,

BizServiceID, BizServiceName, BizServiceSeq, BizMethodID, BizMethodName, BizMethodSeq,

DatServiceID, DatServiceName, DatServiceSeq, DatMethodID, DatMethodName, DatMethodSeq,

SqlScriptID, SqlScriptName, SqlScriptSeq

)

SELECT DISTINCT

C.ExecuteSeq, 0 OrderSeq,

A.PgmID, A.Caption PgmName, A.PgmSeq, B.PgmMethodName AS PgmMethodID, B.PgmMethodSeq,

D.ServiceID BizServiceID, D.ServiceName BizServiceName, D.ServiceSeq BizServiceSeq, E.MethodID BizMethodID, E.MethodName BizMethodName, E.MethodSeq BizMethodSeq,

'' DatServiceID, '' DatServiceName, 0 DatServiceSeq, '' DatMethodID, '' DatMethodName, 0 DatMethodSeq,

J.SqlScriptID SqlScriptID, J.SqlScriptName SqlScriptName, J.SqlScriptSeq SqlScriptSeq

FROM SYSDEVCommon.._TCAPgm AS A

JOIN SYSDEVCommon.._TCAPgmMethod AS B ON A.CompanySeq = B.CompanySeq AND A.PgmSeq = B.PgmSeq

JOIN SYSDEVCommon.._TCAPgmMethodItem AS C ON B.CompanySeq = C.CompanySeq AND B.PgmSeq = C.PgmSeq AND B.PgmMethodSeq = C.PgmMethodSeq AND C.ServiceSeq \<> 0 AND ISNULL(C.ExecuteYN, '') = '1'

JOIN SYSDEVCommon.._TCAService AS D ON C.CompanySeq = D.CompanySeq AND C.ServiceSeq = D.ServiceSeq AND D.ServiceLayer = 130001

JOIN SYSDEVCommon.._TCAServiceMethod AS E ON C.CompanySeq = E.CompanySeq AND C.ServiceSeq = E.ServiceSeq AND C.MethodSeq = E.MethodSeq AND ISNULL(E.WFType, 0) \<> 228002 -- SQL_WF

--JOIN SYSDEVCommon.._TCAServiceMethodKWF AS F ON E.CompanySeq = F.CompanySeq AND E.ServiceSeq = F.ServiceSeq AND E.MethodSeq = F.MethodSeq AND F.ProcessServiceSeq \<> 0

--JOIN SYSDEVCommon.._TCAService AS G ON F.CompanySeq = G.CompanySeq AND F.ProcessServiceSeq = G.ServiceSeq

--JOIN SYSDEVCommon.._TCAServiceMethod AS H ON F.CompanySeq = H.CompanySeq AND F.ServiceSeq = H.ServiceSeq AND F.ProcessMethodSeq = H.MethodSeq

JOIN SYSDEVCommon.._TCAServiceMethodScript AS I ON E.CompanySeq = I.CompanySeq AND E.ServiceSeq = I.ServiceSeq AND E.MethodSeq = I.MethodSeq AND ISNULL(I.SqlScriptSeq, 0) \<> 0

JOIN SYSDEVCommon.._TCASQLScripts AS J ON I.CompanySeq = J.CompanySeq AND I.SqlScriptSeq = J.SqlScriptSeq

WHERE A.CompanySeq = 0

AND (@OriPgmSeq = 0 OR A.PgmSeq = @OriPgmSeq)

AND (@OriServiceSeq = 0 OR C.ServiceSeq = @OriServiceSeq)

AND (@OriSqlScriptSeq = 0 OR J.SqlScriptSeq = @OriSqlScriptSeq)

INSERT INTO #BIZ_OUT_DataBlock

(

Sort1, Sort2,

PgmID, PgmName, PgmSeq, PgmMethodID, PgmMethodSeq,

BizServiceID, BizServiceName, BizServiceSeq, BizMethodID, BizMethodName, BizMethodSeq,

DatServiceID, DatServiceName, DatServiceSeq, DatMethodID, DatMethodName, DatMethodSeq,

SqlScriptID, SqlScriptName, SqlScriptSeq

)

SELECT DISTINCT

C.ExecuteSeq, F.OrderSeq,

A.PgmID, A.Caption PgmName, A.PgmSeq, B.PgmMethodName AS PgmMethodID, B.PgmMethodSeq,

D.ServiceID BizServiceID, D.ServiceName BizServiceName, D.ServiceSeq BizServiceSeq, E.MethodID BizMethodID, E.MethodName BizMethodName, E.MethodSeq BizMethodSeq,

'' DatServiceID, '' DatServiceName, 0 DatServiceSeq, '' DatMethodID, '' DatMethodName, 0 DatMethodSeq,

J.SqlScriptID SqlScriptID, J.SqlScriptName SqlScriptName, J.SqlScriptSeq SqlScriptSeq

FROM SYSDEVCommon.._TCAPgm AS A

JOIN SYSDEVCommon.._TCAPgmMethod AS B ON A.CompanySeq = B.CompanySeq AND A.PgmSeq = B.PgmSeq

JOIN SYSDEVCommon.._TCAPgmMethodItem AS C ON B.CompanySeq = C.CompanySeq AND B.PgmSeq = C.PgmSeq AND B.PgmMethodSeq = C.PgmMethodSeq AND C.ServiceSeq \<> 0 AND ISNULL(C.ExecuteYN, '') = '1'

JOIN SYSDEVCommon.._TCAService AS D ON C.CompanySeq = D.CompanySeq AND C.ServiceSeq = D.ServiceSeq AND D.ServiceLayer = 130002

JOIN SYSDEVCommon.._TCAServiceMethod AS E ON C.CompanySeq = E.CompanySeq AND C.ServiceSeq = E.ServiceSeq AND C.MethodSeq = E.MethodSeq AND ISNULL(E.WFType, 0) = 228002 -- SQL_WF

JOIN SYSDEVCommon.._TCAServiceMethodKWF AS F ON E.CompanySeq = F.CompanySeq AND E.ServiceSeq = F.ServiceSeq AND E.MethodSeq = F.MethodSeq AND F.SqlScriptSeq \<> 0 AND ISNULL(F.NotUse, '') \<> '1'

--JOIN SYSDEVCommon.._TCAService AS G ON F.CompanySeq = G.CompanySeq AND F.ProcessServiceSeq = G.ServiceSeq

--JOIN SYSDEVCommon.._TCAServiceMethod AS H ON F.CompanySeq = H.CompanySeq AND F.ServiceSeq = H.ServiceSeq AND F.ProcessMethodSeq = H.MethodSeq

--JOIN SYSDEVCommon.._TCAServiceMethodScript AS I ON H.CompanySeq = I.CompanySeq AND H.ServiceSeq = I.ServiceSeq AND H.MethodSeq = I.MethodSeq

JOIN SYSDEVCommon.._TCASQLScripts AS J ON F.CompanySeq = J.CompanySeq AND F.SqlScriptSeq = J.SqlScriptSeq

WHERE A.CompanySeq = 0

AND (@OriPgmSeq = 0 OR A.PgmSeq = @OriPgmSeq)

AND (@OriServiceSeq = 0 OR C.ServiceSeq = @OriServiceSeq)

AND (@OriSqlScriptSeq = 0 OR J.SqlScriptSeq = @OriSqlScriptSeq)

SELECT A.*

FROM #BIZ_OUT_DataBlock AS A

ORDER BY PgmID, PgmName, PgmSeq, PgmMethodID, PgmMethodSeq, Sort1,

BizServiceID, BizServiceName, BizServiceSeq, BizMethodID, BizMethodName, BizMethodSeq, Sort2,

DatServiceID, DatServiceName, DatServiceSeq, DatMethodID, DatMethodName, DatMethodSeq,

SqlScriptID, SqlScriptName, SqlScriptSeq

RETURN

GO
