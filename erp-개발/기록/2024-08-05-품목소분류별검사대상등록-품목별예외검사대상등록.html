FROM _TCOMEnv AS A WITH(NOLOCK)

LEFT OUTER JOIN _TDAWH AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.EnvValue = B.WHSeq

WHERE A.CompanySeq = 1

AND A.EnvSeq = 2000062 -- 검사품입고창고_biont

SELECT A.ItemSeq,

C.IsInQC,

C.IsAfterQCPAM

FROM _TDAItem AS A

JOIN _TDAItemClass AS B WITH(NOLOCK) ON B.ItemSeq = A.ItemSeq

AND B.UMajorItemClass IN (2001, 2004)

LEFT OUTER JOIN _TPDQAItemClassQCKind AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.UMItemClass = B.UMItemClass

WHERE A.ItemSeq = 25359

--SELECT * FROM _TPDQAItemClassQCKind WHERE UMItemClass = 2001035

SELECT A.ItemSeq,

CASE WHEN B.IsUse = '1' THEN '1' ELSE '0' END AS IsInQC,

CASE WHEN C.IsUse = '1' THEN '1' ELSE '0' END AS IsAfterQCPAM

FROM _TDAItem AS A

LEFT OUTER JOIN _TPDQAItemQCKind AS B WITH(NOLOCK) ON B.CompanySeq = 1

AND B.ItemSeq = A.ItemSeq

AND B.SMQcKind = 6018002

LEFT OUTER JOIN _TPDQAItemQCKind AS C WITH(NOLOCK) ON C.CompanySeq = 1

AND C.ItemSeq = A.ItemSeq

AND C.SMQcKind = 6018003

WHERE A.ItemSeq = 25359
