CREATE PROCEDURE dbo.sys_SWPDBOMCfmChangeBOMQuery_TTEST

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

--==================================================

-- 변수선언

--==================================================

DECLARE @ItemSeq INT,

@BOMReqSeq INT,

@ItemBomRev NCHAR(2)

--==================================================

-- 변수에 담기

--==================================================

SELECT TOP 1

@ItemSeq = ISNULL(ItemSeqTree, 0)

, @BOMReqSeq = ISNULL(BOMReqSeq , 0)

FROM #BIZ_IN_DATABLOCK1

SELECT @ItemBomRev = A.ItemBOMRevRemark

FROM sys_TPDBOMReqManagement AS A WITH(NOLOCK)

WHERE A.CompanySeq = @CompanySeq

AND A.ItemSeq = @ItemSeq

AND A.BOMReqSeq = @BOMReqSeq

SELECT @ItemBOMRev

SELECT @ItemBOMRev = RIGHT('00' + CONVERT(NVARCHAR, CONVERT(INT, @ItemBOMRev) -1),2)

CREATE TABLE #ReversedBOM

(

CompanySeq INT NOT NULL,

Seq INT NOT NULL,

ParentSeq INT NOT NULL,

ItemSeq INT NOT NULL,

Sort INT NOT NULL,

TreeLevel INT NOT NULL,

ItemRev CHAR(4) NOT NULL,

NeedQtyNumerator DECIMAL(20,10) null,

NeedQtyDenominator DECIMAL(20,10) null,

TextLevel NVARCHAR(60) null

)

INSERT INTO #ReversedBOM

SELECT @CompanySeq, 1, 0, @ItemSeq, 1, 1, @ItemBOMRev, 1, 1, '1'

EXEC _SPDReversBOM @CompanySeq, @ItemSeq, @ItemBOMRev, 1, 1, 0, '1'

DELETE #ReversedBOM FROM #ReversedBOM WHERE NeedQtyDenominator = 0 -- 소요량이 필수이기 때문에 분모가 0인 건은 삭제

SELECT A.Seq AS Seq, A.ParentSeq AS ParentSeq, A.Sort, A.TreeLevel,

ISNULL(A.ItemSeq, 0) AS ItemSeq,

CASE B.SMStatus WHEN 2001002 THEN '◈폐기◈' ELSE '' END +

ISNULL(B.ItemName, '') + '(' + RTRIM(ISNULL(B.ItemNo, '')) + ') [' + LTRIM(RTRIM(ISNULL(A.ItemRev, ''))) + ']' + ' 수량 : ' +

CAST(CASE ISNULL(A.ItemSeq, 0) WHEN @ItemSeq THEN CAST((ISNULL(NeedQtyNumerator, 0) / ISNULL(NeedQtyDenominator, 0)) AS DECIMAL(20, 10))

ELSE CAST((ISNULL(NeedQtyDenominator, 0) / ISNULL(NeedQtyNumerator, 0)) AS DECIMAL(20, 10))

END AS NVARCHAR(100) ) AS NodeName,

ISNULL(NeedQtyNumerator, 0) AS NeedQtyNumerator ,

ISNULL(NeedQtyDenominator, 0) AS NeedQtyDenominator,

CASE D.SMAssetGrp WHEN 6008002 THEN 'ProdItem'

WHEN 6008004 THEN 'SemiItem'

WHEN 6008005 THEN 'AssyItem'

ELSE 'MatItem'

END AS NodeImg, -- 2013.03.28 김권우

CASE ISNULL(A.ItemSeq, 0) WHEN @ItemSeq THEN ISNULL(NeedQtyNumerator, 0) / ISNULL(NeedQtyDenominator, 0)

ELSE ISNULL(NeedQtyDenominator, 0) / ISNULL(NeedQtyNumerator, 0) END AS Qty,

-- 시트 조회 항목 추가 -- 11.12.09 BY 김세호

LTRIM(RTRIM(ISNULL(A.ItemRev, ''))) AS ItemBOMRev,

CASE B.SMStatus WHEN 2001002 THEN '◈폐기◈' ELSE '' END + +B.ItemName AS ItemName,

B.ItemNo AS ItemNo,

B.Spec AS Spec,

C.UnitSeq AS UnitSeq,

C.UnitName AS UnitName,

D.SMAssetGrp AS AssetSeq,

E.MinorName AS AssetName

FROM #ReversedBOM AS A WITH(NOLOCK)

LEFT OUTER JOIN _TDAItem AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.ItemSeq = B.ItemSeq

LEFT OUTER JOIN _TDAUnit AS C WITH(NOLOCK) ON B.CompanySeq = C.CompanySeq

AND B.UnitSeq = C.UnitSeq

LEFT OUTER JOIN _TDAItemAsset AS D WITH(NOLOCK) ON B.CompanySeq = D.CompanySeq

AND B.AssetSeq = D.AssetSeq

LEFT OUTER JOIN _TDASMinor AS E WITH(NOLOCK) ON D.CompanySeq = E.CompanySeq

AND D.SMAssetGrp = E.MinorSeq

ORDER BY A.TextLevel

RETURN

GO
