24.11.11 단암 불출요청건 프로젝트코드 업데이트

SELECT DISTINCT M.OutReqSeq,M.OUTREQNO, R.PJTSEQ,Q.PJTNO

FROM _TPDMMOutReqM AS M

JOIN _TPDMMOutReqItem AS R WITH(NOLOCK) ON M.CompanySeq = R.CompanySeq

AND M.OutReqSeq = R.OutReqSeq

LEFT OUTER JOIN _TPJTProject AS Q WITH(NOLOCK) ON R.CompanySeq = Q.CompanySeq

AND R.PJTSeq = Q.PJTSeq

WHERE M.OUTREQNO IN('202410300025','202410300032','202410310032')

AND R.PJTSEQ = 15362

SELECT * FROM _TPJTProject WHERE PJTNO = 'GA210605AS002' -- 15362

SELECT * FROM _TPJTProject WHERE PJTNO = 'GA210605AS001' -- 15273

BEGIN TRAN

UPDATE _TPDMMOutReqItem

SET PJTSeq = 15273

FROM _TPDMMOutReqItem

WHERE CompanySeq = 1

AND OutReqSeq IN (54085,54092,54130)

AND PJTSeq = 15362

SELECT DISTINCT PJTSeq FROM _TPDMMOutReqItem WHERE OutReqSeq IN (54085,54092,54130)

ROLLBACK TRAN
