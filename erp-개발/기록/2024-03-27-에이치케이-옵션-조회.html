SELECT c.itemseq,

ISNULL(F.ItemClassSSeq , 0) AS UMItemClass

FROM _TDAItem AS C

LEFT OUTER JOIN _TDAItemClass AS E WITH(NOLOCK) ON E.CompanySeq = C.CompanySeq

AND E.ItemSeq = C.ItemSeq

AND E.UMajorItemClass IN (2001, 2004)

LEFT OUTER JOIN _VDAItemClass AS F WITH(NOLOCK) ON F.CompanySeq = E.CompanySeq

AND F.ItemClassSSeq = E.UMItemClass

LEFT OUTER JOIN _TDAItemAsset AS G WITH(NOLOCK) ON G.CompanySeq = C.CompanySeq

AND G.AssetSeq = C.AssetSeq

WHERE ItemNo ='PHQ01516DS00013100A0D3D00000120'
