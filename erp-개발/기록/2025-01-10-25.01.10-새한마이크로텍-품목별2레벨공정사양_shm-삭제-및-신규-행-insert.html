25.01.10 새한마이크로텍 품목별2레벨공정사양_shm 삭제 및 신규 행 insert

BEGIN TRAN

DECLARE @CompanySeq INT = 1

--=============================================

-- 삭제

--=============================================

DELETE A

FROM shm_TPPWorkLevel A WITH (NOLOCK)

JOIN _TDAItem AS B ON A.CompanySeq = B.CompanySeq AND A.ItemSeq = B.ItemSeq

LEFT OUTER JOIN shm_TPPLevelMaster AS C ON C.CompanySeq = A.CompanySeq

AND C.ItemSeq = A.ItemSeq

AND C.LvlMSerl = A.LvlMSerl

WHERE A.CompanySeq = 1

--AND A.ItemSeq = @ItemSeq

--AND A.LvlMSerl \<> 3

AND C.UMWorkTypeSeq = 2100381004

AND HeadNm LIKE '세척 STEP%'

AND (

B.ItemNo LIKE 'LWSA' + '%'

OR B.ItemNo LIKE 'LWMA' + '%'

OR B.ItemNo LIKE 'LRSA' + '%'

OR B.ItemNo LIKE 'LRMA' + '%'

OR B.ItemNo LIKE 'SWSA' + '%'

OR B.ItemNo LIKE 'SWMA' + '%'

OR B.ItemNo LIKE 'SRSA' + '%'

OR B.ItemNo LIKE 'SRMA' + '%'

)

CREATE TABLE #TMP(

IDX_NO INT IDENTITY(1,1)

,HeadNm NVARCHAR(100)

,PlanValue NVARCHAR(100)

,MeasurementValue NVARCHAR(200)

)

INSERT INTO #TMP(HeadNm, PlanValue,MeasurementValue)

SELECT 'Step 1-1','1) 과수10% 45℃ 1분 -> 가성소다 상온 1분(P)/3분(L)' ,'1) 순수 초음파 2분 2번 -> 아세톤 초음파 상온 2분'

UNION ALL SELECT 'Step 1-2','2) 순수 헹굼 상온 10초 -> 순수 초음파 2분 2번' ,'2) 드라이 1분 -> 건조기'

UNION ALL SELECT 'Step 1-3','3) 아세톤 초음파 상온 2분 -> 드라이 1분 -> 건조기' ,'3) 샌딩 공정 진행'

UNION ALL SELECT 'Step 1-4','4) 1日 변색 발현 대기' ,''

UNION ALL SELECT 'Step 2-1','1) 과수10% 45℃ 1분 -> 가성소다 상온 1분' ,'1) 과수10% 45℃ 1분 -> 가성소다 상온 3분'

UNION ALL SELECT 'Step 2-2','2) 순수 헹굼 상온 10초 -> 순수 초음파 2분 2번' ,'2) 순수 헹굼 상온 10초 -> 순수 초음파 2분 2번'

UNION ALL SELECT 'Step 2-3','3) 아세톤 초음파 상온 2분 -> 드라이 1분 -> 건조기' ,'3) 아세톤 초음파 상온 2분 -> 드라이 1분 -> 건조기'

--=============================================

-- WorkOrderSerl 채번하기위한 임시테이블 생성

--=============================================

SELECT A.ItemSeq,

A.LvlMSerl,

MAX(A.WorkOrderSerl) AS WorkOrderSerl

INTO #Result

FROM shm_TPPWorkLevel A WITH (NOLOCK)

JOIN _TDAItem AS B ON A.CompanySeq = B.CompanySeq AND A.ItemSeq = B.ItemSeq

LEFT OUTER JOIN shm_TPPLevelMaster AS C ON C.CompanySeq = A.CompanySeq

AND C.ItemSeq = A.ItemSeq

AND C.LvlMSerl = A.LvlMSerl

WHERE A.CompanySeq = 1

AND C.UMWorkTypeSeq = 2100381004

AND (

B.ItemNo LIKE 'LWSA' + '%'

OR B.ItemNo LIKE 'LWMA' + '%'

OR B.ItemNo LIKE 'LRSA' + '%'

OR B.ItemNo LIKE 'LRMA' + '%'

OR B.ItemNo LIKE 'SWSA' + '%'

OR B.ItemNo LIKE 'SWMA' + '%'

OR B.ItemNo LIKE 'SRSA' + '%'

OR B.ItemNo LIKE 'SRMA' + '%'

)

GROUP BY A.ItemSeq, A.LvlMSerl

INSERT INTO shm_TPPWorkLevel

(

CompanySeq

,ItemSeq

,LvlMSerl

,WorkOrderSerl

,HeadNm

,PlanValue

,LastUserSeq

,LastDateTime

,MeasurementValue

)

SELECT 1 AS CompanySeq

,A.ItemSeq AS ItemSeq

,A.LvlMSerl AS LvlMSerl

,A.WorkOrderSerl + B.IDX_NO AS WorkOrderSerl

,B.HeadNm AS HeadNm

,B.PlanValue AS PlanValue

,1 AS LastUserSeq

,GETDATE() AS LastDateTime

,B.MeasurementValue AS MeasurementValue

FROM #Result AS A

LEFT OUTER JOIN #TMP AS B ON 1=1

ORDER BY A.ItemSeq, WorkOrderSerl

COMMIT TRAN
