JOIN emk_TACMRForm1 AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.Form1Seq = A.Form1Seq

AND B.UMFormulaCalcKind = 1000677002 -- 설정방법 : 계정과목

JOIN _TDAAccount AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.AccSeq = A.FSItemSeq

AND ISNULL(C.IsSlip, '') = '1' -- 기표하는계정만

AND ISNULL(C.IsNotSlip, '') \<> '1' -- 분개전표에서 설정불가 인 것 제외

GROUP BY C.AccSeq

SELECT ISNULL(F.CFSCoName, '') AS CFSCoName, -- 법인명

ISNULL(F.CFSCoNo, '') AS CFSCoNo, -- 법인번호

ISNULL(F.CFSCoUnit, 0) AS CFSCoUnit, -- 법인내부코드(추출여부에 체크된 관계사, 즉 현재법인)

ISNULL(D.CFSCoNo, '') AS CustCFSCoNo, -- 관계사법인번호

ISNULL(A.CashItemSeq, 0) AS CashItemSeq, -- 현금흐름과목내부코드

SUM((ISNULL(A.DrAmt, 0) - ISNULL(A.CrAmt, 0)) * ISNULL(G.SMDrOrCr, 0))

AS Amt, -- 금액

AND C.CFSCoUnit = D.CFSCoUnit

AND C.CustSeq = E.CustSeq

ISNULL(C.CFSCoUnit, 0) ,

ISNULL(H.MinorName, '') ,

ISNULL(G.SMAccKind, 0) ,

ISNULL(G.AccName, '') ,

ISNULL(G.AccNo, '') ,

ISNULL(A.AccSeq, 0) ,

ISNULL(I.CashItemName, ''),

ISNULL(A.CashItemSeq, 0)

,ISNULL(J.FSItemYardiCode,'')

,ISNULL(J.FSItemName,'')

,ISNULL(A.CurrSeq , 0)

,ISNULL(K.CurrName ,'')

HAVING (SUM((ISNULL(A.DrAmt, 0) - ISNULL(A.CrAmt, 0)) * ISNULL(G.SMDrOrCr, 0)) \<> 0 -- 금액이 있는 건만 조회합니다.

OR ISNULL(SUM(A.CrAmt), 0) \<> 0)

--ISNULL(A.RemAmt, 0) AS Amt, -- 금액(BS계정 잔액)

ISNULL(A.IniAmt, 0) AS Amt, -- 금액(BS계정 잔액) 24.10.08

ISNULL(A.DrAmt, 0) AS DrAmt, -- 차변금액

--ISNULL(G.SMAccKind, 0) IN (4018004, 4018005) -- PL계정(계정대분류 비용, 수익)

--OR

(ISNULL(G.SMAccKind, 0) IN (4018001, 4018002, 4018003) AND -- BS계정(계정대분류 자산, 자본, 부채)은 계정과목에 현금흐름과목종류가 설정된 건만 조회합니다.

ISNULL(G.SMCashItemKind, 0) \<> 0) AND

ISNULL(A.CashItemSeq, 0) \<> 0)

AND LEFT(ISNULL(A.AccDate, ''), 6) BETWEEN @AccDateFr AND @AccDateTo

AND (@CFSCoName = '' OR ISNULL(F.CFSCoName, '') LIKE @CFSCoName + '%')

AND (@CustCFSCoName = '' OR ISNULL(D.CFSCoName, '') LIKE @CustCFSCoName + '%')

AND F.CFSCoNo \<> D.CFSCoNo -- 자회사 건 조회안되도록 추가 24.05.23 SJPARK

GROUP BY ISNULL(F.CFSCoName, '') ,

ISNULL(F.CFSCoNo, '') ,

ISNULL(F.CFSCoUnit, 0) ,

ISNULL(D.CFSCoName, '') ,

ISNULL(D.CFSCoNo, '') ,

ISNULL(C.CFSCoUnit, 0) ,

ISNULL(H.MinorName, '') ,

ISNULL(G.SMAccKind, 0) ,

ISNULL(G.AccName, '') ,

ISNULL(G.AccNo, '') ,

ISNULL(A.AccSeq, 0) ,

ISNULL(I.CashItemName, ''),

ISNULL(A.CashItemSeq, 0)

,ISNULL(J.FSItemYardiCode,'')

,ISNULL(J.FSItemName,'')

,ISNULL(A.CurrSeq , 0)

,ISNULL(K.CurrName ,'')

HAVING (SUM((ISNULL(A.DrAmt, 0) - ISNULL(A.CrAmt, 0)) * ISNULL(G.SMDrOrCr, 0)) \<> 0 -- 금액이 있는 건만 조회합니다.

OR ISNULL(SUM(A.DrAmt), 0) \<> 0

OR ISNULL(SUM(A.CrAmt), 0) \<> 0)

UNION ALL

-- BS계정(계정대분류 자산, 자본, 부채)에 대한 잔액 조회

SELECT ISNULL(F.CFSCoName, '') AS CFSCoName, -- 법인명

ISNULL(F.CFSCoNo, '') AS CFSCoNo, -- 법인번호

ISNULL(F.CFSCoUnit, 0) AS CFSCoUnit, -- 법인내부코드(추출여부에 체크된 관계사, 즉 현재법인)

ISNULL(D.CFSCoName, '') AS CustCFSCoName, -- 관계사법인명

ISNULL(D.CFSCoNo, '') AS CustCFSCoNo, -- 관계사법인번호

ISNULL(C.CFSCoUnit, 0) AS CustCFSCoUnit, -- 관계사법인내부코드(계열사의 거래처내부코드)

ISNULL(H.MinorName, '') AS SMAccKindName, -- 계정대분류

ISNULL(G.SMAccKind, 0) AS SMAccKind, -- 계정대분류내부코드

ISNULL(G.AccName, '') AS AccName, -- 계정과목

ISNULL(G.AccNo, '') AS AccNo, -- 계정과목번호

ISNULL(A.AccSeq, 0) AS AccSeq, -- 계정과목내부코드

'' AS CashItemName, -- 현금흐름과목

0 AS CashItemSeq, -- 현금흐름과목내부코드

--ISNULL(A.RemAmt, 0) AS Amt, -- 금액(BS계정 잔액)

ISNULL(A.IniAmt, 0) AS Amt, -- 금액(BS계정 잔액) 24.10.08

ISNULL(A.DrAmt, 0) AS DrAmt, -- 차변금액

ISNULL(A.CrAmt, 0) AS CrAmt -- 대변금액

,ISNULL(J.FSItemYardiCode,'') AS FSItemYardiCode -- 24.05.20 SJPARK

,ISNULL(J.FSItemName, '') AS FSItemName -- Yardi_계정명 -- 2024.05.28 ghwi

,ISNULL(K.CurrName ,'') AS CurrName -- 화폐 추가 24.06.04 SJPARK

,ISNULL(A.CurrSeq , 0) AS CurrSeq -- 화폐 추가 24.06.04 SJPARK

FROM #SlipSum AS A

JOIN _TACCFSCompanyCust AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq -- 연결자료추출환경설정 화면의 우측 거래처 시트

AND A.RemValSeq1 = C.CustSeq

JOIN _TACCFSCompany AS D WITH(NOLOCK) ON C.CompanySeq = D.CompanySeq -- 연결자료추출환경설정 화면

AND C.CFSCoUnit = D.CFSCoUnit

JOIN _TDACust AS E WITH(NOLOCK) ON C.CompanySeq = E.CompanySeq

AND C.CustSeq = E.CustSeq

JOIN _TACCFSCompany AS F WITH(NOLOCK) ON F.CompanySeq = @CompanySeq -- 연결자료추출환경설정 화면

AND F.IsSelf = '1' -- 추출회사여부

LEFT OUTER JOIN _TDAAccount AS G WITH(NOLOCK) ON G.CompanySeq = @CompanySeq

AND A.AccSeq = G.AccSeq

LEFT OUTER JOIN _TDASMinor AS H WITH(NOLOCK) ON G.CompanySeq = H.CompanySeq

AND G.SMAccKind = H.MinorSeq

LEFT OUTER JOIN #TMP_CodeYardi AS J WITH(NOLOCK) ON J.AccSeq = A.AccSeq

LEFT OUTER JOIN _TDACurr AS K WITH(NOLOCK) ON K.CompanySeq = @CompanySeq

AND K.CurrSeq = A.CurrSeq

WHERE ISNULL(G.SMAccKind, 0) IN (4018001, 4018002, 4018003,4018004, 4018005) -- BS계정(계정대분류 자산, 자본, 부채)

AND ISNULL(G.IsCash, '0') \<> '1' -- 현금성여부에 체크된 계정은 조회대상에서 제외합니다.

AND A.RemSeq1 = 1017

AND (@CFSCoName = '' OR ISNULL(F.CFSCoName, '') LIKE @CFSCoName + '%')

AND (@CustCFSCoName = '' OR ISNULL(D.CFSCoName, '') LIKE @CustCFSCoName + '%')

AND F.CFSCoNo \<> D.CFSCoNo -- 자회사 건 조회안되도록 추가 24.05.23 SJPARK

ORDER BY CashItemName, ISNULL(F.CFSCoName, ''), ISNULL(G.AccNo, ''), ISNULL(D.CFSCoNo, '')

RETURN

GO
