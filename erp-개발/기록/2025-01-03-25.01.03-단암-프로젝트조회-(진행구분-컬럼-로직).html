SELECT A.ISComplete,

A.ISStop ,

B.CfmCode,

A.ISComplete,

A.Companyseq, A.PJTSeq, CASE WHEN A.ISComplete = '1' THEN 7027004 -- 완료

WHEN A.ISStop = '1' THEN 7027002 -- 중단

WHEN B.CfmCode = 1 AND ISNULL(A.ISComplete,'') in ( '','0') THEN 7027003 -- 확정 -- select * from _TDASminor where majorseq = 7027

ELSE 7027001 END -- 작성

FROM _TPJTProject AS A

LEFT OUTER JOIN _TPJTProject_Confirm AS B ON A.CompanySeq = B.CompanySeq

AND A.PJTSeq = B.CfmSeq

AND A.PJTRevSeq = B.CfmSerl
