25.09.05 사원 근무유형 조회

SELECT F.EMPNAME ,

ISNULL(K.ValueText ,'0') AS IsWk

, E.FullTime                                 AS FullTime

, E.BegTime                                 AS BegTime

, E.EndTime                                 AS EndTime

, D.WkTypeSeq                         AS WkTypeSeq

FROM _TDAEmpIn AS A WITH(NOLOCK)

LEFT OUTER JOIN _THRAdmOrdEmp AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.EmpSeq = C.EmpSeq

AND C.OrdDate \< A.RetireDate -- 퇴직일자 이전발령기준

AND C.OrdDate \<= CASE WHEN ISNULL(A.RetireDate, '') > CONVERT(NCHAR(8),GETDATE(),112) THEN CONVERT(NCHAR(8),GETDATE(),112) ELSE CONVERT(NCHAR(8), DATEADD(dd, -1, ISNULL(A.RetireDate, '')), 112) END

AND C.OrdEndDate >= CASE WHEN ISNULL(A.RetireDate, '') > CONVERT(NCHAR(8),GETDATE(),112) THEN CONVERT(NCHAR(8),GETDATE(),112) ELSE CONVERT(NCHAR(8), DATEADD(dd, -1, ISNULL(A.RetireDate, '')), 112) END

AND C.IsOrdDateLast = '1'

LEFT OUTER JOIN _TDAUMinor AS J WITH(NOLOCK) ON A.CompanySeq = J.CompanySeq

AND C.UMJoSeq = J.MinorSeq

LEFT OUTER JOIN _TDAUMinorValue AS K WITH(NOLOCK) ON K.CompanySeq = J.CompanySeq

AND K.MinorSeq = J.MinorSeq

AND K.Serl = 1001

LEFT OUTER JOIN _TXPRWkTypeEmp AS D WITH(NOLOCK) ON D.CompanySeq = A.CompanySeq

AND D.EmpSeq = A.EmpSeq

AND D.EndDate = '99991231'

LEFT OUTER JOIN _TXPRWkTypeTimeSet AS E WITH(NOLOCK) ON E.CompanySeq = D.CompanySeq

AND E.WkTypeSeq = D.WkTypeSeq

AND E.SMWkKindSeq= 5037002

AND E.DayTypeSeq = 1

AND E.WkFormSeq = -1

LEFT OUTER JOIN _TDAEmp AS F WITH(NOLOCK) ON F.CompanySeq = A.CompanySeq

AND F.EmpSeq = A.EmpSeq

WHERE A.CompanySeq = 1
