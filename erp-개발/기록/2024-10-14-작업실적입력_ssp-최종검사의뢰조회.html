작업실적입력_ssp / 최종검사의뢰조회

[작업실적입력_ssp] 저장 시 '최종검사여부' (IsLastQc) = '1'

[Routing 생산실적생성_ssp] 저장 시 '최종검사여부' (IsLastQc) = '0' 으로 저장하고 있습니다.

[최종검사의뢰조회] 시 '최종검사여부' (IsLastQc) = '1' 건만 조회하도록 설정되어 있습니다.

해당 건의 경우 [작업실적입력_ssp]에서 생성된 데이터로 확인하였습니다.

SELECT TOP 100 A.IsLastQc,A.WorkDate,B.MinorSeq, B.MinorName ,C.ProdQty,*

FROM _TPDSFCWorkReport AS A

LEFT OUTER JOIN _TPDBaseWorkCenter AS M WITH(NOLOCK) ON A.WorkCenterSeq = M.WorkCenterSeq

AND A.CompanySeq = M.CompanySeq

LEFT OUTER JOIN _TDASMinor AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.MinorSeq = M.SMWorkCenterType

LEFT OUTER JOIN _TPDSFCGoodIn AS C WITH(NOLOCK) ON C.WorkReportSeq = A.WorkReportSeq

AND C.CompanySeq = A.CompanySeq

--WHERE M.SMWorkCenterType = 6011001

--ORDER BY 1 DESC

WHERE A.WorkReportSeq = 63779

SELECT * FROM _TPDSFCWorkReport AS A WHERE A.WorkReportSeq = 63779

SELECT IsLastQc,PgmSeq,* FROM _TPDSFCWorkReport AS A WHERE A.WorkDate ='20240930' AND A.WorkCenterSeq = 33

SELECT TOP 100 A.IsLastQc,A.WorkDate,B.MinorSeq, B.MinorName ,C.ProdQty,*

FROM _TPDSFCWorkReport AS A

LEFT OUTER JOIN _TPDBaseWorkCenter AS M WITH(NOLOCK) ON A.WorkCenterSeq = M.WorkCenterSeq

AND A.CompanySeq = M.CompanySeq

LEFT OUTER JOIN _TDASMinor AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.MinorSeq = M.SMWorkCenterType

LEFT OUTER JOIN _TPDSFCGoodIn AS C WITH(NOLOCK) ON C.WorkReportSeq = A.WorkReportSeq

AND C.CompanySeq = A.CompanySeq

WHERE M.SMWorkCenterType = 6011001
