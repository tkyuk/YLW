-- 일자별생산계획기준 BOM 풀어서 1레벨 자재 소요량 계산

;WITH TNeedQty(ProdDate, GoodItemSeq, Qty, ItemSeq, ItemBOMRev, SubItemSeq, InLossRate, NeedQtyNumerator, NeedQtyDenominator, NeedQty) AS (

SELECT

A.ProdDate, A.ItemSeq, A.Qty, A.ItemSeq, B.SubItemBomRev, B.SubItemSeq, B.InLossRate, B.NeedQtyNumerator, B.NeedQtyDenominator,

CASE WHEN NeedQtyDenominator \<> 0 THEN (B.NeedQtyNumerator / B.NeedQtyDenominator) ELSE 0 END AS NeedQty

FROM #TPDDailyProdPlan AS A

JOIN _TPDBOM AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq AND A.ItemSeq = B.ItemSeq

JOIN #tmpBOMRev AS C ON B.ItemSeq = C.ItemSeq AND B.ItemBomRev = C.ItemBomRev

--JOIN (SELECT ItemSeq, MAX(ItemBomRev) AS ItemBomRev FROM _TPDBOM WHERE CompanySeq = @CompanySeq GROUP BY ItemSeq) AS C ON B.ItemSeq = C.ItemSeq AND B.ItemBomRev = C.ItemBomRev

JOIN _TDAItem AS D WITH(NOLOCK) ON B.CompanySeq = D.CompanySeq AND B.SubItemSeq = D.ItemSeq

JOIN _TDAItemAsset AS E WITH(NOLOCK) ON D.CompanySeq = E.CompanySeq AND D.AssetSeq = E.AssetSeq

JOIN _TDASMinor AS F WITH(NOLOCK) ON E.CompanySeq = F.CompanySeq AND E.SMAssetGrp = F.MinorSeq AND F.MinorValue = 1 -- 자재

WHERE B.SMDelvType \<> 6032003 -- BOM전개 상세 제외

)

SELECT A.ProdDate

,A.SubItemSeq AS ItemSeq,

SUM(CASE WHEN U.SMDecPointSeq = 1003001 THEN ROUND((A.NeedQty) * A.Qty, @nItemPoint*2, 0) -- 반올림 *2를 한 이유는 소요량은 BOM 자리수로 분자/분모이기 때문에

WHEN U.SMDecPointSeq = 1003002 THEN ROUND((A.NeedQty) * A.Qty, @nItemPoint*2, -1) -- 절사

WHEN U.SMDecPointSeq = 1003003 THEN ROUND((A.NeedQty) * A.Qty + CAST(4 AS DECIMAL(19, 5)) / POWER(CAST(10 AS DECIMAL(19, 5)),(@nItemPoint*2 + 1)), @nItemPoint*2) -- 올림

ELSE ROUND((A.NeedQty) * A.Qty + CAST(4 AS DECIMAL(19, 5)) / POWER(CAST(10 AS DECIMAL(19, 5)),(@nItemPoint*2 + 1)), @nItemPoint*2) END)

AS NeedQty

INTO #TNeedQty

FROM TNeedQty AS A

JOIN #TmpCol AS B ON A.SubItemSeq = B.ItemSeq -- 자재

JOIN _TDAItem AS C WITH(NOLOCK) ON B.ItemSeq = C.ItemSeq

JOIN _TDAItemAsset AS D WITH(NOLOCK) ON C.AssetSeq = D.AssetSeq

LEFT OUTER JOIN _TDAUnit AS U WITH(NOLOCK) ON C.CompanySeq = U.CompanySeq AND C.UnitSeq = U.UnitSeq

GROUP BY A.ProdDate, A.SubItemSeq

SELECT * FROM #TmpCol

SELECT * FROM #TNeedQty

RETURN
