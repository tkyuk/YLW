CREATE PROCEDURE JIS_SWPUKumsTemplatePO

@WorkingTag NVARCHAR(10) = '',

@CompanySeq INT = 1 ,

@LanguageSeq INT = 1 ,

@UserSeq INT = 0 ,

@PgmSeq INT = 0 ,

@MessageDiv CHAR(1) = '',

@MessageType VARCHAR(50) = '',

@TemplateCode VARCHAR(30) = '',

@TemplateContent NVARCHAR(1000)='',

@Param1 NVARCHAR(4000) , -- POSeq

@Param2 NVARCHAR(4000) ,

@Param3 NVARCHAR(4000) ,

@Param4 NVARCHAR(4000) ,

, @RecvValues NVARCHAR(1000) = ''

SELECT @TemplateContent = (SELECT TemplateContent

FROM OTCAKumsTemplate WITH(NOLOCK)

WHERE CompanySeq = @CompanySeq

DECLARE @CompanyName NVARCHAR(100), @PgmName NVARCHAR(200), @UserDept NVARCHAR(100), @UserName NVARCHAR(100), @Remark NVARCHAR(1000), @CurrDate NCHAR(8)

DECLARE @PONo NVARCHAR(20), @DeptName NVARCHAR(100),@EmpName NVARCHAR(100), @PODate NCHAR(10), @ItemName NVARCHAR(100), @UnitName NVARCHAR(100), @Qty DECIMAL(19, 5), @CustName NVARCHAR(100)

SELECT @CurrDate = CONVERT(NCHAR(8), GETDATE(), 112)

SELECT @CompanyName = (SELECT CompanyName FROM _TCACompany WITH(NOLOCK) WHERE CompanySeq = @CompanySeq)

SELECT @PgmName = (SELECT Caption FROM _TCAPgm WITH(NOLOCK) WHERE PgmSeq = @PgmSeq)

SELECT @UserDept = (SELECT ISNULL(D.DeptName, ISNULL(H.DeptName, N''))

FROM _TCAUser AS A WITH(NOLOCK)

LEFT OUTER JOIN _TDAEmp AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.EmpSeq = B.EmpSeq

LEFT OUTER JOIN _THRAdmOrdEmp AS C WITH(NOLOCK) ON B.CompanySeq = C.CompanySeq

AND B.EmpSeq = C.EmpSeq -- 발령일내에 있는 사원.

AND C.OrdDate+RIGHT(N'00000'+CONVERT(NVARCHAR(5),C.IntSeq),5) = (SELECT MAX(OrdDate+RIGHT(N'00000'+CONVERT(NVARCHAR(5),IntSeq),5))

FROM _THRAdmOrdEmp

WHERE CompanySeq = @CompanySeq

AND EmpSeq = C.EmpSeq

AND IsOrdDateLast = N'1'

AND OrdDate \<= @CurrDate) -- 발령일자 기준 이전발령

LEFT OUTER JOIN _TDADept AS D WITH(NOLOCK) ON C.CompanySeq = D.CompanySeq

AND C.DeptSeq = D.DeptSeq

LEFT OUTER JOIN _TDADept AS H WITH(NOLOCK) ON B.CompanySeq = H.CompanySeq --제품군이 WBS인 경우는 _TDAEmpOutContract 테이블 사용하지 않음

AND B.WkDeptSeq = H.DeptSeq

WHERE A.CompanySeq = @CompanySeq AND A.UserSeq = @UserSeq)

SELECT @UserName = (SELECT UserName FROM _TCAUser WHERE UserSeq = @UserSeq AND CompanySeq = @CompanySeq )

SELECT @PONo = A.PONo ,

@DeptName = B.DeptName,

@EmpName = C.EmpName,

@PODate = LEFT(A.PODate, 4) + '-' + SUBSTRING(A.PODate, 5, 2) + '-' + RIGHT(A.PODate, 2),

@Remark = A.Remark,

@CustName = ISNULL(D.CUSTNAME,'')

FROM _TPUORDPO AS A WITH(NOLOCK)

JOIN _TDADept AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.DeptSeq = A.DeptSeq

JOIN _TDAEmp AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.EmpSeq = C.EmpSeq

JOIN _TDACust AS D WITH(NOLOCK) ON A.CustSeq = D.CustSeq

AND A.CompanySeq = D.CompanySeq

WHERE A.CompanySeq = @CompanySeq

AND A.POSeq = @Param1

SELECT @Qty = SUM(B.Qty)

FROM _TPUORDPO AS A WITH(NOLOCK)

JOIN _TPUORDPOItem AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.POSeq = A.POSeq

WHERE A.CompanySeq = @CompanySeq

AND A.POSeq = @Param1

SELECT @ItemName = C.ItemName,

@UnitName = D.UnitName

FROM _TPUORDPO AS A WITH(NOLOCK)

JOIN _TPUORDPOItem AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.POSeq = A.POSeq

JOIN _TDAItem AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.ItemSeq = B.ItemSeq

JOIN _TDAUnit AS D WITH(NOLOCK) ON B.UnitSeq = D.UnitSeq

AND B.CompanySeq = D.CompanySeq

WHERE A.CompanySeq = @CompanySeq

AND A.POSeq = @Param1

GROUP BY C.ItemName, D.UnitName, B.Qty

ORDER BY B.Qty

/*****************************************************************************/

DECLARE @CellPhone NVARCHAR(20),

@UMJpName NVARCHAR(20)

SELECT

@CellPhone = dbo._FWCOMDecrypt(B.Cellphone, N'_TDAEmpIn', 'Cellphone', B.CompanySeq),

@UMJpName = C.UMJpName

FROM _TCAUser AS A

JOIN _TDAEmpIn AS B ON A.CompanySeq = B.CompanySeq AND A.EmpSeq = B.EmpSeq

JOIN _fnAdmEmpOrdEmp(@CompanySeq, @CurrDate, 0) AS C ON A.EmpSeq = C.EmpSeq

WHERE A.CompanySeq = @CompanySeq AND A.UserSeq = @UserSeq

IF ISNULL(@CellPhone,'') = '' SELECT @CellPhone = ''

IF ISNULL(@UMJpName, '') = '' SELECT @UMJpName = ''

SELECT @TemplateContent = REPLACE(@TemplateContent, '#{content}',

'안녕하세요.'+ NCHAR(13)

+ @CompanyName +' '+ @UserDept + ' 입니다.'+ NCHAR(13)

+ NCHAR(13)

+ '첨부와 같이 발주서 송부드립니다.'+ NCHAR(13)

+ NCHAR(13)

--+ '\<p style="color:red; font-size:14pt;">\<u>발주서 상 문제(단가, Delivery date(납기요청일) 내에 납품불가 등)가 있다면 메일로 회신요청드리며,\</u>\</p>'

--+ '\<p style="color:red; font-size:14pt;">\<u>발주서 상 문제 없을 시에도 발주확인완료라고 간단히 메일 회신 부탁드립니다.\</u>\</p>'

+ '\<p style="font-size:13pt">발주서 상 문제(단가, Delivery date(납기요청일) 내에 납품불가 등)가 있다면 메일로 회신요청드리며,\</p>'

+ '\<p style="font-size:13pt">발주서 상 문제 없을 시에도 발주확인완료라고 간단히 메일 회신 부탁드립니다.\</p>'

+ NCHAR(13)

+ '1. 모든 발주는 조인셋의 거래기본계약서에 따른다.'+ NCHAR(13)

+ '4.거래사는 납품 시 거래명세서, 검사성적서 필요시 시료를 제출하여야 납품이 완료된다.'+ NCHAR(13)

+ '5. 제조일로부터 자재보증기간이 지났거나, 임박한(보증기간 제조일로부터 6개월이나 납품시점에 보증기간 2달 남은 자재 등) 자재는 납품 불가한다.'+ NCHAR(13)

+ '성적서는 하기 메일주소로 송부 요청드립니다.'+ NCHAR(13)

-- 출력물명 구매발주서_201905220001.pdf

SELECT @ReportName = ISNULL(Word, N'구매발주서') FROM _TCADictionary WHERE WordSeq = 66988 AND LanguageSeq = @LanguageSeq -- 구매발주서

SELECT @DisplayReportName = @ReportName+'_'+@PONo

/**********************수신자에 기본적으로 거래처의 영업담당자 세팅 시작***********************/

EXEC _SYMMETRICKeyOpen

CREATE TABLE #CustContact(IDX_NO INT IDENTITY(1, 1),CustSeq INT, ContactName NVARCHAR(200), ContactNumber NVARCHAR(400))

INSERT INTO #CustContact(CustSeq, ContactName, ContactNumber)

SELECT A.CustSeq,

C.ContactName,

CASE WHEN @MessageDiv IN ('A','S') THEN ISNULL(dbo._FWCOMDecrypt(C.ContactNumber, N'_TDAContact', N'ContactNumber', @CompanySeq),'') --알림톡,SMS인 경우 인맥의 전화번호

ELSE C.EMAIL END -- Mail인 경우 인맥의 메일

FROM _TDACustEmpInfo AS A WITH(NOLOCK)

JOIN _TPUORDPO AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.CustSeq = B.CustSeq

AND B.POSeq = @Param1

LEFT OUTER JOIN _TDAContact AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.ContactSeq = C.ContactSeq

AND C.ContactType = 500035003

WHERE A.CompanySeq = @CompanySeq

AND A.UMJobRollKind = 8024002

SELECT @IDXNO = MAX(IDX_NO) FROM #CustContact

-- 3번 루프돈다 1.성지원 2.박성호 3.권기석

WHILE @CNT \<= @IDXNO

BEGIN

-- RecvNames= 성지원;박성호;권기석; RecvValues= 01012345678;01012345678;12345645678; 형태로 조회되어야 함

SELECT @RecvNames = @RecvNames + CASE WHEN ISNULL(ContactName,'') = '' THEN ' ' ELSE ContactName END+ ';',

@RecvValues = @RecvValues + CASE WHEN ISNULL(ContactNumber,'') = '' THEN ' ' ELSE ContactNumber END+ ';'

FROM #CustContact

WHERE IDX_NO = @CNT

SELECT @CNT = @CNT + 1

END

-- Mail 보내는 사람 세팅

SELECT @SendEmail = B.ValText FROM _TCAUser AS A JOIN _TDAEmpUserDefine AS B ON A.CompanySeq = B.CompanySeq AND A.EmpSeq = B.EmpSeq AND B.Serl = 1000011 WHERE A.CompanySeq = @CompanySeq AND A.UserSeq = @UserSeq

IF ISNULL(@SendEmail,'') = ''

BEGIN

END

@TemplateContent AS TemplateContent,

@RecvNames AS RecvNames,

@RecvValues AS RecvValues,

'' AS DisplayReportName,

@CustName AS CustName

SELECT @TemplateCode AS TemplateCode,

REPLACE(REPLACE(@TemplateContent, CHAR(10), ''), CHAR(13), ' \<br> ') AS TemplateContent,

@RecvNames AS RecvNames,

@RecvValues AS RecvValues,

@DisplayReportName AS DisplayReportName,

@SendEmail AS SendEmail,

@Subject AS Subject,

@CustName AS CustName

END
