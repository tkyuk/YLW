-- 1. 계열사 -> 본사 본사구매요청입력_ssv

SELECT LastDateTime,* FROM SSTTEST.DBO._TPUORDPOReq WHERE POReqSeq = 594 -- 본사구매요청

SELECT LastDateTime,* FROM SSTTEST.DBO._TPUORDPOReqItem WHERE POReqSeq = 594 -- 본사구매요청

-- 2. 본사에서 계열사 요청 확인 - [구매요청조회(통합)_ssv] 화면 계열사 탭에서 1번의 '본사구매요청' 건 조회 가능 => 거래명세서 점프 및 출고 처리

SELECT LastDateTime,* FROM _TPUORDPOReq WHERE POReqSeq = 594 -- 본사구매요청

SELECT LastDateTime,* FROM _TPUORDPOReqItem WHERE POReqSeq = 594 -- 본사구매요청

-- 3. 계열사출고조회_ssv

SELECT A.LastDateTime,*

FROM ssv_TSLInvoiceItemMCO AS A WITH(NOLOCK)

LEFT OUTER JOIN ssv_TSLOrderItem AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.OrderSeq = A.OrderSeq

AND B.OrderSerl = A.OrderSerl

WHERE A.InvoiceSeq = 3547

-- 4. 본사출고조회_ssv (계열사에서 본사 출고 건 조회) -> 조회 후 납품, 입고 생성

SELECT TOP 10 Z1.LastDateTime , *

FROM _TPDMPSProdReq AS Z1 WITH(NOLOCK)

JOIN _TPDMPSProdReqItem AS Z2 WITH(NOLOCK) ON Z2.CompanySeq = Z1.CompanySeq

AND Z2.ProdReqSeq = Z1.ProdReqSeq

JOIN ssv_TPDMPSProdReqItem AS Z3 WITH(NOLOCK) ON Z3.CompanySeq = Z2.CompanySeq

AND Z3.ProdReqSeq = Z2.ProdReqSeq

AND Z3.Serl = Z2.Serl

JOIN ssv_TSLOrderNo AS Z WITH(NOLOCK) ON Z.CompanySeq = Z3.CompanySeq

AND Z.OrderSeq = Z3.OrderSeq

AND Z.OrderSerl = Z3.OrderSerl

ORDER BY 1 DESC

SELECT LastDateTime, * FROM _TSLInvoice WHERE InvoiceSeq = 3547
