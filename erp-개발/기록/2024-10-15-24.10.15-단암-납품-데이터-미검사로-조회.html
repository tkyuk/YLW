BEGIN TRAN

-- 구매입고테이블에 QC관련 컬럼 업뎃

UPDATE _TPUDElvItem

SET SMQCType = C.SMQcType ,

QcEmpSeq = C.QcEmpSeq,

QcDate = C.QcDate,

QcQty = C.QcQty,

BadQty = C.BadQty,

QcCurAmt = C.QcCurAmt

FROM _TPUDElvItem AS A

--JOIN _TPDQCTestReport AS B ON A.DelvSeq = B.SourceSeq

-- --AND A.DelvSerl = B.SourceSerl

-- AND B.SOURCETYPE = 1

Join danam_TPUDElvItem As C ON C.CompanySeq = 1

AND A.DelvSeq = C.DelvSeq

--AND B.SourceSeq = C.DelvSeq

--AND B.SourceSerl = C.DelvSerl

AND A.ItemSeq = C.ItemSeq

And A.Price = C.Price

AND A.LotNo = C.LotNo

AND ISNULL(A.MakerSeq,0) = ISNULL(c.MakerSeq,0)

JOIN _TPUDelvInItem AS D ON D.CompanySeq = A.CompanySeq

AND D.ProgFromSeq = A.DelvSeq

AND D.ProgFromSerl = A.DelvSerl

AND D.ProgFromTableSeq = 10

JOIN _TPUDelv AS Z ON Z.CompanySeq = A.CompanySeq

AND Z.DelvSeq = A.DelvSeq

WHERE A.CompanySeq = 1

AND A.StdUnitQty = D.StdUnitQty AND A.SMQcType = 6035002

AND LEFT(Z.DelvDate,4) = 2024

ROLLBACK TRAN

SELECT A.*

FROM _TPUDelvItem AS A

JOIN _TPUDelv AS Z ON Z.CompanySeq = A.CompanySeq

AND Z.DelvSeq = A.DelvSeq

JOIN _TDASMinor AS B ON B.CompanySeq = A.CompanySeq

AND B.MinorSeq = A.SMQcType

LEFT OUTER JOIN _TPUDelvInItem AS C ON C.CompanySeq = A.CompanySeq

AND C.ProgFromSeq = A.DelvSeq

AND C.ProgFromSerl = A.DelvSerl

AND C.ProgFromTableSeq = 10

--WHERE DelvSeq = 40922

WHERE

A.StdUnitQty = C.StdUnitQty AND SMQcType = 6035002

AND LEFT(DelvDate,4) = 2024

--A.DelvSeq = 41533

ORDER BY A.DelvSeq, A.DelvSerl
