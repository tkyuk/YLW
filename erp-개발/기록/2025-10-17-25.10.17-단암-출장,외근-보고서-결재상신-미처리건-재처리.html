EXEC dbo._SWCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

6 , -- 중복된 @1 @2가(이) 입력되었습니다. (SELECT * FROM _TCAMessageLanguage WHERE MessageSeq = 6)

@LanguageSeq ,

2755, N'결근자내역' -- SELECT * FROM _TCADictionary WHERE Word like '%결근자내역%'

UPDATE #BIZ_OUT_DataBlock3

SET Result = REPLACE(@Results,'@2', ''),

MessageType = @MessageType,

Status = @Status

FROM #BIZ_OUT_DataBlock3 AS A

INNER JOIN (SELECT S.AbsDate, S.EmpSeq , S.WkItemSeq

FROM (

SELECT A1.AbsDate, A1.EmpSeq , A1.WkItemSeq

FROM #BIZ_OUT_DataBlock3 AS A1

WHERE A1.WorkingTag IN ('A','U')

AND A1.Status = 0

UNION ALL

SELECT A1.AbsDate, A1.EmpSeq , A1.WkItemSeq

FROM _TPRWkAbsEmp AS A1

WHERE A1.CompanySeq = @CompanySeq

AND NOT EXISTS ( SELECT 1

FROM #BIZ_OUT_DataBlock3

WHERE WorkingTag ='D' -- ##01 삭제 시에만 체크 제외해야 하는데 'U'일때도 제외하고 있어 SAVE SP에서 중복키 오류나서 수정

AND Status = 0

AND AbsDate = A1.AbsDate

AND EmpSeq = A1.EmpSeq

AND OldWkItemSeq = A1.WkItemSeq

)

) AS S

GROUP BY S.AbsDate, S.EmpSeq, S.WkItemSeq

HAVING COUNT(1) > 1

) AS B

ON A.AbsDate = B.AbsDate

AND A.EmpSeq = B.EmpSeq

AND A.WkItemSeq = B.WkItemSeq

INSERT INTO #Temp_AbsTime(EmpSeq, AbsDAte, WkItemSeq, DTCnt, FullTime)

SELECT A.EmpSeq, A.AbsDate, A.WkItemSeq, CASE WHEN B.SMDTCType = 3068001 THEN A.DtCnt * ISNULL(C.FullTime, 0) ELSE A.DtCnt END, C.FullTime

FROM _TPRWkAbsEmp AS A WITH(NOLOCK)

JOIN _TPRWkItem AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.WkItemSeq = B.WkItemSeq

LEFT OUTER JOIN #Tmp_EmpWkType AS C ON A.EmpSeq = C.EmpSeq

--

UPDATE A

SET A.DtCnt = CASE WHEN D.SMDTCType = 3068001 THEN B.DtCnt * ISNULL(C.FullTime, 0) ELSE A.DtCnt END

FROM #Temp_AbsTime AS A

AND A.AbsDAte = B.AbsDate

AND A.WkItemSeq = B.WkItemSeq

LEFT OUTER JOIN #Tmp_EmpWkType AS C ON A.EmpSeq = C.EmpSeq

AND A.AbsDate = C.WKDate

JOIN _TPRWkItem AS D WITH(NOLOCK) ON D.CompanySeq = @CompanySeq

AND A.WkItemSeq = D.WkItemSeq

WHERE B.Status = 0 AND B.WorkingTag IN ('A', 'U') AND D.SMAbsWkSort IN (3069001, 3069002, 3069003, 3069011) --##02

AND D.SMDTCType IN (3068001, 3068002)

INSERT INTO #Temp_AbsTime(EmpSeq, AbsDAte, WkItemSeq, DTCnt, FullTime)

SELECT A.EmpSeq, A.AbsDate, A.WkItemSeq, CASE WHEN B.SMDTCType = 3068001 THEN A.DtCnt * ISNULL(C.FullTime, 0) ELSE A.DtCnt END, C.FullTime

FROM #BIZ_OUT_DataBlock3 AS A WITH(NOLOCK)

JOIN _TPRWkItem AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND A.WkItemSeq = B.WkItemSeq

LEFT OUTER JOIN #Tmp_EmpWkType AS C ON A.EmpSeq = C.EmpSeq

AND A.AbsDate = C.WKDate

WHERE A.Status = 0 AND A.WorkingTag IN ('A', 'U') AND B.SMAbsWkSort IN (3069001, 3069002, 3069003, 3069011) --##02

AND B.SMDTCType IN (3068001, 3068002)

AND NOT EXISTS (SELECT 1 FROM #Temp_AbsTime WHERE EmpSeq = A.EmpSeq AND AbsDate = A.AbsDate AND WkItemSeq = A.WkItemSeq)

--select '#Temp_AbsTime', * from #Temp_AbsTime

UPDATE A

SET A.Result = @Results,

A.MessageType = @MessageType,

A.Status = @Status

FROM #BIZ_OUT_DataBlock3 AS A

JOIN (

SELECT EmpSeq, AbsDate, SUM(DtCnt) AS DtCnt, MAX(FullTime) AS FullTime

FROM #Temp_AbsTime

GROUP BY EmpSeq, AbsDate --, WkItemSeq --다른 근태항목이더라도 결근을 집계 했을때 기준만근시간보다 초과하면 체크 함 --##02

HAVING SUM(DtCnt) > MAX(FullTime)

) AS B ON A.EmpSeq = B.EmpSeq AND A.AbsDAte = B.AbsDate -- AND A.WkItemSeq = B.WkItemSeq

WHERE A.Status = 0 AND A.WorkingTag IN ('A', 'U')

WHERE A.CompanySeq = @CompanySeq

AND A.BizTripSeq IN(SELECT BizTripSeq FROM #BIZ_OUT_DataBlock3)

SELECT

X_OUTER.BizTripSeq, -- 1. 그룹화 키

STUFF((

-- 2. 해당 BizTripSeq에 속하는 모든 Cost 상세 정보를 문자열로 합치는 서브쿼리

SELECT

', ' +

-- 목적 (R18)

CASE WHEN ISNULL(R18.ValueText, '0') = '0' THEN '' ELSE CASE WHEN B.BizTripType = '2' THEN LEFT(ISNULL(B.EduName, ''), 15) ELSE LEFT(ISNULL(X.Purpose, ''), 10) END END

-- 분류명 (R19)

+ CASE WHEN ISNULL(R19.ValueText, '0') = '0' THEN '' ELSE CASE WHEN ISNULL(R18.ValueText, '0') = '0' THEN '' ELSE ' ' END + ISNULL(K.ValueText, '') END

-- 출장기간 (R20)

+ CASE WHEN ISNULL(R20.ValueText, '0') = '0' THEN '' ELSE CASE WHEN ISNULL(R18.ValueText, '0') = '0' AND ISNULL(R19.ValueText, '0') = '0' THEN '' ELSE ' ' END

+ CASE WHEN X.TripFrDate = X.TripToDate THEN SUBSTRING(X.TripFrDate, 5, 2) + '/' + RIGHT(X.TripFrDate, 2)

ELSE SUBSTRING(X.TripFrDate, 5, 2) + '/' + RIGHT(X.TripFrDate, 2) + '~' + SUBSTRING(X.TripToDate, 5, 2) + '/' + RIGHT(X.TripToDate, 2) END END

-- 사용일 (R21)

+ CASE WHEN ISNULL(R21.ValueText, '0') = '0' THEN '' ELSE CASE WHEN ISNULL(C.UseDate, '') = '' THEN '' ELSE ' ' +

CASE WHEN B.BizTripType IN ('4', '5') THEN C.UseDate ELSE SUBSTRING(C.UseDate, 5, 2) + '/' + SUBSTRING(C.UseDate, 7, 2) END END END

-- 출장지 (R22)

+ CASE WHEN ISNULL(R22.ValueText, '0') = '0' THEN '' ELSE CASE WHEN B.BizTripType IN ('1', '3')

THEN CASE WHEN ISNULL(X.TripPlace, '') = '' THEN '' ELSE ' ' + X.TripPlace END

ELSE CASE WHEN ISNULL(B.Eduplace, '') = '' THEN '' ELSE ' ' + B.Eduplace END END END

-- 출장업체 (R23)

+ CASE WHEN ISNULL(R23.ValueText, '0') = '0' THEN '' ELSE CASE WHEN B.BizTripType IN ('1', '3')

THEN CASE WHEN ISNULL(X.TripCust , '') = '' THEN '' ELSE ' ' + X.TripCust END

ELSE CASE WHEN ISNULL(B.Institute, '') = '' THEN '' ELSE ' ' + B.Institute END END END

-- 거래처 (R24)

+ CASE WHEN ISNULL(R24.ValueText, '0') = '0' THEN '' ELSE CASE WHEN ISNULL(C.CustName, '') = ''

THEN '' ELSE ' ' + C.CustName END END

-- 식대명단 (R25)

+ CASE WHEN ISNULL(R25.ValueText, '0') = '0' THEN '' ELSE CASE WHEN ISNULL(C.PaymealName, '') = ''

THEN '' ELSE ' ' + C.PaymealName END END

-- 근무장소 (R29)

+ CASE WHEN ISNULL(R29.ValueText, '0') = '0' THEN '' ELSE CASE WHEN ISNULL(C.WorkPlace, '') = '' THEN '' ELSE ' ' + C.WorkPlace END END

FROM

_TARBizTripCost AS X WITH(NOLOCK) -- 내부 쿼리 테이블 X

LEFT OUTER JOIN das_TARBizTripReport AS B WITH(NOLOCK) ON X.CompanySeq = B.CompanySeq AND X.BizTripSeq = B.BizTripSeq

LEFT OUTER JOIN das_TARBizTripReportAmt AS C WITH(NOLOCK) ON C.CompanySeq = X.CompanySeq AND C.BizTripSeq = X.BizTripSeq

LEFT OUTER JOIN _TDAUMinor AS J WITH(NOLOCK) ON C.CompanySeq = J.CompanySeq AND C.UMCostSeq = J.MinorSeq AND J.MajorSeq = 1110352

LEFT OUTER JOIN _TDAUMinorValue AS K WITH(NOLOCK) ON J.CompanySeq = K.CompanySeq AND J.MinorSeq = K.MinorSeq AND K.Serl = 15000012

LEFT OUTER JOIN _TDAUMinorValue AS R18 WITH(NOLOCK) ON J.CompanySeq = R18.CompanySeq AND J.MinorSeq = R18.MinorSeq AND R18.Serl = 15000018

LEFT OUTER JOIN _TDAUMinorValue AS R19 WITH(NOLOCK) ON J.CompanySeq = R19.CompanySeq AND J.MinorSeq = R19.MinorSeq AND R19.Serl = 15000019

LEFT OUTER JOIN _TDAUMinorValue AS R20 WITH(NOLOCK) ON J.CompanySeq = R20.CompanySeq AND J.MinorSeq = R20.MinorSeq AND R20.Serl = 15000020

LEFT OUTER JOIN _TDAUMinorValue AS R21 WITH(NOLOCK) ON J.CompanySeq = R21.CompanySeq AND J.MinorSeq = R21.MinorSeq AND R21.Serl = 15000021

LEFT OUTER JOIN _TDAUMinorValue AS R22 WITH(NOLOCK) ON J.CompanySeq = R22.CompanySeq AND J.MinorSeq = R22.MinorSeq AND R22.Serl = 15000022

LEFT OUTER JOIN _TDAUMinorValue AS R23 WITH(NOLOCK) ON J.CompanySeq = R23.CompanySeq AND J.MinorSeq = R23.MinorSeq AND R23.Serl = 15000023

LEFT OUTER JOIN _TDAUMinorValue AS R24 WITH(NOLOCK) ON J.CompanySeq = R24.CompanySeq AND J.MinorSeq = R24.MinorSeq AND R24.Serl = 15000024

LEFT OUTER JOIN _TDAUMinorValue AS R25 WITH(NOLOCK) ON J.CompanySeq = R25.CompanySeq AND J.MinorSeq = R25.MinorSeq AND R25.Serl = 15000025

LEFT OUTER JOIN _TDAUMinorValue AS R29 WITH(NOLOCK) ON J.CompanySeq = R29.CompanySeq AND J.MinorSeq = R29.MinorSeq AND R29.Serl = 15000029

WHERE

-- 상관관계 조건: 외부 쿼리의 BizTripSeq와 일치하는 행만 처리

X.BizTripSeq = X_OUTER.BizTripSeq

AND X.CompanySeq = X_OUTER.CompanySeq

-- 기존 필터 조건 유지

AND X.CompanySeq = 1

AND X.BizTripSeq IN (SELECT BizTripSeq FROM #BIZ_OUT_DataBlock3)

ORDER BY X.BizTripSeq, X.TripFrDate -- 합치기 순서 보장

FOR XML PATH('')

), 1, 2, '') AS SlipRemark -- 최종 합쳐진 문자열

INTO #Tmp_SlipRemark

FROM

_TARBizTripCost AS X_OUTER WITH(NOLOCK) -- 외부 쿼리 테이블 (BizTripSeq를 가져옴)

WHERE

X_OUTER.CompanySeq = 1

AND X_OUTER.BizTripSeq IN (SELECT BizTripSeq FROM #BIZ_OUT_DataBlock3)

GROUP BY

X_OUTER.BizTripSeq, X_OUTER.CompanySeq -- BizTripSeq 별로 그룹화

ORDER BY

X_OUTER.BizTripSeq

UPDATE das_TARBizTripReport

SET SlipRemark = B.SlipRemark

FROM das_TARBizTripReport AS A

JOIN #Tmp_SlipRemark AS B ON B.BizTripSeq = A.BizTripSeq

WHERE A.CompanySeq = @CompanySeq

END

RETURN
