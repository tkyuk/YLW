--(아마 영림원 서버를 이용하면서 발생한 문제로 사료)

-- 강제로 체크아웃

USE DANAMCommon

SELECT * FROM _TCAService WHERE ServiceID = 'Angkor.Ylw.Sales.DatLGLotMaster_danam'

update _TCAService

set CheckOutYN         = '1' ,

CheckOutUserSeq = '1' ,

CheckOutDateTime = GETDATE()

where ServiceSeq = 1036523

-- _TCAServiceHistory 여기에는 안해도 작업하는데 문제는 없음

-- 체크인

update _TCAService

set CheckOutYN         = '0' ,

CheckOutUserSeq = '0' ,

CheckOutDateTime = NULL

where ServiceSeq = 1036523
