제뉴인 채번

IF EXISTS (SELECT 1 FROM #TPDSFCWorkReport AS A WHERE A.WorkingTag = 'A' AND A.Status = 0)

BEGIN

SELECT @Date = WorkDate

FROM #TPDSFCWorkReport

WHERE WorkingTag = 'A'

AND Status = 0

AND RealLotNo = ''

ORDER BY DataSeq

-- LotNo 채번

SELECT @MaxSerl = RIGHT('00' + CAST(RIGHT(MAX(REPLACE(LOTNO,'-','')),3) AS NVARCHAR), 3)

FROM _TLGLotMaster

WHERE REPLACE(LOTNO,'-','') LIKE REPLACE(@ToolNo + RIGHT(@Date,6) + @WorkStartTime,' ','')+'%'

IF @MaxSerl IS NULL

SELECT @MaxSerl = '001'

ELSE

EXEC _SDANextSerialNo @MaxSerl OUTPUT, 3 -- 3은 자리수

UPDATE #TPDSFCWorkReport

SET RealLotNo = RTRIM(@ToolNo)+'-' + RIGHT(@Date,6) + @WorkStartTime +'-'+@MaxSerl

WHERE WorkingTag ='A'

END
