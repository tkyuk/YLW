DROP PROC dbo.GCOS_CUSTTEST

GO

CREATE PROC dbo.GCOS_CUSTTEST

@CustName NVARCHAR(100), -- 기업명

@BizNo NVARCHAR(20), -- 사업자번호

@Status INT OUTPUT,

@Results NVARCHAR(250) OUTPUT

AS

DECLARE @Result NVARCHAR(10) = ''

,@MessageType INT

,@PgmSeq INT = 9999

,@CompanySeq INT = 1

,@LanguageSeq INT = 1

,@UserSeq INT = 0

,@CustSeq INT = 0

,@CondSeq INT = 0

,@Message NVARCHAR(200)

EXEC _SYMMETRICKeyOpen

--=================================

-- 환경설정

--=================================

DECLARE @Env8063 INT,

@Env8064 INT,

@Env8054 INT

EXEC dbo._SWCOMEnv @CompanySeq,8063 ,0,@@PROCID,@Env8063 OUTPUT -- [매출기준] 매출회수조건기준

EXEC dbo._SWCOMEnv @CompanySeq,8064 ,0,@@PROCID,@Env8064 OUTPUT -- [매출기준] 매출청구방법

EXEC dbo._SWCOMEnv @CompanySeq,8054 ,0,@@PROCID,@Env8054 OUTPUT -- [매출기준] 청구처매출시점

IF ISNULL(@Env8063, 0) = 0 SELECT @Env8063 = ISNULL(@Env8063, 0)

IF ISNULL(@Env8064, 0) = 0 SELECT @Env8064 = ISNULL(@Env8064, 0)

IF ISNULL(@Env8054, 0) = 0 SELECT @Env8054 = ISNULL(@Env8054, 0)

-- 거래처

CREATE TABLE #Temp_Cust

(

DataSeq INT IDENTITY(1,1),

CompanySeq INT,

CustSeq INT,

CondSeq INT,

CustNo NVARCHAR(100),

CustName NVARCHAR(100),

TrunName NVARCHAR(100),

BizNo NVARCHAR(40) ,

Status INT ,

Result NVARCHAR(500)

)

INSERT INTO #Temp_Cust

(

CompanySeq , CustSeq , CondSeq , CustNo , CustName ,

TrunName , BizNo , Status , Result

)

SELECT @CompanySeq , 0 , 0 , @BizNo , @CustName,

@CustName, @BizNo , 0 , ''

--=================================

-- 사업자번호(BizNo) 공백여부 체크

--=================================

EXEC dbo._SWCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

1 , -- 필수입력 데이타를 입력하세요.(SELECT * FROM _TCAMessageLanguage WHERE MessageSeq = 1)

@LanguageSeq ,

640,'' -- SELECT * FROM _TCADictionary WHERE Word like '%사업자번호%'

UPDATE #Temp_Cust

SET Result = @Results,

Status = @Status

FROM #Temp_Cust

WHERE BizNo = ''

--=================================

-- 사업자번호 기준 거래처 등록여부 체크

--=================================

-- 사업자번호 중복체크

EXEC dbo._SWCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

6 , -- 중복된 @1 @2이(가) 입력되었습니다.

@LanguageSeq ,

6 , N'거래처'

UPDATE #Temp_Cust

SET Result = REPLACE(@Results,'@2',A.CustName),

Status = @Status

FROM #Temp_Cust AS A JOIN _TDACust AS B WITH(NOLOCK)

ON B.CompanySeq = A.CompanySeq

AND B.BizNo = A.BizNo

WHERE B.BizNo \<> ''

-- 시스템제공코드

-- (INPUT) 시스템제공코드 대분류

CREATE TABLE #TDASMajorValue(

MajorSeq INT

)

-- (OUTPUT) 시스템제공코드 소분류

CREATE TABLE #TDASMinorValue(

CompanySeq INT

, MajorSeq INT

, MinorSeq INT

, MinorName NVARCHAR(200)

, MinorValue NVARCHAR(50)

, MinorSort INT

, IsUse NCHAR(1)

)

INSERT INTO #TDASMajorValue(MajorSeq)

SELECT 100151 -- 결제대행사

-- 시스템제공코드 가져오기

EXEC dbo._SWUTGetSMinorValue @CompanySeq = @CompanySeq

, @LanguageSeq = @LanguageSeq

--===========================

-- SM 100151 결제대행사

--===========================

CREATE TABLE #TDAPaymentAgency(

CompanySeq INT

, PaymentAgencyBizNo NVARCHAR(50)

, PaymentAgency NVARCHAR(200)

, IsPaymentAgency NCHAR(1)

)

INSERT INTO #TDAPaymentAgency(CompanySeq, PaymentAgencyBizNo, PaymentAgency, IsPaymentAgency)

SELECT A.CompanySeq, A.MinorValue, A.MinorName, N'1'

FROM #TDASMinorValue AS A WITH(NOLOCK)

WHERE A.CompanySeq = @CompanySeq

AND A.MajorSeq = 100151

EXEC dbo._SWCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

1281 , -- @1로(으로) 등록된 @2입니다. @3을(를) 입력하세요.

@LanguageSeq ,

80645 , N'결제대행사',

6 , N'거래처',

91860 , N'실거래처'

UPDATE #Temp_Cust

SET Result = @Results,

Status = @Status

FROM #Temp_Cust AS A

LEFT OUTER JOIN #TDAPaymentAgency AS PA ON PA.CompanySeq = A.CompanySeq

AND PA.PaymentAgencyBizNo = A.BizNo -- 결제대행사

WHERE PA.CompanySeq IS NOT NULL

-- 에러가 발생하면 전부다 에러

IF EXISTS(SELECT TOP 1 1 FROM #Temp_Cust WHERE Status \<> 0)

BEGIN

SELECT @Status = Status,

@Results = Result

FROM #Temp_Cust

WHERE Status \<> 0

SELECT @Status, @Results

RETURN

END

--=================================

-- CustName, TrunName 값 변환처리 -- 거래처명 생략어 제거

--=================================

DECLARE @MaxDataSeq INT,

@DataSeq INT,

@CutName NVARCHAR(50)

IF NOT EXISTS(SELECT TOP 1 1 FROM #Temp_Cust WHERE Status \<> 0)

BEGIN

SELECT @CutName = ''

SELECT @CutName = LTRIM(RTRIM(CutName)) FROM _TDACustCutName WITH(NOLOCK) WHERE CompanySeq = @CompanySeq AND CutSeq = @DataSeq

IF ISNULL(@CutName,'') \<> ''

BEGIN

UPDATE #Temp_Cust

SET CustName = LTRIM(REPLACE(CustName, @CutName, '')),

TrunName = LTRIM(REPLACE(TrunName, @CutName, ''))

FROM #Temp_Cust

END

-- 공백제거

UPDATE #Temp_Cust

SET TrunName = REPLACE(TrunName, ' ', '')

FROM #Temp_Cust

--=================================

-- 채번

--=================================

EXEC @CustSeq = _SWCOMCreateSeq @CompanySeq, '_TDACust', 'CustSeq', 1

EXEC @CondSeq = _SWCOMCreateSeq @CompanySeq, '_TDACustSalesReceiptCond', 'CondSeq', 1

UPDATE #Temp_Cust

SET CustSeq = @CustSeq + 1,

CondSeq = @CondSeq + 1

FROM #Temp_Cust

END

--=================================

-- TRY - CATCH 시작

--=================================

BEGIN TRY

--=================================

-- 트랜잭션 시작

--- CATCH에서 ROLLBACK 처리하기 때문에 TRY 시작에 TRAN 시작

--=================================

BEGIN TRAN

-- 오류가 발생하지 않은 경우

IF NOT EXISTS(SELECT TOP 1 1 FROM #Temp_Cust WHERE Status \<> 0)

BEGIN

--=================================

-- INSERT

--=================================

INSERT INTO _TDACust (

CompanySeq, CustSeq, CustName, TrunName, CustNo

, UMCredLevel, SMDomFor, SMCustStatus, FullName, LastUserSeq

, LastDateTime, UMCountrySeq, FrDate, DeptSeq, EmpSeq

, RegUserSeq, RegDateTime, BizNo, Owner, LawRegNo

, BizAddrSeq, BizKind, BizType, TelNo, MinorBizNo, SMBizPers

, IsFwbBill, IsRevBill, SMCustRelation, PassportNo, IsSumBill

)

SELECT @CompanySeq, CustSeq, CustName, TrunName, N'',

0, 1013001, 2004004, CustName, @UserSeq ,

GETDATE(), 0, CONVERT(NCHAR(8),GETDATE(),112), 0, 0,

@UserSeq, GETDATE(), BizNo, '', N'',

0, '', '', '', N'', 4019001,

N'0', N'0', 8194001, N'', N'0'

FROM #Temp_Cust

WHERE Status = 0

-- 8014001 대표거래처

INSERT INTO _TDACustGroup ( CompanySeq , CustSeq , UMCustGroup , UpperCustSeq , LastUserSeq , LastDateTime)

SELECT @CompanySeq , CustSeq , 8014001 , CustSeq , @UserSeq , GETDATE()

FROM #Temp_Cust

WHERE Status = 0

-- 8014002 청구처

INSERT INTO _TDACustGroup ( CompanySeq , CustSeq , UMCustGroup , UpperCustSeq , LastUserSeq , LastDateTime)

SELECT @CompanySeq , CustSeq , 8014002 , CustSeq , @UserSeq , GETDATE()

FROM #Temp_Cust

WHERE Status = 0

-- 8004 유통구조

INSERT INTO _TDACustClass (CompanySeq , CustSeq , UMajorCustClass , UMCustClass , LastUserSeq , LastDateTime)

SELECT @CompanySeq , CustSeq , 8004 , 0 , @UserSeq , GETDATE()

FROM #Temp_Cust

WHERE Status = 0

-- 거래처종류 (기타거래처)

INSERT INTO _TDACustKind (CompanySeq , CustSeq , UMCustKind , LastUserSeq , LastDateTime)

SELECT @CompanySeq , CustSeq , 1004023 , @UserSeq , GETDATE()

FROM #Temp_Cust

WHERE Status = 0

-- 청구처매출회수조건

INSERT INTO _TDACustSalesReceiptCond (CompanySeq, CondSeq, CustSeq, SMCondStd, SMReceiptMethod,

SMSalesPoint, IsCust, LastUserSeq, LastDateTime, ReceiptCustSeq,

DueDays, SMReceiveMethod, CurrSeq, SMReceiptDateMethod

)

SELECT @CompanySeq, CondSeq, CustSeq, @Env8063, @Env8064,

@Env8054, N'', @UserSeq, GETDATE(), 0,

0, 0, 0, 0

FROM #Temp_Cust

WHERE Status = 0

IF @Env8063 = 8018002 -- 청구처회수조건

BEGIN

-- 매출회수기준

INSERT INTO _TDACustSalesReceiptStd (CompanySeq, CondSeq, CondSerl, SalesCloseDate, BillMonth,

BillDate, ReceiptMonth, ReceiptDate, SMReceiptKind, NotExpDays,

Remark, LastUserSeq, LastDateTime, CashTerms

)

SELECT @CompanySeq, CondSeq, 1, 31, 0,

0, 0, 0, 0, 0,

N'', @UserSeq, GETDATE(), 0

FROM #Temp_Cust

WHERE Status = 0

END

END

--=================================

-- 최종조회

--@Status INT = 0 OUTPUT, -- 처리상태

--@Results NVARCHAR(500) = '' OUTPUT, -- 처리결과

--=================================

COMMIT TRAN

--==============================================================

-- 결과출력 (DataBlock1 : 결과)

--==============================================================

END TRY

BEGIN CATCH

ROLLBACK TRAN

--오류 날 경우

SELECT @Result = 'ERROR', @Status = 999999

SELECT @Status AS Status ,

@Result AS Results

END CATCH

----------------------- OPEN API Log 데이터 생성 ---------------------

IF @Result = '' SELECT @Result = 'OK'

RETURN

GO
