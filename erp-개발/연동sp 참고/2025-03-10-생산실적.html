생산실적

CREATE OR ALTER VIEW gcos_VMESIFWorkReport

AS

SELECT A.CompanySeq AS CompanySeq -- 법인코드

,A.WorkReportSeq AS WorkReportSeq -- 작업실적코드

,A.WorkDate AS WorkDate -- 작업일자

,ISNULL(C.WorkOrderNo,'') AS WorkOrderNo -- 작업지시번호

,I.ItemName AS ItemName -- 품목명

,I.ItemNo AS ItemNo -- 품목번호

,I.Spec AS Spec -- 규격

,A.ProdQty AS ProdQty -- 생산수량

,A.OKQty AS OKQty -- 양품수량

,A.BadQty AS BadQty -- 불량수량

,U.UnitName AS UnitName -- 단위

,A.Remark AS Remark -- 비고

,ISNULL(E.OrderSeq , 0) AS OrderSeq -- 수주코드

,ISNULL(G.CustName ,'') AS OrderCustName -- 수주거래처명

FROM _TPDSFCWorkReport AS A

LEFT OUTER JOIN _TCACompany AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

LEFT OUTER JOIN _TPDSFCWorkOrder AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.WorkOrderSeq = A.WorkOrderSeq

AND C.WorkOrderSerl = A.WorkOrderSerl

LEFT OUTER JOIN _TDAItem AS I WITH(NOLOCK) ON I.CompanySeq = A.CompanySeq

AND I.ItemSeq = A.GoodItemSeq

LEFT OUTER JOIN _TDAUnit AS U WITH(NOLOCK) ON U.CompanySeq = I.CompanySeq

AND U.UnitSeq = I.UnitSeq

OUTER APPLY (SELECT D.CompanySeq, D.ToSeq, MAX(FromSeq) AS FromSeq

FROM _TCOMSourceDailyTrace AS D WITH(NOLOCK)

WHERE D.CompanySeq = A.CompanySeq

AND D.ToSeq = A.WorkReportSeq

AND D.ToSerl = 0

AND D.ToTableSeq = 6 -- 작업실적

AND D.FromTableSeq = 19 -- 수주

GROUP BY D.CompanySeq, D.ToSeq

) AS D

LEFT OUTER JOIN _TSLOrder AS E WITH(NOLOCK) ON E.CompanySeq = D.CompanySeq

AND E.OrderSeq = D.FromSeq

LEFT OUTER JOIN _TDACust AS G WITH(NOLOCK) ON G.CompanySeq = E.CompanySeq

AND G.CustSeq = E.CustSeq

--GO
