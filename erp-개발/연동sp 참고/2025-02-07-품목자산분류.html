품목자산분류

CREATE OR ALTER VIEW mcm_VMESDAItemAsset

AS

SELECT ISNULL(A.CompanySeq , 0) AS CompanySeq , -- 법인내부코드

ISNULL(B.CompanyName ,'') AS CompanyName , -- 법인명

ISNULL(A.AssetSeq , 0) AS AssetSeq , -- 품목자산분류내부코드

ISNULL(A.AssetName ,'') AS AssetName , -- 품목자산분류

ISNULL(A.SMAssetGrp , 0) AS SMAssetGrp , -- 재고자산종류내부코드

ISNULL(C.MinorName ,'') AS SMAssetGrpName , -- 재고자산종류

CASE WHEN A.IsQty = '1' THEN 'YES' ELSE 'NO' END AS IsQty , -- 재고수량관리안함

CASE WHEN A.IsAmt = '1' THEN 'YES' ELSE 'NO' END AS IsAmt , -- 재고금액관리안함

CASE WHEN A.IsMinus = '1' THEN 'YES' ELSE 'NO' END AS IsMinus , -- 재고부족통제

CASE WHEN A.IsSet = '1' THEN 'YES' ELSE 'NO' END AS IsSet , -- 세트품목

ISNULL(A.LastUserSeq , 0) AS LastUserSeq , -- 최종등록자

ISNULL(A.LastDateTime,'') AS LastDateTime -- 최종등록일자

FROM _TDAItemAsset AS A WITH(NOLOCK)

LEFT OUTER JOIN _TCACompany AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

LEFT OUTER JOIN _TDASMinor AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.MinorSeq = A.SMAssetGrp
