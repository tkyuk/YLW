(파라미터) 신규 테이블 연동

DROP PROC smc_MESPDQCReportSave

GO

CREATE PROC smc_MESPDQCReportSave

@WorkingTag NCHAR(1) , -- 저장상태값 / A:삽입 D:삭제

@ReportSeq INT OUTPUT, -- 검사내부코드 / A(삽입)일경우 숫자 0으로 보내주시고, D(삭제)일 경우 검사내부코드

@SourceType INT , -- 서비스처리내부코드 / 생산실적:1 /서비스처리:2

@SourceSeq INT , -- 서비스처리내부순번 / 작업실적내부코드 OR 서비스처리내부코드

@SMTestResult INT , -- 검사결과 / 합격: 6035003 / 불합격: 6035004

@SMChargeType INT , -- 유무상구분 / 유상:1 / 무상:2

@EmpSeq INT , -- 담당자내부코드 /

@UserSeq INT , -- 등록자내부코드 /

@Status NVARCHAR(10) OUTPUT, -- 에러상태코드 / OK / 그 외

@Results NVARCHAR(250) OUTPUT -- 에러메시지 /

AS

SET NOCOUNT ON

SET XACT_ABORT ON

-- 들어온 파라미터 저장

INSERT smc_TMESPDQCReport_Param

(

LogDateTime ,

WorkingTag ,

ReportSeq ,

SourceType ,

SourceSeq ,

SMTestResult,

SMChargeType,

EmpSeq ,

UserSeq ,

Status ,

Results

)

SELECT GETDATE() ,

@WorkingTag ,

@ReportSeq ,

@SourceType ,

@SourceSeq ,

@SMTestResult,

@SMChargeType,

@EmpSeq ,

@UserSeq ,

@Status ,

@Results

SELECT @ReportSeq = ISNULL(@ReportSeq, 0),

@Status = 'OK',

@Results = ''

DECLARE @MaxSeq INT,

@CompanySeq INT = 1,

@PgmSeq INT = 9999,

@CurrDate NCHAR(8),

@DeptSeq INT,

@ERROR_TEXT NVARCHAR(100)

DECLARE @CheckItemSeq INT,

@IsFinalQC NVARCHAR(50)

SELECT @CurrDate = CONVERT(NCHAR(8), GETDATE(), 112)

SELECT @DeptSeq = WkDeptSeq

FROM _TDAEmp

WHERE CompanySeq = @CompanySeq

AND EmpSeq = @EmpSeq

-- 삭제 로그용 테이블

CREATE TABLE #smc_TPDQCReportLog

(

IDX_NO INT,

WorkingTag NVARCHAR(10),

ReportSeq INT,

SourceType INT,

SourceSeq INT,

SMTestResult INT,

SMChargeType INT,

EmpSeq INT,

UserSeq INT,

Status INT,

Result NVARCHAR(250),

MessageType INT

)

-- 체크

IF @WorkingTag = 'A'

BEGIN

IF ISNULL(@SourceType , 0) = 0 OR

ISNULL(@SourceSeq , 0) = 0 OR

ISNULL(@SMTestResult, 0) = 0 OR

ISNULL(@SMChargeType, 0) = 0 OR

ISNULL(@EmpSeq , 0) = 0 OR

ISNULL(@UserSeq , 0) = 0

BEGIN

SELECT @ERROR_TEXT = CASE WHEN ISNULL(@SourceType , 0) = 0 THEN '서비스처리내부코드'

WHEN ISNULL(@SourceSeq , 0) = 0 THEN '서비스처리내부순번'

WHEN ISNULL(@SMTestResult , 0) = 0 THEN '검사결과'

WHEN ISNULL(@SMChargeType , 0) = 0 THEN '유무상구분'

WHEN ISNULL(@EmpSeq , 0) = 0 THEN '담당자내부코드'

WHEN ISNULL(@UserSeq , 0) = 0 THEN '등록자내부코드'

END

SELECT @Status = 'ERROR',

@Results = '필수 값(' + @ERROR_TEXT + ')이(가) 입력되지 않았습니다.'

GOTO RESULT_PROC

END

IF ISNULL(@SourceType, 0) NOT IN (1, 2)

BEGIN

SELECT @Status = 'ERROR',

@Results = '서비스처리내부코드가 정상적으로 입력되지 않았습니다.'

GOTO RESULT_PROC

END

IF ISNULL(@SMTestResult, 0) NOT IN (6035003, 6035004)

BEGIN

SELECT @Status = 'ERROR',

@Results = '검사결과가 정상적으로 입력되지 않았습니다.'

GOTO RESULT_PROC

END

IF ISNULL(@SMChargeType, 0) NOT IN (1, 2)

BEGIN

SELECT @Status = 'ERROR',

@Results = '유무상구분이 정상적으로 입력되지 않았습니다.'

GOTO RESULT_PROC

END

-- 최종검사여부 대상 품목인지 사전 확인

IF @SourceType = 1 -- 생산실적

BEGIN

SELECT @CheckItemSeq = AssyItemSeq

FROM _TPDSFCWorkReport WITH(NOLOCK)

WHERE CompanySeq = @CompanySeq

AND WorkReportSeq = @SourceSeq

END

ELSE IF @SourceType = 2 -- 서비스처리

BEGIN

SELECT @CheckItemSeq = B.ItemSeq

FROM _TSIASServiceProc AS A WITH(NOLOCK)

JOIN _TSIASServiceReq AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.ReceiptSeq = A.ReceiptSeq

WHERE A.CompanySeq = @CompanySeq

AND A.ProcSeq = @SourceSeq

END

-- 가져온 ItemSeq로 사용자정의값(1000021:최종검사여부) 확인

SELECT @IsFinalQC = ISNULL(MngValText, '0')

FROM _TDAItemUserDefine WITH(NOLOCK)

WHERE CompanySeq = @CompanySeq

AND ItemSeq = @CheckItemSeq

AND MngSerl = 1000021

-- 값이 '1'이 아니면 에러 리턴

IF @IsFinalQC \<> '1'

BEGIN

SELECT @Status = 'ERROR',

@Results = CASE WHEN @SourceType = 1 THEN '생산실적' ELSE '서비스처리' END + '의 품목이 최종검사 대상이 아닙니다.'

GOTO RESULT_PROC

END

END

ELSE IF @WorkingTag = 'D'

BEGIN

IF ISNULL(@ReportSeq, 0) = 0

BEGIN

SELECT @Status = 'ERROR',

@Results = '검사내부코드가 입력되지 않았습니다.'

GOTO RESULT_PROC

END

INSERT #smc_TPDQCReportLog

(

IDX_NO ,

WorkingTag ,

ReportSeq ,

SourceType ,

SourceSeq ,

SMTestResult,

SMChargeType,

EmpSeq ,

UserSeq ,

Status ,

Result ,

MessageType

)

SELECT 1 ,

@WorkingTag ,

@ReportSeq ,

@SourceType ,

@SourceSeq ,

@SMTestResult,

@SMChargeType,

@EmpSeq ,

@UserSeq ,

0 ,

@Results ,

0

-- 로그테이블 남기기(마지막 파라메터는 반드시 한줄로 보내기)

EXEC _SCOMXmlLog @CompanySeq ,

@UserSeq ,

'smc_TPDQCReport' , -- 원테이블명

'#smc_TPDQCReportLog', -- 템프테이블명

'ReportSeq' , -- 키가 여러개일 경우는 , 로 연결한다.

@PgmSeq, ''

END

BEGIN TRY

BEGIN TRAN

-- DELETE

IF @WorkingTag = 'D'

BEGIN

-- 하위 테이블 삭제

DELETE smc_TPDQCReportItemTC WHERE ReportSeq = @ReportSeq

DELETE smc_TPDQCReportItemHJ WHERE ReportSeq = @ReportSeq

DELETE smc_TPDQCReportVGBreakdown WHERE ReportSeq = @ReportSeq

DELETE smc_TPDQCReportVGRepair WHERE ReportSeq = @ReportSeq

DELETE smc_TPDQCReportVGBefCali WHERE ReportSeq = @ReportSeq

DELETE smc_TPDQCReportVGAftCali WHERE ReportSeq = @ReportSeq

DELETE smc_TPDQCReportMFCItem1 WHERE ReportSeq = @ReportSeq

DELETE smc_TPDQCReportMFCItem2 WHERE ReportSeq = @ReportSeq

DELETE smc_TPDQCReportMFCItem3 WHERE ReportSeq = @ReportSeq

DELETE smc_TPDQCReportMFCItem4 WHERE ReportSeq = @ReportSeq

DELETE smc_TPDQCReportMFCItem5 WHERE ReportSeq = @ReportSeq

DELETE smc_TPDQCReportTool WHERE ReportSeq = @ReportSeq

DELETE smc_TPDQCReportImage WHERE ReportSeq = @ReportSeq

-- 테이블 삭제

DELETE smc_TPDQCReport WHERE ReportSeq = @ReportSeq

END

ELSE IF @WorkingTag = 'A'

BEGIN

-- 번호 채번

SELECT @MaxSeq = ISNULL(MAX(A.ReportSeq), 0)

FROM smc_TPDQCReport AS A WITH(UPDLOCK)

WHERE A.CompanySeq = @CompanySeq

SELECT @ReportSeq = @MaxSeq + 1

IF @SourceType = 1 -- 생산실적

BEGIN

INSERT INTO smc_TPDQCReport

(

CompanySeq, ReportSeq, SourceType, SourceSeq, SourceSerl, SerialNo, LotNo, CustSeq, ItemSeq, QCDate, QCEmpSeq, QCDeptSeq, SMTestResult, Remark, UseToolNo, DrawingSN, UseTemp,

TrendNo, RetedVoltage, AccuRacy, IsTYLAN, Gas, CF, InletPressure, HeadPressure, Description, FullScale, LastUserSeq, LastDateTime, SMChargeType

)

SELECT @CompanySeq, @ReportSeq, @SourceType, @SourceSeq, 0, B.SerialNo, A.RealLotNo, E.CustSeq, A.AssyItemSeq, @CurrDate, @EmpSeq, @DeptSeq, @SMTestResult, '', '', '', '',

'', '', '', '0', '', '', '', '', '', '', @UserSeq, GETDATE(), @SMChargeType

FROM _TPDSFCWorkReport AS A

JOIN smc_TPDSFCWorkReportSerial AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.WorkReportSeq = A.WorkReportSeq

LEFT OUTER JOIN _TPDSFCWorkOrder AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.WorkOrderSeq = A.WorkOrderSeq

AND C.WorkOrderSerl = A.WorkOrderSerl

LEFT OUTER JOIN _TPDMPSDailyProdPlan AS D WITH(NOLOCK) ON D.CompanySeq = C.CompanySeq

AND D.ProdPlanSeq = C.ProdPlanSeq

LEFT OUTER JOIN _TPDMPSProdReqItem AS E WITH(NOLOCK) ON E.CompanySeq = D.CompanySeq

AND E.ProdReqSeq = D.SourceSeq

AND E.Serl = D.SourceSerl

WHERE A.CompanySeq = @CompanySeq

AND A.WorkReportSeq = @SourceSeq

END

ELSE IF @SourceType = 2 -- 서비스처리

BEGIN

INSERT INTO smc_TPDQCReport

(

CompanySeq, ReportSeq, SourceType, SourceSeq, SourceSerl, SerialNo, LotNo, CustSeq, ItemSeq, QCDate, QCEmpSeq, QCDeptSeq, SMTestResult, Remark, UseToolNo, DrawingSN, UseTemp,

TrendNo, RetedVoltage, AccuRacy, IsTYLAN, Gas, CF, InletPressure, HeadPressure, Description, FullScale, LastUserSeq, LastDateTime, SMChargeType

)

SELECT @CompanySeq, @ReportSeq, @SourceType, @SourceSeq, 0, B.SerialNo, A.ProcNo, B.CustSeq, B.ItemSeq, @CurrDate, @EmpSeq, @DeptSeq, @SMTestResult, '', '', '', '',

'', '', '', '0', '', '', '', '', '', '', @UserSeq, GETDATE(), @SMChargeType

FROM _TSIASServiceProc AS A

LEFT OUTER JOIN _TSIASServiceReq AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.ReceiptSeq = A.ReceiptSeq

WHERE A.CompanySeq = @CompanySeq

AND A.ProcSeq = @SourceSeq

END

END

COMMIT TRAN

END TRY

BEGIN CATCH

IF @@TRANCOUNT > 0

ROLLBACK TRAN

-- 에러 정보 수집

SELECT @Status = 'ERROR',

@Results = ERROR_MESSAGE() -- 실제 SQL 에러 메시지 반환

END CATCH

RESULT_PROC:

INSERT smc_TMESPDQCReport_Result

(

LogDateTime ,

WorkingTag ,

ReportSeq ,

SourceType ,

SourceSeq ,

SMTestResult,

SMChargeType,

EmpSeq ,

UserSeq ,

Status ,

Results

)

SELECT GETDATE() ,

@WorkingTag ,

@ReportSeq ,

@SourceType ,

@SourceSeq ,

@SMTestResult,

@SMChargeType,

@EmpSeq ,

@UserSeq ,

@Status ,

@Results

RETURN
