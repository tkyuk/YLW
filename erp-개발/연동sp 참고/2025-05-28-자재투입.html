CREATE OR ALTER PROC mcm_SMESWorkReportMatSave

@WorkingTag NVARCHAR(2) ,

@WorkReportSeq INT , -- 파라미터값이 0이면 사용자정의코드의 기본값으로 처리

@ItemSeq NVARCHAR(MAX),

@Qty NVARCHAR(MAX),

@UserSeq INT,

@Status NVARCHAR(10) OUTPUT, -- 상태값

@Results NVARCHAR(250) OUTPUT -- 에러메세지

SET NOCOUNT ON

DECLARE @CompanySeq INT,

@PgmSeq INT,

@Status2 INT,

@Results1 NVARCHAR(500),

@DeptSeq INT,

@CurrDateTime DATETIME,

@Status1 INT,

@WorkDate NCHAR(8),

@MatDec INT

SELECT @PgmSeq = 52920053, -- MES 시스템 처리

@LanguageSeq = 1

SELECT @WorkingTag = LTRIM(RTRIM(ISNULL(@WorkingTag , ''))),

@WorkReportSeq = LTRIM(RTRIM(ISNULL(@WorkReportSeq, 0))),

@ItemSeq = LTRIM(RTRIM(ISNULL(@ItemSeq ,''))),

@Qty = LTRIM(RTRIM(ISNULL(@Qty ,''))),

@UserSeq = LTRIM(RTRIM(ISNULL(@UserSeq , 0))),

@CompanySeq = 1,

@Status = 'OK',

@Results = CASE WHEN @WorkingTag = 'A' THEN '자재투입내역이 생성되었습니다.' ELSE '자재투입내역이 삭제되었습니다.' END,

@CurrDateTime = GETDATE()

SELECT @LastUserSeq = @UserSeq

SELECT @EmpSeq = ISNULL(B.EmpSeq,0),

@DeptSeq = A.DeptSeq

FROM _TCAUser AS B WITH(NOLOCK)

JOIN _fnAdmEmpOrd(@CompanySeq, '') AS A ON A.EmpSeq = B.EmpSeq

WHERE B.CompanySeq = @CompanySeq

AND B.UserSeq = @UserSeq

SELECT @LastUserSeq = ISNULL(@LastUserSeq,0),

@EmpSeq = ISNULL(@EmpSeq ,0),

@DeptSeq = ISNULL(@DeptSeq ,0)

EXEC dbo._SWCOMEnv @CompanySeq, 5, 0, @@PROCID, @MatDec                 OUTPUT -- 20230303.구매/자재 수량 소숫점 자릿수 추가

ADD_PROC:

IF @WorkReportSeq = 0

BEGIN

SELECT @Status = 'ERR', @Results = '작업실적 내부코드가 없습니다..'

GOTO ERR_PROC

END

-- 처리자 체크

IF NOT EXISTS (SELECT 1 FROM _TDAEmp WHERE CompanySeq = @CompanySeq AND EmpSeq = @EmpSeq)

BEGIN

SELECT @Status = 'ERR',

@Results = '처리자가 잘못 되었습니다.'

GOTO ERR_PROC

END

-- 처리자 체크

IF NOT EXISTS (SELECT 1 FROM _TPDSFCWorkReport WHERE CompanySeq = @CompanySeq AND WorkReportSeq = @WorkReportSeq)

BEGIN

SELECT @Status = 'ERR',

@Results = '작업실적 건이 존재하지 않습니다.'

GOTO ERR_PROC

END

CREATE TABLE #MESData

(

IDX_NO INT IDENTITY(1,1),

WorkReportSeq INT,

ItemSeq INT,

InputQty DECIMAL(19,5),

WHSeq INT,

UnitSeq INT

)

INSERT INTO #MESData(WorkReportSeq, ItemSeq, InputQty)

SELECT @WorkReportSeq, A.Code, C.Code

FROM _FCOMSplit(@ItemSeq, '_/') AS A

JOIN _FCOMSplit(@Qty, '_/') AS C ON A.IDX_No = C.IDX_No

IF @@ROWCOUNT = 0

BEGIN

SELECT @Status = 'ERR', @Results = '처리할 데이터가 없습니다.'

GOTO ERR_PROC

RETURN

END

UPDATE A

SET UnitSeq = B.UnitSeq

FROM #MESData AS A

JOIN _TDAItem AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND A.ItemSeq = B.ItemSeq

--==================================================

-- 창고 가져오기

--==================================================

UPDATE #MESData

SET WHSeq = (CASE

WHEN P.SMOutKind \<> 6005002 THEN A.FieldWhSeq -- 자동출고 아닌 경우 현장창고

WHEN S.SMAssetGrp = 6008005 THEN A.FieldWhSeq -- 공정품

WHEN U.MinorValue = N'1' THEN B.MatOutWhSeq -- 자재

ELSE B.GoodOutWHSeq

END)

FROM #MESData AS T

JOIN _TPDSFCWorkReport AS A WITH(NOLOCK) ON A.WorkReportSeq = @WorkReportSeq

JOIN _TPDBaseWorkCenter AS B WITH(NOLOCK) ON A.WorkcenterSeq = B.WorkcenterSeq

AND A.CompanySeq = B.CompanySeq

JOIN _TDAItem AS I WITH(NOLOCK) ON T.ItemSeq = I.ItemSeq

AND I.CompanySeq = @CompanySeq

JOIN _TDAItemAsset AS S WITH(NOLOCK) ON I.AssetSeq = S.AssetSeq

AND I.CompanySeq = S.CompanySeq

LEFT OUTER JOIN _TDASMinor AS U WITH(NOLOCK) ON S.SMAssetGrp = U.MinorSeq

AND U.MajorSeq = 6008

AND S.CompanySeq = U.CompanySeq

LEFT OUTER JOIN _TDAItemProduct AS P WITH(NOLOCK) ON T.ItemSeq = P.ItemSeq

AND P.CompanySeq = @CompanySeq

WHERE A.WorkReportSeq = @WorkReportSeq

AND A.CompanySeq = @CompanySeq

-- MCM에 적용되어있는 창고 가져오는 로직 추가

-- 1. 소요자재조회, 자동투입시 투입자재의 현장창고 로직 수정요청합니다.

-- - 현재는 워크센터의 자재출고창고, 제상품출고창고가 적용됩니다.

-- - 품목의 출고구분이 "자재 투입 시 자동출고"면 품목등록 재고정보의 창고가 적용되도록 수정 요청합니다.(TOP 1)

UPDATE T

SET WHSeq = C.WHSeq

FROM #MESData AS T

JOIN _TPDSFCWorkReport AS A WITH(NOLOCK) ON A.WorkReportSeq = @WorkReportSeq

JOIN _TDAItem AS I WITH(NOLOCK) ON T.ItemSeq = I.ItemSeq

AND I.CompanySeq = @CompanySeq

JOIN _TDAItemProduct AS B WITH(NOLOCK) ON I.CompanySeq = B.CompanySeq

AND I.ItemSeq = B.ItemSeq

AND B.SMOutKind = 6005002 -- 자재투입시 자동출고

OUTER APPLY (SELECT TOP 1 WHSeq

FROM _TDAWHItem WITH(NOLOCK)

WHERE ItemSeq = I.ItemSeq

AND CompanySeq = @CompanySeq) AS C

WHERE A.CompanySeq = @CompanySeq

AND A.WorkReportSeq = @WorkReportSeq

SELECT @WorkDate = WorkDate

FROM _TPDSFCWorkReport WITH(NOLOCK)

WHERE CompanySeq = @CompanySeq

AND WorkReportSeq = @WorkReportSeq

-- 수불마감체크

TRUNCATE TABLE #TLGClosingCheck

INSERT INTO #TLGClosingCheck

SELECT A.IDX_NO AS IDX_NO, LEFT(@WorkDate,6), A.ItemSeq, A.WHSeq,

0, 0, ''

FROM #MESData AS A

WHERE A.InputQty \<> 0

EXEC _SWLGClosingCheck @CompanySeq, @LanguageSeq

BEGIN

@Results = Result

FROM #TLGClosingCheck

WHERE Status \<> 0

GOTO ERR_PROC

END

IF EXISTS(SELECT TOP 1 1 FROM #MESData AS A LEFT OUTER JOIN _TDAWH B ON B.CompanySeq = @CompanySeq AND A.WHSeq = B.WHSeq WHERE B.WHSeq IS NULL)

BEGIN

SELECT @Status = 'ERR', @Results = 'ERP에 해당 창고코드가 존재하지 않습니다.'

GOTO ERR_PROC

END

IF EXISTS(SELECT TOP 1 1 FROM #MESData AS A WHERE A.InputQty \<= 0)

BEGIN

SELECT @Status = 'ERR', @Results = '수량이 입력되지 않았습니다.'

GOTO ERR_PROC

END

INSERT INTO #TMESMatInput

(

CompanySeq, WorkingTag , WorkReportSeq, ItemSerl,

InputDate , MatItemSeq , MatUnitSeq , Qty ,

ProcSeq , LastUserSeq, Remark , WHSeq ,

ProcDateTime,Status,Result

)

SELECT @CompanySeq, @WorkingTag, @WorkReportSeq, ROW_NUMBER() OVER(PARTITION BY @WorkReportSeq ORDER BY @WorkReportSeq),

@WorkDate , A.ItemSeq , A.UnitSeq , dbo._FWPDGetDecPointQty(A.InputQty, @MatDec, B.SMDecPointSeq), -- 20230303.수량 소수점 세팅값으로 변경

C.ProcSeq , @UserSeq , '(MES 투입처리)' AS Remark, A.WHSeq AS WHSeq ,

GETDATE(), 0, ''

FROM #MESData AS A WITH(NOLOCK)

LEFT OUTER JOIN _TDAUnit AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND A.UnitSeq = B.UnitSeq

JOIN _TPDSFCWorkReport AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

AND C.WorkReportSeq = @WorkReportSeq

DECLARE @MAXItemSerl INT

SELECT @MAXItemSerl = MAX(ItemSerl)

FROM _TPDSFCMatinput WITH(NOLOCK)

WHERE WorkReportSeq = @WorkReportSeq

UPDATE #TMESMatInput

SET ItemSerl = ISNULL(@MAXItemSerl,0) + ItemSerl

BEGIN TRAN

INSERT INTO _TPDSFCMatInput

(

CompanySeq , WorkReportSeq , ItemSerl , InputDate , MatItemSeq ,

MatUnitSeq , Qty , StdUnitQty , RealLotNo , SerialNoFrom ,

ProcSeq , AssyYn , IsConsign , GoodItemSeq , InputType ,

IsPaid , IsPjt , PjtSeq , WBSSeq , LastUserSeq ,

LastDateTime , Remark , ProdWRSeq , IsReInput , WHSeq ,

PgmSeq

)

SELECT A.CompanySeq , A.WorkReportSeq, A.ItemSerl , A.InputDate , A.MatItemSeq ,

A.MatUnitSeq , A.Qty , A.Qty , '' AS RealLotNo , '' AS SerialNoFrom ,

A.ProcSeq , CASE WHEN D.IsLastProc = '1' THEN '0' ELSE '1' END, '' AS IsConsign, D.GoodItemSeq , 6042002 AS InputType,

'0' AS IsPaid, '0' AS IsPjt , 0 AS PjtSeq , 0 AS WBSSeq , ISNULL(A.LastUserSeq,0) ,

GETDATE() , A.Remark , 0 AS ProdWRSeq , '0' AS IsReInput, A.WHSeq ,

80420010 AS PgmSeq

FROM #TMESMatInput AS A

JOIN _TPDSFCWorkReport AS D WITH(NOLOCK) ON A.CompanySeq = D.CompanySeq

AND A.WorkReportSeq = D.WorkReportSeq

IF @@ERROR \<> 0

BEGIN

SELECT @Status = 'ERR',

@Results = '(자재투입 생성) 오류가 발생 하였습니다. (' + ERROR_MESSAGE() + ')'

ROLLBACK TRAN

GOTO ERR_PROC

END

-- 2-2) 자재투입 수불처리

(

GOTO ERR_PROC

END

SELECT @Status = 'OK',

@Results = '투입처리 되었습니다.'

-- Log Insert

UPDATE A

SET Status = @Status,

Results = @Results

FROM mcm_SMESWorkReportMatProclog AS A

WHERE (1=1)

AND A.LogDateTime = @CurrDateTime

COMMIT TRAN

RETURN
