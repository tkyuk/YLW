-- 같은 데이터라도 신규 입력하면 MES_PK_ProductionOrder가 새로 생성되어 넘어오기 때문에 MES_PK_ProductionOrder 기준 JOIN

-- 20240904 MES에서 계속 중복된 MES_PK_ProductionOrder로 넘어오기 때문에, 실테이블에 존재하는 WORKREPORTSEQ로 넣어줘야한다.

UPDATE A

SET WorkReportSeq = C.ERPWorkReportSeq

FROM #TMESWorkReportInterface AS A

JOIN gnst_TMESWorkReportIF AS B ON B.SETID = A.SETID

JOIN gnst_TMESWorkReportIF AS C ON C.MES_PK_ProductionOrder = B.MES_PK_ProductionOrder

AND C.ERPStatus = 1 -- 정상처리된 건.

JOIN _TPDSFCWorkReport AS D WITH(NOLOCK) ON C.CompanySeq = D.CompanySeq

AND C.ERPWorkReportSeq = D.WorkReportSeq -- 존재하는 건.

WHERE A.WorkReportSeq = 0
