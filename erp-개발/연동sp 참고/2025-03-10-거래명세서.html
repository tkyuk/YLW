거래명세서

CREATE VIEW smc_VMESTSLInvoiceItem

AS

SELECT A.CompanySeq AS CompanySeq, -- 법인내부코드

C.CompanyName AS CompanyName, -- 법인명

A.InvoiceSeq AS InvoiceSeq, -- 거래명세서내부코드

B.InvoiceSerl AS InvoiceSerl, -- 거래명세서순번

A.InvoiceNo AS InvoiceNo, -- 거래명세서번호

A.InvoiceDate AS InvoiceDate, -- 거래명세서일

D.MinorName AS SMExpKindName, -- Local구분

E.MinorName AS UMOutKindName, -- 출고구분

F.DeptName AS DeptName, -- 부서

G.EmpName AS EmpName, -- 담당자

CASE WHEN ISNULL(J.CustName,'') = ''

THEN H.CustName

ELSE J.CustName

END AS BillCustName, -- 청구처

H.CustName AS CustName, -- 거래처

K.ItemName AS ItemName, -- 품명

K.ItemNo AS ItemNo, -- 품번

K.Spec AS Spec, -- 규격

B.ItemSeq AS ItemSeq, -- 품번내부코드

L.UnitName AS UnitName, -- 판매단위

M.WHName AS WHName, -- 창고

B.Remark AS RemarkI, -- 품목특이사항

ISNULL(N.ToQty, 0) AS SalesQty, -- 매출수량

O.AssetName AS AssetName, -- 품목자산분류

B.Dummy2 AS Location, -- 본사/광주구분

R.PONo AS PONo, -- P/O No.

B.Dummy5 AS SerialNo -- SerialNo

FROM _TSLInvoice AS A

JOIN _TSLInvoiceItem AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.InvoiceSeq = A.InvoiceSeq

LEFT OUTER JOIN _TCACompany AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

LEFT OUTER JOIN _TDASMinor AS D WITH(NOLOCK) ON D.CompanySeq = A.CompanySeq

AND D.MinorSeq = A.SMExpKind

LEFT OUTER JOIN _TDAUMinor AS E WITH(NOLOCK) ON E.CompanySeq = A.CompanySeq

AND E.MinorSeq = A.UMOutKind

LEFT OUTER JOIN _TDADept AS F WITH(NOLOCK) ON F.CompanySeq = A.CompanySeq

AND F.DeptSeq = A.DeptSeq

LEFT OUTER JOIN _TDAEmp AS G WITH(NOLOCK) ON G.CompanySeq = A.CompanySeq

AND G.EmpSeq = A.EmpSeq

LEFT OUTER JOIN _TDACust AS H WITH(NOLOCK) ON H.CompanySeq = A.CompanySeq

AND H.CustSeq = A.CustSeq

LEFT OUTER JOIN _TDACustGroup AS I WITH(NOLOCK) ON I.CompanySeq = A.CompanySeq

AND I.CustSeq = A.CustSeq

AND I.UMCustGroup = 8014002

LEFT OUTER JOIN _TDACust AS J WITH(NOLOCK) ON J.CompanySeq = I.CompanySeq

AND J.CustSeq = I.UpperCustSeq

LEFT OUTER JOIN _TDAItem AS K WITH(NOLOCK) ON K.CompanySeq = B.CompanySeq

AND K.ItemSeq = B.ItemSeq

LEFT OUTER JOIN _TDAUnit AS L WITH(NOLOCK) ON L.CompanySeq = B.CompanySeq

AND L.UnitSeq = B.UnitSeq

LEFT OUTER JOIN _TDAWH AS M WITH(NOLOCK) ON M.CompanySeq = B.CompanySeq

AND M.WHSeq = B.WHSeq

LEFT OUTER JOIN (SELECT CompanySeq, FromSeq, FromSerl, FromTableSeq, ToTableSeq , SUM(ADD_DEL * ToQty) AS ToQty

FROM _TCOMSourceDaily

GROUP BY CompanySeq, FromSeq, FromSerl, FromTableSeq, ToTableSeq

) AS N ON N.CompanySeq = B.CompanySeq

AND N.FromSeq = B.InvoiceSeq

AND N.FromSerl = B.InvoiceSerl

AND N.FromTableSeq = 18

AND N.ToTableSeq = 20

LEFT OUTER JOIN _TDAItemAsset AS O WITH(NOLOCK) ON O.CompanySeq = K.CompanySeq

AND O.AssetSeq = K.AssetSeq

LEFT OUTER JOIN _TCOMSourceDaily AS P WITH(NOLOCK) ON P.CompanySeq = B.CompanySeq

AND P.ToSeq = B.InvoiceSeq

AND P.ToSerl = B.InvoiceSerl

AND P.FromTableSeq = 19

AND P.ToTableSeq = 18

LEFT OUTER JOIN _TSLOrder AS R WITH(NOLOCK) ON R.CompanySeq = P.CompanySeq

AND R.OrderSeq = P.FromSeq

WHERE A.IsDelvCfm = '0' -- 미처리
