사업자번호

SELECT A.CompanySeq,

A.TaxUnit,

A.TaxNo,

A.TaxNoAlias,

A.TaxName,

A.TaxEngName,

A.Owner,

A.OwnerEngName,

A.BizType,

A.BizItem,

A.Zip,

ISNULL(dbo._FWCOMDecrypt(A.TelNo, '_TDATaxUnit', 'TelNo', A.CompanySeq), '') AS TelNo,

A.FaxNo,

D.Addr,

E.Addr AS AddrEng,

A.VATRptAddr,

A.BizRegDate,

A.TaxSumPaymentNo,

A.TaxNoSerl,

A.Remark,

A.LastDateTime

FROM _TDATaxUnit AS A WITH(NOLOCK)

LEFT OUTER JOIN _VWCOMAddress AS D ON A.ConTactAddressSeq = D.AddressSeq AND D.SMAddressTypeSeq = 500052002 AND A.Companyseq = D.CompanySeq

LEFT OUTER JOIN _VWCOMAddress AS E ON A.ConTactAddressSeq = E.AddressSeq AND E.SMAddressTypeSeq = 500052003 AND A.Companyseq = E.CompanySeq
