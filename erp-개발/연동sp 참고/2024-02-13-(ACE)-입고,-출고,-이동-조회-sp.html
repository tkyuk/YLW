(ACE) 입고, 출고, 이동 조회 sp

/*

1. 업무구분에 따라 입고대상, 출고대상, 이동대상을 조회해야 합니다.

(1) 입고대상

- 구매발주 미납내역 조회

- 외주발주 미납내역 조회

- 수입BL 미입고내역 조회

(2) 출고대상

- 출하의뢰,수출출하의뢰 미진행내역 조회(거래명세서, 수출Invoice 미진행)

(3) 이동대상

- 이동입력, 자재이동입력 미진행내역 조회

*/

CREATE PROCEDURE TNT_SLGPDAProcReqQry

@CompanySeq INT = 1, -- 법인코드 법인코드 (1:디폴트)

@WorkKind INT , -- 구분 업무구분 (1:입고, 2:출고, 3:이동)

@DateFr NCHAR(8) , -- 일자From 일자From

@DateTo NCHAR(8) , -- 일자To 일자To

@QryText NVARCHAR(200) , -- 조회값 조회값(입고,출고:거래처, 이동: 입고창고)

@PDANo NVARCHAR(30) -- PDA일련번호 ERP의 PDA 기준 번호로 출고창고 체크용도 (PDA 환경설정의 번호)

AS

-- PDA연동 대상조회 SP                                

SET NOCOUNT ON

SELECT @CompanySeq = ISNULL(@CompanySeq , 0),

@WorkKind = ISNULL(@WorkKind , 0),

@DateFr = ISNULL(@DateFr , ''),

@DateTo = ISNULL(@DateTo , ''),

@QryText = ISNULL(@QryText , ''),

@PDANo = ISNULL(@PDANo , '')

IF @CompanySeq = 0 SELECT @CompanySeq = 1

IF @DateTo = '' SELECT @DateTo = '29991231'

DECLARE @WHSeq INT

SELECT @WHSeq = B.ValueSeq

FROM _TDAUMinor AS A WITH(NOLOCK)

LEFT OUTER JOIN _TDAUMinorValue AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq -- 소속창고

AND A.MinorSeq = B.MinorSeq

AND B.Serl = 15000001

WHERE A.CompanySeq = @CompanySeq

AND A.MajorSeq = 1001005 -- tnt_PDA일련번호

AND A.MinorName = @PDANo

IF @WorkKind = 1 -- 입고

BEGIN

SELECT A.PODate AS ReqDate ,

A.PONo AS ReqNo ,

C.CustName AS CustName

FROM _TPUORDPO AS A WITH(NOLOCK)

JOIN _TPUORDPOItem AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.POSeq = B.POSeq

LEFT OUTER JOIN _TDACust AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.CustSeq = C.CustSeq

OUTER APPLY (SELECT SUM(Z.ToSTDQty) AS InQty

FROM _TCOMSourceDailyTrace AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = A.CompanySeq

AND Z.FromSeq = A.POSeq

AND Z.FromSerl = B.POSerl

AND Z.FromTableSeq = 13 -- _TPUORDPOItem         구매발주(수입Order)

AND Z.ToTableSeq = 9 -- _TPUDelvInItem 구매입고

) AS E

WHERE A.CompanySeq = @CompanySeq

AND A.PODate BETWEEN @DateFr AND @DateTo

AND A.SMImpType IN (8008001, 8008002, 8008003)

AND B.WhSeq = @WHSeq

AND (B.StdUnitQty - ISNULL(E.InQty,0)) > 0

AND (@QryText = '' OR C.CustName LIKE @QryText + '%')

GROUP BY A.PODate ,

A.PONo ,

C.CustName

UNION ALL

SELECT A.PODate AS ReqDate ,

A.OSPPONo AS ReqNo ,

C.CustName AS CustName

FROM _TPDOSPPO AS A WITH(NOLOCK)

JOIN _TPDOSPPOItem AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.OSPPOSeq = B.OSPPOSeq

LEFT OUTER JOIN _TDACust AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.CustSeq = C.CustSeq

OUTER APPLY (SELECT SUM(Z.ToSTDQty) AS InQty

FROM _TCOMSourceDailyTrace AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = A.CompanySeq

AND Z.FromSeq = A.OSPPOSeq

AND Z.FromSerl = B.OSPPOSerl

AND Z.FromTableSeq = 4        -- _TPDOSPPOItem 외주발주

AND Z.ToTableSeq = 2        -- _TPDOSPDelvInItem 외주입고

) AS E

WHERE A.CompanySeq = @CompanySeq

AND A.PODate BETWEEN @DateFr AND @DateTo

AND B.WhSeq = @WHSeq

AND (B.StdUnitQty - ISNULL(E.InQty,0)) > 0

AND (@QryText = '' OR C.CustName LIKE @QryText + '%')

GROUP BY A.PODate ,

A.OSPPONo,

C.CustName

UNION ALL

SELECT A.BLDate AS ReqDate ,

A.BLNo AS ReqNo ,

C.CustName AS CustName

FROM _TUIImpBL AS A WITH(NOLOCK)

JOIN _TUIImpBLItem AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.BLSeq = B.BLSeq

LEFT OUTER JOIN _TDACust AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.CustSeq = C.CustSeq

OUTER APPLY (SELECT SUM(Z.ToSTDQty) AS InQty

FROM _TCOMSourceDailyTrace AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = A.CompanySeq

AND Z.FromSeq = A.BLSeq

AND Z.FromSerl = B.BLSerl

AND Z.FromTableSeq = 43        -- _TUIImpBLItem        수입BL

AND Z.ToTableSeq = 49        -- _TUIImpDelvItem 수입입고

) AS E

WHERE A.CompanySeq = @CompanySeq

AND A.BLDate BETWEEN @DateFr AND @DateTo

AND B.WhSeq = @WHSeq

AND (B.StdQty - ISNULL(E.InQty,0)) > 0

AND (@QryText = '' OR C.CustName LIKE @QryText + '%')

GROUP BY A.BLDate ,

A.BLNo ,

C.CustName

ORDER BY 1,2,3

END

ELSE IF @WorkKind = 2 -- 출고

BEGIN

SELECT A.DVReqDate AS ReqDate ,

A.DVReqNo AS ReqNo ,

C.CustName AS CustName

FROM _TSLDVReq AS A WITH(NOLOCK)

JOIN _TSLDVReqItem AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.DVReqSeq = B.DVReqSeq

LEFT OUTER JOIN _TDACust AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.CustSeq = C.CustSeq

OUTER APPLY (SELECT SUM(Z.ToSTDQty) AS InQty

FROM _TCOMSourceDailyTrace AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = A.CompanySeq

AND Z.FromSeq = A.DVReqSeq

AND Z.FromSerl = B.DVReqSerl

AND Z.FromTableSeq = 16        -- _TSLDVReqItem        출하의뢰(반품요청)

AND Z.ToTableSeq = 18        -- _TSLInvoiceItem        거래명세서(수출Invoice)

) AS E

WHERE A.CompanySeq = @CompanySeq

AND A.DVReqDate BETWEEN @DateFr AND @DateTo

AND B.WhSeq = @WHSeq

AND (B.StdQty - ISNULL(E.InQty,0)) > 0

AND (@QryText = '' OR C.CustName LIKE @QryText + '%')

GROUP BY A.DVReqDate ,

A.DVReqNo ,

C.CustName

ORDER BY 1,2,3

END

ELSE IF @WorkKind = 3 -- 이동

BEGIN

SELECT A.ReqDate AS ReqDate ,

A.ReqNo AS ReqNo ,

C.WHName AS CustName

FROM _TLGInOutReq AS A WITH(NOLOCK)

JOIN _TLGInOutReqItem AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.ReqSeq = B.ReqSeq

LEFT OUTER JOIN _TDAWH AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.InWHSeq = C.WHSeq

OUTER APPLY (SELECT SUM(Z.ToSTDQty) AS InQty

FROM _TCOMSourceDailyTrace AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = A.CompanySeq

AND Z.FromSeq = A.ReqSeq

AND Z.FromSerl = B.ReqSerl

AND Z.FromTableSeq = 14        -- _TLGInOutDailyItem        입출고

AND Z.ToTableSeq = 15        -- _TLGInOutReqItem        입출고요청

) AS E

WHERE A.CompanySeq = @CompanySeq

AND A.ReqDate BETWEEN @DateFr AND @DateTo

AND A.OutWHSeq = @WHSeq

AND (B.StdQty - ISNULL(E.InQty,0)) > 0

AND A.InOutReqType IN (80,82) -- 80 이동요청 82 자재이동요청

AND (@QryText = '' OR C.WHName LIKE @QryText + '%')

GROUP BY A.ReqDate ,

A.ReqNo ,

C.WHName

ORDER BY 1,2,3

END

RETURN

/*

GO

EXEC TNT_SLGPDAProcReqQry

@CompanySeq = 1, -- 법인코드 법인코드 (1:디폴트)

@WorkKind = 2, -- 구분 업무구분 (1:입고, 2:출고, 3:이동)

@DateFr = '20210101', -- 일자From 일자From

@DateTo = '20240207', -- 일자To 일자To

@QryText = '' , -- 조회값 조회값(입고,출고:거래처, 이동: 입고창고)

@PDANo = 'PDA0001'

*/
