WITH CTE_OrgDept(CompanySeq, OrgType, DeptSeq, UppDeptSeq, Seq, BegDate, EndDate, UMDeptLevel, Remark, DispSeq, OrgCd)

AS(

SELECT A.CompanySeq, A.OrgType, A.DeptSeq, A.UppDeptSeq, A.Seq, A.BegDate, A.EndDate, A.UMDeptLevel, A.Remark, A.DispSeq, CAST('000' as nvarchar(50))

FROM _THROrgDept AS A WITH(NOLOCK)

WHERE A.BegDate \<= CONVERT(NCHAR(8),GETDATE(),112)

AND A.EndDate >= CONVERT(NCHAR(8),GETDATE(),112)

AND A.UppDeptSeq = 0

UNION ALL

SELECT A.CompanySeq, A.OrgType, A.DeptSeq, A.UppDeptSeq, A.Seq, A.BegDate, A.EndDate, A.UMDeptLevel, A.Remark, A.DispSeq,

CAST(CTE.OrgCd+right('000'+convert(nvarchar(3),A.DispSeq),3) as nvarchar(50))

FROM _THROrgDept A

INNER JOIN CTE_OrgDept AS CTE ON CTE.DeptSeq = A.UppDeptSeq

AND CTE.OrgType = A.OrgType

AND CTE.CompanySeq = A.CompanySeq

WHERE A.BegDate \<= CONVERT(NCHAR(8),GETDATE(),112)

AND A.EndDate >= CONVERT(NCHAR(8),GETDATE(),112)

)

-- 조직도 조회 쿼리

SELECT A.CompanySeq AS CompanySeq -- 법인코드

,A.OrgType AS OrgType -- 조직구분(조직구분등록화면)

,ISNULL(C.DeptName , '') AS DeptName -- 부서

,ISNULL(A.DeptSeq , 0) AS DeptSeq -- 부서코드

,ISNULL(A.UppDeptSeq, 0) AS UppDeptSeq -- 상위부서코드

,ISNULL(D.DeptName, '') AS UppDeptName -- 상위부서

,ISNULL(A.Remark , '') AS Remark -- 비고

,ISNULL(A.Seq , 0) AS Seq                 -- 일련번호

,ISNULL(A.DispSeq         , 0) AS DispSeq         -- 순번

,A.LastDateTime                        AS LastDateTime        -- 최종시간

,B.OrgCd AS OrgCd -- 조직레벨코드

FROM _THROrgDept                 AS A

INNER JOIN CTE_OrgDept AS B ON A.CompanySeq = B.CompanySeq

AND A.OrgType = B.OrgType

AND A.DeptSeq = B.DeptSeq

LEFT JOIN _TDADeptHist AS C WITH(NOLOCK) ON A.DeptSeq = C.DeptSeq

AND A.CompanySeq = C.CompanySeq

AND (C.BegDate \<= CONVERT(NCHAR(8),GETDATE(),112) AND C.EndDate >= CONVERT(NCHAR(8),GETDATE(),112))

LEFT JOIN _TDADeptHist AS D WITH(NOLOCK) ON A.UppDeptSeq = D.DeptSeq

AND A.CompanySeq = D.CompanySeq

AND (D.BegDate \<= CONVERT(NCHAR(8),GETDATE(),112) AND D.EndDate >= CONVERT(NCHAR(8),GETDATE(),112))

WHERE (A.BegDate \<= CONVERT(NCHAR(8),GETDATE(),112) AND A.EndDate >= CONVERT(NCHAR(8),GETDATE(),112))
