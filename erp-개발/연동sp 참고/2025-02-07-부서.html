CREATE OR ALTER VIEW mcm_VMESDADept

AS

SELECT ISNULL(A.CompanySeq , 0) AS CompanySeq , -- 법인코드

ISNULL(B.CompanyName ,'') AS CompanyName , -- 법인명

ISNULL(A.DeptSeq , 0) AS DeptSeq , -- 부서내부코드

ISNULL(A.DeptName ,'') AS DeptName , -- 부서명

ISNULL(A.AbrDeptName ,'') AS AbrDeptName , -- 부서약명R

ISNULL(A.EngDeptName ,'') AS EngDeptName , -- 영문명

ISNULL(A.AbrEngDeptName,'') AS AbrEngDeptName, -- 영문약명

ISNULL(A.BegDate ,'') AS BegDate , -- 부서시작일

ISNULL(A.EndDate ,'') AS EndDate , -- 부서종료일

ISNULL(A.BizUnit , 0) AS BizUnit , -- 사업단위 - View_사업단위

ISNULL(C.BizUnitName ,'') AS BizUnitName , -- 사업단위명

ISNULL(A.LastUserSeq , 0) AS LastUserSeq , -- 최종등록자

ISNULL(A.LastDateTime ,'') AS LastDateTime -- 최종등록일자

FROM _TDADept AS A WITH(NOLOCK)

LEFT OUTER JOIN _TCACompany AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

LEFT OUTER JOIN _TDABizUnit AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.BizUnit = C.BizUnit
