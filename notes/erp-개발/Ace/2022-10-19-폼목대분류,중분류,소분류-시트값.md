LEFT OUTER JOIN \_TDAItem                 AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.ItemSeq = A.ItemSeq

LEFT OUTER JOIN \_TDAItemClass AS E WITH(NOLOCK) ON E.CompanySeq = B.CompanySeq

AND E.ItemSeq = B.ItemSeq

AND E.UMajorItemClass IN (2001, 2004)

LEFT OUTER JOIN \_VDAItemClass AS D WITH(NOLOCK) ON D.CompanySeq = E.CompanySeq

AND D.ItemClassSSeq = E.UMItemClass

WHERE (@CustSeq = 0 OR @CustSeq = A.CustSeq ) -- 거래처

AND (@AssetSeq = 0 OR @AssetSeq = B.AssetSeq ) -- 품목자산분류

AND (@ItemClassLSeq = 0 OR @ItemClassLSeq = D.ItemClassLSeq ) -- 대분류

AND (@ItemClassMSeq = 0 OR @ItemClassMSeq = D.ItemClassMSeq ) -- 중분류

AND (@ItemClassSSeq = 0 OR @ItemClassSSeq = D.ItemClassSSeq ) -- 소분류

AND (@ItemName = '' OR B.ItemName LIKE @ItemName + N'%') -- 품명

AND (@ItemNo = '' OR B.ItemNo LIKE @ItemNo + N'%') -- 품번

AND (@Spec = '' OR B.Spec LIKE @Spec + N'%') -- 규격

AND (CASE WHEN @ShortQtyQuery = '1' THEN A.Qty ELSE 1 END)

\< (CASE WHEN @ShortQtyQuery = '1' THEN 0 ELSE 2 END) -- - 부족수량만 조회 버튼 클릭시 재고가 (-)인 항목만 조회

TextBox        품목대분류        DataBlock3        UMItemClassLName                품목대분류        DIS        100

FloatBox        품목대분류코드        DataBlock3        UMItemClassL                품목대분류코드        DIS;HID        100

TextBox        품목중분류        DataBlock3        UMItemClassMName                품목중분류        DIS        100

FloatBox        품목중분류코드        DataBlock3        UMItemClassM                품목중분류코드        DIS;HID        100

TextBox        품목소분류        DataBlock3        UMItemClassName                품목소분류        DIS        100

FloatBox        품목소분류코드        DataBlock3        UMItemClass                품목소분류코드        DIS;HID        100

TextBox        품목자산분류        DataBlock3        AssetName                품목자산분류        DIS        100

FloatBox        품목자산분류코드        DataBlock3        AssetSeq                품목자산분류코드        DIS;HID        100

 

ISNULL(F.ItemClassLName, '') AS UMItemClassLName,

ISNULL(F.ItemClassLSeq , 0) AS UMItemClassL ,

ISNULL(F.ItemClassMName, '') AS UMItemClassMName,

ISNULL(F.ItemClassMSeq , 0) AS UMItemClassM ,

ISNULL(F.ItemClassSName, '') AS UMItemClassName ,

ISNULL(F.ItemClassSSeq , 0) AS UMItemClass ,

ISNULL(G.AssetName , '') AS AssetName ,

LEFT OUTER JOIN \_TDAItemClass AS E WITH(NOLOCK) ON E.CompanySeq = C.CompanySeq

AND E.ItemSeq = C.ItemSeq

AND E.UMajorItemClass IN (2001, 2004)

LEFT OUTER JOIN \_VDAItemClass AS F WITH(NOLOCK) ON F.CompanySeq = E.CompanySeq

AND F.ItemClassSSeq = E.UMItemClass

LEFT OUTER JOIN \_TDAItemAsset AS G WITH(NOLOCK) ON G.CompanySeq = C.CompanySeq

AND G.AssetSeq = C.AssetSeq
