# 번호 매기기

UPDATE A

SET CustSeq = ISNULL(B.MaxSerl, 0) + ISNULL(B.Serl, 0)

FROM \#BIZ_OUT_DataBlock2 AS A

JOIN ( SELECT A.IDX_NO,

A.QuoteSeq,

MAX(ISNULL(B.CustSeq, 0)) AS MaxSerl,

ROW_NUMBER() OVER(PARTITION BY A.QuoteSeq ORDER BY A.QuoteSeq) AS Serl

FROM \#BIZ_OUT_DataBlock2 AS A

LEFT OUTER JOIN gcos_TPUQuoteReqCust AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND A.QuoteSeq = B.QuoteSeq

WHERE A.WorkingTag = 'A'

AND A.Status = 0

GROUP BY A.IDX_NO, A.QuoteSeq

)

AS B ON A.IDX_NO = B.IDX_NO

WHERE A.WorkingTag = 'A'

AND A.Status = 0
