# 서비스 요청 패치

작업단위 : 서비스요청_sys

 

프로그램        FrmSYSServiceReq_sys

비즈니스서비스        Genuine.sys.Call.BisSYSServiceReq_sys

스크립트(SP,TR,View)        sys_SSYSServiceReqGetSSO

 

추가        TCTService_sys        서비스테이블        업무        IF NOT EXISTS (SELECT \* FROM Sysobjects WHERE Name = 'TCTService_sys' AND xtype = 'U' )BEGIN

CREATE TABLE TCTService_sys

(

CustSeq CHAR(12)

)

END

추가        TCTService_sys        서비스테이블        업무        INSERT INTO TCTService_sys(CustSeq) SELECT 'S20230000025'

 

<span class="mark">TCTService_sys 테이블에 고객사에 맞는 거래처코드가 들어가야함</span>

 

 

SELECT CUSTCD,CustName,SiteInit FROM sys_TSYSCust

WHERE (CustName LIKE '%녹수%' OR CustName LIKE '%삼신%' )

AND CustName LIKE '%ACE%'

ORDER BY 2

 

 

 

UPDATE SSV.DBO.TCTService_sys

SET CustSeq = 'S20230000034'

 

UPDATE SSK.DBO.TCTService_sys

SET CustSeq = 'S20230000035'

 

UPDATE SSD.DBO.TCTService_sys

SET CustSeq = 'S20230000036'

 

UPDATE SST.DBO.TCTService_sys

SET CustSeq = 'S20230000037'
