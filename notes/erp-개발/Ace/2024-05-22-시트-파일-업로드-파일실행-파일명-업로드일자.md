# 시트 파일 업로드 / 파일실행 / 파일명 / 업로드일자

-- 조회 시 파일에 대한 업데이트문 필요

DECLARE @SiteInit NVARCHAR(100),

@SQL NVARCHAR(MAX)

 

SELECT @SiteInit = LTRIM(RTRIM(SiteInit))

FROM \_TCACompanyConfig WITH(NOLOCK)

WHERE CompanySeq = @CompanySeq

 

SELECT @SQL = ''

 

SELECT @SQL = @SQL + 'UPDATE A '

SELECT @SQL = @SQL + ' SET FileName = ISNULL(B.FileName , '''') , FileDate = ISNULL(B.AttachDateTime , '''') '

SELECT @SQL = @SQL + ' FROM \#Result AS A '

SELECT @SQL = @SQL + ' OUTER APPLY(SELECT MAX(Z.FileName) AS FileName , MAX(Z.AttachDateTime) AS AttachDateTime '

SELECT @SQL = @SQL + ' FROM ' + @SiteInit + 'common.dbo.\_TCAAttachFileData AS Z WITH(NOLOCK) '

SELECT @SQL = @SQL + ' WHERE Z.AttachFileSeq = A.FileSeq'

SELECT @SQL = @SQL + ' ) AS B'

 

EXEC(@SQL)

 

SELECT A.YM AS YM

,A.PUQty AS PUQty

,A.RemainQty AS RemainQty

,A.SalesQty AS SalesQty

,A.Gravity AS Gravity

,A.FileSeq AS FileSeq

,A.FileName AS FileName

,CASE WHEN ISNULL(A.FileName, '') = '' THEN '0' ELSE '1' END AS IsFile

,CASE WHEN ISNULL(A.FileName, '') = '' THEN NULL ELSE FileDate END AS FileDate -- 1900-01-01로 조회되서 NULL로 수정 JHMOON

,ISNULL(A.Seq , 0) AS Seq

FROM \#Result AS A

ORDER BY A.YM
