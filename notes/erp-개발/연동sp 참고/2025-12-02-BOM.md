# BOM

ALTER VIEW gnst_VPDBOMView

AS

SELECT A.CompanySeq,

A.BizUnit,

A.ProcRev,

A.ItemSeq,

B.ItemName,

B.ItemNo,

B.Spec,

A.ProcSeq,

C.ProcName,

C.ProcSName AS ProcCode,

E.WorkCenterSeq,

E.WorkCenterName,

G.UnitSeq,

G.UnitName,

A.MatItemSeq AS SubItemSeq,

H.ItemName AS SubItemName,

H.ItemNo AS SubItemNo,

H.Spec AS SubItemSpec,

I.UnitSeq AS SubUnitSeq,

I.UnitName AS SubUnitName,

A.NeedQtyNumerator,

A.NeedQtyDenominator,

'' AS KeyMaterialYN,

A.InLossRate,

A.OutLossRate,

A.Remark,

ISNULL(J1.Empid, '') AS LastEmpID,

A.LastDateTime,

--(SELECT UserName FROM \_TCAUser WITH(NOLOCK) WHERE CompanySeq = A.CompanySeq AND UserSeq = A.LastUserSeq) AS LastUserName

A.ItemSerl

FROM \_TPDROUItemProcMat AS A WITH(NOLOCK) JOIN \_TDAItem AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.ItemSeq = B.ItemSeq

LEFT OUTER JOIN \_TPDBaseProcess AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.ProcSeq = C.ProcSeq

LEFT OUTER JOIN \_TPDROUItemProcWC AS D WITH(NOLOCK) ON A.CompanySeq = D.CompanySeq

AND A.ItemSeq = D.ItemSeq

AND A.ProcRev = D.ProcRev

AND A.ProcSeq = D.ProcSeq

AND A.Serl = D.Serl

LEFT OUTER JOIN \_TPDBaseWorkCenter AS E WITH(NOLOCK) ON D.CompanySeq = E.CompanySeq

AND D.WorkcenterSeq = E.WorkCenterSeq

JOIN \_TDAItemDefUnit AS F WITH(NOLOCK) ON B.CompanySeq = F.CompanySeq

AND B.ItemSeq = F.ItemSeq

AND F.UMModuleSeq = 1116005 -- 생산 SELECT \* FROM \_TDASMinor where MajorSeq = 1116

JOIN \_TDAUnit AS G WITH(NOLOCK) ON F.CompanySeq = G.CompanySeq

AND F.STDUnitSeq = G.UnitSeq

JOIN \_TDAItem AS H WITH(NOLOCK) ON A.CompanySeq = H.CompanySeq

AND A.MatItemSeq = H.ItemSeq

JOIN \_TDAUnit AS I WITH(NOLOCK) ON A.CompanySeq = I.CompanySeq

AND A.UnitSeq = I.UnitSeq

LEFT OUTER JOIN \_TCAUser AS J WITH(NOLOCK) ON A.CompanySeq = J.CompanySeq

AND A.LastUserSeq = J.UserSeq

LEFT OUTER JOIN \_TDAEmp AS J1 WITH(NOLOCK) ON J.CompanySeq = J1.CompanySeq

AND J.EmpSeq = J1.EmpSeq
