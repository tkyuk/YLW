EXEC dbo.\_SWCOMEnv @CompanySeq, 6218, @UserSeq, @@PROCID, @Env6218 OUTPUT

EXEC dbo.\_SWCOMEnv @CompanySeq, 5, 0, @@PROCID, @MatDec OUTPUT -- 20230303.구매/자재 수량 소숫점 자릿수 추가

ADD_PROC:

SELECT @WorkReportSeq = 0

 

IF @WorkOrderSeq = 0 OR @WorkOrderSerl = 0

BEGIN

SELECT @Status = 'ERR', @Results = '작업지시 내부코드가 없습니다..'

GOTO ERR_PROC

END

 

-- 처리일자 체크

IF (IsDate(@WorkDate) \<\> 1 )

BEGIN

SELECT @Status = 'ERR',

@Results = '작업일자가 잘못된 형식입니다.'

GOTO ERR_PROC

END

 

-- 처리자 체크

IF NOT EXISTS (SELECT 1 FROM \_TDAEmp WHERE CompanySeq = 1 AND EmpSeq = @EmpSeq)

BEGIN

SELECT @Status = '2',

@Results = '처리자가 잘못 되었습니다.'

GOTO ERR_PROC

END

 

-- 작업지시 데이터 체크

IF NOT EXISTS (SELECT TOP 1 1

FROM \_TPDSFCWorkOrder WITH(NOLOCK)

WHERE CompanySeq = @CompanySeq

AND WorkOrderSeq = @WorkOrderSeq

AND WorkOrderSerl = @WorkOrderSerl

)

BEGIN

SELECT @Status = 'ERR',

@Results = '존재하지 않는 작업지시 데이터입니다.'

GOTO ERR_PROC

END

 

 

IF (SELECT WorkOrderDate

FROM \_TPDSFCWorkOrder WITH(NOLOCK)

WHERE CompanySeq = @CompanySeq

AND WorkOrderSeq = @WorkOrderSeq

AND WorkOrderSerl = @WorkOrderSerl

) \> @WorkDate

BEGIN

SELECT @Status = 'ERR',

@Results = '작업일이 작업지시일보다 같거나 이후여야 합니다.'

GOTO ERR_PROC

END

 

IF ISNULL(@ProdQty,0) \<= 0

BEGIN

SELECT @Status = 'ERR', @Results = '수량이 입력되지 않았습니다.'

GOTO ERR_PROC

END

 

 

-- 수불마감체크

TRUNCATE TABLE \#TLGClosingCheck

 

INSERT INTO \#TLGClosingCheck

SELECT A.WorkOrderSerl AS IDX_NO, LEFT(@WorkDate,6), A.GoodItemSeq,

CASE WHEN D.SMAssetGrp = 6008004 AND ISNULL(E.ProdInSemiWhSeq,0) \> 0 THEN E.ProdInSemiWhSeq

WHEN D.SMAssetGrp = 6008005 THEN E.FieldWhSeq

ELSE E.ProdInWhSeq

END,

0, 0, ''

FROM \_TPDSFCWorkOrder AS A

JOIN \_TDAItem AS C WITH(NOLOCK) ON A.GoodItemSeq = C.ItemSeq

JOIN \_TDAItemAsset AS D WITH(NOLOCK) ON C.CompanySeq = D.CompanySeq

AND C.AssetSeq = D.AssetSeq

JOIN \_TPDBaseWorkCenter AS E WITH(NOLOCK) ON A.CompanySeq = E.CompanySeq

AND A.WorkCenterSeq = E.WorkCenterSeq

WHERE A.CompanySeq = @CompanySeq

AND A.WorkOrderSeq = @WorkOrderSeq

AND A.WorkOrderSerl = @WorkOrderSerl

AND @ProdQty \<\> 0

 

EXEC \_SWLGClosingCheck @CompanySeq, @LanguageSeq

 

IF EXISTS(SELECT TOP 1 1 FROM \#TLGClosingCheck WHERE Status \<\> 0)

BEGIN

SELECT @Status = 'ERR',

@Results = Result

FROM \#TLGClosingCheck

WHERE Status \<\> 0

 

GOTO ERR_PROC

END
