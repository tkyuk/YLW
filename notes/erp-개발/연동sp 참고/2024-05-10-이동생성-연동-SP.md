# 이동생성 연동 SP

DROP PROC apro_SPDAMoveSave

GO

CREATE PROC apro_SPDAMoveSave

@CompanySeq INT,

@UserSeq INT,

@ReqSeq INT

AS

SET NOCOUNT ON;

 

INSERT INTO apro_TPDAParameterLog

(

LogType, LogDateTime, CompanySeq, UserSeq, ReqSeq

)

SELECT 'MOVE' AS LogType, GETDATE() AS LogDateTime, @CompanySeq, @UserSeq, @ReqSeq

DECLARE @Ststus INT,

@Message NVARCHAR(255),

@CurrDate NVARCHAR(8),

@PgmSeq INT,

@GroupSeq INT,

@InOutSeq INT,

@InOutNo NVARCHAR(20),

@InOutDate NVARCHAR(8),

@LotNo NVARCHAR(30),

@EmpSeq INT,

@DeptSeq INT

SELECT @CurrDate = CONVERT(NVARCHAR(8), GETDATE(), 112),

@PgmSeq = 999999

 

SELECT @Ststus = 0, @Message = '처리 되었습니다.',

@CompanySeq = ISNULL(@CompanySeq, 0),

@UserSeq = ISNULL(@UserSeq , 0),

@ReqSeq = ISNULL(@ReqSeq , 0)

SELECT @LotNo = LEFT(LTRIM(RTRIM(ISNULL(A.EnvValue, ''))), 30)

FROM \_TCOMEnv AS A WITH(NOLOCK)

WHERE A.EnvSeq = 2000347

AND A.CompanySeq = @CompanySeq

SELECT @EmpSeq = A.EmpSeq, @DeptSeq = A.DeptSeq

FROM \_TCAUser AS A WITH(NOLOCK)

WHERE A.UserSeq = @UserSeq

 

IF @CompanySeq = 0 OR @UserSeq = 0 OR @ReqSeq = 0

BEGIN

SELECT @Ststus = 99999, @Message = '필수입력 항목을 입력하지 않았습니다.'

GOTO RESULT_PROC

RETURN

END

 

CREATE TABLE \#TLGInOutDailyItem

(

IDX_NO INT IDENTITY(1, 1),

WorkingTag NVARCHAR(1),

Status INT,

MessageType INT,

Result NVARCHAR(255),

IsMstGroup NVARCHAR(1),

GroupSeq INT,

GroupSerl INT,

 

ReqSeq INT,

ReqSerl INT,

InOutSeq INT,

InOutSerl INT,

InOutType INT,

InOutNo NVARCHAR(20),

 

InOutDate NVARCHAR(8),

BizUnit INT,

ReqBizUnit INT,

DeptSeq INT,

EmpSeq INT,

OutWHSeq INT,

InWHSeq INT,

IsTrans NVARCHAR(1),

 

PJTSeq INT,

ItemSeq INT,

UnitSeq INT,

Qty DECIMAL(19,5),

STDUnitSeq INT,

STDQty DECIMAL(19,5),

InOutKind INT,

InOutDetailKind INT,

LotNo NVARCHAR(30),

Remark NVARCHAR(500)

)

INSERT INTO \#TLGInOutDailyItem

(

WorkingTag, Status, MessageType, Result, IsMstGroup,

GroupSeq,

GroupSerl,

ReqSeq, ReqSerl, InOutSeq, InOutSerl, InOutType, InOutNo,

InOutDate, BizUnit, DeptSeq, EmpSeq,

ReqBizUnit,

OutWHSeq,

InWHSeq, IsTrans,

PJTSeq, ItemSeq, UnitSeq, Qty, STDUnitSeq, STDQty,

InOutKind, InOutDetailKind, LotNo, Remark

)

SELECT 'A' AS WorkingTag, 0 AS Status, 0 AS MessageType, '' AS Result, '0' AS IsMstGroup,

DENSE_RANK() OVER(ORDER BY A.IsTrans, A.BizUnit, A.TransBizUnit, A.OutWHSeq, A.InWHSeq, A.ReqSeq) AS GroupSeq,

ROW_NUMBER() OVER(PARTITION BY A.IsTrans, A.BizUnit, A.TransBizUnit, A.OutWHSeq, A.InWHSeq, A.ReqSeq

ORDER BY A.IsTrans, A.BizUnit, A.TransBizUnit, A.OutWHSeq, A.InWHSeq, A.ReqSeq) AS GroupSerl,

A.ReqSeq, B.ReqSerl, 0 AS InOutSeq, 0 AS InOutSerl, A.InOutReqType AS InOutType, '' AS InOutNo,

@CurrDate AS InOutDate, A.BizUnit, @DeptSeq, @EmpSeq,

CASE WHEN ISNULL(A.TransBizUnit, 0) = 0 THEN A.BizUnit ELSE A.TransBizUnit END AS ReqBizUnit,

CASE WHEN ISNULL(A.OutWHSeq, 0) = 0 THEN D.WHSeq ELSE A.OutWHSeq END AS OutWHSeq,

A.InWHSeq, A.IsTrans,

CONVERT(INT, B.Dummy7) AS PJTSeq, B.ItemSeq, B.UnitSeq, B.Qty, C.UnitSeq AS STDUnitSeq, B.STDQty,

B.InOutReqKind AS InOutKind, B.InOutReqDetailKind AS InOutDetailKind, B.LotNo, B.Remark

FROM \_TLGInOutReq AS A WITH(NOLOCK)

JOIN \_TLGInOutReqItem AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.ReqSeq = A.ReqSeq

JOIN \_TDAItem AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.ItemSeq = B.ItemSeq

JOIN \_TDAWH AS D WITH(NOLOCK) ON D.CompanySeq = A.CompanySeq

AND D.BizUnit = A.BizUnit

AND D.SMWHKind = 8002026 -- \_TDASMinor WHERE MajorSeq = 8002

WHERE A.CompanySeq = @CompanySeq

AND A.ReqSeq = @ReqSeq

AND A.InOutReqType IN (80, 81, 82, 83) -- 80:제품이동/81:제품적송/82:자재이동/83:자재적송

IF @@ROWCOUNT = 0

BEGIN

SELECT @Ststus = 99999, @Message = '이동요청건이 없습니다.'

GOTO RESULT_PROC

RETURN

END

 

 

IF EXISTS(SELECT TOP 1 1

FROM \#TLGInOutDailyItem AS A

JOIN \_TCOMSourceDailyTrace AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.ToTableSeq = 14 -- \_TCOMProgTable WHERE ProgTableName = '\_TLGInOutDailyItem'

AND B.FromTableSeq = 15 -- \_TCOMProgTable WHERE ProgTableName = '\_TLGInOutReqItem'

AND B.FromSeq = A.ReqSeq

AND B.FromSerl = A.ReqSerl

WHERE A.Status = 0

)

BEGIN

SELECT @Ststus = 99999, @Message = '이미 이동생성 된 건입니다 .'

GOTO RESULT_PROC

RETURN

END

 

UPDATE A

SET LotNo = CASE WHEN LTRIM(RTRIM(ISNULL(C.PJTNo, ''))) \<\> '' THEN LTRIM(RTRIM(ISNULL(C.PJTNo, '')))

ELSE @LotNo

END

FROM \#TLGInOutDailyItem AS A

JOIN \_TDAItemStock AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.ItemSeq = A.ItemSeq

LEFT OUTER JOIN \_TPJTProject AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

AND C.PJTSeq = A.PJTSeq

WHERE A.Status = 0

AND B.IsLotMng = '1'

AND LTRIM(RTRIM(ISNULL(A.LotNo, ''))) = '' -- LOTNo가 없으면!

 

-- 수불 마감

CREATE TABLE \#TLGClosingCheck

(

IDX_NO INT ,

InOutYM NCHAR(6) ,

ItemSeq INT ,

WHSeq INT ,

MessageType INT ,

Status INT ,

Result NVARCHAR(255)

)

 

INSERT INTO \#TLGClosingCheck

(

IDX_NO, InOutYM, ItemSeq, WHSeq,

MessageType, Status, Result

)

SELECT A.IDX_NO, LEFT(A.InOutDate,6), A.ItemSeq, A.OutWHSeq,

0 AS MessageType, 0 AS Status, '' AS Result

FROM \#TLGInOutDailyItem AS A

WHERE A.Status = 0

UNION

SELECT A.IDX_NO, LEFT(A.InOutDate,6), A.ItemSeq, A.InWHSeq,

0 AS MessageType, 0 AS Status, '' AS Result

FROM \#TLGInOutDailyItem AS A

WHERE A.Status = 0

 

EXEC \_SWLGClosingCheck @CompanySeq, 1

 

UPDATE A

SET Result = B.Result,

MessageType = B.MessageType,

Status = B.Status

FROM \#TLGInOutDailyItem AS A

JOIN \#TLGClosingCheck AS B WITH(NOLOCK) ON A.IDX_NO = B.IDX_NO

WHERE A.Status = 0

AND B.Status \<\> 0

DROP TABLE \#TLGClosingCheck

-- 납품 키값생성!

SELECT @GroupSeq = 0 -- 초기화

WHILE 1=1

BEGIN

SELECT TOP 1

@GroupSeq = A.GroupSeq,

@InOutDate = A.InOutDate

FROM \#TLGInOutDailyItem AS A

WHERE A.WorkingTag = 'A'

AND A.Status = 0

AND A.GroupSeq \> @GroupSeq

ORDER BY A.GroupSeq

IF @@ROWCOUNT = 0 BREAK

EXEC \_SWCOMCreateNo 'LG', '\_TLGInOutDaily', @CompanySeq, '', @InOutDate, @InOutNo OUTPUT -- InOutNo 생성

EXEC @InOutSeq = \_SWCOMCreateSeq @CompanySeq, '\_TLGInOutDaily', 'InOutSeq', 1 -- InOutSeq 생성

 

UPDATE A

SET InOutNo = @InOutNo,

InOutSeq = @InOutSeq + 1,

InOutSerl = A.GroupSerl

FROM \#TLGInOutDailyItem AS A

WHERE A.WorkingTag = 'A'

AND A.Status = 0

AND A.GroupSeq = @GroupSeq

END

 

UPDATE A

SET IsMstGroup = '1'

FROM \#TLGInOutDailyItem AS A

CROSS APPLY(SELECT TOP 1 Z.GroupSerl

FROM \#TLGInOutDailyItem AS Z

WHERE Z.GroupSeq = A.GroupSeq

ORDER BY Z.GroupSerl ASC

) AS B

WHERE A.WorkingTag IN ('A', 'U')

AND A.Status = 0

AND A.GroupSerl = B.GroupSerl

BEGIN TRAN

INSERT INTO \_TLGInOutDaily

(

CompanySeq, InOutType, InOutSeq, BizUnit, InOutNo, FactUnit, ReqBizUnit,

DeptSeq, EmpSeq, InOutDate, WCSeq, ProcSeq, CustSeq, OutWHSeq, InWHSeq, DVPlaceSeq,

IsTrans,

IsCompleted,

CompleteDeptSeq,

CompleteEmpSeq,

CompleteDate,

InOutDetailType, Remark, Memo, IsBatch,

LastUserSeq, LastDateTime, UseDeptSeq, PgmSeq, IsAuto, FileSeq,

MDummy1, MDummy2, MDummy3, MDummy4, MDummy5,

MDummy6, MDummy7, MDummy8, MDummy9, MDummy10

)

SELECT @CompanySeq, A.InOutType, A.InOutSeq, A.BizUnit, A.InOutNo, 0 AS FactUnit, A.ReqBizUnit,

A.DeptSeq, A.EmpSeq, A.InOutDate, 0 AS WCSeq, 0 AS ProcSeq, 0 AS CustSeq, A.OutWHSeq, A.InWHSeq, 0 AS DVPlaceSeq,

A.IsTrans,

CASE WHEN A.IsTrans \<\> '1' THEN '1' ELSE '0' END AS IsCompleted,

CASE WHEN A.IsTrans \<\> '1' THEN A.DeptSeq ELSE 0 END AS CompleteDeptSeq,

CASE WHEN A.IsTrans \<\> '1' THEN A.EmpSeq ELSE 0 END AS CompleteEmpSeq,

CASE WHEN A.IsTrans \<\> '1' THEN A.InOutDate ELSE '' END AS CompleteDate,

0 AS InOutDetailType, '\[PDA 생성\]' AS Remark, '' AS Memo, '' AS IsBatch,

@UserSeq AS LastUserSeq, GETDATE() AS LastDateTime, 0 AS UseDeptSeq, @PgmSeq AS PgmSeq, '' AS IsAuto, 0 AS FileSeq,

NULL AS MDummy1, NULL AS MDummy2, NULL AS MDummy3, NULL AS MDummy4, NULL AS MDummy5,

NULL AS MDummy6, NULL AS MDummy7, NULL AS MDummy8, NULL AS MDummy9, NULL AS MDummy10

FROM \#TLGInOutDailyItem AS A

WHERE A.WorkingTag IN ('A', 'U')

AND A.Status = 0

AND A.IsMstGroup = '1'

IF @@ERROR \<\> 0

BEGIN

SELECT @Ststus = 99999, @Message = 'INSERT FAIL.'

ROLLBACK TRAN

GOTO RESULT_PROC

RETURN

END

INSERT INTO \_TLGInOutDailyItem

(

CompanySeq, InOutType, InOutSeq, InOutSerl, ItemSeq, InOutRemark, CCtrSeq, DVPlaceSeq,

InWHSeq, OutWHSeq, UnitSeq, Qty, STDQty, Amt, EtcOutAmt, EtcOutVAT,

InOutKind, InOutDetailKind, LotNo, SerialNo, IsStockSales,

OriUnitSeq, OriItemSeq, OriQty, OriSTDQty,

LastUserSeq, LastDateTime, PJTSeq, OriLotNo, PgmSeq, WorkOrderSeq,

Dummy1, Dummy2, Dummy3, Dummy4, Dummy5,

Dummy6, Dummy7, Dummy8, Dummy9, Dummy10, ValiDate

)

SELECT @CompanySeq, A.InOutType, A.InOutSeq, A.InOutSerl, A.ItemSeq, A.Remark AS InOutRemark, 0 AS CCtrSeq, 0 AS DVPlaceSeq,

A.InWHSeq, A.OutWHSeq, A.UnitSeq, A.Qty, A.STDQty, 0 AS Amt, 0 AS EtcOutAmt, 0 AS EtcOutVAT,

A.InOutKind, A.InOutDetailKind, A.LotNo, '' AS SerialNo, '' AS IsStockSales,

0 AS OriUnitSeq, 0 AS OriItemSeq, 0 AS OriQty, 0 AS OriSTDQty,

@UserSeq AS LastUserSeq, GETDATE() AS LastDateTime, A.PJTSeq, '' AS OriLotNo, @PgmSeq, NULL AS WorkOrderSeq,

'' AS Dummy1, '' AS Dummy2, '' AS Dummy3, '' AS Dummy4, '' AS Dummy5,

'' AS Dummy6, 0 AS Dummy7, 0 AS Dummy8, 0 AS Dummy9, 0 AS Dummy10, '' AS ValiDate

FROM \#TLGInOutDailyItem AS A

WHERE A.WorkingTag IN ('A', 'U')

AND A.Status = 0

IF @@ERROR \<\> 0

BEGIN

SELECT @Ststus = 99999, @Message = 'INSERT FAIL.'

ROLLBACK TRAN

GOTO RESULT_PROC

RETURN

END

-- 구매납품 / 구매발주 진행 추가

TRUNCATE TABLE \#SComSourceDailyBatch

INSERT INTO \#SComSourceDailyBatch

(

ToTableName, ToSeq, ToSerl, ToSubSerl,

FromTableName, FromSeq, FromSerl, FromSubSerl,

ToQty, ToStdQty, ToAmt, ToVAT, ToDOMAmt, ToDOMVAT,

FromQty, FromSTDQty, FromAmt, FromVAT, FromDOMAmt, FromDOMVAT

)

SELECT '\_TLGInOutDailyItem' ToTableName, B.InOutSeq ToSeq, B.InOutSerl ToSerl, B.InOutType ToSubSerl,

'\_TLGInOutReqItem' FromTableName, C.ReqSeq FromSeq, C.ReqSerl FromSerl, 0 FromSubSerl,

B.Qty ToQty, B.STDQty ToStdQty, 0 ToAmt, 0 ToVAT, 0 ToDOMAmt, 0 ToDOMVAT,

C.Qty FromQty, C.STDQty FromSTDQty, 0 FromAmt, 0 FromVAT, 0 FromDOMAmt, 0 FromDOMVAT

FROM \#TLGInOutDailyItem AS A

JOIN \_TLGInOutDailyItem AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.InOutSeq = A.InOutSeq

AND B.InOutSerl = A.InOutSerl

AND B.InOutType = A.InOutType

JOIN \_TLGInOutReqItem AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

AND C.ReqSeq = A.ReqSeq

AND C.ReqSerl = A.ReqSerl

WHERE A.WorkingTag IN ('A', 'U')

AND A.Status = 0

 

EXEC dbo.\_SWComSourceDailyBatch N'A', @CompanySeq, @UserSeq

-- 이동처리

TRUNCATE TABLE \#TLGInOutDailyBatch

TRUNCATE TABLE \#TLGInOutMonth

TRUNCATE TABLE \#TLGInOutMonthLot

INSERT INTO \#TLGInOutDailyBatch

(

WorkingTag, Status, MessageType, Result,

InOutType, InOutSeq, InOutSerl, InOutSubSerl, DataKind

)

SELECT A.WorkingTag, A.Status, A.MessageType, A.Result,

A.InOutType, A.InOutSeq, A.InOutSerl, 0 AS InOutSubSerl, 0 AS DataKind

FROM \#TLGInOutDailyItem AS A

WHERE A.Status = 0

 

EXEC dbo.\_SWLGInOutDailyDELETE @CompanySeq = @CompanySeq, @Level = 2, @LanguageSeq = 1 -- 수불 삭제

EXEC dbo.\_SWLGInOutDailyINSERT @CompanySeq = @CompanySeq, @Level = 2, @LanguageSeq = 1 -- 수불 생성

IF EXISTS(SELECT 1 FROM \#TLGInOutDailyBatch WHERE Status \<\> 0)

BEGIN

SELECT TOP 1 @Message = A.Result, @Ststus = A.Status

FROM \#TLGInOutDailyBatch AS A

WHERE A.Status \<\> 0

ROLLBACK TRAN

GOTO RESULT_PROC

RETURN

END

ELSE

BEGIN

-- 창고재고 집계

EXEC dbo.\_SWLGWHStockUPDATE @CompanySeq = @CompanySeq

-- (-)재고 체크

EXEC dbo.\_SWLGInOutMinusCheck @CompanySeq = @CompanySeq, @TableName = N'#TLGInOutDailyBatch', @LanguageSeq = 1

EXEC dbo.\_SWLGInOutLotMinusCheck @CompanySeq = @CompanySeq, @TableName = N'#TLGInOutDailyBatch', @LanguageSeq = 1

IF EXISTS(SELECT 1 FROM \#TLGInOutDailyBatch WHERE Status \<\> 0)

BEGIN

SELECT TOP 1 @Message = A.Result, @Ststus = A.Status

FROM \#TLGInOutDailyBatch AS A

WHERE A.Status \<\> 0

ROLLBACK TRAN

GOTO RESULT_PROC

RETURN

END

END

COMMIT TRAN
