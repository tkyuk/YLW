MESPOSerl NVARCHAR(100) , -- MES구매발주내부순번 필수

MESKey NVARCHAR(100) , -- MES입고내부코드 필수

MESKeySerl NVARCHAR(100) , -- MES입고내부순번 필수

BizUnit INT , -- 사업부문 디폴트 1

SMImpType INT , -- 내외자구분 8008001 내수 8008004 수입

PJTSeq INT , -- 프로젝트내부코드

--ReqQty DECIMAL(19,5) , -- 검사의뢰수량

--RejectQty DECIMAL(19,5) , -- 불합격수량

--DisuseQty DECIMAL(19,5) , -- 검사소요폐기수량

--ReqInQty DECIMAL(19,5) , -- 입고의뢰수량

--QCEmpSeq INT , -- 검사자

--QCDeptSeq INT , -- 검사부서

DelvNo NVARCHAR(100) , -- 구매입고번호

QCSeq INT -- QC내부코드 0으로 전송

,AccSeq INT

,SMQcType INT

,DelvSeq INT

,DelvSerl INT

,POSeq INT

,POSerl INT

,BuyingAccSeq INT

,DomPrice DECIMAL(19,5)

,CurVAT DECIMAL(19,5)

,DomAmt DECIMAL(19,5)

,DomVAT DECIMAL(19,5)

,IsVAT NVARCHAR(1)

)

-- WorkingTag 는 A, D 만 존재 U(수정 없음)

INSERT INTO \#MESDelvIn(

WorkingTag ,CompanySeq ,MESPOKey ,MESPOSerl ,MESKey ,MESKeySerl ,BizUnit ,SMImpType ,

DelvInDate ,DelvDate ,DeptSeq ,EmpSeq ,CustSeq ,Remark ,IsReturn ,CurrSeq ,ExRate ,

LastUserSeq ,LastDateTime ,ItemSeq ,UnitSeq ,Qty ,StdUnitSeq ,StdUnitQty ,WHSeq ,LOTNo ,PJTSeq ,

--ReqQty ,TestStartDate ,TestEndDate ,SMTestResult ,PassedQty ,RejectQty ,DisuseQty ,ReqInQty ,QCEmpSeq ,QCDeptSeq ,

ERPProcYn ,ERPProcDateTime,Status ,Result

 

,DelvInSeq

,DelvInSerl

,POSeq

,POSerl

,QCSeq

,AccSeq

,SMQcType

,DelvSeq

,DelvSerl

,BuyingAccSeq

,Price

,DomPrice

,CurAmt

,CurVAT

,DomAmt

,DomVAT

,IsVAT

)

SELECT A.WorkingTag ,@CompanySeq ,A.MESPOKey ,MESPOSerl ,MESKey ,MESKeySerl ,@BizUnit ,A.SMImpType ,

A.DelvInDate ,A.DelvInDate ,A.DeptSeq ,A.EmpSeq ,A.CustSeq ,ISNULL(A.Remark,''),A.IsReturn ,@CurrSeq ,1 AS ExRate ,

A.LastUserSeq ,GETDATE() ,A.ItemSeq ,A.UnitSeq ,A.Qty ,A.UnitSeq ,A.Qty ,A.WHSeq ,A.LOTNo ,A.PJTSeq ,

--ReqQty ,TestStartDate ,TestEndDate ,SMTestResult ,PassedQty ,RejectQty ,DisuseQty ,ReqInQty ,QCEmpSeq ,QCDeptSeq ,

0 ,'' ,0 ,'' ,

 

CASE WHEN A.WorkingTag = 'D' THEN B.DelvInSeq ELSE 0 END,

CASE WHEN A.WorkingTag = 'D' THEN B.DelvInSerl ELSE 0 END

,D.POSeq AS POSeq

,D.POSerl AS POSerl

,0 AS QCSeq

,ISNULL(J.AccSeq, 0) AS AccSeq

,CASE WHEN ISNULL(G.IsInQc, '0') = '0' THEN 6035001 ELSE 6035002 END AS SMQcType

,CASE WHEN A.WorkingTag = 'D' THEN B.ProgFromSeq ELSE 0 END AS DelvSeq --@@@@

,CASE WHEN A.WorkingTag = 'D' THEN B.ProgFromSerl ELSE 0 END AS DelvSerl --@@@@

,0 AS BuyingAccSeq

,D.Price AS Price

,D.DomPrice AS DomPrice,

CASE WHEN A.Qty = D.Qty

THEN D.CurAmt

ELSE CASE WHEN ISNULL(D.IsVAT, '') = '1'

THEN (ISNULL(A.Qty, 0) \* ISNULL(D.Price, 0)) \* 100 / (100 + 10)

ELSE ISNULL(A.Qty, 0) \* ISNULL(D.Price, 0)

END

END AS CurAmt ,

CASE WHEN A.Qty = D.Qty

THEN D.CurVAT

ELSE CASE WHEN A.SMImpType = 8008001

THEN CASE WHEN ISNULL(D.IsVAT, '') = '1'

THEN (ISNULL(A.Qty, 0) \* ISNULL(D.Price, 0)) - (ISNULL(A.Qty, 0) \* ISNULL(D.Price, 0)) \* 100 / (100 + 10)

ELSE ISNULL(A.Qty, 0) \* ISNULL(D.Price, 0) \* 0.1

END

ELSE 0

END

END AS CurVAT ,

CASE WHEN A.Qty = D.Qty

THEN D.CurAmt

ELSE CASE WHEN ISNULL(D.IsVAT, '') = '1'

THEN (ISNULL(A.Qty, 0) \* ISNULL(D.DomPrice, 0)) \* 100 / (100 + 10)

ELSE ISNULL(A.Qty, 0) \* ISNULL(D.DomPrice, 0)

END

END AS DomAmt ,

CASE WHEN A.Qty = D.Qty

THEN D.CurVAT

ELSE CASE WHEN A.SMImpType = 8008001

THEN CASE WHEN ISNULL(D.IsVAT, '') = '1'

THEN (ISNULL(A.Qty, 0) \* ISNULL(D.DomPrice, 0)) - (ISNULL(A.Qty, 0) \* ISNULL(D.DomPrice, 0)) \* 100 / (100 + 10)

ELSE ISNULL(A.Qty, 0) \* ISNULL(D.DomPrice, 0) \* 0.1

END

ELSE 0

END

END AS DomVAT

,D.IsVAT AS IsVAT

FROM IF_TPUDelvIn_CYMC AS A WITH(NOLOCK)

LEFT OUTER JOIN \_TPUDelvInItem AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND A.DelvInSeq = B.DelvInSeq

AND A.DelvInSerl = B.DelvInSerl

LEFT OUTER JOIN \_TDAItem AS F WITH(NOLOCK) ON A.CompanySeq = F.CompanySeq

AND A.ItemSeq = F.ItemSeq

LEFT OUTER JOIN \_TPDBaseItemQCType AS G WITH(NOLOCK) ON G.CompanySeq = @CompanySeq

AND G.ItemSeq = A.ItemSeq

LEFT OUTER JOIN IF_TPUORDPO_CYMC AS H WITH(NOLOCK) ON A.CompanySeq = @CompanySeq

AND A.MESPOKey = H.CD_PO

AND A.MESPOSerl = H.CD_PODTL

LEFT OUTER JOIN \_TPUORDPOItem AS D WITH(NOLOCK) ON D.CompanySeq = @CompanySeq

AND D.POSeq = H.POSeq

AND D.POSerl = H.POSerl

--LEFT OUTER JOIN \_TPUORDPOItem AS D WITH(NOLOCK) ON A.CompanySeq = @CompanySeq

-- AND A.MESPOKey = D.Memo1

-- AND A.MESPOSerl = D.Memo2

LEFT OUTER JOIN \_TPUORDPO AS C WITH(NOLOCK) ON D.CompanySeq = C.CompanySeq

AND D.POSeq = C.POSeq

LEFT OUTER JOIN \_TDAItemAssetAcc AS I WITH(NOLOCK) ON F.CompanySeq = I.CompanySeq

AND F.AssetSeq = I.AssetSeq

AND I.AssetAccKindSeq = 1

LEFT OUTER JOIN \_TDAAccount AS J WITH(NOLOCK) ON I.CompanySeq = J.CompanySeq

AND I.AccSeq = J.AccSeq

WHERE A.MESKey = @MESKey

AND ISNULL(A.ERPProcYn,0) = 0 -- 처리 안된 건 기준

ORDER BY A.WorkingTag, A.MESKeySerl

 

 

 

IF @@ROWCOUNT = 0

BEGIN

SELECT @Status = 9999,

@Results = '대상건 이 존재하지 않습니다.'

 

RETURN

 

END

@Results = Result

FROM \#MESDelvIn

WHERE Status \<\> 0

INSERT INTO \#Tmp_TCOMCloseItemCheck

(

WorkingTag, IDX_NO, DataSeq, MessageType, Status, Result,

ServiceSeq, MethodSeq, ItemSeq,

BizUnit, Date, DeptSeq,

BizUnitOld, DateOld, DeptSeqOld

)

SELECT A.WorkingTag, A.IDX_NO, A.IDX_NO DataSeq, 0 MessageType, A.Status, A.Result,

2669 ServiceSeq, 0 MethodSeq,

ISNULL(E.ItemSeq, ISNULL(B.ItemSeq, 0)),

ISNULL(G.BizUnit, ISNULL(D.BizUnit, 0)) BizUnit, ISNULL(A.DelvDate, '') Date, ISNULL(F.DeptSeq, ISNULL(C.DeptSeq, 0)) DeptSeq,

ISNULL(G.BizUnit, ISNULL(D.BizUnit, 0)) BizUnitOld, ISNULL(A.DelvDate, '') DateOld, ISNULL(F.DeptSeq, ISNULL(C.DeptSeq, 0)) DeptSeqOld

FROM \#MESDelvIn AS A

LEFT OUTER JOIN \_TPUORDPOItem AS B ON B.CompanySeq = @CompanySeq

AND A.POSeq = B.POSeq

AND A.POSerl = B.POSerl

LEFT OUTER JOIN \_TPUORDPO AS C ON B.CompanySeq = C.CompanySeq

AND B.POSeq = C.POSeq

LEFT OUTER JOIN \_TDADept AS D ON C.CompanySeq = D.CompanySeq

AND C.DeptSeq = D.DeptSeq

LEFT OUTER JOIN \_TPUDelvItem AS E ON E.CompanySeq = @CompanySeq

AND A.DelvSeq = E.DelvSeq

LEFT OUTER JOIN \_TPUDelv AS F ON E.CompanySeq = F.CompanySeq

AND E.DelvSeq = F.DelvSeq

LEFT OUTER JOIN \_TDADept AS G ON F.CompanySeq = G.CompanySeq

AND F.DeptSeq = G.DeptSeq

WHERE A.Status = 0

DECLARE @XmlData NVARCHAR(MAX)

SELECT @XmlData = CONVERT(NVARCHAR(MAX),(

SELECT \*

FROM \#Tmp_TCOMCloseItemCheck

FOR XML RAW ('DataBlock2'), ROOT('ROOT'), ELEMENTS

))

@xmlDocument = @XmlData,

@xmlFlags = 2,

@ServiceSeq = 2639,

@WorkingTag = N'',

@CompanySeq = @CompanySeq,

@LanguageSeq = 1,

@UserSeq = 1,

@PgmSeq = @PgmSeq

 

IF EXISTS(SELECT TOP 1 1 FROM \#Tmp_TCOMCloseItemCheck WHERE Status \<\> 0)

BEGIN

SELECT (SELECT TOP 1 Status FROM \#Tmp_TCOMCloseItemCheck WHERE Status \<\> 0) AS Status, (SELECT TOP 1 Result FROM \#Tmp_TCOMCloseItemCheck WHERE Status \<\> 0) AS Results

RETURN

END
