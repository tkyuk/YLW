-- 추후 'D'에서 오류가 발생하였을 경우, 동일 수입입고에 대한 'U'건에도 Status, Result를 Update하기 위해 구분하여 담는다.

INSERT INTO \#TempDelete(IDXSeq, Kind, SETID, PermitSeq, PermitSerl, DelvSeq, DelvSerl)

SELECT DENSE_RANK() OVER(ORDER BY A.PermitSeq, A.PermitSerl, A.DelvSeq, A.DelvSerl), A.Kind, A.SETID, A.PermitSeq, A.PermitSerl, A.DelvSeq, A.DelvSerl

FROM \#ImpDelvDEL AS A

JOIN (SELECT PermitSeq, PermitSerl, DelvSeq, DelvSerl, SUM(Kind) AS Kind

FROM \#ImpDelvDEL

GROUP BY PermitSeq, PermitSerl, DelvSeq, DelvSerl

HAVING SUM(Kind) \> 0) AS B ON A.PermitSeq = B.PermitSeq

AND A.PermitSerl = B.PermitSerl

AND A.DelvSeq = B.DelvSeq

AND A.DelvSerl = B.DelvSerl

-- 'U'인 건에 대해 삭제 대상에서 DELETE

DELETE A

FROM \#ImpDelvDEL AS A

JOIN \#TempDelete AS B ON A.SETID = B.SETID

AND B.Kind = 0

-- 'U'인 건에 대해 생성 대상에서 DELETE

DELETE A

FROM \#ImpDelvDEL AS A

JOIN \#TempDelete AS B ON A.SETID = B.SETID

AND B.Kind = 0
