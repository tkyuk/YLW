# 창고

CREATE OR ALTER VIEW mcm_VMESDAWH

AS

 

SELECT ISNULL(A.CompanySeq , 0) AS CompanySeq , -- 법인내부코드

ISNULL(B.CompanyName ,'') AS CompanyName , -- 법인명

ISNULL(A.WHSeq , 0) AS WHSeq , -- 창고코드

ISNULL(A.WHName ,'') AS WHName , -- 창고명

ISNULL(A.MngDeptSeq , 0) AS DeptSeq , -- 부서코드

ISNULL(D.DeptName ,'') AS DeptName , -- 부서명

ISNULL(A.SMWHKind , 0) AS SMWHKind , -- 창고구분코드

ISNULL(C.MinorName ,'') AS SMWHKindName, -- 창고구분

ISNULL(A.IsNotUse , 0) AS IsNotUse , -- 미사용여부

ISNULL(A.LastUserSeq , 0) AS LastUserSeq , -- 최종등록자

ISNULL(A.LastDateTime,'') AS LastDateTime -- 최종등록일자

FROM \_TDAWH AS A WITH(NOLOCK)

LEFT OUTER JOIN \_TCACompany AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

LEFT OUTER JOIN \_TDASMinor AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.MinorSeq = SMWHKind

LEFT OUTER JOIN \_TDADept AS D WITH(NOLOCK) ON A.CompanySeq = D.CompanySeq

AND A.MngDeptSeq = D.DeptSeq
