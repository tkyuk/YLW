@DelvDate = LTRIM(RTRIM(ISNULL(@DelvDate , ''))),

@DeptSeq = ISNULL(@DeptSeq ,0),

TRUNCATE TABLE \#SComSourceDailyBatch

 

INSERT INTO \#SComSourceDailyBatch

SELECT '\_TPUDelvInItem', A.DelvInSeq, A.DelvInSerl, 0,

'\_TPUDelvItem', A.DelvSeq, A.DelvSerl, 0,

A.Qty, A.StdUnitQty, A.CurAmt, A.CurVAT, A.DOMAmt, A.DOMVAT,

A.Qty, A.StdUnitQty, A.CurAmt, A.CurVAT, A.DOMAmt, A.DOMVAT

FROM \#TPUDelvItem AS A

WHERE ISNULL(A.DelvInSeq,0) \<\> 0

AND ISNULL(A.DelvInSerl,0) \<\> 0

 

-- 진행연결

EXEC \_SWComSourceDailyBatch 'A', @CompanySeq, @UserSeq

IF @@ERROR \<\> 0

BEGIN

--PRINT 'INSERT ERROR : SourceDaily'

ROLLBACK TRAN

RETURN

END
