# (ACE) MES 거래명세서 출고 생성 및 삭제

DROP PROCEDURE dbo.mcm_SSLInvoiceProc

GO

CREATE PROCEDURE dbo.mcm_SSLInvoiceProc

@WorkingTag NVARCHAR(1) , -- A:삽입(판매출고) D:삭제(판매출고취소)

@InvoiceSeq INT , -- 필수 \_/로 묶어서 전송 ex 한 건이 경우 1\_/, 여러 건인 경우 1\_/2\_/3…

@UserSeq INT , -- 필수

@Status NVARCHAR(10) OUTPUT, -- 상태값

@Results NVARCHAR(250) OUTPUT -- 에러메세지

AS

 

DECLARE @CompanySeq INT = 1 ,

@LanguageSeq INT = 1 ,

@CurrDateTime DATETIME

SELECT @WorkingTag = LTRIM(RTRIM(ISNULL(@WorkingTag , '')))

,@UserSeq = LTRIM(RTRIM(ISNULL(@UserSeq , 0)))

,@Status = 'OK'

,@Results = CASE WHEN @WorkingTag = 'A' THEN '출고처리가 완료되었습니다.' ELSE '출고처리 취소되었습니다.' END

,@CurrDateTime = GETDATE()

 

-- 호출 Log 남기기

IF NOT EXISTS (SELECT \* FROM Sysobjects where Name = 'mcm_TMESInvoiceProclog' AND xtype = 'U' )

BEGIN

CREATE TABLE mcm_TMESInvoiceProclog

(

LogSeq INT IDENTITY(1,1) NOT NULL,

WorkingTag NVARCHAR(1) ,

InvoiceSeq NVARCHAR(20) ,

UserSeq INT ,

Status NVARCHAR(10) NULL,

Results NVARCHAR(250) NULL,

LogDateTime DATETIME NOT NULL

)

END

 

INSERT mcm_TMESInvoiceProclog

(

WorkingTag ,

InvoiceSeq ,

UserSeq ,

Status ,

Results ,

LogDateTime

)

SELECT @WorkingTag,

@InvoiceSeq,

@UserSeq,

@Status,

@Results,

@CurrDateTime

 

IF @WorkingTag NOT IN ('A', 'D')

BEGIN

SELECT @Status = 'ERR', @Results = '잘못된 작업코드 입니다.'

GOTO ERR_PROC

END

 

-- 재고반영

CREATE TABLE \#TLGInOutMonth

(

InOutYM NCHAR(6),

WHSeq INT,

ItemSeq INT

)

CREATE TABLE \#TLGInOutMonthLot

(

InOutYM NCHAR(6),

WHSeq INT,

LotNo NVARCHAR(30),

ItemSeq INT

)

CREATE TABLE \#TLGInOutDailyBatch

(

WorkingTag NVARCHAR(10),

InOutType INT,

InOutSeq INT,

InOutSerl INT,

InOutSubSerl INT,

DataKind INT,

MessageType INT,

Result NVARCHAR(250),

Status INT

)

 

 

CREATE TABLE \#MESData

(

IDX_NO INT,

CompanySeq INT,

WorkingTag NCHAR(1),

InvoiceSeq INT,

InOutType INT,

BizUnit INT,

InvoiceDate NCHAR(8),

SMSalesCrtKind INT,

IsDelvCfm NCHAR(1),

 

MessageType INT,

Status INT,

Result NVARCHAR(250)

)

 

INSERT INTO \#MESData

(

IDX_NO,CompanySeq,WorkingTag,InvoiceSeq,InOutType,

BizUnit,InvoiceDate,SMSalesCrtKind,IsDelvCfm,

MessageType,Status,Result

)

SELECT 1 AS IDX_NO, @CompanySeq, @WorkingTag, A.InvoiceSeq, 10,

A.BizUnit, A.InvoiceDate, 0, CASE WHEN @WorkingTag = 'A' THEN '1' ELSE '0' END,

0 AS MessageType, 0 AS Status, '' AS Result

FROM \_TSLInvoice AS A

WHERE A.CompanySeq = @CompanySeq

AND A.InvoiceSeq = @InvoiceSeq

 

-- 3) 담은 대상 없으면 RETURN

IF NOT EXISTS ( SELECT 1 FROM \#MESData)

BEGIN

SELECT @Status = 9999

, @Results = '거래명세서 처리 대상이 없습니다.'

 

-- Log Insert

UPDATE mcm_TMESInvoiceProclog

SET Status = @Status,

Results = @Results,

InvoiceSeq = @InvoiceSeq

WHERE LogDateTime = @CurrDateTime

 

RETURN

END

 

 

 

-- 기처리된 내역인데, 처리로 올 경우 에러

UPDATE A

SET Status = 2,

Result = '이미 출고처리된 내역입니다.'

FROM \#MESData AS A

JOIN \_TSLInvoice AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.InvoiceSeq = B.InvoiceSeq

WHERE Status = 0

AND A.IsDelvCfm = '1'

AND B.IsDelvCfm = '1'

 

UPDATE A

SET Status = 2,

Result = '이미 출고취소된 내역입니다.'

FROM \#MESData AS A

JOIN \_TSLInvoice AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.InvoiceSeq = B.InvoiceSeq

WHERE Status = 0

AND A.IsDelvCfm = '0'

AND B.IsDelvCfm = '0'

 

-------------------------------------------------

-- 0. 수불마감체크 필요

-------------------------------------------------

CREATE TABLE \#TLGClosingCheck

(

IDX_NO INT ,

InOutYM NCHAR(6) ,

ItemSeq INT ,

WHSeq INT ,

MessageType INT ,

Status INT ,

Result NVARCHAR(255)

)

 

INSERT INTO \#TLGClosingCheck

SELECT A.IDX_NO, LEFT(A.InvoiceDate,6), B.ItemSeq, B.WHSeq, 0, 0, ''

FROM \#MESData AS A

LEFT OUTER JOIN \_TSLInvoiceItem AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.InvoiceSeq = B.InvoiceSeq

WHERE A.Status = 0

AND B.Qty \<\> 0

EXEC \_SWLGClosingCheck @CompanySeq, @LanguageSeq

 

UPDATE A

SET Result = B.Result,

Status = 2

FROM \#MESData AS A

JOIN \#TLGClosingCheck AS B WITH(NOLOCK) ON A.IDX_NO = B.IDX_NO

WHERE A.Status = 0

AND B.Status \<\> 0

 

IF EXISTS (SELECT TOP 1 1 FROM \#MESData WHERE Status \<\> 0)

BEGIN

SELECT @Status = Status

, @Results = Result

FROM \#MESData

WHERE Status \<\> 0

 

UPDATE mcm_TMESInvoiceProclog

SET Status = @Status,

Results = @Results,

InvoiceSeq = @InvoiceSeq

WHERE LogDateTime = @CurrDateTime

 

RETURN

END

 

 

--===================================================================

-- 트랜잭션 Start 구간

--===================================================================

BEGIN TRAN

 

 

-- 로그테이블 남기기(마지막 파라메터는 반드시 한줄로 보내기)

EXEC \_SCOMXmlLog @CompanySeq ,

@UserSeq ,

'\_TSLInvoice', -- 원테이블명

'#MESData', -- 템프테이블명

'InvoiceSeq' , -- 키가 여러개일 경우는 , 로 연결한다.

999999, ''

 

UPDATE \_TSLInvoice

SET IsDelvCfm = B.IsDelvCfm,

DelvCfmEmpSeq= CASE WHEN B.WorkingTag = 'D' THEN 0 ELSE @UserSeq END,

DelvCfmDate = CASE WHEN B.WorkingTag = 'D' THEN '' ELSE GETDATE() END,

LastUserSeq = @UserSeq,

LastDateTime = GETDATE()

FROM \_TSLInvoice AS A

JOIN \#MESData AS B ON A.InvoiceSeq = B.InvoiceSeq

WHERE B.Status = 0

AND A.CompanySeq = @CompanySeq

 

IF @@ERROR \<\> 0

BEGIN

SELECT @Status = 9999, @Results = '출고 작업시 오류발생하였습니다.'

ROLLBACK TRAN

GOTO ERR_PROC

END

-----------------------------------------------------------------------

-- 출고취소

-----------------------------------------------------------------------

IF EXISTS ( SELECT 1 FROM \#MESData WHERE IsDelvCfm = '0') -- 출고취소

BEGIN

TRUNCATE TABLE \#TLGInOutDailyBatch

 

-----------------------------

-- 2-2) 수불삭제 (거래명세서)

-----------------------------

INSERT INTO \#TLGInOutDailyBatch

(

WorkingTag , InOutType , InOutSeq , InOutSerl , InOutSubSerl ,

DataKind , MessageType , Result , Status

)

SELECT DISTINCT

N'D'

, 10 -- 거래명세표

, D.InvoiceSeq

, 0

, 0

, 0, 0, '', 0

FROM \#MESData AS D

WHERE D.InvoiceSeq \> 0

-- 수불 반영 (DELETE)

IF EXISTS ( SELECT 1 FROM \#TLGInOutDailyBatch WHERE WorkingTag IN (N'U', N'D') )

BEGIN

EXEC dbo.\_SWLGInOutDailyDELETE @CompanySeq = @CompanySeq

, @Level = 1

, @LanguageSeq = @LanguageSeq

IF @@ERROR \<\> 0

BEGIN

SELECT @Status = 9999, @Results = '수불삭제시 오류발생하였습니다.'

ROLLBACK TRAN

GOTO ERR_PROC

END

 

IF NOT EXISTS ( SELECT 1 FROM \#TLGInOutDailyBatch WHERE Status \<\> 0 )

BEGIN

-- 창고재고 집계

EXEC dbo.\_SWLGWHStockUPDATE @CompanySeq = @CompanySeq

 

IF @@ERROR \<\> 0

BEGIN

SELECT @Status = 9999, @Results = '창고재고집계시 오류발생하였습니다.'

ROLLBACK TRAN

GOTO ERR_PROC

END

END

ELSE

BEGIN

SELECT TOP 1

@Status = Status,

@Results = Result

FROM \#TLGInOutDailyBatch

WHERE Status \<\> 0

ROLLBACK TRAN

GOTO ERR_PROC

END

END

END

-----------------------------------------------------------------------

-- 'D' / 'U' 인 경우 삭제 END

-----------------------------------------------------------------------

-----------------------------------------------------------------------

-- 출고처리

-----------------------------------------------------------------------

IF EXISTS ( SELECT 1 FROM \#MESData WHERE IsDelvCfm = '1') -- 출고처리

BEGIN

IF NOT EXISTS ( SELECT 1 FROM \#TLGInOutDailyBatch WHERE Status \<\> 0 )

BEGIN

-- 창고재고 집계

EXEC dbo.\_SWLGWHStockUPDATE @CompanySeq = @CompanySeq

 

IF @@ERROR \<\> 0

BEGIN

SELECT @Status = 9999, @Results = '창고재고 집계시 오류발생하였습니다.'

ROLLBACK TRAN

GOTO ERR_PROC

END

END

ELSE

BEGIN

SELECT TOP 1

@Status = Status,

@Results = Result

FROM \#TLGInOutDailyBatch

WHERE Status \<\> 0

 

ROLLBACK TRAN

GOTO ERR_PROC

RETURN

END

 

 

EXEC \_SWLGInOutMinusCheck @CompanySeq, '#TLGInOutDailyBatch', @LanguageSeq

-- 위에서 '-'재고 체크를 하여 오류가 발생 했으면 직접 return

-- 구지 다음 단계까지 탈 이유가 없음, 속도 위하여

IF EXISTS ( SELECT TOP 1 1 FROM \#TLGInOutDailyBatch WHERE Status \<\> 0 )

BEGIN

SELECT TOP 1

@Status = Status,

@Results = Result

FROM \#TLGInOutDailyBatch

WHERE Status \<\> 0

 

ROLLBACK TRAN

GOTO ERR_PROC

END

EXEC \_SWLGInOutLotMinusCheck @CompanySeq, '#TLGInOutDailyBatch', @LanguageSeq

IF EXISTS ( SELECT TOP 1 1 FROM \#TLGInOutDailyBatch WHERE Status \<\> 0 )

BEGIN

SELECT TOP 1

@Status = Status,

@Results = Result

FROM \#TLGInOutDailyBatch

WHERE Status \<\> 0

 

ROLLBACK TRAN

GOTO ERR_PROC

END

END

COMMIT TRAN

 

RETURN

 

 

ERR_PROC:

 

-- Log Insert

UPDATE mcm_TMESInvoiceProclog

SET Status = @Status,

Results = @Results,

InvoiceSeq = @InvoiceSeq

WHERE LogDateTime = @CurrDateTime
