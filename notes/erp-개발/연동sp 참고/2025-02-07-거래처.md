CREATE OR ALTER VIEW mcm_VMESDACust

AS

 

SELECT ISNULL(A.CompanySeq , 0) AS CompanySeq, -- 법인내부코드

ISNULL(B.CompanyName ,'') AS CompanyName, -- 법인명

ISNULL(A.CustSeq , 0) AS CustSeq, -- 거래처내부코드

ISNULL(A.CustName ,'') AS CustName, -- 거래처명

ISNULL(A.SMDomFor , 0) AS SMDomFor, -- 국내외구분

ISNULL(C.MinorName ,'') AS DomForName, -- 국내외구분명

ISNULL(A.SMCustStatus, 0) AS SMCustStatus, -- 거래처상태

ISNULL(D.MinorName ,'') AS CustStatusName, -- 거래처상태명

ISNULL(A.FullName ,'') AS FullName, -- 상호

ISNULL(A.BizNo ,'') AS BizNo, -- 사업자번호

ISNULL(A.PersonId ,'') AS PersonId, -- 주민등록번호

ISNULL(A.Owner ,'') AS Owner, -- 대표자성명

ISNULL(A.LawRegNo ,'') AS LawRegNo, -- 법인번호

ISNULL(A.BizAddr ,'') AS BizAddr, -- 사업자주소

ISNULL(A.BizKind ,'') AS BizKind, -- 종목

ISNULL(A.BizType ,'') AS BizType, -- 업태

ISNULL(dbo.\_FWCOMDecrypt(A.TelNo, N'\_TDACust', N'TelNo', A.CompanySeq),'') AS TelNo, -- 전화번호

ISNULL(E.MobileNo ,'') AS MobileNo, -- 모바일번호

--F.UMCustKind AS UMCustKind, -- 거래처종류

--ISNULL(G.MinorName, '') AS CustKindName, -- 거래처종류명

ISNULL(H.FAX ,'') AS FAX, -- 팩스번호

ISNULL(A.UMCountrySeq, 0) AS UMCountrySeq, -- 국가

ISNULL(I.MinorName ,'') AS CountryName, -- 국가명

ISNULL(H.ZipCode ,'') AS ZipCode, -- 우편번호

ISNULL(A.LastUserSeq , 0) AS LastUserSeq, -- 최종등록자

ISNULL(A.LastDateTime,'') AS LastDateTime -- 최종등록일자

FROM \_TDACust AS A WITH(NOLOCK)

LEFT OUTER JOIN \_TCACompany AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

LEFT OUTER JOIN \_TDASMinor AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.MinorSeq = A.SMDomFor

LEFT OUTER JOIN \_TDASMinor AS D WITH(NOLOCK) ON D.CompanySeq = A.CompanySeq

AND D.MinorSeq = A.SMCustStatus

LEFT OUTER JOIN \_TDACustEmpInfo AS E WITH(NOLOCK) ON E.CompanySeq = A.CompanySeq

AND E.CustSeq = A.CustSeq

AND E.IsStd = '1'

--LEFT OUTER JOIN \_TDACustKind AS F WITH(NOLOCK) ON F.CompanySeq = A.CompanySeq

-- AND F.CustSeq = A.CustSeq

--LEFT OUTER JOIN \_TDAUMinor AS G WITH(NOLOCK) ON G.CompanySeq = F.CompanySeq

-- AND G.MinorSeq = F.UMCustKind

LEFT OUTER JOIN \_TDACustAdd AS H WITH(NOLOCK) ON H.CompanySeq = A.CompanySeq

AND H.CustSeq = A.CustSeq

LEFT OUTER JOIN \_TDAUMinor AS I WITH(NOLOCK) ON I.CompanySeq = A.CompanySeq

AND I.MinorSeq = A.UMCountrySeq
