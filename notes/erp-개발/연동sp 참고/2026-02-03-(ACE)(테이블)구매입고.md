-- 동일한 발주, 납품 건에 대해 WorkingTag가 'U' Or 'D'가 함께 들어왔을 때 우선순위는 'D'가 더 높다.

> -- 따라서 'U'는 삭제/생성 대상에서 제외하기 위해 따로 담는다.

> -- 추후 'D'에서 오류가 발생하였을 경우, 동일 발주/납품에 대한 'U'건에도 Status, Result를 Update하기 위해 구분하여 담는다.

INSERT INTO \#TempDelete(IDXSeq, Kind, SETID, DelvSeq, DelvSerl, DelvInSeq, DelvInSerl)

SELECT DENSE_RANK() OVER(ORDER BY A.DelvSeq, A.DelvSerl, A.DelvInSeq, A.DelvInSerl), A.Kind, A.SETID, A.DelvSeq, A.DelvSerl, A.DelvInSeq, A.DelvInSerl

FROM \#DelvAutoDEL AS A

JOIN (SELECT DelvSeq, DelvSerl, DelvInSeq, DelvInSerl, SUM(Kind) AS Kind

FROM \#DelvAutoDEL

GROUP BY DelvSeq, DelvSerl, DelvInSeq, DelvInSerl

HAVING SUM(Kind) \> 0) AS B ON A.DelvSeq = B.DelvSeq

AND A.DelvSerl = B.DelvSerl

AND A.DelvInSeq = B.DelvInSeq

AND A.DelvInSerl = B.DelvInSerl

 

> -- 'U'인 건에 대해 삭제 대상에서 DELETE

DELETE A

FROM \#DelvAutoDEL AS A

JOIN \#TempDelete AS B ON A.SETID = B.SETID

> AND         B.kind = 0

-- 'U'인 건에 대해 생성 대상에서 DELETE

DELETE A

FROM \#DelvAutoINS AS A

JOIN \#TempDelete AS B ON A.SETID = B.SETID

> AND         B.kind = 0

CREATE TABLE \#NewItemSeq_gnst(

ItemSeq INT,

Qty DECIMAL(19,5)

)

INSERT INTO \#NewItemSeq_gnst(ItemSeq, Qty)

SELECT A.ItemSeq

,SUM(A.Qty) -- 품목별로 수량 SUM

FROM \#TPUDelvItem AS A

WHERE ISNULL(A.DelvInSeq,0) \<\> 0

AND ISNULL(A.DelvInSerl,0) \<\> 0

GROUP BY A.ItemSeq

 

CREATE TABLE \#ShotUpdateData(

ToolSeq INT

,ItemSeq INT

,AddShot DECIMAL(19,5)

,OldShot DECIMAL(19,5)

)

INSERT INTO \#ShotUpdateData(ToolSeq, ItemSeq, AddShot, OldShot)

SELECT C.ToolSeq

,A.ItemSeq

,ISNULL(A.Qty, 0) /

CASE WHEN ISNULL(C.Cavity,0) \> 0 THEN C.Cavity ELSE 1

END AS AddShot -- 증감 값

,ISNULL(C.TotalShot, 0) AS OldShot -- 직전 값

FROM \#NewItemSeq_gnst AS A

JOIN gnst_TPDToolMoldItem AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.Type = 2 -- 자재

AND B.ItemSeq = A.ItemSeq -- 자재가 같은 것

JOIN gnst_TPDToolMold AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.ToolSeq = B.ToolSeq

WHERE ISNULL(C.IsStopMatCnt, '') \<\> '1' -- 자재집계중단여부 체크 안된 것

 

 

CREATE TABLE \#AddShotSum(

ToolSeq INT

,AddShotSum DECIMAL(19,5)

,OldShot DECIMAL(19,5)

)

INSERT INTO \#AddShotSum(ToolSeq, AddShotSum, OldShot)

SELECT A.ToolSeq

,SUM(A.AddShot) -- ToolSeq별로 Shot합산

,MIN(A.OldShot)

FROM \#ShotUpdateData AS A

GROUP BY A.ToolSeq

 

--select \* from \#NewItemSeq_gnst

UPDATE A

SET TotalShot = ISNULL(A.TotalShot,0) + ISNULL(B.AddShotSum,0), -- 실제 테이블에 UPDATE

LastUserSeq = @UserSeq, -- 수정자/일시 업데이트 권장

LastDateTime = GETDATE()

FROM gnst_TPDToolMold AS A

JOIN \#AddShotSum AS B ON B.ToolSeq = A.ToolSeq

WHERE A.CompanySeq = @CompanySeq
