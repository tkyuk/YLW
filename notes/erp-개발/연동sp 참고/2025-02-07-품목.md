# 품목

CREATE OR ALTER VIEW mcm_VMESDAItem

AS

 

SELECT A.CompanySeq AS CompanySeq, -- 법인코드

B.CompanyName AS CompanyName, -- 법인명

A.ItemSeq AS ItemSeq, -- 품목코드

A.ItemName AS ItemName, -- 품명

A.ItemNo AS ItemNo, -- 품번

A.Spec AS Spec, -- 규격

A.AssetSeq AS AssetSeq, -- 품목자산분류코드

C.AssetName AS AssetName, -- 품목자산분류

E.ItemClassLSeq AS ItemClassLSeq, -- 품목대분류코드

E.ItemClassLName AS ItemClassLName, -- 품목대분류

E.ItemClassMSeq AS ItemClassMSeq, -- 품목중분류코드

E.ItemClassMName AS ItemClassMName, -- 품목중분류

E.ItemClassSSeq AS ItemClassSSeq, -- 품목소분류코드

E.ItemClassSName AS ItemClassSName, -- 품목소분류

A.UnitSeq AS UnitSeq, -- 단위코드

F.UnitName AS UnitName, -- 단위

A.SMStatus AS SMStatus, -- 품목상태

G.MinorName AS SMStatusName, -- 품목상태명

ISNULL(H.IsInQC , '0') AS IsInQC, -- 수입검사여부

ISNULL(H.IsFinQC, '0') AS IsLastQc, -- 최종검사여부

CASE WHEN C.SMAssetGrp IN (6008005, 6008006, 6008010) THEN '1' ELSE '0' END AS IsMat, -- 자재여부

A.LastUserSeq AS LastUserSeq, -- 최종등록자

A.LastDateTime AS LastDateTime -- 최종등록일자

FROM \_TDAItem AS A WITH(NOLOCK)

LEFT OUTER JOIN \_TCACompany AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

LEFT OUTER JOIN \_TDAItemAsset AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.AssetSeq = A.AssetSeq

LEFT OUTER JOIN \_TDAItemClass AS D WITH(NOLOCK) ON D.CompanySeq = A.CompanySeq

AND D.ItemSeq = A.ItemSeq

AND D.UMajorItemClass IN (2001, 2004)

LEFT OUTER JOIN \_VDAItemClass AS E WITH(NOLOCK) ON E.CompanySeq = D.CompanySeq

AND E.ItemClassSSeq = D.UMItemClass

LEFT OUTER JOIN \_TDAUnit AS F WITH(NOLOCK) ON F.CompanySeq = A.CompanySeq

AND F.UnitSeq = A.UnitSeq

LEFT OUTER JOIN \_TDASMinor AS G WITH(NOLOCK) ON G.CompanySeq = A.CompanySeq

AND G.MinorSeq = A.SMStatus

LEFT OUTER JOIN \_TPDQAItemClassQCKind AS H WITH(NOLOCK) ON H.CompanySeq = D.CompanySeq

AND H.UMItemClass = D.UMItemClass;
