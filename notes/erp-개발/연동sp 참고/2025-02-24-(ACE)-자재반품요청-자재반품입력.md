CREATE TABLE \#Temp_Prog

(

IDX_NO INT ,

OutQty DECIMAL(19, 5),

OutSTDQty DECIMAL(19, 5),

OutAmt DECIMAL(19, 5)

)

 

INSERT INTO \#Temp_Prog(IDX_NO, OutQty, OutSTDQty, OutAmt)

SELECT A.IDX_NO, SUM(CASE IDInOut WHEN 1 THEN A.Qty ELSE 0 END),

SUM(CASE IDInOut WHEN 1 THEN A.STDQty ELSE 0 END),

SUM(CASE IDInOut WHEN 1 THEN A.Amt ELSE 0 END)

FROM \#TCOMProgressTracking AS A

GROUP BY A.IDX_NO

 

SELECT A.OutReqSeq ,

A.OutReqItemSerl ,

A.Qty ,

A.STDQty,

A.Amt ,

A.ProgQty ,

A.ProgSTDQty ,

A.ProgAmt ,

ISNULL(B.OutQty , 0) AS OutQty ,

ISNULL(B.OutSTDQty, 0) AS OutSTDQty,

ISNULL(B.OutAmt , 0) AS OutAmt ,

CASE WHEN B.IDX_NO IS NULL THEN '0'

ELSE '1' END AS IsExists

INTO \#TempProgCheck

FROM \#Temp_SourceItem AS A

LEFT OUTER JOIN \#Temp_Prog AS B ON A.IDX_NO = B.IDX_NO

 

 

IF EXISTS ( SELECT TOP 1 1

FROM \#TempProgCheck AS C

WHERE ((@SMProgCheckKind = 1045001 AND @IsOverFlow \<\> '1' AND C.Qty \< C.ProgQty + C.OutQty ) OR -- 수량

(@SMProgCheckKind = 1045002 AND @IsOverFlow \<\> '1' AND C.STDQty \< C.ProgSTDQty + C.OutSTDQty) OR -- 기준단위수량

(@SMProgCheckKind = 1045003 AND @IsOverFlow \<\> '1' AND C.Amt \< C.ProgAmt + C.OutAmt ) OR -- 금액

(@SMProgCheckKind = 1045004 AND C.IsExists = '1'))

)

BEGIN

SELECT @Status = '6',

@Results = '해당 건 진행정보가 초과되었습니다.'

GOTO ERR_PROC

RETURN

END
