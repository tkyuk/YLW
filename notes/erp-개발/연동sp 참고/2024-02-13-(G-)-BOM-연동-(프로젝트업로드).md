# \(G\) BOM 연동 (프로젝트업로드)

ALTER PROC dat_SIFBOMProc

@PDMKey NVARCHAR(200),

@PJTNo NVARCHAR(60),

@Status INT OUTPUT,

@Results NVARCHAR(250) OUTPUT

AS

 

SET NOCOUNT ON

 

DECLARE @CompanySeq INT,

@ItemSeq INT,

@PONo NVARCHAR(12),

@UserSeq INT,

@PgmSeq INT,

@PODate NCHAR(8),

@CurrSeq INT,

@LanguageSeq INT

 

SELECT @CompanySeq = 1,

@UserSeq = 1,

@LanguageSeq = 1

 

DECLARE @PJTSeq INT = 0,

@MessageType INT

 

IF NOT EXISTS (SELECT TOP 1 1 FROM dat_IF_TPJTBOM AS A

WHERE A.PDMKey = @PDMKey

AND A.PJTNo = @PJTNo

AND A.FLAG = 'N')

BEGIN

SELECT @Status = 9999,

@Results = '이미 처리된 건 입니다.'

RETURN

END

 

 

SELECT @PJTSeq = B.PJTSeq

FROM dat_IF_TPJTBOM AS A WITH(NOLOCK)

JOIN \_TPJTProject AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.PJTNo = A.PJTNo

WHERE A.PDMKey = @PDMKey

AND A.PJTNo = @PJTNo

 

IF @PJTSeq = 0

BEGIN

SELECT @Status = 9999,

@Results = '프로젝트가 존재하지 않습니다.'

RETURN

END

 

 

 

-- 필수값 체크

IF EXISTS (SELECT TOP 1 1

FROM dat_IF_TPJTBOM AS A

WHERE A.PDMKey = @PDMKey

AND (

ISNULL(A.PJTNo ,'') = '' OR -- 프로젝트번호

ISNULL(A.BOMLevel ,'') = '' OR -- BOM레벨

ISNULL(A.ItemNo ,'') = '' OR -- 품번

ISNULL(A.ItemName ,'') = '' OR -- 품명

ISNULL(A.BasicQty , 0) = 0 OR -- 기본소요량

ISNULL(A.TotQty , 0) = 0 OR -- 총소요량

ISNULL(A.UMSupplyType, 0) = 0 OR -- 조달구분

ISNULL(A.UMRegType , 0) = 0 OR -- 등록구분

ISNULL(A.ItemClassSeq, 0) = 0 OR -- 품목소분류

--ISNULL(A.DWGNo ,'') = '' OR -- 도면번호

ISNULL(A.ProdCnt , 0) = 0 -- 제작대수

))

BEGIN

-- 필수값 체크

UPDATE dat_IF_TPJTBOM

SET Result = '필수입력 항목을 입력하지 않았습니다.',

Status = 9999,

ERPProcDateTime = GETDATE()

FROM dat_IF_TPJTBOM AS A

WHERE A.PDMKey = @PDMKey

AND (

ISNULL(A.PJTNo ,'') = '' OR -- 프로젝트번호

ISNULL(A.BOMLevel ,'') = '' OR -- BOM레벨

ISNULL(A.ItemNo ,'') = '' OR -- 품번

ISNULL(A.ItemName ,'') = '' OR -- 품명

ISNULL(A.BasicQty , 0) = 0 OR -- 기본소요량

ISNULL(A.TotQty , 0) = 0 OR -- 총소요량

ISNULL(A.UMSupplyType, 0) = 0 OR -- 조달구분

ISNULL(A.UMRegType , 0) = 0 OR -- 등록구분

--ISNULL(A.DWGNo ,'') = '' OR -- 도면번호

ISNULL(A.ProdCnt , 0) = 0 -- 제작대수

)

SELECT @Status = 9999,

@Results = '필수입력 항목을 입력하지 않았습니다.'

RETURN

END

SELECT DataSeq

,ROW_NUMBER() OVER (PARTITION BY A.UpperItemNo ORDER BY (SELECT 1)) AS RowCnt

,UpperItemNo

INTO \#Tmp_BOMLevel

FROM \#TPJTBOM AS A WITH(NOLOCK)

--SELECT \* FROM \#TPJTBOM WHERE BOMLevel = '1.1'

--SELECT \* FROM \#TPJTBOM WHERE UpperItemNo = 'RET-00A-A000' RETURN

 

SELECT LEFT(A.BOMLevel,LEN(A.BOMLevel)-CHARINDEX('.',REVERSE(A.BOMLevel))) AS BOMLevel

,MAX(CONVERT(INT,REPLACE(RIGHT(A.BOMLevel,CHARINDEX('.',REVERSE(A.BOMLevel))),'.',''))) AS MaxSerl

,X.ItemNo AS UpperItemNo

INTO \#Tmp_BOMLevel_Max

FROM \_TPJTBOM AS A

JOIN \_TPJTBOM AS Y WITH(NOLOCK) ON A.CompanySeq = Y.CompanySeq

AND A.PJTSeq = Y.PJTSeq

AND Y.BOMSerl \<\> -1

AND A.UpperBOMSerl = Y.BOMSerl

AND ISNULL(Y.BeforeBOMSerl,0) = 0 -- 상위 BOM

JOIN \_TDAItem AS X WITH(NOLOCK) ON Y.CompanySeq = X.CompanySeq

AND Y.ItemSeq = X.ItemSeq

JOIN \#TPJTBOM AS B ON B.UpperItemNo = X.ItemNo

WHERE A.PJTSeq = @PJTSeq

AND LEN(A.BOMLevel) \<\> 1

GROUP BY LEFT(A.BOMLevel,LEN(A.BOMLevel)-CHARINDEX('.',REVERSE(A.BOMLevel))), X.ItemNo

--SELECT UpperItemNo,\* FROM \#TPJTBOM

--select UpperItemNo from \#Tmp_BOMLevel

FROM \#TPJTBOM AS A

JOIN \#Tmp_BOMLevel AS B ON A.DataSeq = B.DataSeq

JOIN \#Tmp_BOMLevel_Max AS C ON C.UpperItemNo = A.UpperItemNo

FROM \_TPJTBOM

WHERE CompanySeq = @CompanySeq

AND PJTSeq = @PJTSeq

AND BOMSerl \<\> -1

AND ISNULL(BeforeBOMSerl,0) = 0 -- 최종차수만..

UPDATE \#Tmp

SET QrySort = CONVERT(INT, RIGHT(SortNum, CHARINDEX('.', REVERSE('.'+SortNum))-1)),

UpperSortNum = SUBSTRING(SortNum, 1, LEN(SortNum) - CHARINDEX('.', REVERSE(SortNum)))

UPDATE \#Tmp

SET UpperBOMSerl = CASE WHEN B.BOMSerl = A.BOMSerl THEN -1 ELSE B.BOMSerl END,

NodeCode = CASE WHEN B.BOMSerl = A.BOMSerl THEN REPLICATE('0',6- len(convert(varchar,A.QrySort)))+convert(varchar,A.QrySort)

ELSE REPLICATE('0',4- len(convert(varchar,A.QrySort)))+convert(varchar,A.QrySort) END -- By 18.04.30 hjlim (4자리까지 가능하도록 수정)

FROM \#Tmp AS A

LEFT OUTER JOIN \_TPJTBOM AS B ON B.CompanySeq = @CompanySeq

AND B.PJTSeq = @PJTSeq

AND A.UpperSortNum = B.BOMLevel AND B.BOMSerl \<\> -1

AND ISNULL(B.BeforeBOMSerl,0) = 0 -- 최종차수만..

UPDATE \#Tmp

SET TreeCode = NodeCode,

TmpSortnum = SortNum

WHERE UpperBOMSerl = -1
