# 자재출고 연동 SP

DROP PROC apro_SPDAOutSave

GO

CREATE PROC apro_SPDAOutSave

@CompanySeq INT,

@UserSeq INT,

@ReqSeq INT

AS

SET NOCOUNT ON;

 

INSERT INTO apro_TPDAParameterLog

(

LogType, LogDateTime, CompanySeq, UserSeq, ReqSeq

)

SELECT 'MATOUT' AS LogType, GETDATE() AS LogDateTime, @CompanySeq, @UserSeq, @ReqSeq

DECLARE @Ststus INT,

@Message NVARCHAR(255),

@CurrDate NVARCHAR(8),

@PgmSeq INT,

@GroupSeq INT,

@OutSeq INT,

@OutNo NVARCHAR(20),

@OutDate NVARCHAR(8),

@EmpSeq INT,

@DeptSeq INT

SELECT @CurrDate = CONVERT(NVARCHAR(8), GETDATE(), 112),

@PgmSeq = 999999

 

SELECT @Ststus = 0, @Message = '처리 되었습니다.',

@CompanySeq = ISNULL(@CompanySeq, 0),

@UserSeq = ISNULL(@UserSeq , 0),

@ReqSeq = ISNULL(@ReqSeq , 0)

 

SELECT @EmpSeq = A.EmpSeq, @DeptSeq = A.DeptSeq

FROM \_TCAUser AS A WITH(NOLOCK)

WHERE A.UserSeq = @UserSeq

 

IF @CompanySeq = 0 OR @UserSeq = 0 OR @ReqSeq = 0

BEGIN

SELECT @Ststus = 99999, @Message = '필수입력 항목을 입력하지 않았습니다.'

GOTO RESULT_PROC

RETURN

END

 

CREATE TABLE \#TPJTOutItem

(

IDX_NO INT IDENTITY(1, 1),

WorkingTag NVARCHAR(1),

Status INT,

MessageType INT,

Result NVARCHAR(255),

IsMstGroup NVARCHAR(1),

GroupSeq INT,

GroupSerl INT,

 

InOutType INT,

ReqSeq INT,

ReqSerl INT,

OutSeq INT,

OutSerl INT,

OutNo NVARCHAR(20),

 

OutDate NVARCHAR(8),

BizUnit INT,

DeptSeq INT,

EmpSeq INT,

PJTSeq INT,

ItemSeq INT,

UnitSeq INT,

Qty DECIMAL(19,5),

STDUnitSeq INT,

STDQty DECIMAL(19,5),

WHSeq INT,

BOMSerl INT,

LotNo NVARCHAR(30),

Remark NVARCHAR(500)

)

 

INSERT INTO \#TPJTOutItem

(

WorkingTag, Status, MessageType, Result, IsMstGroup,

GroupSeq,

GroupSerl,

InOutType, ReqSeq, ReqSerl, OutSeq, OutSerl, OutNo,

OutDate, BizUnit, DeptSeq, EmpSeq,

PJTSeq, ItemSeq, UnitSeq, Qty, STDUnitSeq, STDQty,

WHSeq, BOMSerl, LotNo, Remark

)

SELECT 'A' AS WorkingTag, 0 AS Status, 0 AS MessageType, '' AS Result, '0' AS IsMstGroup,

DENSE_RANK() OVER(ORDER BY A.BizUnit, A.ReqSeq) AS GroupSeq,

ROW_NUMBER() OVER(PARTITION BY A.BizUnit, A.ReqSeq

ORDER BY A.BizUnit, A.ReqSeq) AS GroupSerl,

A.InOutReqType AS InOutType, A.ReqSeq, B.ReqSerl, 0 AS OutSeq, 0 AS OutSerl, '' AS OutNo,

@CurrDate AS OutDate, A.BizUnit, @DeptSeq, @EmpSeq,

B.PJTSeq, B.ItemSeq, B.UnitSeq, B.Qty, B.UnitSeq AS STDUnitSeq, B.STDQty,

CASE WHEN ISNULL(C.WHSeq, 0) = 0 THEN B.WHSeq ELSE C.WHSeq END AS WHSeq, B.BOMSerl, B.LotNo, B.Remark

FROM \_TPJTOutReq AS A WITH(NOLOCK)

JOIN \_TPJTOutReqItem AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.ReqSeq = A.ReqSeq

JOIN \_TDAWH AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.BizUnit = A.BizUnit

AND C.SMWHKind = 8002026 -- \_TDASMinor WHERE MajorSeq = 8002

WHERE A.CompanySeq = @CompanySeq

AND A.ReqSeq = @ReqSeq

IF @@ROWCOUNT = 0

BEGIN

SELECT @Ststus = 99999, @Message = '자재출고요청건이 없습니다.'

GOTO RESULT_PROC

RETURN

END

 

 

IF EXISTS(SELECT TOP 1 1

FROM \#TPJTOutItem AS A

JOIN \_TCOMSourceDailyTrace AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.ToTableSeq = 500008 -- \_TCOMProgTable WHERE ProgTableName = '\_TPJTOutItem'

AND B.FromTableSeq = 500009 -- \_TCOMProgTable WHERE ProgTableName = '\_TPJTOutReqItem'

AND B.FromSeq = A.ReqSeq

AND B.FromSerl = A.ReqSerl

WHERE A.Status = 0

)

BEGIN

SELECT @Ststus = 99999, @Message = '이미 자재출고생성 된 건입니다 .'

GOTO RESULT_PROC

RETURN

END

-- 수불 마감

CREATE TABLE \#TLGClosingCheck

(

IDX_NO INT ,

InOutYM NCHAR(6) ,

ItemSeq INT ,

WHSeq INT ,

MessageType INT ,

Status INT ,

Result NVARCHAR(255)

)

 

INSERT INTO \#TLGClosingCheck

(

IDX_NO, InOutYM, ItemSeq, WHSeq,

MessageType, Status, Result

)

SELECT A.IDX_NO, LEFT(A.OutDate,6), A.ItemSeq, A.WHSeq,

0 AS MessageType, 0 AS Status, '' AS Result

FROM \#TPJTOutItem AS A

WHERE A.Status = 0

 

EXEC \_SWLGClosingCheck @CompanySeq, 1

 

UPDATE A

SET Result = B.Result,

MessageType = B.MessageType,

Status = B.Status

FROM \#TPJTOutItem AS A

JOIN \#TLGClosingCheck AS B WITH(NOLOCK) ON A.IDX_NO = B.IDX_NO

WHERE A.Status = 0

AND B.Status \<\> 0

DROP TABLE \#TLGClosingCheck

-- 납품 키값생성!

SELECT @GroupSeq = 0 -- 초기화

WHILE 1=1

BEGIN

SELECT TOP 1

@GroupSeq = A.GroupSeq,

@OutDate = A.OutDate

FROM \#TPJTOutItem AS A

WHERE A.WorkingTag = 'A'

AND A.Status = 0

AND A.GroupSeq \> @GroupSeq

ORDER BY A.GroupSeq

IF @@ROWCOUNT = 0 BREAK

EXEC \_SWCOMCreateNo 'PMS', '\_TPJTOut', @CompanySeq, '', @OutDate, @OutNo OUTPUT -- OutNo 생성

EXEC @OutSeq = \_SWCOMCreateSeq @CompanySeq, '\_TPJTOut', 'OutSeq', 1 -- OutSeq 생성

 

UPDATE A

SET OutNo = @OutNo,

OutSeq = @OutSeq + 1,

OutSerl = A.GroupSerl

FROM \#TPJTOutItem AS A

WHERE A.WorkingTag = 'A'

AND A.Status = 0

AND A.GroupSeq = @GroupSeq

END

 

UPDATE A

SET IsMstGroup = '1'

FROM \#TPJTOutItem AS A

CROSS APPLY(SELECT TOP 1 Z.GroupSerl

FROM \#TPJTOutItem AS Z

WHERE Z.GroupSeq = A.GroupSeq

ORDER BY Z.GroupSerl ASC

) AS B

WHERE A.WorkingTag IN ('A', 'U')

AND A.Status = 0

AND A.GroupSerl = B.GroupSerl

BEGIN TRAN

-- \_TDAUMinor WHERE MajorSeq = 7028

INSERT INTO \_TPJTOut

(

CompanySeq, InOutType, OutSeq, OutNo, BizUnit, OutDate,

UseType, DeptSeq, EmpSeq, Remark,

LastUserSeq, LastDateTime, PgmSeq

)

SELECT @CompanySeq, A.InOutType, A.OutSeq, A.OutNo, A.BizUnit, A.OutDate,

7028001 AS UseType, A.DeptSeq, A.EmpSeq, '\[PDA생성\]' AS Remark,

@UserSeq AS LastUserSeq, GETDATE() AS LastDateTime, @PgmSeq

FROM \#TPJTOutItem AS A

WHERE A.WorkingTag IN ('A', 'U')

AND A.Status = 0

AND A.IsMstGroup = '1'

IF @@ERROR \<\> 0

BEGIN

SELECT @Ststus = 99999, @Message = 'INSERT FAIL.'

ROLLBACK TRAN

GOTO RESULT_PROC

RETURN

END

INSERT INTO \_TPJTOutItem

(

CompanySeq, InOutType, OutSeq, OutSerl,

PJTSeq, ItemSeq, WHSeq, UnitSeq, Qty, STDUnitSeq, STDQty, LotNo, OutType, Remark,

LastUserSeq, LastDateTime, PgmSeq, HogiItemSeq, BOMSerl,

Memo1, Memo2, Memo3, Memo4, Memo5, Memo6,

Dummy1, Dummy2, Dummy3, Dummy4, Dummy5,

Dummy6, Dummy7, Dummy8, Dummy9, Dummy10

)

SELECT @CompanySeq, A.InOutType, A.OutSeq, A.OutSerl,

A.PJTSeq, A.ItemSeq, A.WHSeq, A.UnitSeq, A.Qty, A.STDUnitSeq, A.STDQty, A.LotNo, 0 AS OutType, A.Remark,

@UserSeq AS LastUserSeq, GETDATE() AS LastDateTime, @PgmSeq, 0 AS HogiItemSeq, A.BOMSerl,

'' AS Memo1, '' AS Memo2, '' AS Memo3, 0 AS Memo4, 0 AS Memo5, 0 AS Memo6,

'' AS Dummy1, '' AS Dummy2, '' AS Dummy3, '' AS Dummy4, '' AS Dummy5,

'' AS Dummy6, 0 AS Dummy7, 0 AS Dummy8, 0 AS Dummy9, 0 AS Dummy10

FROM \#TPJTOutItem AS A

WHERE A.WorkingTag IN ('A', 'U')

AND A.Status = 0

IF @@ERROR \<\> 0

BEGIN

SELECT @Ststus = 99999, @Message = 'INSERT FAIL.'

ROLLBACK TRAN

GOTO RESULT_PROC

RETURN

END

-- 구매납품 / 구매발주 진행 추가

TRUNCATE TABLE \#SComSourceDailyBatch

INSERT INTO \#SComSourceDailyBatch

(

ToTableName, ToSeq, ToSerl, ToSubSerl,

FromTableName, FromSeq, FromSerl, FromSubSerl,

ToQty, ToStdQty, ToAmt, ToVAT, ToDOMAmt, ToDOMVAT,

FromQty, FromSTDQty, FromAmt, FromVAT, FromDOMAmt, FromDOMVAT

)

SELECT '\_TPJTOutItem' ToTableName, B.OutSeq ToSeq, B.OutSerl ToSerl, B.OutType ToSubSerl,

'\_TLGInOutReqItem' FromTableName, C.ReqSeq FromSeq, C.ReqSerl FromSerl, 0 FromSubSerl,

B.Qty ToQty, B.STDQty ToStdQty, 0 ToAmt, 0 ToVAT, 0 ToDOMAmt, 0 ToDOMVAT,

C.Qty FromQty, C.STDQty FromSTDQty, 0 FromAmt, 0 FromVAT, 0 FromDOMAmt, 0 FromDOMVAT

FROM \#TPJTOutItem AS A

JOIN \_TPJTOutItem AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.OutSeq = A.OutSeq

AND B.OutSerl = A.OutSerl

AND B.OutType = A.InOutType

JOIN \_TPJTOutReqItem AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

AND C.ReqSeq = A.ReqSeq

AND C.ReqSerl = A.ReqSerl

WHERE A.WorkingTag IN ('A', 'U')

AND A.Status = 0

 

EXEC dbo.\_SWComSourceDailyBatch N'A', @CompanySeq, @UserSeq

IF @@ERROR \<\> 0

BEGIN

SELECT @Ststus = 99999, @Message = 'INSERT FAIL.'

ROLLBACK TRAN

GOTO RESULT_PROC

RETURN

END

-- 이동처리

TRUNCATE TABLE \#TLGInOutDailyBatch

TRUNCATE TABLE \#TLGInOutMonth

TRUNCATE TABLE \#TLGInOutMonthLot

INSERT INTO \#TLGInOutDailyBatch

(

WorkingTag, Status, MessageType, Result,

InOutType, InOutSeq, InOutSerl, InOutSubSerl, DataKind

)

SELECT A.WorkingTag, A.Status, A.MessageType, A.Result,

A.InOutType, A.OutSeq AS InOutSeq, A.OutSerl AS InOutSerl, 0 AS InOutSubSerl, 0 AS DataKind

FROM \#TPJTOutItem AS A

WHERE A.Status = 0

 

EXEC dbo.\_SWLGInOutDailyDELETE @CompanySeq = @CompanySeq, @Level = 2, @LanguageSeq = 1 -- 수불 삭제

EXEC dbo.\_SWLGInOutDailyINSERT @CompanySeq = @CompanySeq, @Level = 2, @LanguageSeq = 1 -- 수불 생성

IF EXISTS(SELECT 1 FROM \#TLGInOutDailyBatch WHERE Status \<\> 0)

BEGIN

SELECT TOP 1 @Message = A.Result, @Ststus = A.Status

FROM \#TLGInOutDailyBatch AS A

WHERE A.Status \<\> 0

ROLLBACK TRAN

GOTO RESULT_PROC

RETURN

END

ELSE

BEGIN

-- 창고재고 집계

EXEC dbo.\_SWLGWHStockUPDATE @CompanySeq = @CompanySeq

-- (-)재고 체크

EXEC dbo.\_SWLGInOutMinusCheck @CompanySeq = @CompanySeq, @TableName = N'#TLGInOutDailyBatch', @LanguageSeq = 1

EXEC dbo.\_SWLGInOutLotMinusCheck @CompanySeq = @CompanySeq, @TableName = N'#TLGInOutDailyBatch', @LanguageSeq = 1

IF EXISTS(SELECT 1 FROM \#TLGInOutDailyBatch WHERE Status \<\> 0)

BEGIN

SELECT TOP 1 @Message = A.Result, @Ststus = A.Status

FROM \#TLGInOutDailyBatch AS A

WHERE A.Status \<\> 0

ROLLBACK TRAN

GOTO RESULT_PROC

RETURN

END

END

COMMIT TRAN

RETURN
