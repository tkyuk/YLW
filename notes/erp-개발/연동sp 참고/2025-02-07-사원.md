# 사원

CREATE OR ALTER VIEW mcm_VMESDAEmp

AS

SELECT ISNULL(A.CompanySeq , 0) AS CompanySeq, -- 법인코드

ISNULL(B.CompanyName ,'') AS CompanyName, -- 법인명

ISNULL(A.EmpSeq , 0) AS EmpSeq, -- 사원코드

ISNULL(A.EmpName ,'') AS EmpName, -- 사원명

ISNULL(A.Empid ,'') AS EmpID, -- 사번

ISNULL(A.DeptSeq , 0) AS DeptSeq, -- 부서코드

ISNULL(C.DeptName ,'') AS DeptName, -- 부서명

ISNULL(E.MinorName ,'') AS UMJpName, -- 직위

ISNULL(F.MinorName ,'') AS UMJdName, -- 직책

ISNULL(G.Email ,'') AS Email, -- 이메일주소

ISNULL(H.MinorSeq , 0) AS IsRetType, -- 재직퇴직구분

CASE WHEN H.MinorSeq = 3031001 THEN '1' ELSE '0' END AS IsRetYn, -- 퇴사여부

ISNULL(I.UserSeq , 0) AS UserSeq, -- 사용자내부코드

ISNULL(I.UserName ,'') AS UserName, -- 사용자이름

ISNULL(A.LastUserSeq , 0) AS LastUserSeq, -- 최종등록자

ISNULL(A.LastDateTime,'') AS LastDateTime -- 최종등록일자

FROM \_TDAEmp AS A WITH(NOLOCK)

LEFT OUTER JOIN \_TCACompany AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

LEFT OUTER JOIN \_TDADept AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.DeptSeq = A.DeptSeq

LEFT OUTER JOIN \_THRAdmOrdEmp AS D WITH(NOLOCK) ON D.CompanySeq = A.CompanySeq

AND D.EmpSeq = A.EmpSeq

AND D.IsOrdDateLast = '1'

AND D.OrdDate \<= ISNULL(A.RetireDate, '99991231')

AND D.OrdDate \<= CASE WHEN ISNULL(A.RetireDate, '99991231') \> CONVERT(NCHAR(8), GETDATE(), 112)

THEN CONVERT(NCHAR(8), GETDATE(), 112)

ELSE ISNULL(A.RetireDate, '99991231')

END

AND D.OrdEndDate \>= CASE WHEN ISNULL(A.RetireDate, '99991231') \> CONVERT(NCHAR(8), GETDATE(), 112)

THEN CONVERT(NCHAR(8), GETDATE(), 112)

ELSE ISNULL(A.RetireDate, '99991231')

END

LEFT OUTER JOIN \_TDAUMinor AS E WITH(NOLOCK) ON E.CompanySeq = D.CompanySeq

AND E.MinorSeq = D.UMJpSeq

LEFT OUTER JOIN \_TDAUMinor AS F WITH(NOLOCK) ON F.CompanySeq = D.CompanySeq

AND F.MinorSeq = D.UMJdSeq

LEFT OUTER JOIN \_TDAEmpIn AS G WITH(NOLOCK) ON G.CompanySeq = A.CompanySeq

AND G.EmpSeq = A.EmpSeq

LEFT OUTER JOIN \_VWXBMSMinor AS H WITH(NOLOCK) ON H.CompanySeq = A.CompanySeq

AND H.MinorSeq = CASE WHEN ISNULL(A.RetireDate, '99991231') \>= CONVERT(NCHAR(8), GETDATE(), 112)

THEN 3031001

ELSE 3031002

END

AND H.LanguageSeq = 1

LEFT OUTER JOIN \_TCAUser AS I WITH(NOLOCK) ON I.CompanySeq = A.CompanySeq

AND I.EmpSeq = A.EmpSeq

-- View_사원

CREATE VIEW emk_VDAEmpView

AS

-- 사원정보 조회 쿼리

SELECT ISNULL(A.CompanySeq , 0) AS CompanySeq,

> ISNULL(A.EmpSeq , 0) AS EmpSeq , -- 사원내부코드

ISNULL(B.EmpName ,'') AS EmpName , -- 사원명

ISNULL(A.EmpId ,'') AS EmpID         , -- 사원번호

ISNULL(D.DeptName ,'') AS DeptName , -- 부서명

ISNULL(A.EntDate ,'') AS EntDate , -- 입사일

ISNULL(A.RetireDate ,'') AS RetireDate, -- 퇴사일

CASE WHEN ISNULL(A.RetireDate, '') \>= CONVERT(NCHAR(8), GETDATE(), 112) THEN '재직자' ELSE '퇴직자' END AS TypeName , -- 재직/퇴직구분

ISNULL(H.MinorName ,'') AS UMPgName , -- 직급

ISNULL(G.MinorName ,'') AS UMJPName , -- 직위

ISNULL(I.MinorName ,'') AS UMJdName , -- 직책

ISNULL(J.MinorName ,'') AS UMJoName , -- 직종                

ISNULL(A.LastDateTime,'') AS LastDateTime -- 최종수정일시(변경일자)

FROM \_TDAEmpIn AS A WITH(NOLOCK)

JOIN \_TDAEmp AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.EmpSeq = B.EmpSeq

LEFT OUTER JOIN \_THRAdmOrdEmp AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.EmpSeq = C.EmpSeq

AND C.OrdDate \< A.RetireDate -- 퇴직일자 이전발령기준

AND C.OrdDate \<= CASE WHEN ISNULL(A.RetireDate, '') \> CONVERT(NCHAR(8),GETDATE(),112) THEN CONVERT(NCHAR(8),GETDATE(),112) ELSE CONVERT(NCHAR(8), DATEADD(dd, -1, ISNULL(A.RetireDate, '')), 112) END

AND C.OrdEndDate \>= CASE WHEN ISNULL(A.RetireDate, '') \> CONVERT(NCHAR(8),GETDATE(),112) THEN CONVERT(NCHAR(8),GETDATE(),112) ELSE CONVERT(NCHAR(8), DATEADD(dd, -1, ISNULL(A.RetireDate, '')), 112) END

AND C.IsOrdDateLast = '1'

JOIN \_TDADept AS D WITH(NOLOCK) ON A.CompanySeq = D.CompanySeq

AND C.DeptSeq = D.DeptSeq

LEFT OUTER JOIN \_TDAUMinor AS G WITH(NOLOCK) ON A.CompanySeq = G.CompanySeq

AND C.UMJpSeq = G.MinorSeq

LEFT OUTER JOIN \_TDAUMinor AS H WITH(NOLOCK) ON A.CompanySeq = H.CompanySeq

AND C.UMPgSeq = H.MinorSeq

LEFT OUTER JOIN \_TDAUMinor AS I WITH(NOLOCK) ON A.CompanySeq = I.CompanySeq

AND C.UMJdSeq = I.MinorSeq

LEFT OUTER JOIN \_TDAUMinor AS J WITH(NOLOCK) ON A.CompanySeq = J.CompanySeq

AND C.UMJoSeq = J.MinorSeq
