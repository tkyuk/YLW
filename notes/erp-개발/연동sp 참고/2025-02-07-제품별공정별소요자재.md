# 제품별공정별소요자재

CREATE OR ALTER VIEW mcm_VMESPDROUItemProc

AS

 

SELECT A.CompanySeq AS CompanySeq , -- 법인내부코드

B.CompanyName AS CompanyName , -- 법인이름

A.ItemSeq AS ItemSeq , -- 제품코드

A.ProcRev AS ProcRev , -- 공정차수

A.BizUnit AS BizUnit , -- 사업단위내부코드

A.ProcSeq AS ProcSeq , -- 공정내부코드

P.ProcName AS ProcName , -- 공정

I.UnitSeq AS UnitSeq , -- 제품단위내부코드

A.AssyItemSeq AS AssyItemSeq , -- 공정품내부코드

M.MatItemSeq AS MatItemSeq , -- 자재내부코드

A.AssyQty AS AssyQty , -- 공정품소요량

A.LastUserSeq AS LastUserSeq , -- 최종등록자

A.LastDateTime AS LastDateTime -- 최종등록일자

FROM \_TPDROUItemProc AS A WITH(NOLOCK)

LEFT OUTER JOIN \_TCACompany AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

LEFT OUTER JOIN \_TPDROUItemProcMat AS M WITH(NOLOCK) ON A.ItemSeq = M.ItemSeq

AND A.ProcRev = M.ProcRev

AND A.BizUnit = M.BizUnit

AND A.Serl = M.Serl

AND A.CompanySeq = M.CompanySeq

JOIN \_TDAItem AS I WITH(NOLOCK) ON A.ItemSeq = I.ItemSeq

AND A.CompanySeq = I.CompanySeq

JOIN \_TDAUnit AS U WITH(NOLOCK) ON I.UnitSeq = U.UnitSeq

AND I.CompanySeq = U.CompanySeq

JOIN \_TPDBaseProcess AS P WITH(NOLOCK) ON A.ProcSeq = P.ProcSeq

AND A.CompanySeq = P.CompanySeq
