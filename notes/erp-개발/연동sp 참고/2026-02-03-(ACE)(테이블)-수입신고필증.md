> -- 따라서 'U'는 삭제/생성 대상에서 제외하기 위해 따로 담는다.

> -- 추후 'D'에서 오류가 발생하였을 경우, 동일 발주/납품에 대한 'U'건에도 Status, Result를 Update하기 위해 구분하여 담는다.

INSERT INTO \#TempDelete(IDXSeq, Kind, SETID, POSeq, POSerl, DelvSeq, DelvSerl)

SELECT DENSE_RANK() OVER(ORDER BY A.POSeq, A.POSerl, A.DelvSeq, A.DelvSerl), A.Kind, A.SETID, A.POSeq, A.POSerl, A.DelvSeq, A.DelvSerl

FROM \#ImpPermitDEL AS A

JOIN (SELECT POSeq, POSerl, DelvSeq, DelvSerl, SUM(Kind) AS Kind

FROM \#ImpPermitDEL

GROUP BY POSeq, POSerl, DelvSeq, DelvSerl

HAVING SUM(Kind) \> 0) AS B ON A.POSeq = B.POSeq

AND A.POSerl = B.POSerl

AND A.DelvSeq = B.DelvSeq

AND A.DelvSerl = B.DelvSerl

 

> -- 'U'인 건에 대해 삭제 대상에서 DELETE

DELETE A

FROM \#ImpPermitDEL AS A

JOIN \#TempDelete AS B ON A.SETID = B.SETID

> AND         B.kind = 0

-- 'U'인 건에 대해 생성 대상에서 DELETE

DELETE A

FROM \#ImpPermitINS AS A

JOIN \#TempDelete AS B ON A.SETID = B.SETID

> AND         B.kind = 0
