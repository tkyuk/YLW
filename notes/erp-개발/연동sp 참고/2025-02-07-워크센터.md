# 워크센터

CREATE OR ALTER VIEW mcm_VMESPDBaseWorkCenter

AS

SELECT ISNULL(A.CompanySeq , 0) AS CompanySeq, -- 법인내부코드

ISNULL(B.CompanyName ,'') AS CompanyName, -- 법인명

ISNULL(A.WorkCenterSeq , 0) AS WorkCenterSeq, -- 워크센터코드

ISNULL(A.WorkCenterName,'') AS WorkCenterName, -- 워크센터명

ISNULL(A.DeptSeq , 0) AS DeptSeq, -- 부서코드

ISNULL(C.DeptName ,'') AS DeptName, -- 부서명

ISNULL(A.MatOutWhSeq , 0) AS MatOutWhSeq, -- 불출창고코드

ISNULL(D.WHName ,'') AS MatOutWhName, -- 불출창고명

ISNULL(A.GoodOutWHSeq , 0) AS GoodOutWhSeq, -- 자재불출창고코드

ISNULL(G.WHName ,'') AS GoodOutWhName, -- 자재불출창고명

ISNULL(A.FieldWhSeq , 0) AS FieldWhSeq, -- 현장창고코드

ISNULL(E.WHName ,'') AS FieldWhName, -- 현장창고명

ISNULL(A.ProdInWhSeq , 0) AS ProdInWhSeq, -- 입고창고코드

ISNULL(F.WHName ,'') AS ProdInWhName, -- 입고창고명

ISNULL(A.LastUserSeq , 0) AS LastUserSeq, -- 최종등록자

ISNULL(A.LastDateTime ,'') AS LastDateTime -- 최종등록일자

FROM \_TPDBaseWorkCenter AS A WITH(NOLOCK)

LEFT OUTER JOIN \_TCACompany AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

LEFT OUTER JOIN \_TDADept AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.DeptSeq = A.DeptSeq

LEFT OUTER JOIN \_TDAWH AS D WITH(NOLOCK) ON D.CompanySeq = A.CompanySeq

AND D.WHSeq = A.MatOutWhSeq

LEFT OUTER JOIN \_TDAWH AS E WITH(NOLOCK) ON E.CompanySeq = A.CompanySeq

AND E.WHSeq = A.FieldWhSeq

LEFT OUTER JOIN \_TDAWH AS F WITH(NOLOCK) ON F.CompanySeq = A.CompanySeq

AND F.WHSeq = A.ProdInWhSeq

LEFT OUTER JOIN \_TDAWH AS G WITH(NOLOCK) ON G.CompanySeq = A.CompanySeq

AND G.WHSeq = A.GoodOutWHSeq
