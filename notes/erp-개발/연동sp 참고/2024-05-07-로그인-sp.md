# 로그인 sp

DROP PROC apro_SPDALogin

GO

CREATE PROC apro_SPDALogin

@CompanySeq INT,

@ID NVARCHAR(100),

@PassWord NVARCHAR(100)

AS

SET NOCOUNT ON;

 

DECLARE @Ststus INT,

@Message NVARCHAR(255),

@UserSeq INT,

@UserID NVARCHAR(100),

@UserName NVARCHAR(100)

SELECT @Ststus = 0, @Message = '정상적으로 로그인하였습니다.',

@CompanySeq = ISNULL(@CompanySeq, 0),

@ID = LTRIM(RTRIM(ISNULL(@ID , ''))),

@PassWord = LTRIM(RTRIM(ISNULL(@PassWord , '')))

 

IF EXISTS(SELECT TOP 1 1 FROM \_TCAUser WITH(NOLOCK) WHERE CompanySeq = @CompanySeq AND UserID = @ID)

BEGIN

SELECT @CompanySeq = @CompanySeq -- Dummy

END

ELSE

BEGIN

SELECT @Ststus = 99999, @Message = '등록된 사용자가 아닙니다.'

GOTO RESULT_PROC

RETURN

END

 

IF EXISTS(SELECT TOP 1 1 FROM \_TCAUser WITH(NOLOCK) WHERE CompanySeq = @CompanySeq AND UserID = @ID

AND LoginPwd COLLATE Korean_Wansung_CS_AS \<\> LOWER(CONVERT(VARCHAR(MAX), HASHBYTES ('SHA2_512', CONVERT(VARCHAR(200), @PassWord)), 2))

)

BEGIN

SELECT @Ststus = 99999, @Message = '비밀번호가 잘못되었습니다. 관리자에게 문의하세요.'

GOTO RESULT_PROC

RETURN

END

IF EXISTS(SELECT TOP 1 1 FROM \_TCAUser WITH(NOLOCK) WHERE CompanySeq = @CompanySeq AND UserID = @ID AND LoginStatus = 18004) -- \_TCAMinor WHERE MajorSeq = 18

BEGIN

SELECT @Ststus = 99999, @Message = '계정이 중지되었습니다. 관리자에게 문의 부탁드립니다.'

GOTO RESULT_PROC

RETURN

END

 

SELECT @UserSeq = A.UserSeq,

@UserID = A.UserID,

@UserName = A.UserName

FROM \_TCAUser AS A WITH(NOLOCK)

WHERE A.CompanySeq = @CompanySeq

AND A.UserID = @ID

RESULT_PROC:

 

SELECT @Ststus AS Ststus, @Message AS Message,

@CompanySeq AS CompanySeq, @UserSeq AS UserSeq, @UserID AS UserID, @UserName AS UserName

 

 

RETURN

 

GO
