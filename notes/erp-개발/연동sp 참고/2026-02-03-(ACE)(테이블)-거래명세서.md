-- 'U', 'D' 일 때, InvoiceSeq 존재여부 체크

UPDATE \#TMESInvoceInterface

SET Result = '자료가 등록되어 있지 않습니다.',

Status = 2

FROM \#TMESInvoceInterface                AS A

LEFT OUTER JOIN \#InvoiceDEL AS D ON A.SETID = D.SETID

WHERE A.WorkingTag IN (N'D', N'U')

AND A.Status = 0

AND D.SETID IS NULL

 

 

-- 'U' 인 건 중 오류발생된 건에 대해서 INSERT 항목으로 처리되어 있는 건 삭제처리

DELETE B

FROM \#TMESInvoceInterface AS A

JOIN \#InvoiceINS AS B ON A.SETID = B.SETID

WHERE A.Status = 2

AND A.WorkingTag = 'U'
