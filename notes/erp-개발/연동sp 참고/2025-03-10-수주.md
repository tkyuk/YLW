# 수주

CREATE VIEW smc_VMESTSLOrderItem

AS

SELECT A.CompanySeq AS CompanySeq, -- 법인내부코드

C.CompanyName AS CompanyName, -- 법인명

A.OrderSeq AS OrderSeq, -- 수주내부코드

B.OrderSerl AS OrderSerl, -- 수주내부순번

A.OrderNo AS OrderNo, -- 수주번호

A.OrderDate AS OrderDate, -- 수주일

D.CustName AS CustName, -- 거래처

A.CustSeq AS CustSeq, -- 거래처내부코드

E.MinorName AS UMOrderKindName, -- 수주구분

O.MinorName AS SMExpKindName, -- Local구분

F.DeptName AS DeptName, -- 부서

G.EmpName AS EmpName, -- 담당자

A.EmpSeq AS EmpSeq, -- 담당자내부코드

A.Remark AS Remark, -- 특이사항

I.ItemClassLName AS ItemClassLName, -- 품목대분류

I.ItemClassMName AS ItemClassMName, -- 품목중분류

I.ItemClassSName AS ItemClassSName, -- 품목소분류

J.ItemName AS ItemName, -- 품명

B.ItemSeq AS ItemSeq, -- 품번내부코드

J.ItemNo AS ItemNo, -- 품번

J.Spec AS Spec, -- 규격

K.UnitName AS UnitName, -- 판매단위

L.UnitName AS StdUnitName, -- 기준단위

B.STDQty AS StdQty, -- 기준단위수량

B.Qty AS Qty, -- 수량

M.WHName AS WHName, -- 창고

B.WHSeq AS WHSeq , -- 창고내부코드

N.AssetName AS AssetName, -- 품목자산분류

B.Remark AS ItemRemark, -- 품목특이사항

P.ToQty AS InvQty

FROM \_TSLOrder AS A

JOIN \_TSLOrderItem AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.OrderSeq = A.OrderSeq

LEFT OUTER JOIN \_TCACompany AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

LEFT OUTER JOIN \_TDACust AS D WITH(NOLOCK) ON D.CompanySeq = A.CompanySeq

AND D.CustSeq = A.CustSeq

LEFT OUTER JOIN \_TDAUMinor AS E WITH(NOLOCK) ON E.CompanySeq = A.CompanySeq

AND E.MinorSeq = A.UMOrderKind

LEFT OUTER JOIN \_TDADept AS F WITH(NOLOCK) ON F.CompanySeq = A.CompanySeq

AND F.DeptSeq = A.DeptSeq

LEFT OUTER JOIN \_TDAEmp AS G WITH(NOLOCK) ON G.CompanySeq = A.CompanySeq

AND G.EmpSeq = A.EmpSeq

LEFT OUTER JOIN \_TDAItemClass AS H WITH(NOLOCK) ON H.CompanySeq = B.CompanySeq

AND H.ItemSeq = B.ItemSeq

AND H.UMajorItemClass IN (2001, 2004)

LEFT OUTER JOIN \_VDAItemClass AS I WITH(NOLOCK) ON I.CompanySeq = H.CompanySeq

AND I.ItemClassSSeq = H.UMItemClass

LEFT OUTER JOIN \_TDAItem AS J WITH(NOLOCK) ON J.CompanySeq = B.CompanySeq

AND J.ItemSeq = B.ItemSeq

LEFT OUTER JOIN \_TDAUnit AS K WITH(NOLOCK) ON K.CompanySeq = B.CompanySeq

AND K.UnitSeq = B.UnitSeq

LEFT OUTER JOIN \_TDAUnit AS L WITH(NOLOCK) ON L.CompanySeq = B.CompanySeq

AND L.UnitSeq = B.STDUnitSeq

LEFT OUTER JOIN \_TDAWH AS M WITH(NOLOCK) ON M.CompanySeq = B.CompanySeq

AND M.WHSeq = B.WHSeq

LEFT OUTER JOIN \_TDAItemAsset AS N WITH(NOLOCK) ON N.CompanySeq = J.CompanySeq

AND N.AssetSeq = J.AssetSeq

LEFT OUTER JOIN \_TDASMinor AS O WITH(NOLOCK) ON O.CompanySeq = A.CompanySeq

AND O.MinorSeq = A.SMExpKind

LEFT OUTER JOIN (SELECT CompanySeq, FromSeq, FromSerl, FromTableSeq, ToTableSeq , SUM(ADD_DEL \* ToQty) AS ToQty

FROM \_TCOMSourceDaily

GROUP BY CompanySeq, FromSeq, FromSerl, FromTableSeq, ToTableSeq

) AS P ON P.CompanySeq = B.CompanySeq

AND P.FromSeq = B.OrderSeq

AND P.FromSerl = B.OrderSerl

AND P.FromTableSeq = 19

AND P.ToTableSeq = 18

-- WHERE ISNULL(B.STDQty, 0) \> ISNULL(P.ToQty, 0)
