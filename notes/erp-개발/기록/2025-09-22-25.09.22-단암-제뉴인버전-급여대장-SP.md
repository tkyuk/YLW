-- 사용할 변수를 선언한다.

DECLARE @docHandle INT , -- XML을 핸들할 변수

@PuSeq INT , -- 급여작업군(코드)

@PbYM NCHAR(6) , -- 적용년월

@PayDate NCHAR(8) , -- 지급일

@AdYM NCHAR(6) , -- 개정연월

@CompYM NCHAR(6) , -- 복합급여항목 개정연월

@ExistsSum NCHAR(1) ,

@CompanyName NVARCHAR(100) ,

@PrtDept INT ,

@PrtOrder INT ,

-- XML파싱

EXEC sp_xml_preparedocument @docHandle OUTPUT, @xmlDocument -- @xmlDocument의 XML을 @docHandle로 핸들한다.

 

> -- XML의 DataBlock1으로부터 값을 가져와 변수에 저장한다.

SELECT @PuSeq = ISNULL(PuSeq , 0), -- 급여작업군

@PbYM = ISNULL(PbYM , ''), -- 적용년월

@SerialName = ISNULL(SerialName , ''), -- 급상여차수

@SerialNo = ISNULL(SerialNo , 0), -- 일련번호

@SMPrtType = ISNULL(SMPrtType , 0), -- 출력구분

@DeptSeq = ISNULL(DeptSeq , 0), -- 부서

@EmpSeq = ISNULL(EmpSeq , 0), -- 사원

@UMDeptLevel = ISNULL(UMDeptLevel , 0), -- 부서집계레벨

@SMPayPrtType = ISNULL(SMPayPrtType, 0), -- 급여출력방식(이메일:3207001, 인쇄용지:3207002)

@SMEmpOrDept = ISNULL(SMEmpOrDept , 0), -- 출력대장종류(개인별:3208001, 부서별:3208002)

@SMObjSeq = ISNULL(SMObjSeq , 0) -- 대상구분(전체:0, 재직자:3031001, 퇴직자:3031002)

FROM OPENXML(@docHandle, N'/ROOT/DataBlock1', @xmlFlags) -- XML의 DataBlock1으로부터

WITH (PuSeq INT ,

PbYM NCHAR(6),

SerialName NVARCHAR(100),

SerialNo INT ,

PbSeq INT ,

PtSeq INT ,

SMPrtType INT ,

DeptSeq INT ,

EmpSeq INT ,

UMDeptLevel INT ,

SMPayPrtType INT ,

SMEmpOrDept INT ,

SMObjSeq INT

)

 

 

-- 급상여구분

IF(ISNULL(@PbSeq, 0) = 0)

BEGIN

-- 급상여일자내역에서 급상여구분을 읽는다.

SELECT @PbSeq = PbSeq

FROM \_TPRPayDateDtl

WHERE CompanySeq = @CompanySeq -- 법인번호

AND PuSeq = @PuSeq -- 급여작업군

AND PbYM = @PbYM -- 적용년월

AND SerialNo = @SerialNo -- 일련번호

END

 

 

 

-- 급상여구분별 계좌

SELECT @AccType = UMAccNoType FROM \_TPRBasPb WHERE CompanySeq = @CompanySeq AND PbSeq = @PbSeq

 

 

 

 

-- 출력대상자를 담을 임시테이블(순서대로 들어갔다라고 보고 진행)

CREATE TABLE \#EmpMASter

(

-- 급상여일자내역에서 일자을 읽는다.

SELECT @BegDate = ISNULL(BegDate, ''), -- 시작일

@EndDate = ISNULL(EndDate, ''), -- 종료일

@PayDate = ISNULL(PayDate, '') -- 지급일

FROM \_TPRPayDateDtl

WHERE CompanySeq = @CompanySeq -- 법인번호

AND PuSeq = @PuSeq -- 급여작업군

AND PbYM = @PbYM -- 적용년월

AND SerialNo = @SerialNo -- 일련번호

 

 

-- 급상여출력시 부서(3511)

-- 소속부서:3205001, 근무부서:3205002

 

EXEC dbo.\_SCOMEnv @CompanySeq, 3511, @UserSeq, @@PROCID, @EnvValue OUTPUT

 

SELECT @PrtDept = CONVERT(INT, @EnvValue)

EXEC dbo.\_SCOMEnv @CompanySeq, 3510, @UserSeq, @@PROCID, @EnvValue OUTPUT

SELECT @PrtOrder = CONVERT(INT, @EnvValue)

 

 

 

-- 개인별인경우

IF(@EmpSeq \<\> 0) -- 사원 한 사람 찍기

BEGIN

 

INSERT INTO \#EmpMASter(EmpSeq , EmpName , -- 사원(코드) , 사원

EmpID , EntDate , -- 사번 , 입사일

RetireDate , ResidId , -- 퇴사일 , 주민등록번호

DeptSeq , PosSeq , -- 부서 , 포지션

UMPgSeq , Ps , -- 직급 , 호봉

UMJpSeq , UMJdSeq , -- 직위 , 직책

UMJoSeq , PtSeq , -- 직종 , 급여형태

UMWsSeq , SMSexSeq , -- 근무상태 , 성별

TotPayAmt , TotPrevPayAmt, -- 지급총액 , 기지급총액

TotAllowBnsAmt, TotDeducAmt , -- 인정상여총액, 공제총액

ActPayAmt , BASMmAmt , -- 실지급액 , 기본급기준금액

BASDdAmt , BASTmAmt , -- 기본일급 , 기본시급

GenMmAmt , GenDdAmt , -- 통상임금 , 통상일급

GenTmAmt , OrgCd , -- 통상시급 , 조직순서

BnsRate , AccNo , -- 상여율 , 계좌번호

BankSeq                 , BankName                ,         -- 은행코드        , 은행명

UMEmpType , UMEmpTypeName) -- 사원구분코드,사원구분

 

SELECT A.EmpSeq , C.EmpName ,

B.EmpID , B.EntDate ,

B.RetireDate , dbo.\_FCOMDecrypt(C.ResidID, '\_TDAEmp', 'ResidID', @CompanySeq),

CASE WHEN @PrtDept = 3205002 THEN A.WkDeptSeq ELSE A.DeptSeq END,

A.PosSeq , A.UMPgSeq ,

A.Ps , A.UMJpSeq ,

A.UMJdSeq , A.UMJoSeq ,

A.PtSeq , A.UMWsSeq ,

A.SMSexSeq , A.TotPayAmt ,

A.TotPrevPayAmt, A.TotAllowBnsAmt,

A.TotDeducAmt , A.ActPayAmt ,

A.BASMmAmt , A.BASDdAmt ,

A.BASTmAmt , A.GenMmAmt ,

A.GenDdAmt , A.GenTmAmt ,

'' , A.BnsRate ,

dbo.\_FCOMDecrypt(D.AccNo, '\_THRBasEmpAccNo', 'AccNo', @CompanySeq),

D.PayBankSeq , (SELECT PayBankName FROM \_THRBasBank WHERE CompanySeq = @CompanySeq AND PayBankSeq = D.PayBankSeq),

B.UMEmpType , E.MinorName

 

FROM \_TPRPayResult AS A JOIN \_TDAEmpIN AS B ON A.CompanySeq = B.CompanySeq

AND A.EmpSeq = B.EmpSeq

 

JOIN \_TDAEmp AS C ON A.CompanySeq = C.CompanySeq

AND A.EmpSeq = C.EmpSeq

 

LEFT OUTER JOIN \_THRBasEmpAccNo AS D ON A.CompanySeq = D.CompanySeq -- 계좌정보추가

AND A.EmpSeq = D.EmpSeq

AND D.UMAccNoType = @AccType

LEFT OUTER JOIN \_TDAUMinor AS E ON B.CompanySeq = E.CompanySeq

AND B.UMEmpType = E.MinorSeq

AND E.MajorSeq = 3059

 

WHERE A.CompanySeq = @CompanySeq -- 법인번호

AND A.EmpSeq = @EmpSeq -- 사원

AND A.PuSeq = @PuSeq -- 급여작업군

AND A.PbYM = @PbYM -- 적용년월

AND A.SerialNo = @SerialNo -- 일련번호

 

END

ELSE

BEGIN

 

-- 출력순서 때문에 동적쿼리로작성

SET @Sql = ''

SET @Sql = @Sql + 'INSERT INTO \#EmpMASter(EmpSeq, EmpName, EmpID, EntDate, RetireDate, ResidId, DeptSeq, PosSeq, UMPgSeq, Ps, UMJpSeq, UMJdSeq, UMJoSeq,

SET @Sql = @Sql + ' CASE WHEN ' + CAST(@PrtDept AS NVARCHAR) + ' = 3205002 THEN A.WkDeptSeq ELSE A.DeptSeq END, '

SET @Sql = @Sql + ' A.PosSeq, A.UMPgSeq, A.Ps, A.UMJpSeq, A.UMJdSeq, A.UMJoSeq, A.PtSeq, A.UMWsSeq, A.SMSexSeq, A.TotPayAmt,

A.TotPrevPayAmt, A.TotAllowBnsAmt, A.TotDeducAmt, A.ActPayAmt, A.BASMmAmt, A.BASDdAmt, A.BASTmAmt, A.GenMmAmt,

A.GenDdAmt, A.GenTmAmt, ISNULL(D.OrgCd, ''999999'') AS OrgCd, A.BnsRate, dbo.\_FCOMDecrypt(G.AccNo, ''\_THRBasEmpAccNo'', ''AccNo'', ' + CAST(@CompanySeq AS NVARCHAR) + '), G.PayBankSeq, (SELECT PayBankName FROM \_THRBasBank WHERE CompanySeq = ' + CAST(@CompanySeq AS NVARCHAR) + ' AND PayBankSeq = G.PayBankSeq), B.UMEmpType, H.MinorName

FROM \_TPRPayResult AS A WITH(NOLOCK) JOIN \_TDAEmpIN AS B WITH(NOLOCK) ON (A.CompanySeq = B.CompanySeq AND A.EmpSeq = B.EmpSeq)

JOIN \_TDAEmp AS C WITH(NOLOCK) ON (A.CompanySeq = C.CompanySeq AND A.EmpSeq = C.EmpSeq)

LEFT OUTER JOIN \_TDAUMinor AS H WITH(NOLOCK) ON B.CompanySeq = H.CompanySeq AND B.UMEmpType = H.MinorSeq AND H.MajorSeq = 3059

LEFT OUTER JOIN \_THRBasEmpAccNo AS G ON A.CompanySeq = G.CompanySeq AND A.EmpSeq = G.EmpSeq AND G.UMAccNoType = ' + CAST(@AccType AS NVARCHAR)

SET @Sql = @Sql + ' LEFT OUTER JOIN \_fnOrgDept(' + CAST(@CompanySeq AS NVARCHAR) + ', 3059001,' + CAST(@DeptSeq AS NVARCHAR) + ',''' + @EndDate + ''') D ON '

SET @Sql = @Sql + ' (CASE WHEN '+CAST(@PrtDept AS NVARCHAR)+' = 3205002 THEN A.WkDeptSeq ELSE A.DeptSeq END = D.DeptSeq) '

 

IF @PrtOrder IN (3204002, 3204004, 3204005, 3204008, 3204009) -- 직위가 있는

SET @Sql = @Sql + ' LEFT OUTER JOIN \_TDAUMinor E WITH(NOLOCK) ON (A.CompanySeq = E.CompanySeq AND A.UMJpSeq = E.MinorSeq) '

ELSE IF @PrtOrder IN (3204003, 3204006, 3204007) -- 직급이 있는

SET @Sql = @Sql + ' AND A.PbYM = ''' + @PbYM + ''''

SET @Sql = @Sql + ' AND A.SerialNo = ' + CAST(@SerialNo AS NVARCHAR)

SET @Sql = @Sql + ' AND (D.DeptSeq = ' + CAST(@DeptSeq AS NVARCHAR) + ' OR ' + CAST(@DeptSeq AS NVARCHAR) + ' = 0) '

ELSE IF @PrtOrder IN (3204008) -- 직위(내림차순)+사번

ELSE IF @PrtOrder IN (3204009) -- 직위(오름차순)+사번

IF @@ERROR \<\> 0
