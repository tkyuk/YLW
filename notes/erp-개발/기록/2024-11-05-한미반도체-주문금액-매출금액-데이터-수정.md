BEGIN TRAN

-- 백업 데이터 생성

SELECT A.\*

INTO TSASOItem_BAK241105

FROM TSASOItem AS A

JOIN \#TMP AS B ON B.SONO = A.SONO

ORDER BY IDX_NO

 

 

-- 수정 전 데이터 확인

SELECT A.PRICE, A.KorPrice,A.OriginPrice ,

A.Amt , A.KorAmt , B.Amt AS AfterAmt,

A.Vat , A.KorVat , B.Vat AS AfterVat

FROM TSASOItem AS A

JOIN \#TMP AS B ON B.SONO = A.SONO

ORDER BY IDX_NO

 

-- UPDATE

UPDATE TSASOItem

SET PRICE = B.Amt,

KorPrice = B.Amt,

OriginPrice = B.Amt,

Amt = B.Amt,

KorAmt = B.Amt,

Vat = B.Vat,

KorVat = B.Vat

FROM TSASOItem AS A

JOIN \#TMP AS B ON B.SONO = A.SONO

-- 수정 후 데이터

SELECT A.PRICE, A.KorPrice,A.OriginPrice ,

A.Amt , A.KorAmt , B.Amt AS AfterAmt,

FROM TSASOItem AS A

JOIN \#TMP AS B ON B.SONO = A.SONO

ORDER BY IDX_NO

ROLLBACK TRAN
