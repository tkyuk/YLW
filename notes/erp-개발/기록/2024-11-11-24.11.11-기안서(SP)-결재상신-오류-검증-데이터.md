# 24.11.11 기안서(SP) 결재상신 오류 검증 데이터

begin tran

exec \_SCOMGroupWareStatus @pWorkingTag=N'E',@CompanySeq=N'1',@LanguageSeq=N'1',@UserSeq=N'1',@WorkKind=N'PJTSvc',@TblKey=N'166',@TopUserName=N'장종우',@TopUserPosition=N'',@RevUserName=N'',@RevUserPosition=N'S20180103',@DocID=N'77202',@SiteInit=N'danam'

 

rollback tran

 

 

 

SELECT LastDateTime,\* FROM \_TPJTProject WHERE PJTSeq = 14033 -- SB230501BS001

SELECT LastDateTime,\* FROM \_TPJTProject WHERE PJTno = 'SB230501BS009'

 

 

SELECT \* FROM \_TCAPgm WHERE PgmSeq = 1029162 -- 제품프로젝트일괄등록_danam

 

 

SELECT LastDateTime,\* FROM danam_TPJTSVCProject WHERE SVCSeq = 166

 

SELECT LastDateTime,\* FROM danam_TPJTSVCProjectPJT WHERE PJTNo = 'SB230501BS001'

 

SELECT LastDateTime,\* FROM danam_TPJTSVCProjectPJT WHERE SVCSeq = 166 AND TabNo = 1 -- SB230501BS009

SELECT LastDateTime,\* FROM danam_TPJTSVCProjectPJTLog WHERE SVCSeq = 166 AND TabNo = 1

 

 

SELECT B.PJTNo, B.PJTSeq,B.\*

FROM danam_TPJTSVCProject AS A WITH(NOLOCK)

JOIN danam_TPJTSVCProjectPJT AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.SVCSeq = B.SVCSeq

WHERE A.SVCSeq = 166

AND TabNo = 1

 

SELECT \*

FROM \_TPJTProject

WHERE PJTNo = 'SB230501BS009'

 

 

 

SELECT \* FROM danam_TPJTSVCProjectPJTLog WHERE SVCSeq = 166
