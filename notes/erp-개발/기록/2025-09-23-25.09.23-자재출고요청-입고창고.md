# 25.09.23 자재출고요청 입고창고

-- das_SWMatOutCombineReqQuery

-- 자재출고처리(통합)\_das

SELECT B.WorkCenterSeq,UseType,H.FieldWhSeq AS InWHSeq,

CASE WHEN A.UseType = 6044001 THEN 1

WHEN A.UseType = 6044002 THEN 2

WHEN A.UseType = 6044004 THEN 2

ELSE 0 END,

A.OutReqSeq, B.OutReqItemSerl, A.OutReqNo, A.ReqDate,

A.EmpSeq, A.DeptSeq, A.Remark, B.Memo4, D.GoodItemSeq,

B.ItemSeq, ISNULL(G.COCLevelSeq, F.COCLevelSeq), B.Qty, 0, 0,

0, 0, B.WorkDate, B.Memo5, '0',

D.WorkOrderNo, 0, CASE WHEN IA.SMAssetGrp = 6008005 THEN H.FieldWhSeq -- 공정품

WHEN SM.MinorValue = N'1' THEN H.MatOutWhSeq -- 자재

ELSE H.GoodOutWHSeq -- 제상품

END AS OutWHSeq,

H.FieldWhSeq AS InWHSeq

FROM \_TPDMMOutReqM AS A WITH(NOLOCK)

JOIN \_TPDMMOutReqItem AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.OutReqSeq = B.OutReqSeq

LEFT OUTER JOIN \_TPJTProject AS C WITH(NOLOCK) ON B.CompanySeq = C.CompanySeq

AND B.Memo4 = C.PJTSeq

LEFT OUTER JOIN \_TPDSFCWorkOrder AS D WITH(NOLOCK) ON B.CompanySeq = D.CompanySeq

AND B.WorkOrderSeq = D.WorkOrderSeq

AND B.WorkOrderSerl = D.WorkOrderSerl

LEFT OUTER JOIN \_TDAItem AS E WITH(NOLOCK) ON D.CompanySeq = E.CompanySeq

AND D.GoodItemSeq = E.ItemSeq

LEFT OUTER JOIN das_TPJTProject AS F WITH(NOLOCK) ON B.CompanySeq = F.CompanySeq

AND B.Memo4 = F.PJTSeq

LEFT OUTER JOIN DAS_TPDQAPjtItemCOCLevel AS G WITH(NOLOCK) ON B.CompanySeq = G.CompanySeq

AND B.Memo4 = G.PJTSeq

AND B.ItemSeq = G.ItemSeq

LEFT OUTER JOIN \_TPDBaseWorkCenter AS H WITH(NOLOCK) ON B.CompanySeq = H.CompanySeq

AND B.WorkCenterSeq = H.WorkCenterSeq

JOIN \_TDAItem AS I WITH(NOLOCK) ON B.CompanySeq = I.CompanySeq

AND B.ItemSeq = I.ItemSeq

JOIN \_TDAItemAsset AS IA WITH(NOLOCK) ON I.CompanySeq = IA.CompanySeq

AND I.AssetSeq = IA.AssetSeq

JOIN \_TDASMinor AS SM WITH(NOLOCK) ON IA.SMAssetGrp = SM.MinorSeq

AND SM.MajorSeq = 6008

AND IA.CompanySeq = SM.CompanySeq

WHERE A.CompanySeq = 1

AND A.UseType IN (6044001, 6044002, 6044004) -- 생산출고, 생산추가출고 + 외주추가출고

AND A.OutReqNo LIKE '202509180002'

 

 

-- 자재출고요청품목조회

select UseType,P.WorkCenterSeq,CWH.WHSeq,M.UseType,

CASE

WHEN M.UseType = 6044004 THEN CWH.WHSeq -- 외주추가인 경우 거래처의 생산외주창고

ELSE P.FieldWhSeq

END AS InWHSeq

, CASE

WHEN M.UseType = 6044004 THEN CWH.WHName -- 외주추가인 경우 거래처의 생산외주창고

ELSE IWH.WHName

END AS InWHName

FROM \_TPDMMOutReqM AS M WITH(NOLOCK)

JOIN \_TPDMMOutReqItem AS R WITH(NOLOCK) ON M.CompanySeq = R.CompanySeq

AND M.OutReqSeq = R.OutReqSeq

LEFT OUTER JOIN \_TDAItem AS RI WITH(NOLOCK) ON M.CompanySeq = RI.CompanySeq

AND R.ItemSeq = RI.ItemSeq

LEFT OUTER JOIN \_TPDSFCWorkOrder AS WO WITH(NOLOCK) ON M.CompanySeq = WO.CompanySeq

AND R.WorkOrderSeq = WO.WorkOrderSeq

AND R.WorkOrderSerl = WO.WorkOrderSerl

LEFT OUTER JOIN \_TPDBaseWorkCenter AS P WITH(NOLOCK) ON WO.CompanySeq = P.CompanySeq

AND WO.WorkCenterSeq= P.WorkCenterSeq

LEFT OUTER JOIN \_TDAWH AS IWH WITH(NOLOCK) ON P.CompanySeq = IWH.CompanySeq

AND P.FieldWHSeq = IWH.WHSeq

LEFT OUTER JOIN \_TDAWH AS CWH WITH(NOLOCK) ON R.CustSeq = CWH.CommissionCustSeq

AND M.BizUnit = CWH.BizUnit

AND CWH.SMWHKind = 8002024 -- 생산외주창고

AND CWH.IsNotUse \<\> N'1'

WHERE M.CompanySeq = 1

AND M.UseType IN (6044001, 6044002, 6044004) -- 생산출고, 생산추가출고 + 외주추가출고

AND M.OutReqNo LIKE '202509180002'
