# 25.09.16 단암 자재등록요청 품목코드 업데이트

BEGIN TRAN

 

DECLARE @Seq INT

,@Seq2 INT

> ,@Seq3 INT

SELECT @Seq = max(ItemSeq)

from das_TDAItemReq

 

SELECT @Seq2 = max(ItemSeq)

from \_TDAItem

 

if @Seq \> @Seq2

select @Seq3 = @Seq

else

select @Seq3 = @Seq2

select @Seq3

SELECT \* FROM das_TDAItemReq WHERE ITEMSEQ = 189213

SELECT \* FROM das_TDAItemReq WHERE ITEMSEQ = 189214

SELECT \* FROM das_TDAItemReq WHERE ITEMSEQ = 189219

UPDATE das_TDAItemReq

SET ITEMSEQ = @Seq3 + 1

FROM das_TDAItemReq

WHERE ITEMSEQ = 189213

 

UPDATE das_TDAItemReq

SET ITEMSEQ = @Seq3 + 2

FROM das_TDAItemReq

WHERE ITEMSEQ = 189214

 

UPDATE das_TDAItemReq

SET ITEMSEQ = @Seq3 + 3

FROM das_TDAItemReq

WHERE ITEMSEQ = 189219

 

 

UPDATE \_TCOMCreateSeqMax

SET MaxSeq = @Seq3 + 3

WHERE CompanySeq = 1AND TableName = '\_TDAItem'

 

SELECT \* FROM das_TDAItemReq WHERE ITEMSEQ = 189305

SELECT \* FROM das_TDAItemReq WHERE ITEMSEQ = 189306

SELECT \* FROM das_TDAItemReq WHERE ITEMSEQ = 189307

select \* from \_TCOMCreateSeqMax where TableName = '\_TDAItem'

 

commit TRAN
