# 25.09.11 부서 코드도움 사용하는 화면 조회

-- 부서_WBS : 610010

-- 부서_VEN : 2720020

 

 

-- 2. ACE

SELECT A.Caption

,B.ControlName

,B.ControlCaption

,B.DataFieldName

,B.DataFieldCd

,A.\*

FROM dasCommon..\_TCAPgm AS A WITH(NOLOCK)

JOIN dasCommon..\_TCAPgmcontrols AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.PgmSeq = A.PgmSeq

WHERE b.CodeHelpConst = 610010

AND A.PgmId LIKE '%DAS%'

ORDER BY 1

/\*

출장/교육/외근/시험근무보고서 전표처리_das
