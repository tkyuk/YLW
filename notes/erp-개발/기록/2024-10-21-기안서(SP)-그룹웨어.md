DECLARE @CompanySeq INT = 1

,@TblKey INT = 168

 

/\*

DROP Table \#TmpSVC

Create Table \#TmpSVC

(

WorkingTag NCHAR(1),

IDX_NO INT Identity,

DataSeq INT,

Status INT,

Selected INT,

PJTTypeSeq INT,

PJTTypeName NVARCHAR(50),

PJTSeq INT,

ProdItemSeq INT,

CustSeq INT,

PJTName NVARCHAR(100),

PJTNo NVARCHAR(50),

CustName NVARCHAR(100),

ProdItemName NVARCHAR(100),

ProdItemNo NVARCHAR(100),

Qty DECIMAL(19,5),

Price DECIMAL(19,5),

Amt DECIMAL(19,5),

Vat DECIMAL(19,5),

UnitFr INT,

UnitTo INT,

ItemGubunName NVARCHAR(100),

DelvDueDate NCHAR(8),

ItemGubunSeq INT,

EstDeliveryDate NCHAR(8),

SortNum INT,

PlanFrDate NCHAR(8),

PlanToDate NCHAR(8),

ContractFrDate NCHAR(8),

ContractToDate NCHAR(8),

IsWorkTimeEnd NCHAR(1),

Etc NVARCHAR(1000),

ConDelvDate NCHAR(8),

TABLE_NAME NVARCHAR(100),

BizPJTSeq INT

)

 

 

Insert Into \#TmpSVC

(

WorkingTag ,

DataSeq , Status , Selected , PJTTypeSeq , PJTTypeName ,

PJTSeq , ProdItemSeq , CustSeq , PJTName , PJTNo ,

CustName , ProdItemName , ProdItemNo , Qty , Price ,

Amt , Vat , UnitFr , UnitTo , ItemGubunName ,

DelvDueDate , ItemGubunSeq , EstDeliveryDate , SortNum , PlanFrDate ,

PlanToDate , ContractFrDate , ContractToDate , IsWorkTimeEnd , Etc ,

ConDelvDate , TABLE_NAME , BizPJTSeq

)

 

Select Case When Isnull(P.PJTSeq, 0) = 0 Then 'A' Else 'U' End AS WorkingTag,

1 AS DataSeq , 0 AS Status , 1 AS Selected , B.PJTTypeSeq , '서비스' AS PJTTypeName,

B.PJTSeq , IsNull(P5.ItemSeq, 0) AS ProdItemSeq, A.CustSeq , Rtrim(IsNull(I5.ItemName, '')) + '\_' + Right(B.PJTNo, 3) + '\_SP' AS PJTName, B.PJTNo ,

C.CustName , B.PJTNo , B.PJTNo, 1 AS Qty , IsNull(P2.ItemDomAmt, 0) + IsNull(D.Amt, 0),

IsNull(P2.ItemDomAmt, 0) + IsNull(D.Amt, 0), (IsNull(P2.ItemDomAmt, 0) + IsNull(D.Amt, 0)) \* 0.1 As Vat, 0 AS UnitFr, 0 AS UnitTo , '제작' AS ItemGubunName,

A.PlanToDate, 1011243001 , '' , IsNull(P3.SortNum, 0) , A.PlanFrDate AS PlanFrDate,

A.PlanToDate AS PlanToDate, '' AS ContractFrDate, '' AS ContractToDate, '0' , '' AS Etc ,

A.PlanToDate, 'DataBlock1' , A.BizPJTSeq

From danam_TPJTSVCProject AS A with(nolock) Join danam_TPJTSVCProjectPJT AS B with(nolock)

On A.CompanySeq = B.CompanySeq

And A.SVCSeq = B.SVCSeq

Left Outer Join \_TDACust AS C with(nolock)

On A.CompanySeq = C.CompanySeq

And A.CustSeq = C.CustSeq

Left Outer Join (Select CompanySeq, SVCSeq, PJTSeq, Sum(Amt) AS Amt

From danam_TPJTSVCProjectItem with(nolock)

Where CompanySeq = @CompanySeq

And SVCSeq = @TblKey

Group by CompanySeq, SVCSeq, PJTSeq) AS D

On B.CompanySeq = D.CompanySeq

And B.SVCSeq = D.SVCSeq

And B.PJTSeq = D.PJTSeq

Left Outer Join \_TPJTProject AS P with(nolock)

On B.CompanySeq = P.CompanySeq

And B.PJTNo = P.PJTNo

And P.PJTTypeSeq = 14 --서비스프로젝트

Left Outer Join (Select CompanySeq, PJTSeq, Sum(ItemDomAmt) AS ItemDomAmt

From \_TPJTProjectItem A with(nolock)

Group by CompanySeq, PJTSeq) AS P2

On P.CompanySeq = P2.CompanySeq

And P.PJTSeq = P2.PJTSeq

Left Outer Join danam_TPJTProject AS P3 with(nolock)

On P.CompanySeq = P3.CompanySeq

And P.PJTSeq = P3.PJTSeq

Left Outer Join (Select a.CompanySeq, MAX(a.PJTSeq) AS PJTSeq, Left(a.PJTNo, 9) AS PJTNo

From \_TPJTProject a with(nolock) Join danam_TPJTSVCProjectPJT b with(nolock)

on a.CompanySeq = b.CompanySeq

and Left(a.PJTNo, 9) = Left(b.PJTNo, 9)

Left Outer Join \_TPJTProjectItem AS P5 with(nolock)

On a.CompanySeq = P5.CompanySeq

And a.PJTSeq = P5.PJTSeq

And P5.PJTItemSerl = 1

Where a.CompanySeq = @CompanySeq

and b.SVCSeq = @TblKey

Group by a.CompanySeq, Left(a.PJTNo, 9)

) AS P4

On B.CompanySeq = P4.CompanySeq

and Left(B.PJTNo, 9) = P4.PJTNo

Left Outer Join \_TPJTProjectItem AS P5 with(nolock)

On P4.CompanySeq = P5.CompanySeq

And P4.PJTSeq = P5.PJTSeq

And P5.PJTItemSerl = 1

Left Outer Join \_TDAItem AS I5 with(nolock)

ON P5.CompanySeq = I5.CompanySeq

and P5.ItemSeq = I5.ItemSeq

Where A.CompanySeq = @CompanySeq

And A.SVCSeq = Convert(INT, @TblKey)

 

 

\*/

SELECT PJTNo,\* FROM \#TmpSVC

-- SA221801AS003

SELECT B.\*

FROM \#TmpSVC AS A JOIN ( SELECT S.PJTNo

FROM (

SELECT A1.PJTNo

FROM \#TmpSVC AS A1

WHERE A1.WorkingTag IN ('A','U')

AND A1.Status = 0

UNION ALL

SELECT A1.PJTNo

FROM \_TPJTProject AS A1

WHERE A1.Companyseq = @CompanySeq

AND A1.PJTSeq NOT IN (SELECT PJTSeq

FROM \#TmpSVC

WHERE WorkingTag IN ('U','D')

AND Status = 0)

) AS S

GROUP BY S.PJTNo

HAVING COUNT(1) \> 1

) AS B ON (A.PJTNo = B.PJTNo)

 

 

 

SELECT S.PJTNo ,COUNT(\*)

FROM (

SELECT A1.PJTNo

FROM \#TmpSVC AS A1

WHERE A1.WorkingTag IN ('A','U')

AND A1.Status = 0

UNION ALL

SELECT A1.PJTNo

FROM \_TPJTProject AS A1

WHERE A1.Companyseq = @CompanySeq

AND A1.PJTSeq NOT IN (SELECT PJTSeq

FROM \#TmpSVC

WHERE WorkingTag IN ('U','D')

AND Status = 0)

) AS S

GROUP BY S.PJTNo

HAVING COUNT(\*) \> 1
