# 품목소분류별검사대상등록 / 품목별예외검사대상등록

SELECT A.EnvValue, B.WHName

FROM \_TCOMEnv AS A WITH(NOLOCK)

LEFT OUTER JOIN \_TDAWH AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.EnvValue = B.WHSeq

WHERE A.CompanySeq = 1

AND A.EnvSeq = 2000062 -- 검사품입고창고_biont

 

 

DECLARE @CompanySeq INT = 1

 

 

/\*

SELECT DISTINCT

A.MinorSeq AS UMItemClass,

A.MinorName AS UMItemClassName,

B.IsNotQC,

C.ItemClassMName AS UMItemClassNameM,

C.ItemClassLName AS UMItemClassNameL,

B.IsFinQC,

B.IsInQC,

B.IsOutQC,

> CASE WHEN A.MajorSeq = 2001 THEN (SELECT MinorName FROM \_TDASMinor WITH(NOLOCK) WHERE CompanySeq = @CompanySeq AND MinorSeq = 8010001)
>
> ELSE (SELECT MinorName FROM \_TDASMinor WITH(NOLOCK) WHERE CompanySeq = @CompanySeq AND MinorSeq = 8010002) END AS SMItemTypeName,
>
> CASE WHEN A.MajorSeq = 2001 THEN 8010001 ELSE 8010002 END AS SMItemType,

B.IsAfterQCPAM,

B.IsPurQC,

B.IsOspQC,

B.IsPMSOutQc

FROM \_TDAUMinor AS A WITH(NOLOCK)

LEFT OUTER JOIN \_TPDQAItemClassQCKind AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.MinorSeq = B.UMItemClass

LEFT OUTER JOIN \_VDAItemClass AS C WITH(NOLOCK) ON A.MinorSeq = C.ItemClassSSeq

AND A.CompanySeq = C.CompanySeq

WHERE A.CompanySeq = @CompanySeq

AND A.MajorSeq IN (2001, 2004)

ORDER BY CASE WHEN A.MajorSeq = 2001 THEN 8010001 ELSE 8010002 END, C.ItemClassLName, C.ItemClassMName, A.MinorName                -- By 18.05.28 hjlim 수정 (품목구분, 대분류명, 중분류명, 소분류명 순으로 정렬)

 

 

 

SELECT \*

FROM \_TPDQAItemQCKind

 

SELECT \* FROM \_TDASMinor WHERE MajorSeq = 6018

\*/

-- 인수검사 IsInQC        

-- 최종검사 IsFinQC        

-- 출하검사 IsOutQC        

 

 

-- 입고후시험 IsAfterQCPAM

-- 외주검사 IsOspQC

-- 공정시험 IsPMSOutQC

 

/\*

6018001 공정검사

6018002 인수검사

6018003 최종검사

6018004 출하검사

6018005 구매검사

6018006 외주검사

6018008 입고후시험

\*/

 

 

 

 

SELECT A.ItemSeq,

C.IsInQC,

C.IsAfterQCPAM

FROM \_TDAItem AS A

JOIN \_TDAItemClass AS B WITH(NOLOCK) ON B.ItemSeq = A.ItemSeq

AND B.UMajorItemClass IN (2001, 2004)

LEFT OUTER JOIN \_TPDQAItemClassQCKind AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.UMItemClass = B.UMItemClass

WHERE A.ItemSeq = 25359

 

 

--SELECT \* FROM \_TPDQAItemClassQCKind WHERE UMItemClass = 2001035

 

 

SELECT A.ItemSeq,

CASE WHEN B.IsUse = '1' THEN '1' ELSE '0' END AS IsInQC,

CASE WHEN C.IsUse = '1' THEN '1' ELSE '0' END AS IsAfterQCPAM

FROM \_TDAItem AS A

LEFT OUTER JOIN \_TPDQAItemQCKind AS B WITH(NOLOCK) ON B.CompanySeq = 1

AND B.ItemSeq = A.ItemSeq

AND B.SMQcKind = 6018002

LEFT OUTER JOIN \_TPDQAItemQCKind AS C WITH(NOLOCK) ON C.CompanySeq = 1

AND C.ItemSeq = A.ItemSeq

AND C.SMQcKind = 6018003

WHERE A.ItemSeq = 25359
