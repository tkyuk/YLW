# 25.03.31 단암 월근태현황 INSERT 데이터 확인

/\*

생성 sp : danam_ADTWorkHistoryCreate

테이블 : \[daou_gw\].\[do\].\[public\].adt_workhistory_new

OutTime 컬럼의 9번째 자리가 0인 경우 WorkDate에서 하루 뺀 날짜를 입력

(9번째 자리 = 시간

ex. 20250203082743

-\> 2025-0203-08:27:43 =\> 08시 27분 43초의 맨 앞자리인 0

=

\*/

 

--select \* from \[daou_gw\].\[do\].\[public\].adt_workhistory_new where workdate ='20250202' and employee_number ='20170601'

 

--select \* from ADT_WorkHistory_danam where empseq = 433 and WkDate ='20250202'

--select \* from ADT_WorkHistory_danam where empseq = 433 and WkDate ='20250203'

 

 

Select IsNull(B.EmpSeq, 0) As EmpSeq

, A.Employee_number

, A.WorkDate

, 'out' AS WorkType

, Max(Right(A.OutTime, 6)) AS WorkTime

, Case When Substring(A.OutTime, 9, 1) = '0' Then convert(NCHAR(8), DATEADD(Day, -1, A.WorkDate), 112) Else A.WorkDate End AS WkDate

From \[daou_gw\].\[do\].\[public\].adt_workhistory_new a Left Outer Join \_TDAEmp b on b.CompanySeq = 1 and a.Employee_number = b.EmpId

Where Isnull(A.OutTime, '') \<\> ''

--and A.WorkDate Between convert(NCHAR(8), DATEADD(Day, -1, Convert(NCHAR(8), GETDATE(), 112)), 112) and Convert(NCHAR(8), GETDATE(), 112)

and a.workdate like '202502%'

and employee_number ='20170601'

Group by IsNull(B.EmpSeq, 0), A.Employee_number, A.WorkDate,

Case When Substring(A.OutTime, 9, 1) = '0' Then convert(NCHAR(8), DATEADD(Day, -1, A.WorkDate), 112) Else A.WorkDate End,

Case When Left(A.Employee_number, 1) ='G' Then Convert(INT, Right(Rtrim(A.Employee_number), 3)) Else 0 End

Order by a.WorkDate
