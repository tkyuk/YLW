SELECT TOP 100 FromSeq,ToSeq,B.SMResrcType,C.SMResrcType

FROM \_TCOMSourceDaily AS A

JOIN \_TPJTSupplyContDelvItem AS B ON B.SupplyDelvSeq = A.FromSeq

JOIN \_TPJTResultExpenseItem AS C ON C.ExpenseSeq = A.ToSeq

WHERE FromTableSeq = 62

AND ToTableSeq = 63

ORDER BY A.LastDateTime DESC

SELECT B.ExpenseDate,\*

FROM \_TPJTResultExpenseItem AS A

JOIN \_TPJTResultExpenseM AS B ON B.ExpenseSeq = A.ExpenseSeq

WHERE ExpenseDate = '20240131'

SELECT B.ExpenseDate,A.ExpenseSumAmt,C.SupplyDelvDomSumAmt,\*

FROM \_TPJTResultExpenseItem AS A

JOIN \_TPJTResultExpenseM AS B ON B.ExpenseSeq = A.ExpenseSeq

JOIN \_TPJTSupplyContDelvItem AS C ON C.CompanySeq = A.CompanySeq

AND C.SupplyDelvSeq = A.ProgFromSeq

AND C.SupplyDelvSerl = A.ProgFromSerl

WHERE ExpenseDate = '20240131'

ORDER BY 1 DESC
