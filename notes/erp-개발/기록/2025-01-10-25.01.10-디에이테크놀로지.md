# 25.01.10 디에이테크놀로지

--\[활동센터별전표발행조회\]

-- 전표집계 (전표번호별로 비용항목계정인 전표들을 따로 EsM용 테이블에 저장시킨다)

SELECT DISTINCT LEFT(AccDate,6)

FROM \_TESMGSlipCost

WHERE LEFT(AccDate,6) = '202402'

AND AccSeq = 418

 

--\[활동센터별비용집계조회\]

-- 활동센터별금액집계

SELECT \*

FROM \_TESMCProfCostSum AS A WITH(NOLOCK)

JOIN \_TESMDCostKey AS K WITH(NOLOCK) ON A.CompanySeq = K.CompanySeq

AND A.CostKeySeq = K.CostKeySeq

WHERE K.CostYm = '202402'

AND A.CostAccSeq = 247
