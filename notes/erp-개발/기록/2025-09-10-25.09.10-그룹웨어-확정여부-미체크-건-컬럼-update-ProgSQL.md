# 25.09.10 그룹웨어 확정여부 미체크 건 컬럼 update ProgSQL

BEGIN TRAN

UPDATE B

SET B.CfmCode = '1'

FROM \_TCOMConfirm AS B WITH(NOLOCK)

LEFT OUTER JOIN \_TCOMGroupWare AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.TblKey = B.CfmSeq

WHERE B.ConfirmSeq = 15000169

AND C.WorkKind = 'ExpendReq'

AND C.IsEnd = '1'

AND B.CfmCode \<\> '1'

ROLLBACK TRAN
