ssv_TXPRDeposit

begin tran

 

SELECT B.EMPNAME,A.\*

FROM ssv_TXPRDeposit AS A

JOIN \#TMP AS B ON A.CompanySeq = 1

AND A.DepositYM = '202401'

AND A.DepositType = 0

AND A.EmpSeq = B.EmpSeq

AND A.Amt = B.OldAmt

 

 

UPDATE ssv_TXPRDeposit

SET Amt = B.Amt

FROM ssv_TXPRDeposit AS A

JOIN \#TMP AS B ON A.CompanySeq = 1

AND A.DepositYM = '202401'

AND A.DepositType = 0

AND A.EmpSeq = B.EmpSeq

 

SELECT B.EMPNAME,OLDAMT,A.\*

FROM ssv_TXPRDeposit AS A

JOIN \#TMP AS B ON A.CompanySeq = 1

AND A.DepositYM = '202401'

AND A.DepositType = 0

AND A.EmpSeq = B.EmpSeq

 

rollback tran
