# 25.01.07 엘에스머트리얼즈 합의된 건 신규 저장 시 시트에 에러 반환 안됨

DROP PROC sys_SWPDBOMReqMngSave

GO

CREATE PROC sys_SWPDBOMReqMngSave

@xmlDocument NVARCHAR(MAX) ,

@xmlFlags INT = 0,

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0

AS

DECLARE @MessageType INT,

@Status INT,

@Results NVARCHAR(250),

@EmpSeq INT

-- 서비스 마스타 등록 생성

CREATE TABLE \#TPDBOMManagement (WorkingTag NCHAR(1) NULL)

EXEC dbo.\_SCAOpenXmlToTemp @xmlDocument, @xmlFlags, @CompanySeq, @ServiceSeq, 'DataBlock4', '#TPDBOMManagement'

IF @@ERROR \<\> 0 RETURN

-- BINHVD 20230426

SELECT @EmpSeq = EmpSeq FROM \_TCAUser WHERE UserSeq = @UserSeq

SELECT \*

INTO \#\_TDAUMinorValue

FROM \_TDAUMinorValue

WHERE CompanySeq = @CompanySeq

AND MajorSeq = 1027281

UPDATE \#TPDBOMManagement

SET Result = '승인완료된 건은 수정, 삭제 불가능 합니다.',

MessageType = 17009,

Status = 9999

FROM \#TPDBOMManagement AS A

JOIN sys_TPDBOMReqManagement AS B ON B.CompanySeq = @CompanySeq

AND B.ItemSeq = A.ItemSeq

AND B.ItemBOMRevRemark = A.ItemBOMRevRemark

AND B.BOMReqSeq = A.BOMReqSeq

JOIN sys_TPDBOMReqManagement_Confirm AS C ON C.CompanySeq = @CompanySeq

AND C.CfmSeq = A.BOMReqSeq

WHERE C.CfmCode = '1'

 

--SELECT \*

--FROM \_TDAUMinorValue

--WHERE MajorSeq = 1027281

EXEC dbo.\_SCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

6 , -- 중복된 @1 @2가(이) 입력되었습니다.(SELECT \* FROM \_TCAMessageLanguage WHERE MessageSeq = 6)

@LanguageSeq ,

0,'' -- SELECT \* FROM \_TCADictionary WHERE Word like '%거래처%'

UPDATE \#TPDBOMManagement

SET Result = CASE WHEN ISNULL(A.Agree1,'0') \<\> '0' THEN N'합의1 컨트롤에 체크할 권한이 없습니다.'

WHEN ISNULL(A.Agree2,'0') \<\> '0' THEN N'합의2 컨트롤에 체크할 권한이 없습니다.'

WHEN ISNULL(A.Agree3,'0') \<\> '0' THEN N'합의3 컨트롤에 체크할 권한이 없습니다.'

--WHEN ISNULL(A.CheckDelvTeamManager,'0') \<\> '0' THEN N'개발팀장 컨트롤에 체크할 권한이 없으십니다.'

ELSE N'컨트롤에 체크할 권한이 없습니다.' END,

MessageType = @MessageType,

Status = @Status

FROM \#TPDBOMManagement AS A

WHERE WorkingTag = 'A'

AND Status = 0

AND ISNULL(BOMReqSeq,0) = 0

AND ((ISNULL(Agree1,'0') \<\> '0' AND @EmpSeq NOT IN (SELECT ValueSeq FROM \#\_TDAUMinorValue WHERE Serl = 1000001))

OR (ISNULL(Agree2,'0') \<\> '0' AND @EmpSeq NOT IN (SELECT ValueSeq FROM \#\_TDAUMinorValue WHERE Serl = 1000002))

OR (ISNULL(Agree3,'0') \<\> '0' AND @EmpSeq NOT IN (SELECT ValueSeq FROM \#\_TDAUMinorValue WHERE Serl = 1000003)) )

--OR (ISNULL(CheckDelvTeamManager,'0') \<\> '0' AND @EmpSeq NOT IN (SELECT ValueSeq FROM \#\_TDAUMinorValue WHERE Serl = 1000004)))

 

EXEC dbo.\_SCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

6 , -- 중복된 @1 @2가(이) 입력되었습니다.(SELECT \* FROM \_TCAMessageLanguage WHERE MessageSeq = 6)

@LanguageSeq ,

0,'' -- SELECT \* FROM \_TCADictionary WHERE Word like '%거래처%'

UPDATE \#TPDBOMManagement

SET Result = CASE WHEN ISNULL(A.Col,0) = 1 THEN N'합의1 컨트롤에 체크할 권한이 없습니다.'

WHEN ISNULL(A.Col,0) = 2 THEN N'합의2 컨트롤에 체크할 권한이 없습니다.'

WHEN ISNULL(A.Col,0) = 3 THEN N'합의3 컨트롤에 체크할 권한이 없습니다.'

--WHEN ISNULL(A.Col,0) = 4 THEN N'개발팀장 컨트롤에 체크할 권한이 없습니다.'

ELSE N'컨트롤에 체크할 권한이 없습니다.' END,

MessageType = @MessageType,

Status = @Status

FROM \#TPDBOMManagement AS A

JOIN sys_TPDBOMReqManagement AS B ON B.CompanySeq = @CompanySeq

AND B.ItemSeq = A.ItemSeq

AND B.ItemBOMRevRemark = A.ItemBOMRevRemark

AND B.BOMReqSeq = A.BOMReqSeq

WHERE WorkingTag \<\> 'D'

AND Status = 0

AND ((ISNULL(A.Col,0) = 1 AND @EmpSeq NOT IN (SELECT ValueSeq FROM \#\_TDAUMinorValue WHERE Serl = 1000001))

OR (ISNULL(A.Col,0) = 2 AND @EmpSeq NOT IN (SELECT ValueSeq FROM \#\_TDAUMinorValue WHERE Serl = 1000002))

OR (ISNULL(A.Col,0) = 3 AND @EmpSeq NOT IN (SELECT ValueSeq FROM \#\_TDAUMinorValue WHERE Serl = 1000003))

--OR (ISNULL(A.Col,0) = 4 AND @EmpSeq NOT IN (SELECT ValueSeq FROM \#\_TDAUMinorValue WHERE Serl = 1000004))

)

EXEC dbo.\_SCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

6 , -- 중복된 @1 @2가(이) 입력되었습니다.(SELECT \* FROM \_TCAMessageLanguage WHERE MessageSeq = 6)

@LanguageSeq ,

0,''

UPDATE \#TPDBOMManagement

SET Result = N'중복된 BOM차수명이 입력되었습니다.' ,

MessageType = @MessageType ,

Status = @Status

FROM \#TPDBOMManagement AS A

JOIN sys_TPDBOMReqManagement AS B ON B.CompanySeq = @CompanySeq

AND B.ItemSeq = A.ItemSeq

AND B.ItemBOMRevRemark = A.ItemBOMRevRemark

AND B.BOMReqSeq \<\> A.BOMReqSeq

WHERE Status = 0

AND A.WorkingTag IN ('A','U')

EXEC dbo.\_SCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

9 , -- 중복된 @1 @2가(이) 입력되었습니다.(SELECT \* FROM \_TCAMessageLanguage WHERE MessageSeq = 9)

@LanguageSeq ,

0,''

--UPDATE \#TPDBOMManagement

-- SET Result = N'작성미완료만 수정/삭제 가능합니다.' ,

-- MessageType = @MessageType ,

-- Status = @Status

--FROM \#TPDBOMManagement AS A

-- JOIN sys_TPDBOMReqManagement AS B ON B.CompanySeq = @CompanySeq

-- AND B.ItemSeq = A.ItemSeq

-- AND A.RegCompleteYn = B.RegCompleteYn

-- WHERE Status = 0

-- AND A.RegCompleteYn = '1'

-- 작성이 아닌, 합의 1,2,3에 대해서 체크 23.12.29 SJPARK

UPDATE \#TPDBOMManagement

SET Result = N'합의되지 않은 건만 수정/삭제 가능합니다.' ,

MessageType = @MessageType ,

Status = @Status

FROM \#TPDBOMManagement AS A

JOIN sys_TPDBOMReqManagement AS B ON B.CompanySeq = @CompanySeq

AND B.ItemSeq = A.ItemSeq

AND A.ItemBomRev = B.ItemBOMRevRemark

WHERE Status = 0

AND (A.Agree1 = B.Agree1) -- 합의가 바뀌는 경우 제외

AND (A.Agree2 = B.Agree2) -- 합의가 바뀌는 경우 제외

AND (A.Agree3 = B.Agree3) -- 합의가 바뀌는 경우 제외

AND (ISNULL(B.Agree1,0) = 1 OR ISNULL(B.Agree2,0) = 1 OR ISNULL(B.Agree3,0) = 1)

--AND A.WorkingTag IN ('A','U')

IF EXISTS ( SELECT 1 FROM \#TPDBOMManagement WHERE Status \<\> 0)

BEGIN

-- 합의1,2,3에 대해 미 반영 시 체크 해제 될 수 있도록 UPDATE문 추가 23.12.28 SJPARK

UPDATE \#TPDBOMManagement

SET Agree1 = CASE WHEN A.Col = '1' THEN B.Agree1 ELSE A.Agree1 END,

Agree2 = CASE WHEN A.Col = '2' THEN B.Agree2 ELSE A.Agree2 END,

Agree3 = CASE WHEN A.Col = '3' THEN B.Agree3 ELSE A.Agree3 END,

RegCompleteYn = CASE WHEN A.Col = '4' THEN B.RegCompleteYn ELSE A.RegCompleteYn END

FROM \#TPDBOMManagement AS A

LEFT OUTER JOIN sys_TPDBOMReqManagement AS B ON B.CompanySeq = @CompanySeq

AND B.BOMReqSeq = A.BOMReqSeq

SELECT \* FROM \#TPDBOMManagement

RETURN

END

 

-- 20240712.합의1,2,3 중에 하나라도 되어있을 경우 작성취소하면 에러

UPDATE A

SET Result = '합의가 되어있어 작성취소할 수 없습니다.',

MessageType = 12007,

Status = 9999

FROM \#TPDBOMManagement AS A

JOIN sys_TPDBOMReqManagement AS B ON B.CompanySeq = @CompanySeq

AND B.BOMReqSeq = A.BOMReqSeq

WHERE A.Status = 0

AND (A.RegCompleteYn = '0' AND B.RegCompleteYn = '1')

AND (B.Agree1 = '1' OR B.Agree2 = '1' OR B.Agree3 = '1')

 

 

-- BOM등록화면에서 최초 저장시 WorkingTag 무조건 'A'인데, 기등록된 데이터있으면 'U'로 변경해준다

-- (최초및 최종작성자 때문에 로그를남겨야 하므로) :: 11.10.27 BY 김세호

UPDATE A

SET WorkingTag = 'U'

FROM \#TPDBOMManagement AS A

WHERE A.WorkingTag = 'A'

AND A.Status = 0

AND EXISTS (SELECT 1 FROM sys_TPDBOMReqManagement WHERE CompanySeq = @CompanySeq AND ItemSeq = A.ItemSeq AND BOMReqSeq = A.BOMReqSeq)

IF @@ERROR \<\> 0 RETURN

-- 로그테이블 남기기(마지막 파라메터는 반드시 한줄로 보내기)

EXEC \_SCOMLog @CompanySeq ,

@UserSeq ,

'sys_TPDBOMReqManagement', -- 원테이블명

'#TPDBOMManagement', -- 템프테이블명

'BOMReqSeq' , -- 키가 여러개일 경우는 , 로 연결한다.

'CompanySeq, ItemSeq, BOMReqSeq, BOMReqNo ,RegDate, LastUptDate, FrTermValidity, ToTermValidity, BatchSize, Remark, LastUserSeq, LastDateTime, FileSeq

, ItemBOMRevRemark, RegDateTime, RegCompleteYn', '',@PgmSeq

--Complete

IF EXISTS (SELECT 1 FROM sys_TPDBOMReqManagement AS A

JOIN \#TPDBOMManagement AS B ON A.ItemSeq = B.ItemSeq

AND A.BOMReqSeq = B.BOMReqSeq

WHERE B.WorkingTag IN ('A','U')

AND B.Status = 0

AND A.CompanySeq = @CompanySeq

AND A.RegCompleteYn \<\> B.RegCompleteYn)

BEGIN

-- UPDATE

UPDATE sys_TPDBOMReqManagement

SET RegCompleteYn = B.RegCompleteYn ,

LastUserSeq = @UserSeq ,

LastDateTime = GETDATE()

,Agree1 = B.Agree1

,Agree2 = B.Agree2

,Agree3 = B.Agree3

,CheckDelvTeamManager = B.CheckDelvTeamManager

,Agree1Remark = B.Agree1Remark

,Agree2Remark = B.Agree2Remark

,Agree3Remark = B.Agree3Remark

,DelvTeamManagerRemark = B.DelvTeamManagerRemark

FROM sys_TPDBOMReqManagement AS A

JOIN \#TPDBOMManagement AS B ON A.ItemSeq = B.ItemSeq

AND A.BOMReqSeq = B.BOMReqSeq

WHERE B.WorkingTag IN ('A','U')

AND B.Status = 0

AND A.CompanySeq = @CompanySeq

AND A.RegCompleteYn \<\> B.RegCompleteYn

IF @@ERROR \<\> 0 RETURN

SELECT \* FROM \#TPDBOMManagement

RETURN

END

-- 작업순서 맞추기: DELETE -\> UPDATE -\> INSERT

-- DELETE

IF EXISTS (SELECT TOP 1 1 FROM \#TPDBOMManagement WHERE WorkingTag = 'D' AND Status = 0)

BEGIN

DELETE sys_TPDBOMReqManagement

FROM sys_TPDBOMReqManagement AS A

JOIN \#TPDBOMManagement AS B ON A.ItemSeq = B.ItemSeq

AND A.BOMReqSeq = B.BOMReqSeq

WHERE B.WorkingTag = 'D'

AND B.Status = 0

AND A.CompanySeq = @CompanySeq

IF @@ERROR \<\> 0 RETURN

END

-- 내부코드 채번

DECLARE @Count INT,

@BOMReqSeq INT ,

@Date NCHAR(8),

@MaxNo NVARCHAR(50)

SELECT @Count = COUNT(1) FROM \#TPDBOMManagement WHERE WorkingTag = 'A' AND Status = 0

SELECT @Date = CONVERT(VARCHAR,GETDATE(),112)

IF @Count \> 0

BEGIN

EXEC @BOMReqSeq = \_SCOMCreateSeq @CompanySeq, 'sys_TPDBOMReqManagement', 'BOMReqSeq', @Count

UPDATE \#TPDBOMManagement

SET BOMReqSeq = @BOMReqSeq + DataSeq

FROM \#TPDBOMManagement

WHERE WorkingTag = 'A'

AND Status = 0

SELECT @MaxNo = MAX(BOMReqNo) FROM sys_TPDBOMReqManagement WHERE LEFT(BOMReqNo,8) = @Date AND ISNUMERIC(RIGHT(BOMReqNo,4)) = 1

-- Temp Talbe 에 생성된 키값 UPDATE

UPDATE \#TPDBOMManagement

SET BOMReqNo = @Date + CONVERT(VARCHAR,RIGHT(('0000' + RIGHT((CONVERT(NUMERIC(19,0),ISNULL(@MaxNo,0))+ DataSeq),4)),4))

WHERE WorkingTag = 'A'

AND Status = 0

UPDATE \#TPDBOMManagement SET RegDate = CONVERT(NCHAR(8), GETDATE(), 112)

WHERE WorkingTag = 'A'

AND Status = 0

END

-- UPDATE

UPDATE sys_TPDBOMReqManagement

SET LastUptDate = CONVERT(NCHAR(8), GETDATE(), 112) ,

FrTermValidity = B.FrTermValidity ,

ToTermValidity = B.ToTermValidity ,

BatchSize = B.BatchSize ,

Remark = B.Remark ,

LastUserSeq = @UserSeq ,

LastDateTime = GETDATE() ,

FileSeq = B.FileSeq ,

ItemBOMRevRemark = B.ItemBOMRevRemark

,Agree1 = B.Agree1

,Agree2 = B.Agree2

,Agree3 = B.Agree3

,CheckDelvTeamManager = B.CheckDelvTeamManager

,Agree1Remark = B.Agree1Remark

,Agree2Remark = B.Agree2Remark

,Agree3Remark = B.Agree3Remark

,DelvTeamManagerRemark = B.DelvTeamManagerRemark

FROM sys_TPDBOMReqManagement AS A

JOIN \#TPDBOMManagement AS B ON A.ItemSeq = B.ItemSeq

AND A.BOMReqSeq = B.BOMReqSeq

WHERE B.WorkingTag IN ('A','U')

AND B.Status = 0

AND A.CompanySeq = @CompanySeq

IF @@ERROR \<\> 0 RETURN

-- INSERT

INSERT INTO sys_TPDBOMReqManagement

(CompanySeq , ItemSeq , RegDate , LastUptDate ,

FrTermValidity , ToTermValidity , BatchSize , Remark , LastUserSeq ,

LastDateTime , FileSeq , ItemBOMRevRemark , RegDateTime

, BOMReqSeq , BOMReqNo , RegCompleteYn

,Agree1

,Agree2

,Agree3

,CheckDelvTeamManager

,Agree1Remark

,Agree2Remark

,Agree3Remark

,DelvTeamManagerRemark )

SELECT @CompanySeq ,A.ItemSeq , CONVERT(NCHAR(8),GETDATE(),112),CONVERT(NCHAR(8),GETDATE(),112) ,

A.FrTermValidity ,A.ToTermValidity ,A.BatchSize , A.Remark ,@UserSeq ,

GETDATE() ,A.FileSeq ,A.ItemBOMRevRemark, GETDATE()

,A.BOMReqSeq , A.BOMReqNo , A.RegCompleteYn

,A.Agree1

,A.Agree2

,A.Agree3

,A.CheckDelvTeamManager

,A.Agree1Remark

,A.Agree2Remark

,A.Agree3Remark

,A.DelvTeamManagerRemark

FROM \#TPDBOMManagement AS A

WHERE A.WorkingTag IN ('A','U')

AND A.Status = 0

AND NOT EXISTS (SELECT 1 FROM sys_TPDBOMReqManagement WHERE CompanySeq = @CompanySeq AND ItemSeq = A.ItemSeq AND BOMReqSeq = A.BOMReqSeq)

IF @@ERROR \<\> 0 RETURN

SELECT \* FROM \#TPDBOMManagement

RETURN
