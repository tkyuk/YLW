FROM das_TPJTProject AS A

JOIN \_TPJTProject AS A2 WITH(NOLOCK) ON A.CompanySeq = A2.CompanySeq

AND A.PJTSeq = A2.PJTSeq

JOIN \_TPJTProjectItem AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.PJTSeq = B.PJTSeq

AND B.PJTItemSerl = 1

AND C.PJTSeq = A2.RelationPJTSeq
