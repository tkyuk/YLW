> insert into \_TACSlipSum

SELECT A.\*

FROM JISTEST.dbo.\_TACSlipSum AS A WITH (NOLOCK)

INNER JOIN \_TDAAccount AS B WITH (NOLOCK)

ON B.CompanySeq = A.CompanySeq

AND B.AccSeq = A.AccSeq

AND B.SMAccKind IN (4018001,4018002, 4018003)

WHERE A.CompanySeq = 1

AND A.SMAccStd = 1

AND A.AccUnit = 1

AND A.AccYM = '202401'

AND A.IsIni = '0'
