A.POReqSeq,

A.POReqSerl,

-1, 6036001,

A.IsStop,

A.Qty,

0,

0,

CASE A.UMPOReqType WHEN 6501014 THEN T.MRPDailySeq WHEN 6501015 THEN T1.MRPMonthSeq ELSE 0 END ,

M.UMRegType,

M2.ItemNo,

(SELECT TOP 1 M4.ItemName

AND ISNULL(M1.BeforeBOMSerl,0) = 0 -- 상위 BOM

AND M1.ItemSeq = M2.ItemSeq

LEFT OUTER JOIN \_TPJTBOM AS M3 WITH(NOLOCK) ON A.CompanySeq = M3.CompanySeq

AND A.PJTSeq = M3.PJTSeq

AND ISNULL(M3.BeforeBOMSerl,0) = 0

AND M3.UpperBOMSerl = -1

-- AND LEN(M3.BOMLevel) = 1 최상위BOMLevel을 1자리로 한정하여 넣어놔서 주석처리 김민석(2013) AND ISNUMERIC(REPLACE(M3.BOMLevel,'.','/')) = 1

AND M3.ItemSeq = M4.ItemSeq

AND A.UMPOReqType = 6501014

AND A.SourceSeq = T.MRPDailySeq

AND A.SourceSerl = T.Serl

AND A.SourceSeq = T1.MRPMonthSeq
