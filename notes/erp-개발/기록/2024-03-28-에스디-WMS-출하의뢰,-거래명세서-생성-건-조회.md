DECLARE @DomAmtType INT, -- EnvSeq = 8041 (원화금액소수점처리(판매))

@DomVatType INT, -- EnvSeq = 8040 (원화부가세소수점처리(판매))

@DomAmtDecLength INT, -- EnvSeq = 15 (자국통화금액 소수점 자리수 : 원화)

@AmtDecLength INT, -- EnvSeq = 14 (거래금액 소수점 자리수 : 외화)

@StdDomCurrSeq INT -- EnvSeq = 13 (자국 통화)

-- EnvSeq = 8041 (원화금액소수점처리(판매))

EXEC @DomAmtType = dbo.\_SCOMEnvR @CompanySeq, 8041, 1, @@PROCID

-- EnvSeq = 8040 (원화부가세소수점처리(판매))

EXEC @DomVatType = dbo.\_SCOMEnvR @CompanySeq, 8040, 1, @@PROCID

-- EnvSeq = 15 (자국통화금액 소수점 자리수)

EXEC @DomAmtDecLength = dbo.\_SCOMEnvR @CompanySeq, 15, 1, @@PROCID

-- EnvSeq = 14 (거래금액 소수점 자리수)

EXEC @AmtDecLength = dbo.\_SCOMEnvR @CompanySeq, 14, 1, @@PROCID

-- EnvSeq = 13 (자국 통화)

EXEC @StdDomCurrSeq = dbo.\_SCOMEnvR @CompanySeq, 13, 1, @@PROCID

DECLARE @DomDec INT,

@DomAmtValue INT

IF ISNULL(@DomAmtDecLength,0) \> 0

SELECT @DomDec = POWER(10,@DomAmtDecLength)

ELSE

SELECT @DomDec = 1

--반올림 1003001

--절사 1003002

--올림 1003003

IF @DomAmtType = 1003001 SELECT @DomAmtValue = 0

ELSE IF @DomAmtType = 1003002 SELECT @DomAmtValue = 1

ELSE IF @DomAmtType = 1003003 SELECT @DomAmtValue = 0

 

Update \#TMP_SD_TSLInvoice Set Price = Round(Price,@AmtDecLength)                -- 단가의 소수점를 거래금액 소수점 자리수로 설정함.

> WHERE Status = 0

AND WorkingTag IN ('A','U')

SET CurAmt = CASE WHEN ISNULL(A.IsInclusedVAT,'') \<\> '1'

THEN (CASE WHEN ISNULL(A.CurrSeq,0) \<\> @StdDomCurrSeq

THEN (ROUND(ISNULL(A.Price,0) \* ISNULL(A.Qty,0), @AmtDecLength))

ELSE (ROUND(ISNULL(A.Price,0) \* ISNULL(A.Qty,0), @DomAmtDecLength))

END)

ELSE (CASE WHEN ISNULL(A.CurrSeq,0) \<\> @StdDomCurrSeq

THEN (ROUND(ISNULL(A.Price,0) \* ISNULL(A.Qty,0) / (1. + (ISNULL(A.VATRate,0) / 100.)), @AmtDecLength))

ELSE (ROUND(ISNULL(A.Price,0) \* ISNULL(A.Qty,0) / (1. + (ISNULL(A.VATRate,0) / 100.)), @DomAmtDecLength))

END)

END,

CurVAT = CASE WHEN A.CurVAT = 0 THEN 0 ELSE

CASE WHEN ISNULL(A.IsInclusedVAT,'') \<\> '1'

THEN (CASE WHEN ISNULL(A.CurrSeq,0) \<\> @StdDomCurrSeq

THEN (ROUND(ISNULL(A.Price,0) \* ISNULL(A.Qty,0) \* (ISNULL(A.VATRate,0) / 100.), @AmtDecLength))

ELSE (ROUND(ISNULL(A.Price,0) \* ISNULL(A.Qty,0) \* (ISNULL(A.VATRate,0) / 100.), @DomAmtDecLength))

END)

ELSE (CASE WHEN ISNULL(A.CurrSeq,0) \<\> @StdDomCurrSeq

THEN (ROUND((ISNULL(A.Price,0) \* ISNULL(A.Qty,0)) - (ISNULL(A.Price,0) \* ISNULL(A.Qty,0) / (1. + (ISNULL(A.VATRate,0) / 100.))), @AmtDecLength))

ELSE (ROUND((ISNULL(A.Price,0) \* ISNULL(A.Qty,0)) - (ISNULL(A.Price,0) \* ISNULL(A.Qty,0) / (1. + (ISNULL(A.VATRate,0) / 100.))), @DomAmtDecLength))

END)

END

END

FROM \#TMP_SD_TSLInvoice AS A

WHERE A.Status = 0

AND A.WorkingTag IN ('A','U')

THEN (CEILING(ISNULL(A.CurAmt,0) \* ISNULL(A.ExRate,0) \* @DomDec) / @DomDec)

ELSE (ROUND(A.CurAmt \* A.ExRate, @DomAmtDecLength, @DomAmtValue))

 

 

END) / (CASE ISNULL(C.BasicAmt, 0) WHEN 0 THEN 1 ELSE C.BasicAmt END) ,

DomVAT = CASE WHEN A.DomVAT = 0 THEN 0 ELSE

(CASE WHEN @DomAmtType = 1003003

THEN (CEILING(ISNULL(A.CurVAT,0) \* ISNULL(A.ExRate,0) \* @DomDec) / @DomDec)

ELSE (ROUND(A.CurVAT \* A.ExRate, @DomAmtDecLength, @DomVatType))

END) / (CASE ISNULL(C.BasicAmt, 0) WHEN 0 THEN 1 ELSE C.BasicAmt END)

END
