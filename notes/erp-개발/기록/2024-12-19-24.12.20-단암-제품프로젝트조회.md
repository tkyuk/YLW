/\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*

작성일 - 20160308 단암

수정일 - 2016.07.26 인도예정일 추가

\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*/

CREATE PROC danam_SPJTProjecItemBatchQuery_sys_TTEST

@xmlDocument NVARCHAR(MAX) ,

@xmlFlags INT = 0,

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0

AS

SET TRANSACTION ISOLATION LEVEL READ UNCOMMITTED

DECLARE @docHandle INT,

@BizPJTSeq INT,

@PJTNo NVARCHAR(100), -- 프로젝트번호

@PJTName NVARCHAR(100), -- 프로젝트명

@PlanFrDate NCHAR(8), -- 계획시작일자

@PlanToDate NCHAR(8), -- 계획종료일자

@IsBizPJTType INT, -- 통합프로젝트유형

@IsLotPJTType INT -- 차수프로젝트(제품Lot)

EXEC sp_xml_preparedocument @docHandle OUTPUT, @xmlDocument

SELECT @BizPJTSeq = ISNULL(BizPJTSeq , 0),

@PJTNo = ISNULL(PJTNo , ''),

@PJTName = ISNULL(PJTName , ''),

@PlanFrDate = ISNULL(PlanFrDate, ''),

@PlanToDate = ISNULL(PlanToDate, '')

FROM OPENXML(@docHandle, N'/ROOT/DataBlock1', @xmlFlags)

WITH (BizPJTSeq INT,

PJTNo NVARCHAR(100),

PJTName NVARCHAR(100),

PlanFrDate NCHAR(8),

PlanToDate NCHAR(8)

)

IF @PlanFrDate = '' SELECT @PlanFrDate = '19000101'

IF @PlanToDate = '' SELECT @PlanToDate = '99991231'

 

SELECT @IsBizPJTType = A.MinorSeq

FROM \_TDAUMinorValue AS A WITH(NOLOCK)

WHERE A.CompanySeq = @CompanySeq

AND A.MajorSeq = 1021491

AND A.Serl = 1000003 -- 통합여부

AND A.ValueText = '1'

SELECT @IsLotPJTType = A.ValueSeq

FROM \_TDAUMinorValue AS A

WHERE A.CompanySeq = @CompanySeq

AND A.MajorSeq = 1021148

AND A.Serl = 1000001 -- 프로젝트유형

AND A.MinorSeq = (SELECT MinorSeq

FROM \_TDAUMinorValue

WHERE CompanySeq = @CompanySeq

AND MajorSeq = 1021148

AND Serl = 1000004 -- 차수프로젝트여부(제품Lot여부)

AND ValueText = '1')

 

SELECT A.PJTSeq AS PJTSeq,

A.PJTNo,

A.PJTName,

A.PlanFrDate,

A.PlanToDate,

A.RegDate,

A.PJTAmt,

A.CurrSeq,

A.CurrRate,

I.CurrName as CurrName,

A.PJTTypeSeq,

F.PJTTypeName,

A.CustSeq,

H.CustName AS CustName,

A.CCtrSeq,

J.CCtrName,

A.ETC,

A.Summary,

A.Object,

K.ItemQty AS Qty,

K.ItemPrice AS Price,

K.ItemAmt AS Amt,

K.ItemDomPrice AS DomPrice,

K.ItemDomAmt AS DomAmt,

L.ItemSeq AS ProdItemSeq,

L.ItemName AS ProdItemName,

L.ItemNo AS ProdItemNo,

L.Spec AS ProdItemSpec,

Delv.DelvExpectDate AS DelvDueDate,

--납품추가정보

Delv2.UnitFr,

Delv2.UnitTo,

Delv.DelvVatAmt AS Vat,

A2.ItemGubunSeq,

M.MinorName AS ItemGubunName,

A2.EstDeliveryDate, -- 16.07.26 인도예정일

A.ContractFrDate,

A.ContractToDate,

A2.SortNum, -- 순서 추가

A2.IsWorkTimeEnd, -- 공수마감 추가

A2.ConDelvDate AS ConDelvDate -- 계약납기일

FROM \_TPJTProject A

LEFT OUTER JOIN danam_TPJTProject AS A2 WITH (NOLOCK) ON A2.CompanySeq = A.CompanySeq

AND A2.PJTSeq = A.PJTSeq

LEFT OUTER JOIN \_TDAUMinor AS M WITH (NOLOCK) ON M.CompanySeq = A.CompanySeq

AND M.MinorSeq = A2.ItemGubunSeq

LEFT OUTER JOIN \_TPJTProjectDesc AS D WITH (NOLOCK) ON A.CompanySeq = D.CompanySeq

AND A.PJTSeq = D.PJTSeq

LEFT OUTER JOIN \_TPJTType AS F WITH (NOLOCK) ON A.CompanySeq = F.CompanySeq

AND A.PJTTypeSeq = F.PJTTypeSeq

LEFT OUTER JOIN \_TDACust AS H WITH (NOLOCK) ON A.CompanySeq = H.CompanySeq

AND A.CustSeq = H.CustSeq

LEFT OUTER JOIN \_TDACurr AS I WITH (NOLOCK) ON A.CompanySeq = I.CompanySeq

AND A.CurrSeq = I.CurrSeq

LEFT OUTER JOIN \_TDACCtr AS J WITH (NOLOCK) ON A.CompanySeq = J.CompanySeq

AND A.CCtrSeq = J.CCtrSeq

LEFT OUTER JOIN \_TPJTProjectItem AS K WITH (NOLOCK) ON A.CompanySeq = K.CompanySeq

AND A.PJTSeq = K.PJTSeq

AND K.PJTItemSerl = (SELECT MAX(PJTItemSerl)

FROM \_TPJTProjectItem

WHERE CompanySeq = A.CompanySeq

AND PJTSeq = A.PJTSeq)

LEFT OUTER JOIN \_TDAItem AS L WITH(NOLOCK) ON A.CompanySeq = L.CompanySeq

AND K.ItemSeq = L.ItemSeq

LEFT OUTER JOIN \_TPJTWBSPJTUnion AS P WITH(NOLOCK) ON A.CompanySeq = P.CompanySeq

AND A.PJTSeq = P.JoinPJTSeq

LEFT OUTER JOIN \_TPJTProject AS P2 WITH(NOLOCK) ON P2.CompanySeq = P.CompanySeq

AND P2.PJTSeq = P.PJTSeq

LEFT OUTER JOIN \_TPJTProjectDelivery AS Delv WITH(NOLOCK) ON A.CompanySeq =Delv.CompanySeq

AND A.PJTSeq = Delv.PJTSeq

AND Delv.DelvSerl = (SELECT MAX(DelvSerl)

FROM \_TPJTProjectDelivery

WHERE CompanySeq = A.CompanySeq

AND PJTSeq = A.PJTSeq)

LEFT OUTER JOIN danam_TPJTProjectDelivery AS Delv2 WITH(NOLOCK) ON A.CompanySeq = Delv2.CompanySeq

AND A.PJTSeq = Delv2.PJTSeq

AND Delv2.DelvSerl = (SELECT MAX(DelvSerl)

FROM \_TPJTProjectDelivery

WHERE CompanySeq = A.CompanySeq

AND PJTSeq = A.PJTSeq)

WHERE A.CompanySeq = @CompanySeq

AND (@BizPJTSeq = 0 OR P2.PJTSeq = @BizPJTSeq)

AND (A.PJTNo LIKE @PJTNo + '%')

AND (A.PJTName LIKE @PJTName + '%')

AND (A.PlanFrDate \<= @PlanToDate)

AND (A.PlanToDate = '' OR A.PlanToDate \>= @PlanFrDate)

AND (A.PJTTypeSeq NOT IN (SELECT ValueSeq FROM \_TDAUMinorValue WHERE CompanySeq = @CompanySeq AND MajorSeq = 1021491 AND Serl = 1000001 AND MinorSeq = @IsBizPJTType))

AND (A.PJTTypeSeq \<\> @IsLotPJTType) -- 제품Lot(차수)프로젝트 제외

RETURN
