# 24.12.11 한국알박 품목 연동 SP 오류 건 추출

-- 1. ERP, MES 품번이나 품명 다른 건에 대해 DISTINCT 로 ItemSeq 뽑기

CREATE TABLE \#TMP_ItemSeq(

ItemSeq INT

)

INSERT INTO \#TMP_ItemSeq(ItemSeq)

SELECT DISTINCT A.ItemSeq

FROM uck_TMESITEMMASTER AS A

JOIN \_TDAItem AS B ON B.ItemSeq = A.ItemSeq

WHERE A.ItemNo \<\> B.ItemNo

OR A.ItemName \<\> B.ItemName

 

 

SELECT 'U' AS AUDType ,

GETDATE() AS CrtDateTime ,

'N' AS RecvYn ,

NULL AS RecvDateTime,

A.CompanySeq AS CompanySeq ,

A.ItemSeq AS ItemSeq ,

A.ItemName AS ItemName ,

A.ItemNo AS ItemNo ,

A.Spec AS Spec ,

A.AssetSeq AS AssetSeq ,

ISNULL(C.AssetName, '') AS AssetName,

ISNULL(L.ValueSeq,0) AS ItemClassLSeq ,

CASE WHEN ISNULL(L.ValueSeq,0) = 0 THEN ''

ELSE ( SELECT ISNULL(MinorName,'')

FROM \_TDAUMinor WITH(NOLOCK)

WHERE CompanySeq = 1 AND MinorSeq = L.ValueSeq ) END AS ItemClassLName, -- 품목대분류

ISNULL(K.ValueSeq,0) AS ItemClassMSeq,

CASE WHEN ISNULL(K.ValueSeq,0) = 0 THEN ''

ELSE ( SELECT ISNULL(MinorName,'')

FROM \_TDAUMinor WITH(NOLOCK)

WHERE CompanySeq = 1 AND MinorSeq = K.ValueSeq ) END AS ItemClassMName, -- 품목중분류

ISNULL(B1.UMItemClass,0) AS ItemClassSSeq, -- 품목소분류코드

ISNULL(H.MinorName,'') AS ItemClassSName, -- 품목소분류

A.UnitSeq ,

ISNULL(B.UnitName , '') ,

ISNULL(A.SMStatus,0) AS SMStatus, -- 품목 상태

CASE WHEN ISNULL(A.SMStatus,0) = 0 THEN '' ELSE (SELECT ISNULL(MinorName,'') FROM \_TDASMinor WITH (NOLOCK) WHERE CompanySeq = 1 AND MajorSeq = 2001 AND MinorSeq = A.SMStatus) END AS SMStatusName, -- 품목 상태

ISNULL(F.IsInQc , '0') ,

ISNULL(F.IsLastQc, '0') ,

ISNULL(M.StdTime,0) ,

A.LastUserSeq ,

A.LastDateTime

FROM \#TMP_ItemSeq AS Z

JOIN \_TDAItem AS A WITH(NOLOCK) ON A.CompanySeq = 1

AND A.ItemSeq = Z.ItemSeq

LEFT OUTER JOIN \_TDAUnit AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.UnitSeq = B.UnitSeq

LEFT OUTER JOIN \_TDAItemAsset AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.AssetSeq = C.AssetSeq

LEFT OUTER JOIN \_TDASMinor AS E WITH(NOLOCK) ON A.CompanySeq = E.CompanySeq

AND A.SMStatus = E.MinorSeq

LEFT OUTER JOIN \_TPDBaseItemQCType AS F WITH(NOLOCK) ON A.CompanySeq = F.CompanySeq

AND A.ItemSeq = F.ItemSeq

LEFT OUTER JOIN \_TDAItemClass AS B1 WITH(NOLOCK) ON ( A.CompanySeq = B1.CompanySeq AND A.ItemSeq = B1.ItemSeq AND B1.UMajorItemClass IN (2001,2004) )

LEFT OUTER JOIN \_TDAUMinor AS H WITH(NOLOCK) ON ( B1.CompanySeq = H.CompanySeq AND H.MajorSeq = LEFT( B1.UMItemClass, 4 ) AND B1.UMItemClass = H.MinorSeq )

LEFT OUTER JOIN \_TDAUMinorValue AS K WITH(NOLOCK) ON ( H.CompanySeq = K.CompanySeq AND K.MajorSeq IN (2001,2004) AND H.MinorSeq = K.MinorSeq AND K.Serl IN (1001,2001) )

LEFT OUTER JOIN \_TDAUMinorValue AS L WITH(NOLOCK) ON ( K.CompanySeq = L.CompanySeq AND L.MajorSeq IN (2002,2005) AND K.ValueSeq = L.MinorSeq AND L.Serl = 2001 )

LEFT OUTER JOIN (SELECT ItemSeq, SUM(StdWorkTime) AS StdTime, MAX(LastDateTime) AS LastDateTime FROM \_TPDROUItemProcWC GROUP BY ItemSeq) AS M ON A.ItemSeq = M.ItemSeq

WHERE A.CompanySeq = 1

AND A.AssetSeq IN ('1','6', '10','16','2','4')

 

DROP TABLE \#TMP_ItemSeq
