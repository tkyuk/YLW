DROP TABLE \#TMP

 

CREATE TABLE \#TMP (

ID INT IDENTITY(1,1),

LogSeq INT

)

 

INSERT INTO \#TMP

SELECT LogSeq FROM \_TCOMTableLog where TableName IN ('\_TLGInOutSerialSub','AMT_TLGInOutSerialSub') AND LogDateTime\< DATEADD(MONTH, -6, GETDATE()) ORDER BY 1

 

 

 

 

DECLARE @Start INT ,

@End INT ,

@ABSSumTot DECIMAL(19,5)

SELECT @Start = MIN(A.ID),

@End = MAX(A.ID)

FROM \#TMP AS A

 

 

--반복처리

WHILE (@Start \<= @End)

BEGIN

 

DELETE

FROM \_TCOMTableLog

FROM \_TCOMTableLog AS A

JOIN \#TMP AS B ON B.LogSeq = A.LogSeq

where B.ID = @Start

 

SELECT @Start = @Start +1

END
