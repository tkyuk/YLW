CREATE PROC dbo.SD_SWFPSLShoppingMallSalesUploadQuery_ttest

@ServiceSeq INT = 0 ,

@WorkingTag NVARCHAR(10) = '',

@CompanySeq INT = 1 ,

@LanguageSeq INT = 1 ,

@UserSeq INT = 0 ,

@PgmSeq INT = 0 ,

@IsTransaction BIT = 0

 

AS

DECLARE @SalesYM NCHAR(8) ,

@MallSeq INT ,

@MallOrderNo NVARCHAR(50),

@GoodPriceDecLength INT ,

@CurAmtDecLength INT ,

@DomAmtDecLength INT ,

@DeptSeq INT ,

@EmpSeq INT ,

@SalesState NCHAR(1),

@CustSeq INT

 

SELECT @SalesYM = ISNULL(SalesYM , 0 ),

@MallSeq = ISNULL(MallSeq , 0 ),

@MallOrderNo = ISNULL(MallOrderNo, ''),

@DeptSeq = ISNULL(QDeptSeq,0),

@EmpSeq = ISNULL(QEmpSeq ,0),

@SalesState = ISNULL(SalesState, '0')

FROM \#BIZ_IN_DataBlock1

 

SELECT @CustSeq = CustSeq

FROM \_TFPSLShoppingMall

WHERE MallSeq = @MallSeq

 

IF @SalesState = ''

BEGIN

SELECT @SalesState = '0'

END

 

EXEC dbo.\_SWCOMEnv @CompanySeq, 9 , @UserSeq, @@PROCID, @GoodPriceDecLength OUTPUT -- 판매/제품 원화단가 소수점 자릿수

EXEC dbo.\_SWCOMEnv @CompanySeq, 14, @UserSeq, @@PROCID, @CurAmtDecLength OUTPUT -- 거래금액 소수점 자릿수

EXEC dbo.\_SWCOMEnv @CompanySeq, 15, @UserSeq, @@PROCID, @DomAmtDecLength OUTPUT -- 자국통화금액 소수점 자릿수

 

 

 

-- 1) 진행여부 체크 하기위한 임시테이블

SELECT A.CompanySeq, A.Dummy2, ISNULL(B.ToSeq,0) AS ToSeq, ISNULL(B.ToSerl,0) AS ToSerl, ISNULL(C.BillSeq,0) AS BillSeq, ISNULL(D.UploadSeq,0) AS UploadSeq

INTO \#ProgInoviceItem

FROM \_TSLInvoiceItem AS A WITH(NOLOCK)

JOIN \_TSLInvoice AS A1 WITH(NOLOCK) ON A.CompanySeq = A1.CompanySeq

AND A.InvoiceSeq = A1.InvoiceSeq

LEFT JOIN \_TCOMSourceDaily AS B WITH(NOLOCK, INDEX(IDX1_TCOMSourceDaily)) ON B.FromTableSeq = 18

AND B.ToTableSeq = 20

AND B.FromSeq = A.InvoiceSeq

AND B.FromSerl = A.InvoiceSerl

LEFT JOIN \_TSLSalesBillRelation AS C WITH(NOLOCK) ON B.CompanySeq = C.CompanySeq

AND B.ToSeq = C.SalesSeq

AND B.ToSerl = C.SalesSerl

LEFT JOIN SD_TFPSLShoppingMallSaleUpload AS D WITH(NOLOCK, INDEX(OrderNo_INDEX)) ON REPLACE(A.Dummy2,'\_사은품','') = D.OrderNo

AND ISNULL(C.BillSeq,0) = ISNULL(D.BillSeq,0)

AND D.UploadSeq = 2142

WHERE A1.IsDelvCfm = '1'

AND ISNULL(A.Dummy2,'') \<\> ''

AND A1.InvoiceDate \<= D.SalesDate

 

 

-- 1-1) 조회 시 해당 업로드 건으로 매출 된 건이 삭제되어 있다면 BillSeq = 0으로 변경

UPDATE A1

SET IsProg = '0',

BillSeq = 0

FROM SD_TFPSLShoppingMallSaleUpload AS A1

LEFT JOIN \#ProgInoviceItem AS B WITH(NOLOCK) ON A1.CompanySeq = B.CompanySeq

AND A1.OrderNo = REPLACE(B.Dummy2,'\_사은품','')

AND A1.UploadSeq = B.UploadSeq

WHERE A1.CompanySeq = @CompanySeq

AND A1.IsProg \<\> '0'

AND ISNULL(A1.BillSeq,0) \<\> 0

AND A1.SalesDate = @SalesYM

AND B.ToSeq IS NULL

AND B.UploadSeq IS NULL

 

 

 

-- 2) 위의 진행 키값 정리 후 주문번호별 거래명세서 집계를 위한 임시테이블 생성

SELECT A.\*, ISNULL(B.ToSeq,0) AS ToSeq, ISNULL(B.ToSerl,0) AS ToSerl, ISNULL(C.BillSeq,0) AS BillSeq, ISNULL(D.UploadSeq,0) AS UploadSeq

INTO \#DailyInoviceItem

FROM \_TSLInvoiceItem AS A WITH(NOLOCK)

JOIN \_TSLInvoice AS A1 WITH(NOLOCK) ON A.CompanySeq = A1.CompanySeq

AND A.InvoiceSeq = A1.InvoiceSeq

AND A1.InvoiceDate \<= @SalesYM

LEFT JOIN \_TCOMSourceDaily AS B WITH(NOLOCK, INDEX(IDX1_TCOMSourceDaily)) ON B.FromTableSeq = 18

AND B.ToTableSeq = 20

AND B.FromSeq = A.InvoiceSeq

AND B.FromSerl = A.InvoiceSerl

LEFT JOIN \_TSLSalesBillRelation AS C WITH(NOLOCK) ON B.CompanySeq = C.CompanySeq

AND B.ToSeq = C.SalesSeq

AND B.ToSerl = C.SalesSerl

LEFT JOIN SD_TFPSLShoppingMallSaleUpload AS D WITH(NOLOCK, INDEX(OrderNo_INDEX)) ON REPLACE(A.Dummy2,'\_사은품','') = D.OrderNo

AND ISNULL(C.BillSeq,0) = ISNULL(D.BillSeq,0)

AND D.SalesDate = @SalesYM

WHERE A1.IsDelvCfm = '1'

AND ISNULL(A.Dummy2,'') \<\> ''

AND A1.InvoiceDate \<= D.SalesDate

AND (A1.CustSeq = @CustSeq OR ISNULL(@CustSeq,0) = 0) -- 거래처내부코드 조회조건 추가

AND (D.MallSeq = @MallSeq OR @MallSeq = 0)

 

 

-- 주문번호 - 거래명세서 집계

SELECT SUM(ISNULL(A.CurAmt,0)) AS CurAmt, SUM(ISNULL(A.CurVAT,0)) AS CurVAT,

SUM(ISNULL(A.DomAmt,0)) AS DomAmt, SUM(ISNULL(A.DomVAT,0)) AS DomVAT,

REPLACE(A.Dummy2,'\_사은품','') AS MallOrderNo, ISNULL(A.BillSeq,0) AS BillSeq --, ISNULL(C.UploadSeq,0) AS UploadSeq

INTO \#InvoiceItem

FROM \#DailyInoviceItem AS A WITH(NOLOCK)

JOIN \_TSLInvoice AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.InvoiceSeq = B.InvoiceSeq

WHERE B.CompanySeq = @CompanySeq

AND B.IsDelvCfm = '1'

AND ISNULL(A.Dummy2,'') \<\> ''

GROUP BY REPLACE(A.Dummy2,'\_사은품',''), ISNULL(A.BillSeq,0)--, ISNULL(C.UploadSeq,0)

 

 

SELECT CASE WHEN ISNULL(E.BillNo, '') = '' THEN '0'

ELSE '1' END AS IsSales ,

E.BillNo AS BillNo ,

E.BillID AS BillID ,

B.MallName,

B.MallID AS MallUserID,

ISNULL(C.MallOrderNo,'') AS MallOrderNo_In,

A.SalesDate,

C.DomAmt,C.DomVAT,

ISNULL(C.DomAmt,0) + ISNULL(C.DomVAT,0) AS InvoiceAmt,

--C.CurAmt + C.CurVAT AS InvoiceAmt,

(ISNULL(C.DomAmt,0) + ISNULL(C.DomVAT,0)) - A.SaleAmt AS DiffAmt,

--(C.CurAmt + C.CurVAT) - A.SaleAmt AS DiffAmt,

A.OrderNo AS MallOrderNo_Sale,

A.SaleAmt AS SalesAmt,

A.MallSeq,

A.UploadSeq,

E.SalesSeq AS SalesSeq,

E.BillSeq AS BillSeq,

G.EmpName AS EmpName, -- 사원명

G.EmpSeq, -- 사원내부코드

H.DeptName AS DeptName, -- 부서명

H.DeptSeq -- 부서내부코드

FROM SD_TFPSLShoppingMallSaleUpload AS A WITH(NOLOCK)

JOIN \_TFPSLShoppingMall AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.MallSeq = A.MallSeq

LEFT JOIN \#InvoiceItem AS C ON A.OrderNo = C.MallOrderNo

AND ISNULL(A.BillSeq,0) = ISNULL(C.BillSeq,0)

OUTER APPLY ( SELECT DISTINCT

X.BillNo ,

X.BillSeq ,

X.BillID ,

Y.SalesSeq

FROM \_TCOMSourceDaily AS Z WITH(NOLOCK)

LEFT OUTER JOIN \_TSLSalesBillRelation AS Y WITH(NOLOCK) ON Y.CompanySeq = Z.CompanySeq

AND Y.SalesSeq = Z.ToSeq

AND Y.SalesSerl = Z.ToSerl

LEFT OUTER JOIN \_TSLBill AS X WITH(NOLOCK) ON X.CompanySeq = Y.CompanySeq

AND X.Billseq = Y.BillSeq

LEFT OUTER JOIN \_TSLSalesItem AS W WITH(NOLOCK) ON Y.CompanySeq = W.CompanySeq

AND Y.SalesSeq = W.SalesSeq

AND Y.SalesSerl = W.SalesSerl

WHERE Z.CompanySeq = A.CompanySeq

AND Z.FromTableSeq = 18 -- 거래명세서

AND Z.ToTableSeq = 20 -- 매출

AND Y.BillSeq = C.BillSeq

) AS E

LEFT JOIN \_TCAUser AS F WITH(NOLOCK) ON A.CompanySeq = F.CompanySeq

AND A.LastUserSeq = F.UserSeq

LEFT JOIN \_TDAEmp AS G WITH(NOLOCK) ON F.CompanySeq = G.CompanySeq

AND F.EmpSeq = G.EmpSeq

--LEFT JOIN \_TDADept AS H WITH(NOLOCK) ON F.CompanySeq = H.CompanySeq

-- AND F.DeptSeq = H.DeptSeq

LEFT OUTER JOIN \_FnXHREmpIn(@CompanySeq, '', '0') AS H ON G.EmpSeq = H.EmpSeq

 

WHERE A.CompanySeq = @CompanySeq

AND A.Salesdate = @SalesYM

AND (@MallSeq = 0 OR A.MallSeq = @MallSeq)

AND (@MallOrderNo = '' OR A.OrderNo LIKE @MallOrderNo + '%')

AND (@EmpSeq = 0 OR F.EmpSeq = @EmpSeq)

AND (@DeptSeq = 0 OR F.DeptSeq = @DeptSeq)

AND (@SalesState = '0' OR (@SalesState = '1' AND ISNULL(E.BillNo, '') = '')

OR (@SalesState = '2' AND ISNULL(E.BillNo, '') \<\> ''))

ORDER BY B.MallName, A.SalesDate, A.OrderNo

RETURN

go
