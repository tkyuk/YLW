UPDATE C1

SET IsCfm = '0'

FROM \#TLGInOutDailyItem AS A

JOIN \#BIZ_OUT_DataBlock2 AS B ON A.IDX_NO = B.IDX_NO

JOIN cif_TSLFreeOutReqItemSub AS C1 ON C1.CompanySeq = @CompanySeq

AND B.ReqSeq = C1.ReqSeq AND B.ReqSerl = C1.Serl AND B.ReqSubSerl = C1.SubSerl

WHERE B.Status = 0

AND B.GubnSeq = 2000093001

UPDATE C2

SET IsCfm = '0'

FROM \#TLGInOutDailyItem AS A

JOIN \#BIZ_OUT_DataBlock2 AS B ON A.IDX_NO = B.IDX_NO

JOIN cif_TLGCommissionOutReqItemSub AS C2 ON C2.CompanySeq = @CompanySeq AND B.GubnSeq = 2000093002

AND B.ReqSeq = C2.ReqSeq AND B.ReqSerl = C2.Serl AND B.ReqSubSerl = C2.SubSerl

WHERE B.Status = 0

AND B.GubnSeq = 2000093002

-- 조회

SELECT 2000093001, C.ReqSeq, C.Serl, MIN(C.SubSerl), SUM(C.Qty)

FROM cif_TSLFreeOutReq AS A

JOIN cif_TSLFreeOutReqItem AS B ON A.CompanySeq = B.CompanySeq AND A.ReqSeq = B.ReqSeq

JOIN cif_TSLFreeOutReqItemSub AS C ON B.CompanySeq = C.CompanySeq AND B.ReqSeq = C.ReqSeq AND B.Serl = C.Serl

LEFT OUTER JOIN \_TDAItem AS D WITH(NOLOCK) ON B.CompanySeq = D.CompanySeq AND B.ItemSeq = D.ItemSeq

WHERE A.CompanySeq = @CompanySeq

AND @GubnSeq IN (0, 2000093001)

 

SELECT \* FROM cif_TSLFreeOutReq WHERE ReqSeq = 14885

SELECT DVReqSeq,OutSeq,OutSerl,\* FROM CIF_WMS_TSLINVOICE WHERE DVReqSeq = 14885

SELECT DVReqSeq, \* FROM CIF_WMS_TSLINVOICE WHERE OutSeq = 0 and OutType = 2 order by Seq desc

SELECT \* FROM \_TDAItem

SELECT \* FROM \_TDACust WHERE CustSeq = 23

 

 

SELECT \* FROM \_TSLInvoice WHERE InvoiceSeq = 168227

 

 

SELECT C.ReqNo,B.\*

FROM cif_TSLFreeOutReqItem AS A

JOIN CIF_WMS_TSLINVOICE AS B ON B.DVReqSeq = A.ReqSeq

AND B.DVReqSerl = A.Serl

JOIN cif_TSLFreeOutReq AS C ON A.CompanySeq = C.CompanySeq AND A.ReqSeq = C.ReqSeq

JOIN cif_TSLFreeOutReqItemSub AS D ON D.CompanySeq = A.CompanySeq AND D.ReqSeq = A.ReqSeq AND D.Serl = A.Serl

WHERE OutSeq = 0 and OutType = 2 order by Seq desc
