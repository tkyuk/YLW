# 25.09.11 서비스프로젝트 - 코스트센터 신규 생성

BEGIN TRAN

SELECT TOP 100 LastDateTime,\* FROM \_TDACCtr ORDER BY CCtrSeq DESC

SELECT COUNT(\*) FROM \_TDACCtr

 

DECLARE @CompanySeq INT = 1

,@Seq INT = 0

> ,@Count INT = 0
>
> ,@PgmSeq INT = 128410304

CREATE TABLE \#Result(

DataSeq INT IDENTITY(1,1)

> ,SVCPJTSeq INT
>
> ,ItemCCtrSeq INT

,CCtrSeq         INT

,CCtrName NVARCHAR(200)

> ,UserSeq INT

)

 

> INSERT INTO \#Result(SVCPJTSeq,ItemCCtrSeq, CCtrName, UserSeq)
>
> SELECT A.SVCPJTSeq, F.CCtrSeq, E.PJTNo, E.RegUserSeq

FROM das_TPJTServicePJTItem AS A

LEFT OUTER JOIN \_TPJTProject AS E WITH(NOLOCK) ON E.CompanySeq = A.CompanySeq

AND E.PJTSeq = A.SVCPJTSeq

LEFT OUTER JOIN \_TPJTProject AS F WITH(NOLOCK) ON F.CompanySeq = A.CompanySeq

AND F.PJTSeq = A.ItemPJTSeq

WHERE ISNULL(A.SVCPJTSeq ,0) \<\> 0

> AND ISNULL(E.CCtrSeq,0) = ISNULL(F.CCtrSeq,0)

 

> SELECT \* FROM \#Result

-- 키값생성코드부분 시작

EXEC @Seq = dbo.\_SWCOMCreateSeq @CompanySeq, '\_TDACCtr', 'CCtrSeq', @Count

 

> IF EXISTS (SELECT 1 FROM \_TDACCtr WITH(NOLOCK) WHERE CompanySeq = @CompanySeq AND CCtrSeq \> @Seq)
>
> BEGIN
>
> SELECT @Seq = (SELECT MAX(CCtrSeq) FROM \_TDACCtr WITH(NOLOCK) WHERE CompanySeq = @CompanySeq)
>
>  
>
> UPDATE \_TCOMCreateSeqMax
>
> SET MaxSeq = @Seq + @Count
>
> WHERE CompanySeq = @CompanySeq
>
> AND TableName = '\_TDACCtr'
>
> END

 

-- Temp Talbe 에 생성된 키값 UPDATE

UPDATE \#Result

SET CCtrSeq = @Seq + DataSeq

 

INSERT INTO \_TDACCtr

(

CompanySeq ,CCtrSeq ,CCtrName ,EmpSeq ,DeptSeq ,

UMCCtrKind ,UMCostType ,IsNotUse ,IsNotUseDate,RegDate ,

RegUserSeq ,IsCCtr ,Remark ,SMSourceType,LastUserSeq ,

LastDateTime,AccUnit ,IsTemp ,FactUnit ,BizUnit ,CostUnit ,

ProfCostUnit,DispSeq, PE_IsEDU, Remark2, PgmSeq

)

SELECT @CompanySeq ,A.CCtrSeq ,A.CCtrName ,B.EmpSeq , B.DeptSeq ,

B.UMCCtrKind ,B.UMCostType ,B.IsNotUse ,B.IsNotUseDate, CONVERT(NCHAR(8),GETDATE(),112),

A.UserSeq ,B.IsCCtr ,B.Remark ,B.SMSourceType, A.UserSeq ,

GETDATE() ,B.AccUnit ,B.IsTemp ,B.FactUnit , B.BizUnit , B.CostUnit ,

B.ProfCostUnit,B.DispSeq ,B.PE_IsEDU ,B.Remark2 , @PgmSeq

FROM \#Result AS A

JOIN \_TDACCtr AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.CCtrSeq = A.ItemCCtrSeq

 

SELECT A.CCtrSeq, A.PJTNo

> FROM \_TPJTProject AS A
>
> JOIN \#Result AS C ON A.PJTSeq = C.SVCPJTSeq

ORDER BY A.CCtrSeq DESC

 

 

> UPDATE \_TPJTProject
>
> SET CCtrSeq = C.CCtrSeq
>
> FROM \_TPJTProject AS A
>
> JOIN \#Result AS C ON A.PJTSeq = C.SVCPJTSeq

 

SELECT A.CCtrSeq, A.PJTNo

> FROM \_TPJTProject AS A
>
> JOIN \#Result AS C ON A.PJTSeq = C.SVCPJTSeq

ORDER BY A.CCtrSeq DESC

 

SELECT TOP 100 LastDateTime,\* FROM \_TDACCtr ORDER BY CCtrSeq DESC

SELECT COUNT(\*) FROM \_TDACCtr

DROP TABLE \#Result

COMMIT TRAN
