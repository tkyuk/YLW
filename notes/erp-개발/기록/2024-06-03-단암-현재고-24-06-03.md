CREATE PROC \_SLGGetInOutLotStock_TTEST

@CompanySeq INT = 0, -- 법인코드

@BizUnit INT = 0, -- 사업부문

@FactUnit INT = 0, -- 생산사업장

@DateFr NCHAR(8) = '', -- 조회기간Fr

@DateTo NCHAR(8) = '', -- 조회기간To

@WHSeq INT = 0, -- 창고지정

@SMWHKind INT = 0, -- 창고구분별 조회

@CustSeq INT = 0, -- 수탁거래처

@IsTrustCust NCHAR(1) = '', -- 수탁여부

@IsSubDisplay NCHAR(1) = '', -- 기능창고 조회

@IsUnitQry NCHAR(1) = '', -- 단위별 조회

@QryType NCHAR(1) = '', -- 'S': 1007 실재고, 'B':1008 자산포함, 'A':1011 가용재고

@IsUseDetail NCHAR(1) = '0' -- 입출고상세내역정보를 호출여부

AS

DECLARE @QryTypeSeq INT,

@MonthQry NCHAR(1),

@PrevDate NCHAR(8),

@FromDate NCHAR(6),

@FromYM NCHAR(6),

@PrevYM NCHAR(6),

@ToYM NCHAR(6),

@WHMajor NCHAR(1)

 

IF @QryType = 'S'

SELECT @QryTypeSeq = 1007

ELSE IF @QryType = 'B'

SELECT @QryTypeSeq = 1008

ELSE IF @QryType = 'A'

SELECT @QryTypeSeq = 1011

ELSE

SELECT @QryTypeSeq = 0

IF @DateFr = ''

SELECT @DateFr = '19000101'

IF @DateTo = ''

SELECT @DateTo = '99991231'

-- 조회기간 시작일이 물류시작월이 속하는 회계기간의 시작년월보다 앞선 일자면 ...

-- 조회기간 시작일을 물류시작월이 속하는 회계기간의 시작년월의 1일로 대체함

-- 원인: 기초재고가 물류시작월이 속하는 회계기간의 시작년월의 1일 이월재고로 들어가는데...

-- 만약 그전 일자로 조회한다면 (-)재고로 나타남, 기초재고가 0으로 나오기때문에...

-- 2013.01.10 기초가 입력될 시작년월일을 변량에 담기, START

DECLARE @PreStkDate NVARCHAR(8)

SELECT @PreStkDate = B.FrSttlYM+'01'

FROM \_TCOMEnv AS A WITH(NOLOCK)

JOIN \_TDAAccFiscal AS B WITH(NOLOCK) ON ( A.CompanySeq = B.CompanySeq and A.EnvValue between B.FrSttlYM and B.ToSttlYM )

WHERE A.CompanySeq = @CompanySeq

AND A.EnvSeq = 1006

IF @@ROWCOUNT = 0 SELECT @PreStkDate = ''

SELECT @DateFr = CASE WHEN @PreStkDate \> @DateFr THEN @PreStkDate ELSE @DateFr END

-- 2013.01.10, END

SELECT @PrevDate = LEFT(@FromYM,4) + '0101'

IF LEN(@DateFr) = 6

SELECT @FromDate = @DateFr + '01'

ELSE

SELECT @FromDate = @DateFr

IF LEN(@DateTo) = 6

SELECT @MonthQry = '1'

ELSE

SELECT @MonthQry = '0'

EXEC \_SACGetAccTerm @CompanySeq, @FromDate, @FromYM OUTPUT ,@ToYM OUTPUT

IF NOT EXISTS (SELECT 1 FROM \_TLGLotStock WHERE CompanySeq = @CompanySeq AND StkYM = @FromYM AND (PrevQty \<\> 0 AND STDPrevQty \<\> 0))

BEGIN

SELECT @FromYM = CONVERT(NCHAR(8), DATEADD(yy, -1, @FromYM + '01'), 112)

END

SELECT @FromYM = CASE WHEN LEFT(@PreStkDate,6) \> @FromYM THEN LEFT(@PreStkDate,6) ELSE @FromYM END

SELECT @PrevYM = CONVERT(NCHAR(6), DATEADD(M, -1, @FromDate+'01'), 112),

@PrevDate = CONVERT(NCHAR(8), DATEADD(D, -1, @FromDate+'01'), 112)
