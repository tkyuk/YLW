INSERT INTO \#TMP

SELECT MinorSeq AS UMMngType , -- 기기명

\[15000001\] AS EquipName , -- 검사항목

\[15000002\] AS InspectionItem, -- 단위

\[15000003\] AS UnitName , -- 기준값

\[15000004\] AS STDValue

FROM (SELECT Serl,MinorSeq,ValueText FROM \#Tmp_Minor) AS B

PIVOT (

MIN(ValueText)

FOR Serl

--IN (\[15000001\],\[15000002\],\[15000003\],\[15000004\])

IN (\[15000001\],\[15000002\],\[15000003\],\[15000004\])

 

) AS A

ORDER BY MinorSeq

 

DROP TABLE \#Tmp_Minor

SELECT MAX(Empid) AS Empid, ISNULL(dbo.\_FWCOMDecrypt(A.ResidID, '\_TDAEmp', 'ResidID', 1),'') AS ResidID

INTO \#TDAEmp

FROM \_TDAEmp AS A

GROUP BY ISNULL(dbo.\_FWCOMDecrypt(A.ResidID, '\_TDAEmp', 'ResidID', 1),'')

 

 

SELECT A.Seq AS '페이지 번호'

,A.EmpName AS '사원'        

,B.Empid AS '사번'

,A.ResidID AS '주민등록번호'        

,A.Amt1         AS '퇴직급여'

,A.Amt2 AS '과세대상퇴직급여'

FROM \#TMP AS A

JOIN \#TDAEmp AS B ON B.ResidID = A.ResidID

ORDER BY SEQ
