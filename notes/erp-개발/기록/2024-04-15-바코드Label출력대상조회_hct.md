# 바코드Label출력대상조회_hct

DECLARE @CompanySeq INT

DECLARE @POType2 NVARCHAR(100)

DECLARE @POType NVARCHAR(100)

DECLARE @LastPrintNo INT

 

SELECT @POType2 = EnvValue

FROM \_TCOMEnv WITH(NOLOCK)

WHERE CompanySeq = 1

AND EnvSeq = 2000242 -- 업체코드

SELECT EnvValue,\* FROM \_TCOMEnv WITH(NOLOCK) WHERE CompanySeq = 1 AND EnvSeq = 2000242 -- 업체코드

 

 

SELECT @POType = EnvValue

FROM \_TCOMEnv WITH(NOLOCK)

WHERE CompanySeq = 1

AND EnvSeq = 2000243 -- 납품처코드

 

SELECT EnvValue,\* FROM \_TCOMEnv WITH(NOLOCK) WHERE CompanySeq = 1 AND EnvSeq = 2000243 -- 납품처코드

 

 

SELECT TOP 1

@LastPrintNo = ISNULL(C.LastPrintNo,0) -- 20231218.추가

FROM \_TPDMMOutItem AS A WITH(NOLOCK)

LEFT OUTER JOIN \_TDAItemUserDefine AS C4 WITH(NOLOCK) ON A.CompanySeq = C4.CompanySeq

AND A.ItemSeq = C4.ItemSeq

AND C4.MngSerl = 1000173 -- 제품명_hct

JOIN \_TDAItemProduct AS B WITH(NOLOCK) ON C4.CompanySeq = B.CompanySeq

AND C4.MngValSeq = B.ItemSeq

LEFT OUTER JOIN hct_TPDMMBarcodeLotMax AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.ItemLotNo = C.LotNo

AND A.ItemSeq = C.ItemSeq

WHERE A.CompanySeq = 1

AND A.ItemLotNo \<\> ''

AND A.MatOutSeq = 906

AND A.OutItemSerl \> 1

ORDER BY A.MatOutSeq, A.OutItemSerl

SELECT        C2.MngValText,

@POType2

,@POType

,RTRIM(B.ItemLotNo)

,CASE WHEN 1 + @LastPrintNo \< 10 THEN 'A' + CONVERT(NVARCHAR(10), 1 + @LastPrintNo) ELSE REPLACE(CONVERT(NVARCHAR(10), 1 + @LastPrintNo), '0', 'A') END

,

C2.MngValText + @POType2 + @POType + RTRIM(B.ItemLotNo) +

CASE WHEN 1 + @LastPrintNo \< 10 THEN 'A' + CONVERT(NVARCHAR(10), 1 + @LastPrintNo) ELSE REPLACE(CONVERT(NVARCHAR(10), 1 + @LastPrintNo), '0', 'A') END +

CASE WHEN LEN(RTRIM(CONVERT(NVARCHAR(10), CONVERT(INT, C7.LotSize)))) = 1 THEN '0000' + RTRIM(CONVERT(NVARCHAR(10), CONVERT(INT, C7.LotSize)))

WHEN LEN(RTRIM(CONVERT(NVARCHAR(10), CONVERT(INT, C7.LotSize)))) = 2 THEN '0000' + RTRIM(CONVERT(NVARCHAR(10), CONVERT(INT, C7.LotSize)))

WHEN LEN(RTRIM(CONVERT(NVARCHAR(10), CONVERT(INT, C7.LotSize)))) = 3 THEN '000' + RTRIM(CONVERT(NVARCHAR(10), CONVERT(INT, C7.LotSize)))

WHEN LEN(RTRIM(CONVERT(NVARCHAR(10), CONVERT(INT, C7.LotSize)))) = 4 THEN '00' + RTRIM(CONVERT(NVARCHAR(10), CONVERT(INT, C7.LotSize)))

                         WHEN LEN(RTRIM(CONVERT(NVARCHAR(10), CONVERT(INT, C7.LotSize)))) = 5 THEN '0' + RTRIM(CONVERT(NVARCHAR(10), CONVERT(INT, C7.LotSize)))

                         ELSE RTRIM(CONVERT(NVARCHAR(10), CONVERT(INT, C7.LotSize))) END AS Barcode -- 바코드

FROM \_TPDMMOutM AS A WITH(NOLOCK)

JOIN \_TPDMMOutItem AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.MatOutSeq = B.MatOutSeq

LEFT OUTER JOIN \_TDAItem AS C WITH(NOLOCK) ON B.CompanySeq = C.CompanySeq

AND B.ItemSeq = C.ItemSeq

-- LEFT OUTER JOIN \_TDAItemUserDefine AS C2 WITH(NOLOCK) ON B.CompanySeq = C2.CompanySeq

-- AND B.ItemSeq = C2.ItemSeq

-- AND C2.MngSerl = 1000163 -- 바코드라벨 품목번호

-- LEFT OUTER JOIN \_TDAItemUserDefine AS C3 WITH(NOLOCK) ON B.CompanySeq = C3.CompanySeq

-- AND B.ItemSeq = C3.ItemSeq

-- AND C3.MngSerl = 1000164 -- 바코드라벨 SPECIFICATION

LEFT OUTER JOIN \_TDAItemUserDefine AS C4 WITH(NOLOCK) ON B.CompanySeq = C4.CompanySeq

AND B.ItemSeq = C4.ItemSeq

AND C4.MngSerl = 1000173 -- 제품명_hct

LEFT OUTER JOIN \_TDAItem AS C5 WITH(NOLOCK) ON C4.CompanySeq = C5.CompanySeq

AND C4.MngValSeq = C5.ItemSeq

LEFT OUTER JOIN \_TDAItemProduct AS C7 WITH(NOLOCK) ON C5.CompanySeq = C7.CompanySeq -- 제품명_hct의 제품에 대한 lot적정사이즈 출력

AND C5.ItemSeq = C7.ItemSeq

LEFT OUTER JOIN \_TDAItemUserDefine AS C2 WITH(NOLOCK) ON C5.CompanySeq = C2.CompanySeq

AND C5.ItemSeq = C2.ItemSeq

AND C2.MngSerl = 1000163 -- 바코드라벨 품목번호

LEFT OUTER JOIN \_TDAItemUserDefine AS C3 WITH(NOLOCK) ON C5.CompanySeq = C3.CompanySeq

AND C5.ItemSeq = C3.ItemSeq

AND C3.MngSerl = 1000164 -- 바코드라벨 SPECIFICATION

 

WHERE A.CompanySeq = 1

AND A.MatOutSeq = 906

AND B.OutItemSerl = 1
