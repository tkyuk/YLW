FROM \_TPUORDPOItem AS A

JOIN \_TPUDelvItem AS C ON C.CompanySeq = A.CompanySeq

AND C.ProgFromSeq = A.POSeq

AND C.ProgFromSerl = A.POSerl

AND C.ProgFromTableSeq = 13

JOIN \_TPUDelv AS D ON D.CompanySeq = C.CompanySeq

AND D.DelvSeq = C.DelvSeq

 

JOIN \_TPUDelvInItem AS E ON E.CompanySeq = C.CompanySeq

AND E.ProgFromSeq = C.DelvSeq

AND E.ProgFromSerl = C.DelvSerl

AND E.ProgFromTableSeq = 10

JOIN \_TPUDelvIn AS F ON F.CompanySeq = E.CompanySeq

AND F.DelvInSeq = E.DelvInSeq

WHERE (A.POSeq = 642380 AND A.POSerl in(25,26,27,28))
