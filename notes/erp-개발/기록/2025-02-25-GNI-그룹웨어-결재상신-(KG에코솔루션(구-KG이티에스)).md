# GNI 그룹웨어 결재상신 (KG에코솔루션(구 KG이티에스))

BEGIN TRAN

 

SELECT \*FROM TDAGroupWare WHERE TblKey = 'HESK_0101202502240002'

SELECT \*FROM TDAGroupWareSlip_ESK WHERE Slipno in( '0101202502240002')

SELECT \*FROM TDAGroupWare_ESK WHERE APPRO_KEY= 'HESK_0101202502240002'

 

 

EXEC SGWSlipPrint_esk 'Q','0101202502240002001\_/','251001 ','00019'

go

EXEC SDAGroupWareCfm 'frmGWSlip','HESK_0101202502240002','2511001 ','00019'

go

EXEC SDAGroupWareStatus 'G','frmGWSlip','HESK_0101202502240002','251001 ','00019'

 

SELECT \*FROM TDAGroupWare WHERE TblKey = 'HESK_0101202502240002'

SELECT \*FROM TDAGroupWareSlip_ESK WHERE Slipno in( '0101202502240002')

SELECT \*FROM TDAGroupWare_ESK WHERE APPRO_KEY= 'HESK_0101202502240002'

 

 

ROLLBACK TRAN

 

SELECT \* FROM TASSlip AS A WHERE substring(a.SlipNo,5,8) + '-' + substring(a.SlipNo,14,4)= '20250224-002'

SELECT \* FROM TASSlip AS A WHERE substring(a.SlipNo,5,8) + '-' + substring(a.SlipNo,14,4)= '20250224-004'

 

SELECT \* FROM TASSlipRow AS A WHERE slipno = '0101202502240002'

SELECT \* FROM TASSlipRow AS A WHERE slipno = '0101202502240004'

SELECT TOP 100 \* FROM TDAGroupWare AS A ORDER BY RegDate DESC

BEGIN TRAN

DELETE FROM TDAGroupWare WHERE TblKey = 'HESK_0101202502240002'

DELETE FROM TDAGroupWareSlip_ESK WHERE Slipno in( '0101202502240002')

DELETE FROM TDAGroupWare_ESK WHERE APPRO_KEY= 'HESK_0101202502240002'

 

ROLLBACK TRAN

SELECT \* FROM TDAGroupWareSlip_ESK AS A WHERE Slipno in( '0101202502240002','0101202502240004')

 

 

 

SELECT \* FROM TDAGroupWare_ESK AS A WHERE APPRO_KEY in( 'HESK_0101202502240002','HESK_0101202502240004')

 

SELECT \* FROM TDAGroupWare_ESK AS A WHERE APPRO_KEY= 'HESK_0101202502240002'
