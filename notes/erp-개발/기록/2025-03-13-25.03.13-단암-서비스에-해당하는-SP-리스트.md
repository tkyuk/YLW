# 25.03.13 단암 - 서비스에 해당하는 SP 리스트

SELECT \* FROM \_TCAServiceMethod WHERE ServiceSeq = 1552 AND MethodSeq = 2

 

 

SELECT ProcessSeq,B.\*

FROM \_TCAServiceMethodKWFCache AS A

JOIN \_TCASQLScripts AS B ON B.SqlScriptSeq =A.SqlScriptSeq

WHERE ServiceSeq = 1552

AND MethodSeq = 2
