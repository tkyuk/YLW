# 25.02.04 티모넷 부가세 계산식 추가

CASE WHEN ROUND(B.Vat - ISNULL(BM.NextVat,0),CASE WHEN B.CurrCd = @wWon THEN @iWonAmt ELSE @iForAmt END) \<\> 0

THEN ROUND(B.Vat - ISNULL(BM.NextVat,0),CASE WHEN B.CurrCd = @wWon THEN @iWonAmt ELSE @iForAmt END)

ELSE CASE WHEN ISNULL(B.VatClss, '') = 1 THEN (ROUND(B.Amt - ISNULL(BM.NextAmt,0) - ISNULL(T.tAmt, 0) ,CASE WHEN B.CurrCd = @wWon THEN @iWonAmt ELSE @iForAmt END)) / 1.1 \* 0.1 -- 부가세 포함 금액에서 부가세 계산

WHEN ISNULL(B.VatClss, '') = 0 THEN (ROUND(B.Amt - ISNULL(BM.NextAmt,0) - ISNULL(T.tAmt, 0) ,CASE WHEN B.CurrCd = @wWon THEN @iWonAmt ELSE @iForAmt END)) \* 0.1 -- 부가세 미포함 금액에서 부가세 계산

ELSE 0 END END AS VAT, --버림, 부가세 ㅈ
