# 제뉴인 사이트 패치내역 작업자 UPDATE

SELECT A.DBPatchSeq , A.DBWorkSeq , A.DBPatchListSeq , A.Seq , A.PatchList ,

A.ToDB , A.FromDB , A.IsFail , A.Remark , A.LastUserSeq ,

A.LastDateTime

, ISNULL(B.UserName, N'') AS LastUserName

FROM \_TDEVDBPatchListHistory AS A WITH(NOLOCK)

LEFT OUTER JOIN \_TCAUser AS B WITH(NOLOCK) ON A.LastUserSeq = B.UserSeq

WHERE A.DBPatchSeq = 13

AND A.DBWorkSeq IN (46476,46464,46461)

ORDER BY A.LastDateTime DESC, A.Seq DESC

 

 

 

BEGIN TRAN

 

UPDATE \_TDEVDBPatchListHistory

SET LastUserSeq = 46

FROM \_TDEVDBPatchListHistory

WHERE DBPatchSeq = 13

AND DBWorkSeq IN (46476,46464,46461)

ROLLBACK TRAN
