UPDATE \_TLGWHStock

SET PrevQty = ISNULL(Y.PrevQty,0),

STDPrevQty = ISNULL(Y.STDPrevQty,0)

FROM \_TLGWHStock AS X

JOIN (SELECT @CompanySeq AS CompanySeq, @NextMonth AS StkYM,

A.WHSeq, A.ItemSeq, A.UnitSeq,

(SUM(ISNULL(A.PrevQty, 0)) + SUM(ISNULL(A.InQty, 0)) - SUM(ISNULL(A.OutQty, 0))) AS PrevQty,

(SUM(ISNULL(A.STDPrevQty, 0)) + SUM(ISNULL(A.STDInQty, 0)) - SUM(ISNULL(A.STDOutQty, 0))) AS STDPrevQty

FROM \_TLGWHStock AS A WITH(NOLOCK)

WHERE A.CompanySeq = @CompanySeq

AND A.StkYm BETWEEN @FromYM AND @ToYM

GROUP BY A.WHSeq, A.ItemSeq, A.UnitSeq) AS Y ON X.CompanySeq = Y.CompanySeq

AND X.StkYM = Y.StkYM

AND X.WHSeq = Y.WHSeq

AND X.ItemSeq = Y.ItemSeq

AND X.UnitSeq = Y.UnitSeq

SELECT \*

FROM \_TLGWHStock AS A

LEFT OUTER JOIN \_TLGInOutStock AS B ON B.CompanySeq = A.CompanySeq

AND B.ItemSeq = A.ItemSeq

AND B.WHSeq = A.WHSeq

AND B.InOutYM = A.StkYM

WHERE A.StkYM = '202412'

AND B.CompanySeq IS NULL

AND A.OutQty \<\> 0

ORDER BY A.ItemSeq
