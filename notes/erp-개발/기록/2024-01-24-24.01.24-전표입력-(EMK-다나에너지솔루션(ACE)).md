# 24.01.24 전표입력 (EMK-다나에너지솔루션(ACE))

BEGIN TRAN

EXEC \_SWCOMGroupWarePrint 1 , 1 , 1 , 502082 , 'Slip' , 92999 , 'DN'

ROLLBACK TRAN

--SELECT TOP 100 \* FROM \_TACSlip WHERE SlipMstSeq = 92999

 

 

 

 

select \* from \_TDAAccount WHERE AccSeq = 977 -- 계정과목

select \* from \_TDAAccountSub WHERE AccSeq = 977 -- 계정과목의관리항목

select \* from \_TDAAccountRem WHERE RemSeq IN (3009) -- 관리항목테이블

 

\_TDAAccountSub

SELECT \* FROM \_TACSlip WHERE SlipMstSeq = 92999 -- 전표 M테이블

 

select SlipSeq,\* from \_TACSlipRow

WHERE SlipMstSeq = 92999 -- 전표 D테이블

AND SlipSeq = 263409

 

SELECT SlipSeq,P.RemValText,\*

--,CASE WHEN CHARINDEX(',', ISNULL(P.RemValText , '')) \<\> 0 OR ISNULL(P.RemValText , '') = '' THEN ISNULL(P.RemValText , '')

-- ELSE dbo.\_fnGWAmtCommma(1,0,5,ISNULL(P.RemValText , ''),0,0,2,0)

-- END AS SupplyAmt

FROM \_TACSliprem AS P

WHERE SlipSeq = 263409

select \* from \_TDAAccountSub -- CompanySeq, AccSeq, RemSeq

 

 

select top 100 \* from \_TACSlip -- CompanySeq, SlipMstSeq

select top 100 accseq,\* from \_TACSlipRow -- CompanySeq, SlipMstSeq, SlipSeq

select top 100 RemSeq,\* from \_TACSlipRem --CompanySeq, SlipSeq, RemSeq

select top 100 C.CompanySeq,dbo.\_fnGWAmtCommma(1,0,5,ISNULL(P.RemValText , ''),0,0,2,0),

CASE WHEN CHARINDEX(',', ISNULL(P.RemValText , '')) \<\> 0 OR ISNULL(P.RemValText , '') = '' THEN '11111111'+ISNULL(P.RemValText , '')

ELSE '111111'+dbo.\_fnGWAmtCommma(1,0,5,ISNULL(P.RemValText , ''),0,0,2,0)

END AS SupplyAmt,\*

from \_TACSlipRow as a

LEFT OUTER join \_TACSlipRem as P on P.SlipSeq = a.SlipSeq

AND P.RemSeq = 3009

LEFT OUTER join \_TDAAccountSub as c on c.AccSeq = a.AccSeq

and c.RemSeq = P.RemSeq

WHERE P.SlipSeq = 263409
