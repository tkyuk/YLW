# 25.09.26 직군 연구, 계정과목 판관 -\> 연구 업데이트

-- 직군이 연구인데 판관 계정과목

 

BEGIN TRAN

SELECT A.ACCSEQ,

AccName

FROM das_VARBizTripReport AS A

> JOIN das_TARExpendAuthorityEmp AS B ON B.EmpSeq = A.TripEmpSeq

JOIN das_TARExpendAuthorityM AS C ON C.GroupSeq = B.GroupSeq

WHERE B.GroupSeq = 2

AND A.ACCSEQ IS NOT NULL

 

UPDATE A

SET AccSeq = D.AccSeq3

FROM das_TARBizTripReportAmt AS A

JOIN \_TARBizTripCost AS E ON E.BizTripSeq = A.BizTripSeq

> JOIN das_TARExpendAuthorityEmp AS B ON B.EmpSeq = E.TripEmpSeq

JOIN das_TARExpendAuthorityM AS C ON C.GroupSeq = B.GroupSeq

> JOIN \#tMP AS D ON D.AccSeq2 = A.ACCSEQ
>
> JOIN \_TDAAccount AS F ON F.AccSeq = A.AccSeq

WHERE B.GroupSeq = 2

AND A.ACCSEQ IS NOT NULL

 

 

SELECT A.ACCSEQ,

AccName

FROM das_VARBizTripReport AS A

> JOIN das_TARExpendAuthorityEmp AS B ON B.EmpSeq = A.TripEmpSeq

JOIN das_TARExpendAuthorityM AS C ON C.GroupSeq = B.GroupSeq

WHERE B.GroupSeq = 2

AND A.ACCSEQ IS NOT NULL

ROLLBACK TRAN

 

 

SELECT \*

FROM das_VARBizTripReport AS A

JOIN das_TARExpendAuthorityEmp AS B ON B.EmpSeq = A.TripEmpSeq

JOIN das_TARExpendAuthorityM AS C ON C.GroupSeq = B.GroupSeq

> --JOIN \#tMP AS D ON D.AccSeq2 = A.ACCSEQ

WHERE B.GroupSeq = 2

AND ACCSEQ IS NOT NULL

 

 

CREATE TABLE \#tMP(

> AccSeq1 INT
>
> ,AccSeq2 INT
>
> ,AccSeq3 INT
>
> ,AccName1 NVARCHAR(100)
>
> ,AccName2 NVARCHAR(100)
>
> ,AccName3 NVARCHAR(100)

)

INSERT INTO \#tMP

SELECT DISTINCT L.ValueSeq, M.ValueSeq,N.ValueSeq,A.AccName, B.AccName,C.AccName

FROM \_TDAUMinor AS J

LEFT OUTER JOIN \_TDAUMinorValue AS L WITH(NOLOCK) ON J.CompanySeq = L.CompanySeq

AND J.MinorSeq = L.MinorSeq

AND L.Serl = 15000007 -- 제조비용계정

LEFT OUTER JOIN \_TDAUMinorValue AS M WITH(NOLOCK) ON J.CompanySeq = M.CompanySeq

AND J.MinorSeq = M.MinorSeq

AND M.Serl = 15000008 -- 판관비용계정

LEFT OUTER JOIN \_TDAUMinorValue AS N WITH(NOLOCK) ON J.CompanySeq = N.CompanySeq

AND J.MinorSeq = N.MinorSeq

AND N.Serl = 15000009 -- 연구비용계정

LEFT OUTER JOIN \_TDAAccount AS A WITH(NOLOCK) ON A.CompanySeq = L.CompanySeq

AND A.AccSeq = L.ValueSeq

LEFT OUTER JOIN \_TDAAccount AS B WITH(NOLOCK) ON B.CompanySeq = M.CompanySeq

AND B.AccSeq = M.ValueSeq

LEFT OUTER JOIN \_TDAAccount AS C WITH(NOLOCK) ON C.CompanySeq = N.CompanySeq

AND C.AccSeq = N.ValueSeq

WHERE J.MajorSeq = 1110352
