# 24.12.09 단암 구매입고, 매입정산처리 -\> 납품 수량 다른건 UPDATE

진행 테이블

입고, 매입정산

 

BEGIN TRAN

SELECT A.Qty, B.FromQty, B.ToQty

fROM \_TPUDelvItem AS A

JOIN \_TCOMSourceDaily AS B ON B.CompanySeq = A.CompanySeq

AND B.FromSeq = A.DelvSeq

AND B.FromSerl = A.DelvSerl

AND FromTableSeq = 10

WHERE B.FromSeq = 42461

 

UPDATE B

SET TOQty = A.Qty, FromQty = A.Qty

fROM \_TPUDelvItem AS A

JOIN \_TCOMSourceDaily AS B ON B.CompanySeq = A.CompanySeq

AND B.FromSeq = A.DelvSeq

AND B.FromSerl = A.DelvSerl

AND FromTableSeq = 10

WHERE B.FromSeq = 42461

AND A.Qty \<\> B.FromQty

 

 

SELECT A.Qty, B.FromQty, B.ToQty

fROM \_TPUDelvItem AS A

JOIN \_TCOMSourceDaily AS B ON B.CompanySeq = A.CompanySeq

AND B.FromSeq = A.DelvSeq

AND B.FromSerl = A.DelvSerl

AND FromTableSeq = 10

WHERE B.FromSeq = 42461

 

 

ROLLBACK TRAN

 

BEGIN TRAN

 

SELECT \* FROM \_TPUDelvInItem WHERE ProgFromSeq = 42461

 

UPDATE c

SET Qty = A.Qty

FROM \_TPUDelvItem AS A

JOIN \_TPUDelvInItem AS B ON B.CompanySeq = A.CompanySeq

AND B.ProgFromSeq = A.DelvSeq

AND B.ProgFromSerl = A.DelvSerl

JOIN \_TPUBuyingAcc AS C ON C.CompanySeq = A.CompanySeq

AND C.SourceSeq = B.DelvInSeq

AND C.SourceSerl = B.DelvInSerl

WHERE A.DelvSeq = 42461

AND A.Qty \<\> B.Qty

 

 

UPDATE B

SET Qty = A.Qty

FROM \_TPUDelvItem AS A

JOIN \_TPUDelvInItem AS B ON B.CompanySeq = A.CompanySeq

AND B.ProgFromSeq = A.DelvSeq

AND B.ProgFromSerl = A.DelvSerl

WHERE A.DelvSeq = 42461

AND A.Qty \<\> B.Qty

 

 

SELECT A.DelvSerl , A.Qty,B.Qty,c.Qty

FROM \_TPUDelvItem AS A

JOIN \_TPUDelvInItem AS B ON B.CompanySeq = A.CompanySeq

AND B.ProgFromSeq = A.DelvSeq

AND B.ProgFromSerl = A.DelvSerl

JOIN \_TPUBuyingAcc AS C ON C.CompanySeq = A.CompanySeq

AND C.SourceSeq = B.DelvInSeq

AND C.SourceSerl = B.DelvInSerl

 

WHERE A.DelvSeq = 42461

 

 

ROLLBACK TRAN
