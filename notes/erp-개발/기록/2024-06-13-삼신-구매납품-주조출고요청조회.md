# 삼신 - 구매납품 - 주조출고요청조회

-- 202403110001

 

 

SELECT D.DelvNo, C.DelvSeq,

C.DelvSerl,

B.ToSeq,

A.COReqSeq ,

A.COReqSerl,

B.Qty,

UMToCompany

,C.\*

FROM ssv_TPDCastOutReqItem AS A

JOIN ssv_TCOMProgressCompany AS B WITH(NOLOCK) ON B.CompanySeq = 1

AND B.FromTableSeq = 15000084 --주조출고요청

AND B.ToTableSeq = 10 --구매납품, 이동(입출고)

AND B.FromSeq = A.COReqSeq

AND B.FromSerl = A.COReqSerl

LEFT OUTER JOIN \_TPUDelvItem AS C WITH(NOLOCK) ON C.CompanySeq = 1

AND C.DelvSeq = B.ToSeq

AND C.DelvSerl = B.ToSerl

LEFT OUTER JOIN \_TPUDelv AS D WITH(NOLOCK) ON D.CompanySeq = 1

AND D.DelvSeq = B.ToSeq

 

where D.DelvNo IS NOT NULL

AND ISNULL(B.ProcSeq, 0) = 0

ORDER BY 1

 

 

 

SELECT \* FROM \_TPUDelvItem WHERE DelvSeq = 104 AND DelvSerl = 4

SELECT \* FROM ssv_TPUDelvItem WHERE DelvSeq = 104 AND DelvSerl = 4

SELECT \* FROM ssv_TCOMProgressCompany WHERE ToSeq = 104 AND ToSerl = 4
