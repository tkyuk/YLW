# 체크인 UPDATE문 (제뉴인)

-- common DB에서 해야함

 

-- 서비스 Seq 확인

SELECT \* FROM \_TCAService WHERE ServiceID ='Angkor.danam.PD.DatPDWorkHourResult_danam'

 

-- UPDATE

UPDATE \_TCAService

SET CheckOutYN = 0,

CheckOutDateTime = null,

CheckOutUserSeq = 0,

CheckInDateTime = @CurrDate

WHERE ServiceSeq = @ServiceSeq
