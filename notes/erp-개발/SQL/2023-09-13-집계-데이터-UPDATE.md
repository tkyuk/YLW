# 집계 데이터 UPDATE

UPDATE A

SET A.EmpNum = B.EmpNum

FROM \#Temp_DdFixCol AS A

JOIN (SELECT A.RecruitSeq, COUNT(B.ResumeSeq) AS EmpNum

FROM \#Temp_DdFixCol AS A

JOIN \#EduEmp_NEW AS B ON B.RecruitSeq = A.RecruitSeq

GROUP BY A.RecruitSeq ) AS B ON A.RecruitSeq = B.RecruitSeq
