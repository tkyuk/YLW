DECLARE @AbsDate NVARCHAR(8), @PossibleDays DECIMAL(19,5), @Month NVARCHAR(8), @IDX_NO INT, @UseDays DECIMAL(19,5)

SELECT @AbsDate = '', @UseDays = 0

SELECT '#Result',\* FROM \#Result

SELECT '#Tmp_Useday',\* FROM \#Tmp_Useday

 

WHILE(1=1)

BEGIN

SELECT TOP 1 @UseDays = UseDays,

@AbsDate = AbsDate

FROM \#Tmp_Useday

WHERE AbsDate \> @AbsDate

AND UseDays \> 0

ORDER BY AbsDate

 

IF @@ROWCOUNT = 0 BREAK

 

SELECT @IDX_NO = 0

SELECT @PossibleDays = 0

 

WHILE(1=1)

BEGIN

 

SELECT TOP 1 @IDX_NO = IDX_NO,

@PossibleDays = PossibleDays

FROM \#Result

WHERE @AbsDate BETWEEN UseFrDate AND UseEndDate

AND IDX_NO \> @IDX_NO

AND PossibleDays \> 0

ORDER BY IDX_NO

 

IF @@ROWCOUNT = 0 BREAK

 

IF @PossibleDays \>= @UseDays

BEGIN

UPDATE A

SET UseDays = UseDays + @UseDays, -- 사용일 누적

PossibleDays = PossibleDays - @UseDays -- 잔여일 차감

FROM \#Result AS A

WHERE IDX_NO = @IDX_NO

END

ELSE

BEGIN

UPDATE A

SET UseDays = InitDays,

PossibleDays = 0

FROM \#Result AS A

WHERE IDX_NO = @IDX_NO

END

 

SELECT @UseDays = @UseDays - @PossibleDays

 

IF @UseDays \<= 0 BREAK

END

 

END

 

SELECT \* FROM \#Result

-- Stored Procedure 생성 구문

CREATE PROCEDURE dbo.hcwoo_SWLotCalc

AS

BEGIN

SET NOCOUNT ON;

-- 1. 테스트 데이터 준비

CREATE TABLE \#TempPJTNeedStk (

PJTSeq INT,

ItemSeq INT,

NeedQty DECIMAL(19, 5), -- 총 필요수량

StkQty DECIMAL(19, 5) -- 기 충당된 수량

);

CREATE TABLE \#Tmp_StkQty (

ItemSeq INT,

LotNo NVARCHAR(100),

StkQty DECIMAL(19, 5)

);

INSERT INTO \#TempPJTNeedStk (PJTSeq, ItemSeq, NeedQty, StkQty) VALUES (1, 101, 100, 0);

INSERT INTO \#Tmp_StkQty (ItemSeq, LotNo, StkQty) VALUES

(101, 'LOT-001', 10),

(101, 'LOT-002', 30),

(101, 'LOT-003', 80),

(101, 'LOT-004', 70);

-- 2. Lot별 재고와 필요수량을 결합한 기본 임시테이블 생성

CREATE TABLE \#TempLotQty(

IDX_NO INT,

PJTSeq INT,

ItemSeq INT,

NeedQty DECIMAL(19,5), -- 순 필요수량

RemNeedQty DECIMAL(19,5),

LotNo NVARCHAR(100),

LotStkQty DECIMAL(19,5)

);

INSERT INTO \#TempLotQty(IDX_NO, PJTSeq, ItemSeq, NeedQty, RemNeedQty, LotNo, LotStkQty)

SELECT

ROW_NUMBER() OVER(PARTITION BY A.PJTSeq, A.ItemSeq ORDER BY B.LotNo) AS IDX_NO, -- LotNo 순으로 선입선출 순서 부여

A.PJTSeq,

A.ItemSeq,

ISNULL(A.NeedQty, 0) - ISNULL(A.StkQty, 0) AS NeedQty,

ISNULL(A.NeedQty, 0) - ISNULL(A.StkQty, 0) AS RemNeedQty,

B.LotNo,

B.StkQty AS LotStkQty

FROM \#TempPJTNeedStk AS A WITH(NOLOCK)

LEFT OUTER JOIN \#Tmp_StkQty AS B WITH(NOLOCK) ON A.ItemSeq = B.ItemSeq;

------------------------------------------------------------------------------------

-- 비교 로직 1: \#TempCumulative 임시 테이블 사용 방식

------------------------------------------------------------------------------------

-- 3. 윈도우 함수를 사용하여 누적 재고량 계산

CREATE TABLE \#TempCumulative (

IDX_NO INT,

PJTSeq INT,

ItemSeq INT,

NeedQty DECIMAL(19, 5),

LotNo NVARCHAR(100),

LotStkQty DECIMAL(19, 5),

PrevCumulativeLotQty DECIMAL(19, 5), -- 이전 Lot까지의 누적 재고

CurrentCumulativeLotQty DECIMAL(19, 5) -- 현재 Lot까지의 누적 재고

);

INSERT INTO \#TempCumulative

SELECT

IDX_NO,

PJTSeq,

ItemSeq,

NeedQty,

LotNo,

LotStkQty,

ISNULL(SUM(LotStkQty) OVER (PARTITION BY PJTSeq, ItemSeq ORDER BY IDX_NO ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING), 0) AS PrevCumulativeLotQty,

SUM(LotStkQty) OVER (PARTITION BY PJTSeq, ItemSeq ORDER BY IDX_NO) AS CurrentCumulativeLotQty

FROM \#TempLotQty;

-- 4. 누적 재고량을 기반으로 각 Lot별 배분 수량(AllocatedQty) 계산

SELECT

A.PJTSeq,

A.ItemSeq,

A.NeedQty,

A.LotNo,

A.LotStkQty,

CASE

WHEN A.NeedQty \<= A.PrevCumulativeLotQty THEN 0

WHEN A.NeedQty \> A.CurrentCumulativeLotQty THEN A.LotStkQty

ELSE A.NeedQty - A.PrevCumulativeLotQty

END AS AllocatedQty

FROM \#TempCumulative AS A WITH(NOLOCK)

ORDER BY A.PJTSeq, A.ItemSeq, A.IDX_NO;

------------------------------------------------------------------------------------

-- 비교 로직 2: CTE 사용 방식

------------------------------------------------------------------------------------

CREATE TABLE \#ResultQty (

IDX_NO INT,

PJTSeq INT,

ItemSeq INT,

LotNo NVARCHAR(100),

LotStkQty DECIMAL(19,5),

AllocQty DECIMAL(19,5)

);

WITH CTE AS (

SELECT

T.IDX_NO,

T.PJTSeq,

T.ItemSeq,

T.NeedQty,

T.LotNo,

T.LotStkQty,

SUM(T.LotStkQty) OVER(

PARTITION BY T.PJTSeq, T.ItemSeq

ORDER BY T.IDX_NO

ROWS UNBOUNDED PRECEDING

) AS CumLotQty

FROM \#TempLotQty T

)

INSERT INTO \#ResultQty (IDX_NO, PJTSeq, ItemSeq, LotNo, LotStkQty, AllocQty)

SELECT

IDX_NO,

PJTSeq,

ItemSeq,

LotNo,

LotStkQty,

CASE

WHEN CumLotQty \<= NeedQty

THEN LotStkQty

WHEN CumLotQty - LotStkQty \< NeedQty

THEN NeedQty - (CumLotQty - LotStkQty)

ELSE 0

END AS AllocQty

FROM CTE

ORDER BY PJTSeq, ItemSeq, IDX_NO;

SELECT 111, \* FROM \#ResultQty;

-- 임시 테이블 정리

DROP TABLE \#TempPJTNeedStk;

DROP TABLE \#Tmp_StkQty;

DROP TABLE \#TempLotQty;

DROP TABLE \#TempCumulative;

DROP TABLE \#ResultQty;

SET NOCOUNT OFF;

END;
