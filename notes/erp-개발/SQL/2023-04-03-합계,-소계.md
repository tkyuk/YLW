hsg_SWACCondLoanQuery

DROP PROC hsg_SWACCondLoanQuery

GO

CREATE PROC hsg_SWACCondLoanQuery

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

AS

 

DECLARE @AccUnit INT ,

@AccSeq INT ,

@BankSeq INT ,

@CurrSeq INT ,

@SMKoOrFor INT ,

@UMLoanKind INT ,

@LoanDate NCHAR(8) ,

@LoanDateTo NCHAR(8) ,

@Date NCHAR(8) ,

@MinorSeq1 INT ,

@LoanNo NVARCHAR(50),

@LoanName NVARCHAR(100),

@SMAccStd INT,

@RemSeq INT -- 관리항목코드

 

 

SELECT @AccUnit = ISNULL(AccUnit , 0),

@AccSeq = ISNULL(AccSeq , 0),

@BankSeq = ISNULL(BankSeq , 0),

@CurrSeq = ISNULL(CurrSeq , 0),

@SMKoOrFor = ISNULL(SMKoOrFor , 0),

@UMLoanKind = ISNULL(UMLoanKind , 0),

@LoanDate = ISNULL(LoanDate , ''),

@LoanDateTo = ISNULL(LoanDateTo , ''),

@Date = ISNULL(Date , ''),

@MinorSeq1 = ISNULL(MinorSeq1 , 0),

@LoanNo = ISNULL(LoanNo , ''),

@LoanName = ISNULL(LoanName , '')

FROM \#BIZ_IN_DataBlock1

 

 

 

SELECT @SMAccStd = 1 -- 회계기준단위 GAAP로 우선 세팅 -- 추후제거 -- IFRS 링크드화면 연결로 인해 주석처리 및 아래와 같이 가져오도록 수정 2017.05.08 by sryoun.

 

SELECT @RemSeq = 1007 -- 관리항목 차입(1007)

 

IF @LoanDateTo = ''

BEGIN

SELECT @LoanDateTo = '29991231'

END

 

CREATE TABLE \#Temp_LoanMain ( -- 20070329 by Him 조건에 해당하는 차입번호

LoanSeq INT NULL ,

Amt DECIMAL(19,5) NULL, -- 원화금액

ForAmt DECIMAL(19,5) NULL, -- 외화금액

BfrOpenDate NCHAR(8) NULL)

CREATE CLUSTERED INDEX idx_LoanSeqMain ON \#Temp_LoanMain( LoanSeq )

 

 

CREATE TABLE \#Temp_TACLoanPlan(

LoanSeq INT , -- 차입코드

Amt DECIMAL(19,5) null, -- 원화금액 -- 대변금액

ForAmt DECIMAL(19,5) null, -- 외화금액 -- 대변금액

Amt1 DECIMAL(19,5) null, -- 원화상환금액 -- 차변금액

Amt5 DECIMAL(19,5) null, -- 외화상환금액 -- 차변금액

Amt2 DECIMAL(19,5) null, -- 원화이자지급액

Amt6 DECIMAL(19,5) null, -- 외화이자지급액

Amt4 DECIMAL(19,5) null, -- 원화이자잔액

Amt8 DECIMAL(19,5) null -- 외화이자잔액

)

CREATE CLUSTERED INDEX idx_LoanSeq ON \#Temp_TACLoanPlan( LoanSeq )

 

 

CREATE TABLE \#LoanPlan(

LoanSeq INT,

IsPay NCHAR(01),

SMInterestOrCapital INT,

PayAmt DECIMAL(19,5),

PayForAmt DECIMAL(19,5),

PayInAmt DECIMAL(19,5),

PayForIntAmt DECIMAL(19,5))

CREATE CLUSTERED INDEX idx_LoanSeqPlan ON \#LoanPlan( LoanSeq )

 

 

INSERT INTO \#Temp_LoanMain ( LoanSeq ,Amt , ForAmt, BfrOpenDate)

SELECT B.LoanSeq , ISNULL(B.Amt, 0), ISNULL(B.ForAmt, 0), C.SystemOpenDate

FROM \_TACLoan AS B WITH (NOLOCK)

JOIN \_TDAAccUnit AS C WITH(NOLOCK) ON B.CompanySeq = C.CompanySeq AND B.AccUnit = C.AccUnit

WHERE B.CompanySeq = @CompanySeq

AND (@LoanDate = N'' OR B.LoanDate \>= @LoanDate)

AND (@LoanDateTo = N'' OR B.LoanDate \<= @LoanDateTo)

AND B.LoanDate \<= @Date -- 차입일이 기준일보다 작아야 함

AND (@AccUnit = 0 OR B.AccUnit = @AccUnit )

AND (@AccSeq = 0 OR B.Accseq = @AccSeq )

AND (@BankSeq = 0 OR B.BankSeq = @BankSeq )

AND (@UMLoanKind = 0 OR B.UMLoanKind = @UMLoanKind )

AND (@LoanName = N'' OR B.LoanName LIKE @LoanName + N'%')

 

 

INSERT INTO \#LoanPlan (LoanSeq, IsPay, SMInterestOrCapital, PayAmt, PayForAmt, PayInAmt, PayForIntAmt)

SELECT A.LoanSeq,

CASE WHEN C.AccDate \> @Date THEN N'0' -- 기준일 조회 이후 전표처리 한 경우 제외 by 김성훈 2019.01.30 201901290205

WHEN ISNULL(A.RepaySlipSeq, 0) \<\> 0 AND C.IsSet = N'1' THEN N'1' -- 상환전표를 작성하였더라도 승인되지 않은 건의 경우 상환액으로 조회되지 않도록 수정 (202209130121) 2022.09.15 by 박민경

WHEN A.PayDate \< @Date AND A.PayDate \< B.BfrOpenDate THEN N'1'

ELSE N'0'

END ,

A.SMInterestOrCapital,

A.PayAmt,

A.PayForAmt,

A.PayIntAmt,

A.PayForIntAmt

FROM \_TACLoanPlan AS A WITH (NOLOCK) JOIN \#Temp_LoanMain AS B

ON A.LoanSeq = B.LoanSeq

LEFT OUTER JOIN \_TACSlipRow AS C WITH(NOLOCK)

ON C.CompanySeq = A.CompanySeq

AND C.SlipSeq = A.RepaySlipSeq

WHERE A.CompanySeq = @CompanySeq

 

INSERT INTO \#Temp_TACLoanPlan(LoanSeq,Amt,ForAmt,Amt1,Amt5,Amt2,Amt6,Amt4,Amt8)

SELECT B.LoanSeq,

ISNULL(B.Amt, 0),

ISNULL(B.ForAmt, 0),

SUM(CASE WHEN A.IsPay = N'1' AND A.SMInterestOrCapital IN (4025001,4025003) THEN A.PayAmt ELSE 0 END) Amt1, -- 원화상환금액

SUM(CASE WHEN A.IsPay = N'1' AND A.SMInterestOrCapital IN (4025001,4025003) THEN A.PayForAmt ELSE 0 END) Amt5, -- 외화상환금액

SUM(CASE WHEN A.IsPay = N'1' AND A.SMInterestOrCapital IN (4025002,4025003) THEN A.PayInAmt ELSE 0 END) Amt2, -- 원화이자지급액

SUM(CASE WHEN A.IsPay = N'1' AND A.SMInterestOrCapital IN (4025002,4025003) THEN A.PayForIntAmt ELSE 0 END) Amt6, -- 외화이자지급액

SUM(CASE WHEN A.IsPay = N'0' AND A.SMInterestOrCapital IN (4025002,4025003) THEN A.PayInAmt ELSE 0 END) Amt4, -- 원화이자잔액

SUM(CASE WHEN A.IsPay = N'0' AND A.SMInterestOrCapital IN (4025002,4025003) THEN A.PayForIntAmt ELSE 0 END) Amt8 -- 외화이자잔액

FROM \#Temp_LoanMain AS B LEFT OUTER JOIN \#LoanPlan AS A

ON B.LoanSeq = A.LoanSeq

GROUP BY B.LoanSeq, ISNULL(B.Amt, 0), ISNULL(B.ForAmt, 0)

 

SELECT ISNULL(B.BankName , '') AS BankName , -- 차입금융기관지점BankName

ISNULL(A.LoanName , '') AS LoanName , -- 차입명 LoanName

ISNULL(A.LoanNo , '') AS LoanNo , -- 차입번호 LoasNo

ISNULL(C.MinorName , '') AS MinorName1 , -- 차입종류 MinorName1

ISNULL(A.LoanDate , '') AS LoanDate , -- 차입일 LoanDate

ISNULL(E.Todate , '') AS Date1 , -- 최종상환일 Date1

ISNULL(G.InterestRate , 0) AS InterestRate , -- 이율 InterestRate

ISNULL(A.Amt , 0) AS Amt , -- 원화차입액 Amt -- 대변금액

ISNULL(N.DrAmt , 0) AS Amt1 , -- 원화상환액 Amt1 -- 차변금액

ISNULL(A.Amt, 0) - ISNULL(N.DrAmt, 0) AS Amt3 , -- 원화상환잔액 Amt3 -- 잔액

ISNULL(A.Remark , '') AS Remark , -- 비고 Remark

ISNULL(A.ForAmt , 0) AS ForAmt , -- 외화차입액 ForAmt -- 대변금액

ISNULL(N.DrForAmt , 0) AS Amt5 , -- 외화상환액 Amt5 -- 차변금액

ISNULL(A.ForAmt, 0) - ISNULL(N.DrForAmt, 0) AS Amt7 , -- 외화상환잔액 Amt7 -- 잔액

ISNULL(I.CurrName , '') AS CurrName , -- 통화 CurrName

ISNULL(A.ExRateDate , '') AS ExRateDate , -- 환율일 ExRateDate

ISNULL(A.ExRate , 0) AS ExRate , -- 환율 ExRate

ISNULL(J.AccName , '') AS AccName , -- 차입계정 AccName

ISNULL(K.MinorName , '') AS SMLoanType , -- 차입구분 SMLoanType

A.LoanSeq,

ISNULL(L.AccUnitName , '') AS AccUnitName,

ISNULL(A.BankSeq , 0) AS BankSeq,

ISNULL(M.AccName , '') AS CurrAccSeqName , -- 유동성차입계정 CurrAccSeqName

A.CurrSeq , -- 통화내부코드

A.AccUnit -- 회계단위내부코드

INTO \#Tmp_Result

FROM \_TACLoan AS A WITH(NOLOCK) LEFT OUTER JOIN \_TDABank AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.BankSeq = B.BankSeq

LEFT OUTER JOIN \_TDAUMinor AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.UMLoanKind = C.MinorSeq

LEFT OUTER JOIN (SELECT LoanSeq, MAX(Serl) AS Serl

FROM \_TACLoanRepayOpt WITH(NOLOCK)

WHERE CompanySeq = @CompanySeq

GROUP BY LoanSeq) AS D ON A.LoanSeq = D.LoanSeq -- 차입상환조건 Serl 맥스값 구한것(최종상환일)

LEFT OUTER JOIN \_TACLoanRepayOpt AS E WITH (NOLOCK) ON A.CompanySeq = E.CompanySeq

AND D.LoanSeq = E.LoanSeq

AND D.Serl = E.Serl

LEFT OUTER JOIN (SELECT LoanSeq, MAX(Serl) AS Serl

FROM \_TACLoanInterestOpt WITH(NOLOCK)

WHERE CompanySeq = @CompanySeq

GROUP BY LoanSeq) AS F ON A.LoanSeq = F.LoanSeq -- 차입이자조건 Serl 맥스값 구한것(이자율)

LEFT OUTER JOIN \_TACLoanInterestOpt AS G WITH(NOLOCK) ON A.CompanySeq = G.CompanySeq

AND F.LoanSeq = G.LoanSeq

AND F.Serl = G.Serl

LEFT OUTER JOIN \_TDACurr AS I WITH (NOLOCK) ON A.CompanySeq = I.CompanySeq

AND A.CurrSeq = I.CurrSeq

LEFT OUTER JOIN \_TDAAccount AS J WITH (NOLOCK) ON A.CompanySeq = J.CompanySeq

AND A.AccSeq = J.AccSeq

LEFT OUTER JOIN \_TDASMinor AS K WITH(NOLOCK) ON A.CompanySeq = K.CompanySeq

AND A.SMLoanType = K.MinorSeq

LEFT OUTER JOIN \_TDAAccUnit AS L WITH(NOLOCK) ON A.CompanySeq = L.CompanySeq

AND A.AccUnit = L.AccUnit

LEFT OUTER JOIN \_TDAAccount AS M WITH(NOLOCK) ON A.CompanySeq = M.CompanySeq

AND A.CurrAccSeq = M.AccSeq

OUTER APPLY(SELECT SUM(ISNULL(Z.DrAmt, 0)) AS DrAmt,

SUM(ISNULL(Z.DrForAmt, 0)) AS DrForAmt

FROM \_TACSlipRow AS Z WITH(NOLOCK)

JOIN \_TACSlipRem AS Y WITH(NOLOCK) ON Y.CompanySeq = Z.CompanySeq

AND Y.SlipSeq = Z.SlipSeq

AND Y.RemSeq = 1007

WHERE Z.CompanySeq = A.CompanySeq

AND Z.AccSeq = A.AccSeq

AND Y.RemValSeq = A.LoanSeq

AND Z.AccDate BETWEEN A.LoanDate AND @Date

AND Z.IsSet = '1') AS N

 

 

WHERE A.Companyseq = @CompanySeq

AND (@AccUnit = 0 OR A.AccUnit = @AccUnit )

AND (@AccSeq = 0 OR A.Accseq = @AccSeq )

AND A.LoanDate \>= @LoanDate

AND A.LoanDate \<= RTRIM(@LoanDateTo) + '99999999'

AND A.LoanDate \<= @Date -- 차입일이 기준일보다 작아야 함

AND (@BankSeq = 0 OR A.BankSeq = @BankSeq )

AND (@UMLoanKind = 0 OR A.UMLoanKind = @UMLoanKind )

AND (@SMKoOrFor = 0 OR A.SMKoOrFor = @SMKoOrFor )

AND (@CurrSeq = 0 OR A.CurrSeq = @CurrSeq )

AND (@LoanNo = N'' OR A.LoanNo LIKE @LoanNo + N'%' )

AND (@LoanName = N'' OR A.LoanName LIKE @LoanName + N'%' )

AND ((@MinorSeq1 = 0)

OR (@MinorSeq1 = 4032001 AND ISNULL(A.Amt, 0) - ISNULL(N.DrAmt , 0) \> 0)

OR (@MinorSeq1 = 4032002 AND ISNULL(A.Amt, 0) - ISNULL(N.DrAmt , 0) \<= 0)

)

ORDER BY B.BankName, A.LoanName, A.LoanNo

 

 

CREATE TABLE \#TMP

(

BankSeq INT,

LoanSeq INT,

Amt DECIMAL(19,5),

Amt1 DECIMAL(19,5),

Amt3 DECIMAL(19,5),

ForAmt DECIMAL(19,5),

Amt5 DECIMAL(19,5),

Amt7 DECIMAL(19,5)

)

 

INSERT INTO \#TMP

SELECT A.BankSeq, A.LoanSeq, A.Amt, A.Amt1, A.Amt3, A.ForAmt, A.Amt5, A.Amt7

FROM \#Tmp_Result AS A

 

SELECT BankSeq,

LoanSeq,

IIF(BankSeq IS NULL, '1', '0') AS BankSort,

IIF(LoanSeq IS NULL, '1', '0') AS LoanSort,

SUM(Amt ) AS Amt ,

SUM(Amt1 ) AS Amt1 ,

SUM(Amt3 ) AS Amt3 ,

SUM(ForAmt) AS ForAmt,

SUM(Amt5 ) AS Amt5 ,

SUM(Amt7 ) AS Amt7

INTO \#Result

FROM \#TMP

GROUP BY ROLLUP(BankSeq, LoanSeq)

 

 

SELECT CASE WHEN A.BankSeq IS NULL THEN '합계' ELSE D.BankName END AS BankName, -- 차입금융기관지점BankName

A.BankSeq,

CASE WHEN A.BankSeq IS NULL THEN ''

WHEN A.LoanSeq IS NULL THEN '소계' ELSE C.LoanName END AS LoanName, -- 차입명 LoanName

A.LoanSeq,

B.LoanNo , -- 차입번호 LoasNo

B.MinorName1 , -- 차입종류 MinorName1

B.LoanDate , -- 차입일 LoanDate

B.Date1 , -- 최종상환일 Date1

B.InterestRate , -- 이율 InterestRate

A.Amt , -- 원화차입액 Amt -- 대변금액

A.Amt1 , -- 원화상환액 Amt1 -- 차변금액

A.Amt3 , -- 원화상환잔액 Amt3 -- 잔액

B.Remark , -- 비고 Remark

A.ForAmt , -- 외화차입액 ForAmt -- 대변금액

A.Amt5 , -- 외화상환액 Amt5 -- 차변금액

A.Amt7 , -- 외화상환잔액 Amt7 -- 잔액

B.CurrName , -- 통화 CurrName

B.ExRateDate , -- 환율일 ExRateDate

B.ExRate , -- 환율 ExRate

B.AccName , -- 차입계정 AccName

B.SMLoanType , -- 차입구분 SMLoanType

B.AccUnitName,

B.CurrAccSeqName , -- 유동성차입계정 CurrAccSeqName

B.CurrSeq , -- 통화내부코드

B.AccUnit -- 회계단위내부코드

FROM \#Result AS A

LEFT OUTER JOIN \#Tmp_Result AS B ON A.BankSeq = B.BankSeq

AND A.LoanSeq = B.LoanSeq

LEFT OUTER JOIN \_TACLoan AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

AND C.LoanSeq = A.LoanSeq

LEFT OUTER JOIN \_TDABank AS D WITH(NOLOCK) ON D.CompanySeq = @CompanySeq

AND D.BankSeq = A.BankSeq

ORDER BY A.BankSort, 1, A.LoanSort, 3

 

RETURN
