# 전표승인월마감

emk_SWACMRMontlyReportConfirmSave

 

 

 

DECLARE @ClosingSeq INT,

@AccYM NCHAR(6)

--@StartDate NCHAR(8),

--@EndDate NCHAR(8)

 

SELECT @ClosingSeq = 1146

 

SELECT @AccYM = ISNULL(A.AccYM, '')

FROM \#BIZ_OUT_DataBlock1 AS A

 

 

-- \[회계업무\]전표월마감 테이블에 데이터가 있으면 UPDATE

UPDATE A

SET IsClose = ISNULL(B.IsCfm, '0')

FROM \_TCOMClosingYM AS A

-- 이태욱(2202): DtlUnitSeq 제외하고 조인

JOIN \#BIZ_OUT_DataBlock1 AS B ON A.ClosingYM = @AccYM

AND A.ClosingSeq = @ClosingSeq --마감업무코드

AND A.CompanySeq = @CompanySeq

 

 

IF NOT EXISTS(SELECT TOP 1 1

FROM \_TCOMClosingYM AS A WITH(NOLOCK)

WHERE A.ClosingYM = @AccYM

AND A.ClosingSeq = @ClosingSeq --마감업무코드

AND A.CompanySeq = @CompanySeq)

BEGIN

 

-- \[회계업무\]전표월마감 테이블에 데이터가 없으면 INSERT

INSERT INTO \_TCOMClosingYM (CompanySeq, ClosingSeq , ClosingYM, UnitSeq, DtlUnitSeq,

IsClose , LastUserSeq, LastDateTime)

SELECT @CompanySeq, @ClosingSeq, @AccYM, B.AccUnit, C.SlipUnit,

ISNULL(A.IsCfm, '0'), @UserSeq, GETDATE()

FROM \#BIZ_OUT_DataBlock1 AS A

LEFT OUTER JOIN \_TDAAccUnit AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

LEFT OUTER JOIN \_TACSlipUnit AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

 

END
