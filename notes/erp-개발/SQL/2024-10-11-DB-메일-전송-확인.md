SELECT A.Sent_status,B.event_type,B.log_date,A.recipients,B.description,A.subject,A.body,\*

FROM msdb.dbo.sysmail_allitems AS A

LEFT OUTER JOIN msdb.dbo.sysmail_event_log AS B ON B.mailitem_id = A.mailitem_id

ORDER BY A.mailitem_id desc

 

SELECT \* FROM msdb.dbo.sysmail_event_log order by log_id desc
