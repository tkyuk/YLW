# 시간 더하기 (시, 분)

DROP PROCEDURE dbo.jis_SXPRWkCalcTime

GO

CREATE PROCEDURE dbo.jis_SXPRWkCalcTime

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10) = '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

 

AS

-- 변수 선언

DECLARE @BegTime NVARCHAR(4), -- 시작시간

> @WkDTCnt DECIMAL(19, 5) -- 시간수
>
>  

-- 변수 설정

SELECT @BegTime = ISNULL(BegTime , ''), -- BegTime이 '0000'인 경우도 있어서 ''으로 수정(2021.06.28)

@WkDTCnt = ISNULL(WkDTCnt , 0)

FROM \#BIZ_IN_DataBlock1

 

SELECT @BegTime        AS BegTime,

@WkDTCnt        AS WkDTCnt,

REPLACE(CONVERT(VARCHAR(5), DATEADD(HOUR, FLOOR(@WkDTCnt), DATEADD(MINUTE, (@WkDTCnt - FLOOR(@WkDTCnt)) \* 60, + LEFT(@BegTime,2) + ':' + RIGHT(@BegTime,2))), 108),':','') AS EndTime,

@WkDTCnt AS DTCnt,

@WkDTCnt \* 60 AS MinCnt

RETURN

REPLACE(CONVERT(VARCHAR(5), DATEADD(HOUR, FLOOR(@WkDTCnt), DATEADD(MINUTE, (@WkDTCnt - FLOOR(@WkDTCnt)) \* 60, + LEFT(@BegTime,2) + ':' + RIGHT(@BegTime,2))), 108),':','')
