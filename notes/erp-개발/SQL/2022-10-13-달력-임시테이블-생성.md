# 달력 임시테이블 생성

DECLARE @STARTDATE VARCHAR(10)

DECLARE @ENDDATE VARCHAR(10)

 

SET @STARTDATE = '2019-01-01'

SET @ENDDATE = '2019-01-31'

 

;WITH MONTH AS

(

SELECT CONVERT(VARCHAR(10), @STARTDATE, 120) AS DAY

UNION ALL

SELECT CONVERT(VARCHAR(10), DATEADD(dd, 1, DAY), 120) AS DAY

FROM MONTH

WHERE CONVERT(VARCHAR(10), DAY, 120) \< CONVERT(VARCHAR(10), @ENDDATE, 120)

)

SELECT \* FROM MONTH

WkDate NCHAR(8)

)

 

;WITH DateCTE AS (

SELECT @StartDate AS WkDate

UNION ALL

SELECT DATEADD(DAY, 1, WkDate)

FROM DateCTE

WHERE DATEADD(DAY, 1, WkDate) \<= @EndDate

)

INSERT INTO \#TmpDate (WkDate)

-- 2-1. 조회조건에 해당하는 월의 날짜 담는 임시테이블 생성

CREATE TABLE \#TmpDate (

WKDate NCHAR(8),

DayName NVARCHAR(10)

)

-- 재귀 CTE로 날짜 생성

;WITH DateCTE AS (

SELECT @StartDate AS WKDate

UNION ALL

SELECT DATEADD(DAY, 1, WKDate)

FROM DateCTE

WHERE DATEADD(DAY, 1, WKDate) \<= @EndDate

)

INSERT INTO \#TmpDate (WKDate, DayName)

SELECT

CONVERT(NCHAR(8), WKDate, 112),

CASE (DATEPART(WEEKDAY, WKDate) + @@DATEFIRST - 2) % 7 + 1

WHEN 1 THEN '월요일'

WHEN 2 THEN '화요일'

WHEN 3 THEN '수요일'

WHEN 4 THEN '목요일'

WHEN 5 THEN '금요일'

WHEN 6 THEN '토요일'

WHEN 7 THEN '일요일'

END AS DayName

FROM DateCTE

OPTION (MAXRECURSION 31)
