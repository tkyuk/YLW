# 서비스 INPUT 추가

SELECT D.DataFieldID,D.DataFieldName,B.\*

FROM \_TCAService AS Z

JOIN \_TCAServiceDataBlock AS D ON D.ServiceSeq = Z.ServiceSeq

JOIN \_TCAServiceMethod AS A ON A.ServiceSeq = Z.ServiceSeq

LEFT OUTER JOIN \_TCAServiceMethodLink AS B WITH(NOLOCK) ON B.CompanySeq = 0

AND D.ServiceSeq = B.ServiceSeq

AND A.MethodSeq = B.MethodSeq

AND D.DataFieldSeq = B.DataFieldSeq

WHERE Z.ServiceSeq =2113 AND B.MethodSeq = 13

AND Input = '1'

--AND D.DataFieldID = 'SMCertiType'

 

SELECT \* fROM \_TCAServiceMethodLink

WHERE ServiceSeq = 2113

AND MethodSeq = 13

AND DataFieldSeq = 5

 

 

BEGIN TRAN

UPDATE \_TCAServiceMethodLink

SET Input = '1'

WHERE CompanySeq = 0

AND ServiceSeq = 2113

AND MethodSeq = 13

AND DataFieldSeq = 5

 

UPDATE \_TCAServiceMethodInOutCache

SET Input = '1'

WHERE ServiceSeq =2113 AND MethodSeq = 13 AND DataFieldID = 'SMCertiType'

 

UPDATE \_TCAServiceMethodKWFDataBlockCache

SET Input = '1'

WHERE ServiceSeq =2113 AND MethodSeq = 13 AND SourceDataFieldID = 'SMCertiType'

ROLLBACK TRAN

 

SELECT \* FROM \_TCAServiceMethodInOutCache WHERE ServiceSeq =2113 AND MethodSeq = 13 AND DataFieldID = 'SMCertiType'

SELECT \* FROM \_TCAServiceMethodLink WHERE ServiceSeq =2113 AND MethodSeq = 13 AND DataFieldSeq = 5

SELECT \* FROM \_TCAServiceMethodKWFDataBlockCache WHERE ServiceSeq =2113 AND MethodSeq = 13 AND SourceDataFieldID = 'SMCertiType'
