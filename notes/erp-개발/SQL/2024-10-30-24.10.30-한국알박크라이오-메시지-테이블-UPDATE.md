# 24.10.30 한국알박크라이오 - 메시지 테이블 UPDATE

UPDATE \_TCAMessage

SET Message = '비밀번호는 개인정보를 포함하지 않은 @1자리 이상이고, 다음의 규칙을 준수해야 합니다.'

FROM \_TCAMessage

WHERE MessageSeq = 2145

 

UPDATE \_TCAMessageLanguage

SET Message = '비밀번호는 개인정보를 포함하지 않은 @1자리 이상이고, 다음의 규칙을 준수해야 합니다.'

FROM \_TCAMessageLanguage

WHERE MessageSeq = 2145

 

 

SELECT \* FROM \_TCAMessage WHERE Message LIKE '%비밀번호는 %'

SELECT \* FROM \_TCAMessageLanguage
