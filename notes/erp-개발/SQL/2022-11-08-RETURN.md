# RETURN

SELECT subject, \[1반\],\[2반\],\[3반\]

-- 가져올 컬럼명 열로 변환시킬 행

 

FROM (SELECT class,subject,score FROM \#examresult) AS EXAM

-- 테이블

PIVOT (

AVG(score)

-- 집계 함수(검색할 열)

FOR class

-- 크로스탭의 기준이 되는열

IN (\[1반\],\[2반\],\[3반\])

-- 변환시킬 행

) AS PIVOTTABLE1

-- 피벗 테이블의 별칭

 

-- PIVOT 함수 (행을 열로 바꾸기)

-- 그룹함수를 이용하여 통계를 추출할 때 많이 사용

-- 그룹함수는 SUM(), COUNT(), AVG() 등을 사용할 수 있다.

-- 피벗컬럼값의 대괄호(\[ \])는 존재해야하며, FROM절과 PIVOT절의 별칭(result, pivot_result)은 꼭 붙여줘야 오류가 발생하지 않는다.
