# 부서에 연결된 사업자번호, 사업자 일련번호

ex.  공급받는자는 발주부서의 사업자등록 정보로 조회        

 

SELECT ISNULL(T1.TaxName, T2.TaxName) AS DeptCompanyName, -- 부서에 연결된 사업자 상호

> ISNULL(T1.TaxNoSerl,T2.TaxNoSerl) AS DeptTaxNoSerl , -- 부서에 연결된 사업자 일련번호

\*

FROM \_TPUORDPO AS Z WITH(NOLOCK)

LEFT OUTER JOIN \_TDADept AS J WITH(NOLOCK) ON Z.CompanySeq = J.CompanySeq

AND Z.DeptSeq = J.DeptSeq

LEFT OUTER JOIN \_TDATaxUnitHist AS T1 WITH(NOLOCK) ON T1.CompanySeq = J.CompanySeq

AND T1.TaxUnit = J.TaxUnit

AND (Z.PODate BETWEEN T1.FrDate AND T1.ToDate)

LEFT OUTER JOIN \_TDATaxUnit AS T2 WITH(NOLOCK) ON T2.CompanySeq = J.CompanySeq

AND T2.TaxUnit = J.TaxUnit
