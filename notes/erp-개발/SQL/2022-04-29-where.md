# where

WHERE A.CompanySeq = @CompanySeq

AND (@QuoteDate = '' OR B.QuoteDate \>= @QuoteDate )

AND (@QuoteDateTo = '' OR B.QuoteDate \<= @QuoteDateTo )

AND (@QuoteNo = '' OR B.QuoteNo LIKE @QuoteNo + '%')

AND (@EmpSeq = 0 OR B.EmpSeq = @EmpSeq )

AND (@DeptSeq = 0 OR B.DeptSeq = @DeptSeq )

AND (@QuoteLastDate = '' OR B.QuoteLastDate \>= @QuoteLastDate )

AND (@QuoteLastDateTo = '' OR B.QuoteLastDate \<= @QuoteLastDateTo )

AND (@QuoteDelvDate = '' OR A.QuoteDelvDate \>= @QuoteDelvDate )

AND (@QuoteDelvDateTo = '' OR A.QuoteDelvDate \<= @QuoteDelvDateTo )

AND (@ItemSeq = 0 OR A.ItemSeq = @ItemSeq )

AND (@ItemNameTxt = '' OR A.ItemName LIKE @ItemNameTxt + '%')

AND (@ItemNoTxt = '' OR A.ItemNo LIKE @ItemNoTxt + '%')

AND (@SpecTxt = '' OR A.Spec LIKE @SpecTxt + '%')
