# 품목자산분류설정 값 가져오기

DECLARE @AssetProd TABLE

(

AssetSeq INT

)

-- 생산품목자산분류 담기

INSERT INTO @AssetProd

SELECT DISTINCT

B.AssetSeq

FROM \_TDAItemHelpGroup AS A WITH(NOLOCK)

JOIN \_TDAItemAssetCodeHelp AS B WITH(NOLOCK) ON A.ItemCodeHelpSeq = B.ItemCodeHelpSeq

AND A.CompanySeq = B.CompanySeq

WHERE A.NameCodeHelp = 618011 -- 생산품목

AND A.CompanySeq = @CompanySeq

 

UPDATE \#ProdReqItem

SET IsProdItem = '1'

FROM \#ProdReqItem AS A

JOIN \_TDAItem AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.ItemSeq = A.ItemSeq

WHERE B.AssetSeq IN (SELECT AssetSeq FROM @AssetProd)
