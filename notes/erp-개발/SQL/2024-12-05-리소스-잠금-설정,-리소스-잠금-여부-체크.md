# 리소스 잠금 설정, 리소스 잠금 여부 체크

/\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*

DESCRIPTION : MES 연동용 품목 I/F Data 생성 SP

작 성 일 :

\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*/

CREATE PROCEDURE uck_SMESItemInfo

@CompanySeq INT = 1

AS

 

-- 리소스 잠금 여부 체크 : 잠금 상태이면 SP 종료하도록 한다.

IF APPLOCK_TEST('PUBLIC', 'uck_SMESItemInfo', 'EXCLUSIVE', 'SESSION') = 0

BEGIN

RETURN

END

-- 리소스 잠금 설정 : SP 중복 호출로 인한 중복 Data 생성을 방지하기 위함

EXEC SP_GETAPPLOCK 'uck_SMESItemInfo', 'EXCLUSIVE', 'SESSION'

DECLARE @FinDate DATETIME, -- 최종전송시간

@CrtDate DATETIME -- 전송시간

SELECT @CrtDate = GETDATE()

SELECT @FinDate = MAX(CrtDateTime) FROM uck_TMESITEMMASTER WITH(NOLOCK) WHERE CompanySeq = @CompanySeq

SELECT @FinDate = ISNULL(@FinDate, '')

 

 

--SELECT @FinDate = '20200901'

-- 임시 테이블 생성 --

CREATE TABLE \#ERP_Item

(

WorkingTag NVARCHAR(1) ,

ItemSeq INT,

ItemName NVARCHAR(100) NULL,

ItemNo NVARCHAR(100) NULL,

Spec NVARCHAR(100) NULL,

AssetSeq INT NULL,

AssetName NVARCHAR(100) NULL,

ItemClassLSeq INT NULL,

ItemClassLName NVARCHAR(100) NULL,

ItemClassMSeq INT NULL,

ItemClassMName NVARCHAR(100) NULL,

ItemClassSSeq INT NULL,

ItemClassSName NVARCHAR(100) NULL,

UnitSeq INT NULL,

UnitName NVARCHAR(100) NULL,

SMStatus INT NULL,

SMStatusName NVARCHAR(100) NULL,

IsInQc NCHAR(1) NULL,

IsLastQc NCHAR(1) NULL,

LastUserSeq INT NULL,

LastDateTime DATETIME NULL,

StdTime DECIMAL(19,5) NULL,

Sort NVARCHAR(1)

)

 

CREATE TABLE \#MES_Item

(

ItemSeq INT,

ItemName NVARCHAR(100) NULL,

ItemNo NVARCHAR(100) NULL,

Spec NVARCHAR(100) NULL,

AssetSeq INT NULL,

AssetName NVARCHAR(100) NULL,

ItemClassLSeq INT NULL,

ItemClassLName NVARCHAR(100) NULL,

ItemClassMSeq INT NULL,

ItemClassMName NVARCHAR(100) NULL,

ItemClassSSeq INT NULL,

ItemClassSName NVARCHAR(100) NULL,

UnitSeq INT NULL,

UnitName NVARCHAR(100) NULL,

SMStatus INT NULL,

SMStatusName NVARCHAR(100) NULL,

IsInQc NCHAR(1) NULL,

IsLastQc NCHAR(1) NULL,

StdTime DECIMAL(19,5) NULL,

LastUserSeq INT NULL,

LastDateTime DATETIME NULL

)

 

-- 최종 전송시간 이후 신규로 등록되거나 수정된 Data를 담는다.

INSERT INTO \#ERP_Item

SELECT 'U' ,

A.ItemSeq ,

A.ItemName ,

A.ItemNo ,

A.Spec ,

A.AssetSeq ,

ISNULL(C.AssetName, '') ,

ISNULL(L.ValueSeq,0) ,

CASE WHEN ISNULL(L.ValueSeq,0) = 0 THEN ''

ELSE ( SELECT ISNULL(MinorName,'')

FROM \_TDAUMinor WITH(NOLOCK)

WHERE CompanySeq = @CompanySeq AND MinorSeq = L.ValueSeq ) END AS ItemClassLName, -- 품목대분류

ISNULL(K.ValueSeq,0) ,

CASE WHEN ISNULL(K.ValueSeq,0) = 0 THEN ''

ELSE ( SELECT ISNULL(MinorName,'')

FROM \_TDAUMinor WITH(NOLOCK)

WHERE CompanySeq = @CompanySeq AND MinorSeq = K.ValueSeq ) END AS ItemClassMName, -- 품목중분류

ISNULL(B1.UMItemClass,0) AS ItemClassSSeq, -- 품목소분류코드

ISNULL(H.MinorName,'') AS ItemClassSName, -- 품목소분류

A.UnitSeq ,

ISNULL(B.UnitName , '') ,

ISNULL(A.SMStatus,0) AS SMStatus, -- 품목 상태

CASE WHEN ISNULL(A.SMStatus,0) = 0 THEN '' ELSE (SELECT ISNULL(MinorName,'') FROM \_TDASMinor WITH (NOLOCK) WHERE CompanySeq = @CompanySeq AND MajorSeq = 2001 AND MinorSeq = A.SMStatus) END AS ItemStatus, -- 품목 상태

ISNULL(F.IsInQc , '0') ,

ISNULL(F.IsLastQc, '0') ,

A.LastUserSeq ,

--A.LastDateTime ,

dbo.uck_fnGetMaxDate(ISNULL(A.LastDateTime, '1900-01-01 00:00.000'), ISNULL(B.LastDateTime, '1900-01-01 00:00.000'), ISNULL(F.LastDateTime, '1900-01-01 00:00.000'),

ISNULL(B1.LastDateTime, '1900-01-01 00:00.000'), ISNULL(M.LastDateTime, '1900-01-01 00:00.000'), '1900-01-01 00:00.000') AS LastDateTime,

ISNULL(M.StdTime,0) ,

'0'

FROM \_TDAItem AS A WITH(NOLOCK)

LEFT OUTER JOIN \_TDAUnit AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.UnitSeq = B.UnitSeq

LEFT OUTER JOIN \_TDAItemAsset AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.AssetSeq = C.AssetSeq

LEFT OUTER JOIN \_TDASMinor AS E WITH(NOLOCK) ON A.CompanySeq = E.CompanySeq

AND A.SMStatus = E.MinorSeq

LEFT OUTER JOIN \_TPDBaseItemQCType AS F WITH(NOLOCK) ON A.CompanySeq = F.CompanySeq

AND A.ItemSeq = F.ItemSeq

LEFT OUTER JOIN \_TDAItemClass AS B1 WITH(NOLOCK) ON ( A.CompanySeq = B1.CompanySeq AND A.ItemSeq = B1.ItemSeq AND B1.UMajorItemClass IN (2001,2004) )

LEFT OUTER JOIN \_TDAUMinor AS H WITH(NOLOCK) ON ( B1.CompanySeq = H.CompanySeq AND H.MajorSeq = LEFT( B1.UMItemClass, 4 ) AND B1.UMItemClass = H.MinorSeq )

LEFT OUTER JOIN \_TDAUMinorValue AS K WITH(NOLOCK) ON ( H.CompanySeq = K.CompanySeq AND K.MajorSeq IN (2001,2004) AND H.MinorSeq = K.MinorSeq AND K.Serl IN (1001,2001) )

LEFT OUTER JOIN \_TDAUMinorValue AS L WITH(NOLOCK) ON ( K.CompanySeq = L.CompanySeq AND L.MajorSeq IN (2002,2005) AND K.ValueSeq = L.MinorSeq AND L.Serl = 2001 )

LEFT OUTER JOIN (SELECT ItemSeq, SUM(StdWorkTime) AS StdTime, MAX(LastDateTime) AS LastDateTime FROM \_TPDROUItemProcWC GROUP BY ItemSeq) AS M ON A.ItemSeq = M.ItemSeq

WHERE A.CompanySeq = @CompanySeq

AND ((dbo.uck_fnGetMaxDate(ISNULL(A.LastDateTime , '1900-01-01 00:00.000'), ISNULL(B.LastDateTime, '1900-01-01 00:00.000'), ISNULL(F.LastDateTime, '1900-01-01 00:00.000'),

ISNULL(B1.LastDateTime, '1900-01-01 00:00.000'), ISNULL(M.LastDateTime, '1900-01-01 00:00.000'), '1900-01-01 00:00.000')) \> @FinDate)

AND ((dbo.uck_fnGetMaxDate(ISNULL(A.LastDateTime , '1900-01-01 00:00.000'), ISNULL(B.LastDateTime, '1900-01-01 00:00.000'), ISNULL(F.LastDateTime, '1900-01-01 00:00.000'),

ISNULL(B1.LastDateTime, '1900-01-01 00:00.000'), ISNULL(M.LastDateTime, '1900-01-01 00:00.000'), '1900-01-01 00:00.000')) \<= @CrtDate)

AND A.AssetSeq IN ('1','6', '10','16','2','4')

 

 

 

IF @@ROWCOUNT \<\> 0

BEGIN

INSERT INTO \#MES_Item

SELECT A.ItemSeq ,

A.ItemName ,

A.ItemNo ,

A.Spec ,

A.AssetSeq ,

A.AssetName ,

A.ItemClassLSeq ,

A.ItemClassLName,

A.ItemClassMSeq ,

A.ItemClassMName,

A.ItemClassSSeq ,

A.ItemClassSName,

A.UnitSeq ,

A.UnitName ,

FROM uck_TMESITEMMASTER AS A WITH(NOLOCK)

JOIN (

SELECT Z.ItemNo, MAX(X.SEQ) AS Seq

FROM \#ERP_Item AS Z

JOIN uck_TMESITEMMASTER AS X WITH(NOLOCK) ON X.CompanySeq = @CompanySeq

AND X.ItemNo = Z.ItemNo

WHERE Z.Sort = '0'

GROUP BY Z.ItemNo

) AS B ON A.Seq = B.Seq

SET A.WorkingTag = 'A'

FROM \#ERP_Item AS A

WHERE NOT EXISTS (SELECT 1 FROM \#MES_Item WHERE ItemNo = A.ItemNo)

AND A.Sort = '0'

--AND A.LastUserSeq = B.LastUserSeq

--AND A.LastDateTime = B.LastDateTime

FROM uck_TMESITEMMASTER AS A

WHERE AUDType = 'U'

GROUP BY ItemNo

) AS B ON A.Seq = B.Seq

JOIN \#ERP_Item AS C ON A.ItemNo = C.ItemNo

WHERE CompanySeq = @CompanySeq

AND LogDateTime \> @FinDate

AND LogDateTime \<= @CrtDate

CREATE OR ALTER PROC SJPARKTEST

AS

SET NOCOUNT ON

SET TRANSACTION ISOLATION LEVEL READ UNCOMMITTED

 

-- 1. 중복 실행 방지 (AppLock 체크)

IF APPLOCK_TEST('PUBLIC', 'SJPARKTEST', 'EXCLUSIVE', 'SESSION') = 0

RETURN

 

-- 2. 리소스 잠금 획득

EXEC SP_GETAPPLOCK 'SJPARKTEST', 'EXCLUSIVE', 'SESSION'

 

BEGIN TRY

-- \[트랜잭션 시작\]

BEGIN TRAN

 

-- 정상 데이터 입력 (1번째)

INSERT INTO sjpark_test (ItemSeq)

SELECT 5

 

-- 에러 발생 유도 (2번째: INT 컬럼에 문자열 입력)

INSERT INTO sjpark_test (ItemSeq)

SELECT 'ㄱ'

 

-- 모든 로직 성공 시 커밋

COMMIT TRAN

SELECT 'SUCCESS' AS Result

END TRY

BEGIN CATCH

-- \[에러 발생 시 롤백\]

IF @@TRANCOUNT \> 0

ROLLBACK TRAN

 

-- 에러 내용 출력

SELECT ERROR_MESSAGE() AS ErrorMsg, ERROR_NUMBER() AS ErrorNo

END CATCH

 

-- 3. 리소스 잠금 해제 (TRY/CATCH 외부에서 반드시 수행)

EXEC SP_RELEASEAPPLOCK 'SJPARKTEST', 'SESSION'

RETURN

GO

 

-- 실행 테스트

EXEC SJPARKTEST

-- 결과 확인 (아무 데이터도 들어가지 않아야 정상)

SELECT \* FROM sjpark_test WITH(NOLOCK)
