# 자동전표 환경설정 복사 스크립트

CREATE PROC sys_SDASlipAutoEnvCopy

@FromSlipKindNo NVARCHAR(100),

@ToSlipKindNo NVARCHAR(100)

 

AS

 

DECLARE @FromSlipAutoEnvSeq INT,

@ToSlipAutoEnvSeq INT,

@FromSlipKind INT,

@ToSlipKind INT,

@FromPgmSeq INT,

@ToPgmSeq INT

 

SELECT @FromSlipKind = SlipKind,

@FromPgmSeq = PgmSeq

FROM \_TACSlipKind

WHERE SlipKindNo = @FromSlipKindNo

 

SELECT @ToSlipKind = SlipKind,

@ToPgmSeq = PgmSeq

FROM \_TACSlipKind

WHERE SlipKindNo = @ToSlipKindNo

SELECT @FromSlipAutoEnvSeq = @FromSlipKind,

@ToSlipAutoEnvSeq = @ToSlipKind

 

DELETE \_TACSlipAutoEnv WHERE SlipAutoEnvSeq = @ToSlipAutoEnvSeq

DELETE \_TACSlipPgmControls WHERE PgmSeq = @ToPgmSeq

DELETE \_TACSlipAutoEnvRow WHERE SlipAutoEnvSeq = @ToSlipAutoEnvSeq

DELETE \_TACSlipAutoEnvKey WHERE SlipAutoEnvSeq = @ToSlipAutoEnvSeq

 

INSERT \_TACSlipAutoEnv

SELECT CompanySeq, @ToSlipAutoEnvSeq, @ToSlipKindNo, ControlAccUnit, ControlAccUnitSheet, ControlSlipUnit, ControlSlipUnitSheet, ControlAccDate, ControlRemark1, ControlRemark2, ControlRemark3, LastUserSeq, LastDateTime

FROM \_TACSlipAutoEnv

WHERE SlipAutoEnvSeq = @FromSlipAutoEnvSeq

 

INSERT \_TACSlipPgmControls

SELECT CompanySeq, @ToPgmSeq, Serl, ControlName, ControlType, DataBlockName, DataFieldName, DataFieldCd, CellType, ControlCaption, WordSeq, LastUserSeq, LastDateTime, CodeHelpConst, CodeHelpParams

FROM \_TACSlipPgmControls

WHERE PgmSeq = @FromPgmSeq

 

INSERT \_TACSlipAutoEnvRow

SELECT CompanySeq, @ToSlipAutoEnvSeq, Serl, AccSeq, UMCostType, SMDrOrCr, RowSort, IsAnti, IsDftAcc, IsSumProc, IsUseCash, DefaultEvidSeq, DefaultTaxKindSeq, SMCostItemKind,

ControlAccSeq, ControlAccName, ControlAmt, ControlForAmt, ControlCurrSeq, ControlCurrName, ControlExRate, ControlDivExRate, ControlEvidSeq, ControlEvidName,

ControlUMCostType, ControlUMCostTypeName, ControlCostItemSeq, ControlCostItemName, ControlCashDate, ControlCashMethod, ControlCashMethodName,

ControlSummary1, ControlSummary2, ControlSummary3, ControlOnSlipSeq, ControlOnSlipID, ControlCostDeptSeq, ControlCostDeptName, ControlBgtDeptSeq, ControlBgtDeptName,

ControlCostCCtrSeq, ControlCostCCtrName, ControlBgtCCtrSeq, ControlBgtCCtrName, ControlMtCostDeptSeq, ControlMtCostDeptName, ControlMtCostCCtrSeq, ControlMtCostCCtrName,

ControlMtAmt, ControlSumCol1, ControlSumCol2, ControlSumCol3, ControlSumCol4, ControlSumCol5, ControlSelRow, ControlSourceSeq, ControlSourceID,

DefaultSMCashMethod, ControlCoCustSeq, ControlCoCustName, ProcTypeName, ControlSMProcType, ControlSMProcTypeName, ControlControlSeq, ControlControlName,

ControlCodeHelpSeq, ControlCodeHelpName, ControlProcAccType, ControlProcAccTypeName, ControlEtcCol1, ControlEtcCol2, ControlEtcCol3,

ControlEtcCol4, ControlEtcCol5, ControlCashItemSeq, LastDateTime, LastUserSeq, RegDateTime, RegUserSeq, ControlCostItemSeq2, SMCostItemKind2

FROM \_TACSlipAutoEnvRow

WHERE SlipAutoEnvSeq = @FromSlipAutoEnvSeq

 

INSERT \_TACSlipAutoEnvKey

SELECT CompanySeq, @ToSlipAutoEnvSeq, ControlSeq, TableName, ColumnName, IsUsedQry, IsUsedInput, LastUserSeq, LastDateTime

FROM \_TACSlipAutoEnvKey

WHERE SlipAutoEnvSeq = @FromSlipAutoEnvSeq

 

RETURN

DECLARE @FromSlipKindNo NVARCHAR(100),

@ToSlipKindNo NVARCHAR(100)

 

SELECT @FromSlipKindNo = 'FrmWSLInvoice',

@ToSlipKindNo = 'FrmWSLInvoice_svs'

/\*=================================================================--

@FromSlipKindNo 데이터가 @ToSlipKindNo 로 복사 됨

FrmWSIAFebCardCfm 데이터가 FrmWSIAFebCardCfm_hct로 복사 됨

@ToSlipKindNo에 기존 저장 데이터는 삭제 후 복사하기때문에 조심!!

--=================================================================\*/

 

DECLARE @FromSlipAutoEnvSeq INT,

@ToSlipAutoEnvSeq INT,

@FromSlipKind INT,

@ToSlipKind INT,

@FromPgmSeq INT,

@ToPgmSeq INT,

@JourMethod INT

 

SELECT @FromSlipKind = SlipKind,

@FromPgmSeq = PgmSeq

FROM \_TACSlipKind

WHERE SlipKindNo = @FromSlipKindNo

 

SELECT @ToSlipKind = SlipKind,

@ToPgmSeq = PgmSeq

FROM \_TACSlipKind

WHERE SlipKindNo = @ToSlipKindNo

SELECT @FromSlipAutoEnvSeq = @FromSlipKind,

@ToSlipAutoEnvSeq = @ToSlipKind

 

-- 테이블 방식이 아닌 경우

IF @JourMethod \<\> 4030003

BEGIN

-- 여기서 기존 데이터 삭제

DELETE \_TACSlipAutoEnv WHERE SlipAutoEnvSeq = @ToSlipAutoEnvSeq

DELETE \_TACSlipPgmControls WHERE PgmSeq = @ToPgmSeq

DELETE \_TACSlipAutoEnvRow WHERE SlipAutoEnvSeq = @ToSlipAutoEnvSeq

DELETE \_TACSlipAutoEnvKey WHERE SlipAutoEnvSeq = @ToSlipAutoEnvSeq

DELETE \_TACSlipAutoEnvRowCol WHERE SlipAutoEnvSeq = @ToSlipAutoEnvSeq

 

 

INSERT \_TACSlipAutoEnv

SELECT CompanySeq, @ToSlipAutoEnvSeq, @ToSlipKindNo, ControlAccUnit, ControlAccUnitSheet, ControlSlipUnit, ControlSlipUnitSheet, ControlAccDate, ControlRemark1, ControlRemark2, ControlRemark3, LastUserSeq, LastDateTime

FROM \_TACSlipAutoEnv

WHERE SlipAutoEnvSeq = @FromSlipAutoEnvSeq

 

INSERT \_TACSlipPgmControls

SELECT CompanySeq, @ToPgmSeq, Serl, ControlName, ControlType, DataBlockName, DataFieldName, DataFieldCd, CellType, ControlCaption, WordSeq, LastUserSeq, LastDateTime, CodeHelpConst, CodeHelpParams

FROM \_TACSlipPgmControls

WHERE PgmSeq = @FromPgmSeq

 

INSERT \_TACSlipAutoEnvRow

SELECT CompanySeq, @ToSlipAutoEnvSeq, Serl, AccSeq, UMCostType, SMDrOrCr, RowSort, IsAnti, IsDftAcc, IsSumProc, IsUseCash, DefaultEvidSeq, DefaultTaxKindSeq, SMCostItemKind,

ControlAccSeq, ControlAccName, ControlAmt, ControlForAmt, ControlCurrSeq, ControlCurrName, ControlExRate, ControlDivExRate, ControlEvidSeq, ControlEvidName,

ControlUMCostType, ControlUMCostTypeName, ControlCostItemSeq, ControlCostItemName, ControlCashDate, ControlCashMethod, ControlCashMethodName,

ControlSummary1, ControlSummary2, ControlSummary3, ControlOnSlipSeq, ControlOnSlipID, ControlCostDeptSeq, ControlCostDeptName, ControlBgtDeptSeq, ControlBgtDeptName,

ControlCostCCtrSeq, ControlCostCCtrName, ControlBgtCCtrSeq, ControlBgtCCtrName, ControlMtCostDeptSeq, ControlMtCostDeptName, ControlMtCostCCtrSeq, ControlMtCostCCtrName,

ControlMtAmt, ControlSumCol1, ControlSumCol2, ControlSumCol3, ControlSumCol4, ControlSumCol5, ControlSelRow, ControlSourceSeq, ControlSourceID,

DefaultSMCashMethod, ControlCoCustSeq, ControlCoCustName, ProcTypeName, ControlSMProcType, ControlSMProcTypeName, ControlControlSeq, ControlControlName,

ControlCodeHelpSeq, ControlCodeHelpName, ControlProcAccType, ControlProcAccTypeName, ControlEtcCol1, ControlEtcCol2, ControlEtcCol3,

ControlEtcCol4, ControlEtcCol5, ControlCashItemSeq, LastDateTime, LastUserSeq, RegDateTime, RegUserSeq, ControlCostItemSeq2, SMCostItemKind2

FROM \_TACSlipAutoEnvRow

WHERE SlipAutoEnvSeq = @FromSlipAutoEnvSeq

 

INSERT \_TACSlipAutoEnvKey

SELECT CompanySeq, @ToSlipAutoEnvSeq, ControlSeq, TableName, ColumnName, IsUsedQry, IsUsedInput, LastUserSeq, LastDateTime

FROM \_TACSlipAutoEnvKey

WHERE SlipAutoEnvSeq = @FromSlipAutoEnvSeq

 

 

INSERT INTO \_TACSlipAutoEnvRowCol

SELECT CompanySeq, @ToSlipAutoEnvSeq, Serl, ColKey, EnvControlSeq

FROM \_TACSlipAutoEnvRowCol

WHERE SlipAutoEnvSeq = @FromSlipAutoEnvSeq

 

 

END

ELSE

BEGIN

-- 여기서 기존 데이터 삭제

DELETE \_TACSlipAutoInsertTbl WHERE SlipKindNo = @ToSlipKindNo

DELETE \_TACSlipAutoInsertTblCol WHERE SlipKindNo = @ToSlipKindNo

DELETE \_TACSlipAutoEnvTable WHERE SlipKindNo = @ToSlipKindNo

DELETE \_TACSlipRowAutoEnvTable WHERE SlipKindNo = @ToSlipKindNo

 

INSERT INTO \_TACSlipAutoInsertTbl(CompanySeq,SlipKindNo,TableName,TableAlias,LastUserSeq,LastDateTime)

SELECT CompanySeq,@ToSlipKindNo,TableName,TableAlias,LastUserSeq,GETDATE()

FROM \_TACSlipAutoInsertTbl

WHERE SlipKindNo = @FromSlipKindNo

 

 

INSERT INTO \_TACSlipAutoInsertTblCol(CompanySeq,SlipKindNo,CodeSeq,TableName,CodeName,ColName,SMKeyColumn,IsEnvColumn,JoinTable,JoinCode,JoinName,MajorSeq,CodeSort,Remark,LastUserSeq,LastDateTime,CodeHelpConst,CodeHelpParams)

SELECT CompanySeq,@ToSlipKindNo,CodeSeq,TableName,CodeName,ColName,SMKeyColumn,IsEnvColumn,JoinTable,JoinCode,JoinName,MajorSeq,CodeSort,Remark,LastUserSeq,GETDATE(),CodeHelpConst,CodeHelpParams

FROM \_TACSlipAutoInsertTblCol

WHERE SlipKindNo = @FromSlipKindNo

 

INSERT INTO \_TACSlipAutoEnvTable(CompanySeq,SlipKindNo,AccUnit,SlipUnit,AccDate,RegEmpSeq,RegDeptSeq,Remark,LastUserSeq,LastDateTime)

SELECT CompanySeq,@ToSlipKindNo,AccUnit,SlipUnit,AccDate,RegEmpSeq,RegDeptSeq,Remark,LastUserSeq,GETDATE()

FROM \_TACSlipAutoEnvTable

WHERE SlipKindNo = @FromSlipKindNo

 

INSERT INTO \_TACSlipRowAutoEnvTable(

CompanySeq,SlipKindNo,Serl,RowSort,AccSeq,AccSeqCol,UMCostType,UMCostTypeCol,EvidSeq,SMDrOrCr,IsEnvKey,IsSumProc,SumGroup1,SumGroup2,SumGroup3

,DrAmt,CrAmt,EvidSeqCol,Summary1,Summary2,Summary3,IsCash,SMCashMethod,CashDate,SMInOrOut,OnSlipID,CurrSeq,ExRate,DivExRate,DrForAmt,CrForAmt,BgtDeptSeq

,BgtCCtrSeq,CashItemSeq,SMCostItemKind,CostItemSeq,CoCustSeq,CostDeptSeq,CostCCtrSeq,RemSeq1,RemVal1,RemSeq2,RemVal2,RemSeq3,RemVal3,RemSeq4,RemVal4

,RemSeq5,RemVal5,RemSeq6,RemVal6,RemSeq7,RemVal7,RemSeq8,RemVal8,RemSeq9,RemVal9,RemSeq10,RemVal10,LastUserSeq,LastDateTime,SMProcType,ControlSeq,ProcAccType,StdTableName,ProcTypeName

,Sort1,IsAnti,SMSubCostItemKind,SubCostItemSeq

)

SELECT

CompanySeq,@ToSlipKindNo,Serl,RowSort,AccSeq,AccSeqCol,UMCostType,UMCostTypeCol,EvidSeq,SMDrOrCr,IsEnvKey,IsSumProc,SumGroup1,SumGroup2,SumGroup3

,DrAmt,CrAmt,EvidSeqCol,Summary1,Summary2,Summary3,IsCash,SMCashMethod,CashDate,SMInOrOut,OnSlipID,CurrSeq,ExRate,DivExRate,DrForAmt,CrForAmt,BgtDeptSeq

,BgtCCtrSeq,CashItemSeq,SMCostItemKind,CostItemSeq,CoCustSeq,CostDeptSeq,CostCCtrSeq,RemSeq1,RemVal1,RemSeq2,RemVal2,RemSeq3,RemVal3,RemSeq4,RemVal4

,RemSeq5,RemVal5,RemSeq6,RemVal6,RemSeq7,RemVal7,RemSeq8,RemVal8,RemSeq9,RemVal9,RemSeq10,RemVal10,LastUserSeq,GETDATE(),SMProcType,ControlSeq,ProcAccType,StdTableName,ProcTypeName

,Sort1,IsAnti,SMSubCostItemKind,SubCostItemSeq

FROM \_TACSlipRowAutoEnvTable

WHERE SlipKindNo = @FromSlipKindNo

 

END
