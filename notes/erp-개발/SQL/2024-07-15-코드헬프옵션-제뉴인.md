# 코드헬프옵션 - 제뉴인

CREATE PROCEDURE cymc_SCACodeHelpErrorCode

@WorkingTag NVARCHAR(1),

@LanguageSeq INT,

@CodeHelpSeq INT,

@DefQueryOption INT, -- 2: direct search

@CodeHelpType TINYINT,

@PageCount INT = 20,

@CompanySeq INT = 1,

@Keyword NVARCHAR(50) = '',

@Param1 NVARCHAR(50) = '', --

@Param2 NVARCHAR(50) = '', --

@Param3 NVARCHAR(50) = '',

@Param4 NVARCHAR(50) = '',

> @SubConditionSql        INT = 0

AS

SET ROWCOUNT @PageCount

DECLARE @ErrorCode NVARCHAR(100),

@ErrorCodeSeq INT,

@ModuleSeq INT

SELECT @Param1 = ISNULL(LTRIM(RTRIM(@Param1)), ''),

@Param2 = ISNULL(LTRIM(RTRIM(@Param2)), ''),

@Param3 = ISNULL(LTRIM(RTRIM(@Param3)), ''),

@Param4 = ISNULL(LTRIM(RTRIM(@Param4)), '')

IF @DefQueryOption = 0 SELECT @ErrorCodeSeq = CONVERT(INT, REPLACE(@Keyword, '%', ''))

IF @DefQueryOption = 1 SELECT @ErrorCode = ISNULL(@Keyword, '')

SELECT @ModuleSeq = CONVERT(INT, @Param1)

SELECT A.ErrorCodeSeq AS ErrorCode, A.ErrorCode AS ErrorName, A.Condition

FROM CYMC_TPjtCSErrorCode AS A

WHERE A.CompanySeq = @CompanySeq

AND ((@DefQueryOption = 0 AND (@ErrorCodeSeq = 0 OR A.ErrorCodeSeq = @ErrorCodeSeq)) -- 에러내부코드

OR (@DefQueryOption = 1 AND A.ErrorCode LIKE @ErrorCode + '%')) -- 에러코드

AND (@ModuleSeq = 0 OR A.ModuleSeq = @ModuleSeq)

ORDER BY A.ErrorCodeSeq

SET ROWCOUNT 0

RETURN
