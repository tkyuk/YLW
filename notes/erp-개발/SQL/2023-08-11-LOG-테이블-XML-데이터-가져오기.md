# LOG 테이블 XML 데이터 가져오기

CREATE TABLE \#TempTPDBOMLog(

LogType NVARCHAR(100),

LogSeq NVARCHAR(100),

LogUserSeq NVARCHAR(100),

LogPgmSeq NVARCHAR(100),

LogDateTime NVARCHAR(100)

,CompanySeq INT

,ItemSeq INT

,ItemBomRev NCHAR(4)

,LastUserSeq INT

,LastDateTime NVARCHAR(100)

,Serl INT

,BizUnit INT

)

 

 

DECLARE @XmlData NVARCHAR(MAX),

@DocHandle INT

SELECT @XmlData = ''

SELECT @XmlData = @XmlData +

REPLACE(CONVERT(NVARCHAR(MAX), A.xmlData),

'\<A',

'\<A LogType="' + A.LogType+ '" ' +

'LogSeq="' + CONVERT(NVARCHAR(100), ISNULL(A.LogSeq, 0)) + '" ' +

'LogUserSeq="' + CONVERT(NVARCHAR(100), ISNULL(A.LogUserSeq, 0)) + '" ' +

'LogPgmSeq="' + CONVERT(NVARCHAR(100), ISNULL(A.LogPgmSeq , 0)) + '" ' +

'LogDateTime="' + CONVERT(NVARCHAR(100), A.LogDateTime, 120) + '" ')

FROM \_TCOMTableLog AS A WITH(NOLOCK)

WHERE A.TableName = '\_TPDBOM'

 

 

SELECT @XmlData = '\<ROOT\>' + @XmlData + '\</ROOT\>'

EXEC sp_xml_preparedocument @DocHandle OUTPUT, @XmlData

INSERT INTO \#TempTPDBOMLog

SELECT LogType ,

LogSeq ,

LogUserSeq ,

LogPgmSeq ,

LogDateTime ,

CompanySeq ,

ItemSeq ,

ItemBomRev ,

LastUserSeq ,

LastDateTime,

Serl,

BizUnit

FROM OPENXML (@DocHandle, '/ROOT/A', 1)

WITH (LogType NVARCHAR(100),

LogSeq NVARCHAR(100),

LogUserSeq NVARCHAR(100),

LogPgmSeq NVARCHAR(100),

LogDateTime NVARCHAR(100)

,CompanySeq INT

,ItemSeq INT

,ItemBomRev NCHAR(4)

,LastUserSeq INT

,LastDateTime NVARCHAR(100)

,Serl INT

,BizUnit INT

)
