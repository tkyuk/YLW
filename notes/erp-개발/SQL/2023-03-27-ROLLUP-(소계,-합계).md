# ROLLUP (소계, 합계)

DROP TABLE IF EXISTS \#TMP

DROP TABLE IF EXISTS \#Result

 

CREATE TABLE \#TMP

(

LoanBankName NVARCHAR(100),

BankSeq INT,

LoanName NVARCHAR(100),

Amt DECIMAL(19,5)

)

 

INSERT INTO \#TMP

VALUES

('기업오류동' ,1,'11111' ,276376700)

,('기업은행-TEST' ,2,'차입테스트1' ,10000 )

,('기업은행-TEST' ,2,'차입TEST' ,1 )

,('기업은행-TEST' ,2,'차입TEST' ,1 )

,('기업은행-TEST' ,2,'테스트대출' , 5000000 )

,('산업은행(외화)',3,'외화차입테스트1',100000000)

 

SELECT LoanBankName, BankSeq, LoanName, SUM(Amt)

FROM \#TMP

GROUP BY ROLLUP (BankSeq, LoanBankName, LoanName)

 

SELECT CASE WHEN LoanBankName IS NULL

THEN '합계' ELSE CASE WHEN LoanName IS NULL

THEN '소계' ELSE LoanBankName END END AS LoanBankName,

CASE WHEN BankSeq IS NULL THEN '9999' ELSE CAST(BankSeq AS NVARCHAR(10)) END AS BankSeq,

CASE WHEN LoanName IS NULL THEN '소계' ELSE LoanName END AS LoanName,

SUM(Amt) AS Amt

FROM \#TMP

GROUP BY ROLLUP (BankSeq, LoanBankName, LoanName)

ORDER BY BankSeq, CASE WHEN LoanName IS NULL THEN 1 ELSE 0 END, LoanName

컬럼에 NULL 값이 포함될 경우 GROUPING 사용

컬럼에 NULL 값이 포함 안될 경우 그냥 ISNULL(컬럼,'합계') 로 하면 좋다

 

SELECT CASE WHEN GROUPING(job) = 1 AND GROUPING(deptno) = 1 THEN '합계'

ELSE job END,

CASE WHEN GROUPING(job) = 0 AND GROUPING(deptno) = 1 THEN '소계'

ELSE ISNULL(CONVERT(NVARCHAR(10),deptno),'') END,

SUM(total_sal),

GROUPING(job),

GROUPING(deptno)

FROM \#EMP

GROUP BY job, deptno

WITH ROLLUP

 

DROP TABLE \#TMP

 

CREATE TABLE \#TMP

(

COL1 NCHAR(2),

COL2 INT,

COL3 DECIMAL(19,5)

)

 

INSERT INTO \#TMP VALUES

('A',1,500),

('A',2,600),

('B',1,300),

('B',2,200)

 

SELECT COL1, COL2, SUM(COL3)

FROM \#TMP

GROUP BY COL1, COL2

ORDER BY 1,2

 

SELECT CASE WHEN COL1 IS NULL THEN '합계' ELSE

CASE WHEN ISNULL(COL2,9999) = 9999 THEN '소계' ELSE ISNULL(COL1,'합계') END END,

ISNULL(COL2,9999),

SUM(COL3)

FROM \#TMP

GROUP BY ROLLUP(COL1, COL2)

 

SELECT COL1, COL2, SUM(COL3), GROUPING_ID(COL1,COL2)

FROM \#TMP

GROUP BY ROLLUP(COL1, COL2)
