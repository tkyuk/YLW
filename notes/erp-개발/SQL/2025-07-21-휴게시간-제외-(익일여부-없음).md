# 휴게시간 제외 (익일여부 없음)

CREATE TABLE \#Tmp_BreakTime(

BegTime NCHAR(4)

,EndTime NCHAR(4)

)

 

CREATE TABLE \#Result_WkTime(

IDX_NO INT

,WkTime DECIMAL(19,5)

)

 

 

INSERT INTO \#Tmp_BreakTime(BegTime,EndTime)

SELECT B.BegTime, B.EndTime

FROM \#BIZ_OUT_DataBlock1 AS Z

JOIN \_TDAEmpIn AS A WITH(NOLOCK) ON A.CompanySeq = @CompanySeq

AND A.EmpSeq = Z.TripEmpSeq

LEFT OUTER JOIN \_TXPRWkTypeEmp AS D WITH(NOLOCK) ON D.CompanySeq = A.CompanySeq

AND D.EmpSeq = A.EmpSeq

AND D.EndDate = '99991231'

INSERT INTO \#Result_WkTime(IDX_NO, WkTime)

SELECT A.IDX_NO

-- WkTime에서 겹치는 분(OverlapMin)을 시간으로 환산 후 차감

,CAST(A.WkTime AS FLOAT) - ISNULL(SUM(CASE WHEN WorkEnd \<= BreakStart OR WorkStart \>= BreakEnd THEN 0

ELSE (CASE WHEN WorkEnd \< BreakEnd THEN WorkEnd ELSE BreakEnd END)

\- (CASE WHEN WorkStart \> BreakStart THEN WorkStart ELSE BreakStart END)

END), 0) / 60.0 AS AdjustedWkTime

FROM (SELECT \*,

CAST(LEFT(TimeFrom, 2) AS INT) \* 60 + CAST(RIGHT(TimeFrom, 2) AS INT) AS WorkStart,

CAST(LEFT(TimeTO, 2) AS INT) \* 60 + CAST(RIGHT(TimeTO, 2) AS INT) AS WorkEnd

FROM \#BIZ_OUT_DataBlock2

) A

LEFT OUTER JOIN(SELECT \*,

CAST(LEFT(BegTime, 2) AS INT) \* 60 + CAST(RIGHT(BegTime, 2) AS INT) AS BreakStart,

CAST(LEFT(EndTime, 2) AS INT) \* 60 + CAST(RIGHT(EndTime, 2) AS INT) AS BreakEnd

FROM \#Tmp_BreakTime ) B ON NOT (

(CAST(LEFT(A.TimeTO, 2) AS INT) \* 60 + CAST(RIGHT(A.TimeTO, 2) AS INT))

\<= (CAST(LEFT(B.BegTime, 2) AS INT) \* 60 + CAST(RIGHT(B.BegTime, 2) AS INT))

OR (CAST(LEFT(A.TimeFrom, 2) AS INT) \* 60 + CAST(RIGHT(A.TimeFrom, 2) AS INT))

\>= (CAST(LEFT(B.EndTime, 2) AS INT) \* 60 + CAST(RIGHT(B.EndTime, 2) AS INT))

)

GROUP BY A.IDX_NO,A.TimeFrom, A.TimeTO, A.WkTime

 

UPDATE A

SET WkTime = B.WkTime

FROM \#BIZ_OUT_DataBlock2 AS A

JOIN \#Result_WkTime AS B ON B.IDX_NO = A.IDX_NO

WHERE A.Status = 0

AND A.WorkingTag IN('A','U')
