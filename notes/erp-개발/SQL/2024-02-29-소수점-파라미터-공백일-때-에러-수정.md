# 소수점 파라미터 공백일 때 에러 수정

alter PROCEDURE test240229

@YourNumericParam VARCHAR(100) -- 먼저 문자열로 받음

AS

BEGIN

-- 문자열이 비어 있는지 확인하고, 비어 있다면 NULL로 변환

IF @YourNumericParam = ''

BEGIN

SET @YourNumericParam = NULL;

END

-- 이후 NUMERIC 유형으로 변환

DECLARE @NumericValue NUMERIC(18, 2);

SET @NumericValue = NULLIF(@YourNumericParam, '');

-- 나머지 로직

END

 

go

exec test240229 ''
