# 25.06.20 연박일수 계산

IF OBJECT_ID ('tempdb..#Tmp_Date') IS NOT NULL DROP TABLE \#Tmp_Date

CREATE TABLE \#Tmp_Date (

RegDate NCHAR(8),

IsStayed NCHAR(1) -- 1이면 숙박함, 0이면 안 함

)

INSERT INTO \#Tmp_Date (RegDate, IsStayed)

VALUES

('20250512', 1),

('20250513', 1),

('20250514', 0),

('20250515', 1),

('20250516', 1),

('20250517', 1),

('20250518', 0),

('20250519', 1),

('20250520', 1)

SELECT A.RegDate, ISNULL(B.MaxConsecutiveStays,0)

FROM \#Tmp_Date AS A

LEFT OUTER JOIN (SELECT S1.RegDate,

CASE WHEN S1.IsStayed = 0 THEN 0

ELSE (

SELECT COUNT(\*)

FROM \#Tmp_Date S2

WHERE S2.RegDate \<= S1.RegDate

AND S2.IsStayed = 1

AND NOT EXISTS (

SELECT 1

FROM \#Tmp_Date B

WHERE B.RegDate \> S2.RegDate

AND B.RegDate \<= S1.RegDate

AND B.IsStayed = 0

)

)

END AS MaxConsecutiveStays

FROM \#Tmp_Date S1

) AS B ON DATEADD(DAY,1,B.RegDate) = A.RegDate

ORDER BY A.RegDate
