# 중단 SP 추가

CREATE TABLE \#TCOMStop

(

IDX_NO INT

, StopSeq INT

, StopSerl INT

, StopSubSerl INT

, IsStop NCHAR(1)

, UMStopReason INT

, StopReason NVARCHAR(500)

, Result NVARCHAR(255)

, MessageType INT

, Status INT

, DataSeq INT

)

 

-- SELECT SourceSeq, SourceSerl,InOutType,SourceType FROM \#BIZ_OUT_DATABLOCK2

 

INSERT INTO \#TCOMStop(IDX_NO, StopSeq, StopSerl, StopSubSerl, IsStop, UMStopReason, StopReason, Result, MessageType, Status, DataSeq)

SELECT IDX_NO, SourceSeq, SourceSerl, 0, '1', 0, '', NULL, NULL, 0, DataSeq

FROM \#BIZ_OUT_DATABLOCK2 AS A

WHERE A.Status = 0

AND A.SourceType = '0'

AND A.InOutType = 80

 

EXEC \_SWCOMStop

@CompanySeq = @CompanySeq

, @LanguageSeq = @LanguageSeq

, @UserSeq = @UserSeq
