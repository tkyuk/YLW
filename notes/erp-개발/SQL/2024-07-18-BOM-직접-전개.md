# BOM 직접 전개

CREATE PROCEDURE ssv_SWSLOrderOptionCfmOptQuery

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

AS

DECLARE @OrderSeq INT,

@OrderSerl INT,

@ProdReqSeq INT,

@ProdReqSerl INT,

@IsOptionCfm NCHAR(1),

@PUYN NCHAR(1),

@UMOrderType INT,

@ProcessYN NCHAR(1)

 

SELECT @OrderSeq = ISNULL(OrderSeq , 0),

@OrderSerl = ISNULL(OrderSerl , 0),

@ProdReqSeq = ISNULL(ProdReqSeq , 0),

@ProdReqSerl = ISNULL(ProdReqSerl, 0),

@PUYN = ISNULL(PUYN ,'0'),

@UMOrderType = ISNULL(UMOrderType, 0)

FROM \#BIZ_IN_DataBlock2

 

-- 확정여부

SELECT @IsOptionCfm = ISNULL(IsOptionCfm, '0')

FROM ssv_TPDMPSProdReqItem WITH(NOLOCK)

WHERE CompanySeq = @CompanySeq

AND ProdReqSeq = @ProdReqSeq

AND Serl = @ProdReqSerl

 

-- 가공품여부

SELECT @ProcessYN = IIF(UMOrderType = 1000846003, '1', '0')

FROM ssv_TSLOrderItem WITH(NOLOCK)

WHERE CompanySeq = @CompanySeq

AND OrderSeq = @OrderSeq

AND OrderSerl = @OrderSerl

IF @IsOptionCfm \<\> '1' AND @PUYN \<\> '1'

BEGIN

SELECT @PUYN = ISNULL(PUYN, '0')

FROM ssv_TSLOrderItemOption WITH(NOLOCK)

WHERE CompanySeq = @CompanySeq

AND ProdReqSeq = @ProdReqSeq

AND Serl = @ProdReqSerl

END

 

-- OEM제품 담기

SELECT A.MinorSeq AS UMOrderType

INTO \#OEM

FROM \_TDAUMinor AS A WITH(NOLOCK)

JOIN \_TDAUMinorValue AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.MinorSeq = A.MinorSeq

AND B.Serl = 15000017 -- OEM유무

WHERE A.CompanySeq = @CompanySeq

AND A.MajorSeq = 1000846 -- ssv_수주품목구분

AND ISNULL(B.ValueText, '') = '1'

 

-- 주강품인 품목분류 구하기

SELECT B.MinorSeq AS UMItemType

INTO \#UMItemType

FROM \_TDAUMajor AS A

JOIN \_TDAUMinorValue AS B ON B.CompanySeq = A.CompanySeq

AND B.MajorSeq = A.MajorSeq

AND B.Serl = 15000005 -- ssv_분류

WHERE A.CompanySeq = @CompanySeq

AND A.MajorSeq = 150123 -- ssv_품목분류

AND B.ValueSeq = 1000805001 -- 주강품

(

ItemSeq INT , -- 품목내부코드

ItemBOMRev NCHAR(2) , -- 품목BOM차수

UnitSeq INT , -- 단위내부코드

BOMLevelText NVARCHAR(50) , -- BOM Level

Location NVARCHAR(2000) , -- Location

Remark NVARCHAR(500) , -- 비고

Serl INT , -- 순번

NeedQtyNumerator DECIMAL(19,5) , -- 소요량(분자)

NeedQtyDenominator DECIMAL(19,5) , -- 소요량(분모)

NeedQty DECIMAL(19,10) , -- 소요량

Seq INT IDENTITY(1,1) , -- Seq

ParentSeq INT , -- 모품목Seq

Sort INT , -- Tree Sort

BOMLevel INT , -- Level

BOMLevelTree NVARCHAR(100) , -- Tree Level

ParentItemSeq INT NULL, -- 모품목코드

ParentBomRev NCHAR(2) NULL, -- 모품목차수

InLossRate DECIMAL(19,5) NULL, -- 사내로스율 --2015.06.01 추가 박수영

OutLossRate DECIMAL(19,5) NULL, -- 사외로스율 --2015.06.01 추가 박수영

IsBOMOnly NCHAR(1) NULL, -- BOM조회용 --2015.06.13 추가 박수영 IsIncludeProcMat NCHAR(1) NULL, -- 하위공정소요자재포함여부 --2015.06.13 추가 박수영

)

 

CREATE TABLE \#BOMSpread_Last -- BOM전개

(

OrderSeq INT ,

OrderSerl INT ,

ItemSeq INT , -- 품목내부코드

ItemBOMRev NCHAR(2) , -- 품목BOM차수

UnitSeq INT , -- 단위내부코드

BOMLevelText NVARCHAR(50) , -- BOM Level

Location NVARCHAR(2000) , -- Location

Remark NVARCHAR(500) , -- 비고

Serl INT , -- 순번

NeedQtyNumerator DECIMAL(19,5) , -- 소요량(분자)

NeedQtyDenominator DECIMAL(19,5) , -- 소요량(분모)

NeedQty DECIMAL(19,10) , -- 소요량

Seq INT , -- Seq

ParentSeq INT , -- 모품목Seq

Sort INT , -- Tree Sort

BOMLevel INT , -- Level

BOMLevelTree NVARCHAR(100) , -- Tree Level

ParentItemSeq INT NULL, -- 모품목코드

ParentBomRev NCHAR(2) NULL, -- 모품목차수

InLossRate DECIMAL(19,5) NULL, -- 사내로스율 --2015.06.01 추가 박수영

OutLossRate DECIMAL(19,5) NULL, -- 사외로스율 --2015.06.01 추가 박수영

IsBOMOnly NCHAR(1) NULL, -- BOM조회용 --2015.06.13 추가 박수영

IsIncludeProcMat NCHAR(1) NULL, -- 하위공정소요자재포함여부 --2015.06.13 추가 박수영

UMOption INT NULL,

UMItemType INT NULL,

DWGNo NVARCHAR(100) NULL,

WorkCenterSeq INT ,

ProdReqSeq INT ,

ProdReqSerl INT ,

Size NVARCHAR(100)

)

 

DECLARE @BizUnit INT ,

@ItemSeq INT ,

@ItemBomRev NCHAR(2),

@SemiType NCHAR(1),

@IsReverse NCHAR(1),

@QryLevel INT

SELECT @BizUnit = A.BizUnit ,

@ItemSeq = B.STDItemSeq ,

@ItemBomRev = MAX(C.ItemBOMRev),

@SemiType = '0' ,

@IsReverse = '0' ,

@QryLevel = 0

FROM \_TSLOrder AS A WITH(NOLOCK)

JOIN ssv_TSLOrderItem AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.OrderSeq = A.OrderSeq

LEFT OUTER JOIN \_TPDBOMManagement AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.ItemSeq = B.STDItemSeq

WHERE A.CompanySeq = @CompanySeq

AND A.OrderSeq = @OrderSeq

AND B.OrderSerl = @OrderSerl

AND B.UMOrderType NOT IN (SELECT UMOrderType FROM \#OEM)

GROUP BY B.STDItemSeq, A.BizUnit

ORDER BY B.STDItemSeq

--====================================================================================================

-- BOM전개 직접 처리

--====================================================================================================

CREATE TABLE \#BOMSpreadItem

(

ItemSeq INT,

ItemBOMRev NCHAR(2),

UnitSeq INT,

Location NVARCHAR(2000),

Remark NVARCHAR(500),

Serl INT,

NeedQtyNumerator DECIMAL(19,5),

NeedQtyDenominator DECIMAL(19,5),

NeedQty DECIMAL(19,10),                -- 17.06.19 임희진 수정 (DECIMAL(19,5) -\> DECIMAL(19,10))

ParentSeq INT,

Seq INT IDENTITY(1,1),

ParentItemSeq INT NULL,

ParentBomRev NCHAR(2) NULL,

InLossRate DECIMAL(19,5) NULL,

OutLossRate DECIMAL(19,5) NULL,

IsBOMOnly NCHAR(1),

IsIncludeProcMat NCHAR(1)

)

CREATE TABLE \#BOMtextLevelSeq

(

SpreadItemSeq INT,

LevelSeq INT

)

CREATE TABLE \#ChildCount

(

ParentSeq INT,

LevelLen INT

)

CREATE TABLE \#LevelCount

(

BOMLevel INT,

LevelLen INT

)

 

DECLARE @Seq INT,

@TextLevel NVARCHAR(200),

@BOMLevel INT,

@SeqDiff INT,

@Sort INT,

@NeedQty DECIMAL(19,5),

@ParentSeq INT,

@Env5931 INT,

@Count INT

SELECT @Env5931 = EnvValue

FROM \_TCOMEnv WITH(NOLOCK)

WHERE CompanySeq = @CompanySeq

AND EnvSeq = 5931

 

IF @Env5931 IS NULL

SELECT @Env5931 = 100

 

IF @ItemBomRev = ''

SELECT @ItemBomRev = '00'

 

-- 최상위 품목으로부터 BOM전개

INSERT INTO \#BOMSpread

( ItemSeq , ItemBOMRev, UnitSeq , BOMLevelText , Location ,

Remark , Serl , NeedQtyNumerator , NeedQtyDenominator, NeedQty ,

ParentSeq , Sort , BOMLevel , ParentItemSeq , ParentBomRev,

InLossRate , OutLossRate, IsBOMOnly , IsIncludeProcMat

)

SELECT DISTINCT

@ItemSeq, @ItemBomRev, UnitSeq, '01', '',

'', 0, 1, 1, 1, 0, 1, 1

, @ItemSeq, @ItemBomRev

, 0, 0, '0', '0'

FROM \_TPDBOM WITH(NOLOCK)

WHERE ItemSeq = @ItemSeq

AND ItemBomRev = @ItemBomRev

AND CompanySeq = @CompanySeq

 

 

--2020.09.23 박수영 ::Spread 시 성능부하 에 따른 INDEX 추가

> --2021.05.13 방혁 ::인덱스 2개 추가

IF NOT EXISTS(SELECT 1 FROM tempdb.sys.indexes WHERE OBJECT_ID = OBJECT_ID(N'tempdb..#BOMSpread') AND name IS NOT NULL)

BEGIN

CREATE INDEX NCL_1\_#BOMSpread ON \#BOMSpread(ItemSeq, ItemBOMRev)

CREATE INDEX NCL_2\_#BOMSpread ON \#BOMSpread(ParentSeq)

CREATE INDEX NCL_3\_#BOMSpread ON \#BOMSpread(Seq)

END

-- \#BOMSpread 에 한품목이 여러행으로 들어오는 경우 BOM이 여러셋트로 조회되는 오류가 있어서

WHILE (EXISTS(SELECT 1 FROM \#BOMSpread GROUP BY ItemSeq, ItemBOMRev HAVING COUNT(1) \> 1))

BEGIN

SET ROWCOUNT 1

DELETE A

FROM \#BOMSpread AS A

WHERE EXISTS (SELECT 1 FROM \#BOMSpread

WHERE ItemSeq = A.ItemSeq

AND ItemBOMRev = A.ItemBOMRev

GROUP BY ItemSeq, ItemBOMRev

HAVING COUNT(1) \> 1

)

SET ROWCOUNT 0

END

 

-- 하위품목담기

SET @BOMLevel = 1

SET @Count = 1

WHILE (1 = 1)

BEGIN

INSERT INTO \#BOMSpreadItem

(

ItemSeq , ItemBOMRev , UnitSeq , Location , Remark ,

Serl , NeedQtyNumerator , NeedQtyDenominator, NeedQty , ParentSeq ,

ParentItemSeq , ParentBomRev,

InLossRate , OutLossRate , IsBOMOnly , IsIncludeProcMat

)

SELECT A.SubItemSeq

, A.SubItemBomRev

, A.SubUnitSeq

, A.Location

, A.Remark

, A.Serl

, A.NeedQtyNumerator

, A.NeedQtyDenominator

, B.NeedQty

, B.Seq

, A.ItemSeq

, A.ItemBomRev

, A.InLossRate

, A.OutLossRate

, A.IsBOMOnly

, A.IsIncludeProcMat

FROM \_TPDBOM AS A WITH(NOLOCK)

JOIN \#BOMSpread AS B ON A.ItemSeq = B.ItemSeq

AND A.ItemBomRev = B.ItemBomRev

JOIN \_TDAItem AS I WITH(NOLOCK) ON A.CompanySeq = I.CompanySeq

AND A.ItemSeq = I.ItemSeq

JOIN \_TDAItemAsset AS S WITH(NOLOCK) ON A.CompanySeq = S.CompanySeq

AND I.AssetSeq = S.AssetSeq

LEFT OUTER JOIN ssv_TPDBOM AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.ItemSeq = A.ItemSeq

AND C.ItemBomRev = A.ItemBomRev

AND C.Serl = A.Serl

LEFT OUTER JOIN \_TDAUMinor AS D WITH(NOLOCK) ON D.CompanySeq = C.CompanySeq

AND D.MinorSeq = C.UMItemType

WHERE A.CompanySeq = @CompanySeq

AND B.BOMLevel = @BOMLevel

AND (@BOMLevel = 1 OR S.SMAssetGrp \<\> (CASE WHEN @SemiType = '1' THEN 6008004 ELSE 0 END))

ORDER BY B.ParentSeq, D.MinorSort, A.UserSeq -- 순번 생성시 품목분류 순서 반영

 

IF @@ROWCOUNT = 0 BREAK

 

SET @ParentSeq = 0

WHILE (1 = 1)

BEGIN

SELECT TOP 1

@ParentSeq = ParentSeq

FROM \#BOMSpreadItem

WHERE ParentSeq \> @ParentSeq

ORDER BY ParentSeq

 

IF @@ROWCOUNT = 0 BREAK

 

INSERT INTO \#BOMtextLevelSeq

SELECT Seq, ROW_NUMBER() OVER(ORDER BY Seq)

FROM \#BOMSpreadItem

WHERE ParentSeq = @ParentSeq

ORDER BY Seq

END

 

-- 트리전개를 위한 값 (각 레벨별 순서값)

SELECT @Sort = MAX(Sort) FROM \#BOMSpread WHERE BOMLevel = @BOMLevel

SELECT @Sort = ISNULL(@Sort, 0)

 

-- 조회설정레벨까지만 조회

IF @QryLevel = 0

SELECT @QryLevel = @QryLevel

ELSE IF @BOMLevel = @QryLevel + 1

BREAK

 

SELECT @BOMLevel = @BOMLevel + 1

 

INSERT INTO \#BOMSpread

(

ItemSeq , ItemBOMRev, UnitSeq , BOMLevelText , Location ,

Remark , Serl , NeedQtyNumerator , NeedQtyDenominator, NeedQty ,

ParentSeq , Sort , BOMLevel , ParentItemSeq , ParentBomRev,

InLossRate , OutLossRate, IsBOMOnly , IsIncludeProcMat

)

SELECT ItemSeq , ItemBomRev, UnitSeq , B.LevelSeq , Location ,

Remark , Serl , NeedQtyNumerator , NeedQtyDenominator

, CASE WHEN NeedQtyDenominator \<\> 0 THEN (A.NeedQty \* NeedQtyNumerator / NeedQtyDenominator) ELSE 0 END

, A.ParentSeq

, @Sort + ROW_NUMBER() OVER(ORDER BY Seq)

, @BOMLevel

, A.ParentItemSeq

, A.ParentBomRev

, A.InLossRate

, A.OutLossRate

, A.IsBOMOnly

, A.IsIncludeProcMat

FROM \#BOMSpreadItem AS A

JOIN \#BOMtextLevelSeq AS B ON A.Seq = B.SpreadItemSeq

ORDER BY Seq

 

DELETE \#BOMSpreadItem

 

SELECT @Count = @Count + 1

 

IF @Count \> @Env5931

BREAK

END

 

-- BOM레벨명 구하기

-- BOMLevelText 표현에 필요한 자리수(문자길이)를 구한다.

-- 하위품목이100개이상이면 001,002, 10개이상이면 01, 02 와 같이 표현하기위해

INSERT INTO \#ChildCount

SELECT ParentSeq, CEILING(LOG10(COUNT(Seq) + 1))

FROM \#BOMSpread

GROUP BY ParentSeq

 

INSERT INTO \#LevelCount

SELECT B.BOMLevel, MAX(A.LevelLen)

FROM \#ChildCount AS A

JOIN \#BOMSpread AS B ON A.ParentSeq = B.ParentSeq

GROUP BY B.BOMLevel

 

SET @BOMLevel = 1

WHILE (1 = 1)

BEGIN

SELECT @BOMLevel = @BOMLevel + 1

 

UPDATE A

SET BOMLevelText = P.BOMLevelText + '-' + RIGHT('00000' + A.BOMLevelText, B.LevelLen)

FROM \#BOMSpread AS A

JOIN \#BOMSpread AS P ON A.ParentSeq = P.Seq

JOIN \#LevelCount AS B ON A.BOMLevel = B.BOMLevel

WHERE A.BOMLevel = @BOMLevel

 

IF @@ROWCOUNT = 0

BREAK

END

--====================================================================================================

-- BOM전개 직접 처리 끝

--====================================================================================================

 

INSERT \#BOMSpread_Last

(

OrderSeq ,

OrderSerl ,

ItemSeq ,

ItemBOMRev ,

UnitSeq ,

BOMLevelText ,

Location ,

Remark ,

Serl ,

NeedQtyNumerator ,

NeedQtyDenominator,

NeedQty ,

Seq ,

ParentSeq ,

Sort ,

BOMLevel ,

BOMLevelTree ,

ParentItemSeq ,

ParentBomRev ,

InLossRate ,

OutLossRate ,

IsBOMOnly ,

IsIncludeProcMat ,

UMOption ,

UMItemType ,

DWGNo ,

WorkCenterSeq ,

ProdReqSeq ,

ProdReqSerl ,

Size

)

SELECT @OrderSeq ,

@OrderSerl ,

A.ItemSeq ,

A.ItemBOMRev ,

A.UnitSeq ,

A.BOMLevelText ,

A.Location ,

A.Remark ,

A.Serl ,

A.NeedQtyNumerator ,

A.NeedQtyDenominator,

A.NeedQty ,

A.Seq ,

A.ParentSeq ,

A.Sort ,

A.BOMLevel ,

A.BOMLevelTree ,

A.ParentItemSeq ,

A.ParentBomRev ,

A.InLossRate ,

A.OutLossRate ,

A.IsBOMOnly ,

A.IsIncludeProcMat ,

D.UMOption ,

D.UMItemType ,

D.DWGNo ,

E.WorkCenterSeq ,

@ProdReqSeq ,

@ProdReqSerl ,

D.Size

FROM \#BOMSpread AS A

JOIN \_TDAItem AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.ItemSeq = A.ItemSeq

JOIN \_TDAItemAsset AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

AND C.AssetSeq = B.AssetSeq

LEFT OUTER JOIN ssv_TPDBOM AS D WITH(NOLOCK) ON D.CompanySeq = @CompanySeq

AND D.ItemSeq = A.ParentItemSeq

AND D.ItemBomRev = A.ParentBOMRev

AND D.Serl = A.Serl

LEFT OUTER JOIN ssv_TPDBOMManagement AS E WITH(NOLOCK) ON E.CompanySeq = @CompanySeq

AND E.ItemSeq = A.ItemSeq

AND E.ItemBomRev = A.ItemBomRev

WHERE ((@PUYN = '0' AND C.SMAssetGrp = 6008004) OR -- 반제품만

(@PUYN = '1' AND A.BOMLevel = 2 AND B.AssetSeq IN (SELECT AssetSeq FROM \_TDAItemAssetCodeHelp WHERE CompanySeq = @CompanySeq AND ItemCodeHelpSeq = 24))) -- 구매품만

AND ISNULL(D.UMItemType, 0) NOT IN (SELECT UMItemType FROM \#UMItemType) -- 주조품제외

 

DECLARE @ProcessItem INT

 

-- ssv_수주품목구분 - 가공품

SELECT @ProcessItem = 1000846003

 

-- 수주품목구분이 가공품일 경우 수주에 입력된 bom대분류와 같은 반제품과 하위 반제품만 조회

IF @ProcessItem = (SELECT UMOrderType FROM ssv_TSLOrderItem WITH(NOLOCK) WHERE CompanySeq = @CompanySeq AND OrderSeq = @OrderSeq AND OrderSerl = @OrderSerl)

BEGIN

DECLARE @UMItemType INT

 

SELECT @UMItemType = UMItemType

FROM ssv_TSLOrderItem

WHERE CompanySeq = @CompanySeq

AND OrderSeq = @OrderSeq

AND OrderSerl = @OrderSerl

 

-- 일치하지 않으면 삭제

DELETE A

FROM \#BOMSpread_Last AS A

WHERE ISNULL(A.UMItemType, 0) \<\> @UMItemType

AND ISNULL(A.BOMLevel , 0) = 2

 

-- 상위품목이 없는 경우 삭제

SELECT B.Seq

INTO \#BOMSpread_Last_DELETE

FROM \#BOMSpread_Last AS A

JOIN \#BOMSpread_Last AS B ON A.Seq = B.ParentSeq

 

DELETE A

FROM \#BOMSpread_Last AS A

WHERE A.BOMLevel \<\> 2

AND NOT EXISTS (SELECT TOP 1 1

FROM \#BOMSpread_Last_DELETE AS Z

WHERE A.Seq = Z.Seq)

END

 

-- 진행데이터 찾아오기

SELECT A.ProdReqSeq, A.ProdReqSerl, A.OrderSeq, A.OrderSerl, C.GoodItemSeq AS ItemSeq, C.WorkOrderSeq, C.WorkOrderSerl, C.WorkDate

INTO \#TPDSFCWorkOrder

FROM ssv_TSLOrderItemOption AS A WITH(NOLOCK)

JOIN ssv_TPDSFCWorkOrder        AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.ProdReqSeq = A.ProdReqSeq

AND B.ProdReqSerl = A.ProdReqSerl

JOIN \_TPDSFCWorkOrder        AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.WorkOrderSeq = B.WorkOrderSeq

AND C.WorkOrderSerl = B.WorkOrderSerl

WHERE A.CompanySeq = @CompanySeq

AND A.ItemSeq = C.GoodItemSeq

AND A.ProdReqSeq = @ProdReqSeq

AND A.ProdReqSerl = @ProdReqSerl

 

SELECT A.WorkOrderSeq, A.WorkOrderSerl, SUM(C.ProdQty) AS ProdQty

INTO \#TPDSFCWorkReport

FROM \#TPDSFCWorkOrder AS A

JOIN \_TCOMSourceDailyTrace AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.ToTableSeq = 6 -- 작업실적

AND B.FromTableSeq = 5 -- 작업지시

AND B.FromSeq = A.WorkOrderSeq

AND B.FromSerl = A.WorkOrderSerl

JOIN \_TPDSFCWorkReport AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.WorkReportSeq = B.ToSeq

GROUP BY A.WorkOrderSeq, A.WorkOrderSerl

 

SELECT A.WorkOrderSeq, A.WorkOrderSerl, SUM(C.Qty) AS POQty

INTO \#TPDOSPPOItem

FROM \#TPDSFCWorkOrder AS A

JOIN \_TCOMSourceDailyTrace AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.ToTableSeq = 4 -- 외주발주

AND B.FromTableSeq = 5 -- 작업지시

AND B.FromSeq = A.WorkOrderSeq

AND B.FromSerl = A.WorkOrderSerl

JOIN \_TPDOSPPOItem AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.OSPPOSeq = B.ToSeq

AND C.OSPPOSerl = B.ToSerl

GROUP BY A.WorkOrderSeq, A.WorkOrderSerl

 

SELECT A.WorkOrderSeq, A.WorkOrderSerl, SUM(C.Qty) AS DelvQty

INTO \#TPDOSPDelvItem

FROM \#TPDSFCWorkOrder AS A

JOIN \_TCOMSourceDailyTrace AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.ToTableSeq = 3 -- 외주납품

AND B.FromTableSeq = 5 -- 작업지시

AND B.FromSeq = A.WorkOrderSeq

AND B.FromSerl = A.WorkOrderSerl

JOIN \_TPDOSPDelvItem AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.OSPDelvSeq = B.ToSeq

AND C.OSPDelvSerl = B.ToSerl

GROUP BY A.WorkOrderSeq, A.WorkOrderSerl

 

SELECT A.ProdReqSeq, A.ProdReqSerl, A.OrderSeq, A.OrderSerl, A.ItemSeq, MAX(A.WorkDate) AS WorkDate, ISNULL(SUM(B.ProdQty), 0) AS ProdQty, ISNULL(SUM(C.POQty), 0) AS POQty, ISNULL(SUM(D.DelvQty), 0) AS DelvQty

INTO \#ProgData

FROM \#TPDSFCWorkOrder AS A

LEFT OUTER JOIN \#TPDSFCWorkReport AS B ON A.WorkOrderSeq = B.WorkOrderSeq

AND A.WorkOrderSerl = B.WorkOrderSerl

LEFT OUTER JOIN \#TPDOSPPOItem AS C ON A.WorkOrderSeq = C.WorkOrderSeq

AND A.WorkOrderSerl = C.WorkOrderSerl

LEFT OUTER JOIN \#TPDOSPDelvItem AS D ON A.WorkOrderSeq = D.WorkOrderSeq

AND A.WorkOrderSerl = D.WorkOrderSerl

GROUP BY A.ProdReqSeq, A.ProdReqSerl, A.OrderSeq, A.OrderSerl, A.ItemSeq

 

IF @IsOptionCfm = '0' AND

EXISTS (SELECT TOP 1 1

FROM \#BOMSpread_Last AS A

LEFT OUTER JOIN ssv_TSLOrderItemOption AS B ON B.CompanySeq = @CompanySeq

AND B.ProdReqSeq = @ProdReqSeq

AND B.Serl = @ProdReqSerl

AND B.ItemSeq = A.ItemSeq

WHERE B.CompanySeq IS NULL)

BEGIN

SELECT @IsOptionCfm = '0'

END

 

IF @IsOptionCfm \<\> '1'

BEGIN

--SELECT ISNULL(G.Seq , A.Seq ) AS Seq ,

SELECT ISNULL(A.Seq , 0) AS Seq , -- 기존 저장 데이터랑 중복나기 때문에 BOM기준으로 조회처리 20240206 KHKANG

IIF(ISNULL(G.ParentSeq, A.ParentSeq) = 1, 0, ISNULL(G.ParentSeq, A.ParentSeq)) AS ParentSeq , -- 구매품일때는 항상 0으로 처리

ISNULL(G.ParentSeq , A.ParentSeq) AS ParentSeqOld ,
