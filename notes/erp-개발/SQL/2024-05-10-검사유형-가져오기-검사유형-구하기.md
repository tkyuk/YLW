# 검사유형 가져오기 / 검사유형 구하기

-- 검사유형추가

UPDATE A

SET SMQCType = CASE WHEN D.Serl IS NOT NULL THEN 6035001 -- 무검사

WHEN ISNULL(C.IsUse , '') = N'1' THEN 6035002 -- 미검사

WHEN ISNULL(B.IsInQC, '') = N'1' THEN 6035002 -- 미검사

ELSE 6035001 -- 무검사 \_TDASMinor WHERE MajorSeq = 6035

END

FROM \#TDelvItem AS A

OUTER APPLY(SELECT TOP 1 Y.IsInQC

FROM \_TDAItemClass AS Z WITH(NOLOCK)

JOIN \_TPDQAItemClassQCKind AS Y WITH(NOLOCK) ON Z.CompanySeq = Y.CompanySeq

AND Z.UMItemClass = Y.UMItemClass

WHERE Z.CompanySeq = @CompanySeq

AND Z.ItemSeq = A.ItemSeq

AND Z.UMajorItemClass IN (2001 , 2004)

) AS B -- 품목소분류별 검사대상

OUTER APPLY(SELECT TOP 1 Z.IsUse

FROM \_TPDQAItemQCKind AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = @CompanySeq

AND Z.ItemSeq = A.ItemSeq

AND Z.SMQcKind = 6018002 -- \_TDASMinor WHERE MajorSeq = 6018

) AS C -- 품목별 검사여부 예외등록

OUTER APPLY(SELECT TOP 1 Z.Serl

FROM \_TPDQACustNotQCItem AS Z WITH(NOLOCK)

JOIN \_TPDQACustNotQC AS Y WITH(NOLOCK) ON Z.CompanySeq = Y.CompanySeq

AND Z.CustSeq = Y.CustSeq

AND Z.BizUnit = Y.BizUnit

WHERE Z.CompanySeq = @CompanySeq

AND Z.BizUnit = A.BizUnit

AND Z.CustSeq = A.CustSeq

AND Z.ItemSeq = A.ItemSeq

AND A.DelvDate BETWEEN Z.StartDate AND Z.EndDate

) AS D -- 거래처별 무검사품 등록

WHERE A.WorkingTag IN ('A', 'U')

AND A.Status = 0

AND ISNULL(A.SMQCType, 0) = 0 -- 검사유형이 없다면!
