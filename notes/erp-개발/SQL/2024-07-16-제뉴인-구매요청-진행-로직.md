M4.ItemNo AS TopUnitNo -- 최상위Unit품번 :: 20.02.21 \#임희진 (최종 Select문에서 조회조건 적용되는 부분으로 구문 이동)

FROM \_TPUORDPOReq AS A WITH(NOLOCK)

JOIN \_TPUORDPOReqItem AS B WITH(NOLOCK) ON A.CompanySeq= B.CompanySeq AND A.POReqSeq = B.POReqSeq

LEFT OUTER JOIN \_TDAItem AS C WITH(NOLOCK) ON A.CompanySeq= C.CompanySeq AND B.ItemSeq = C.ItemSeq

LEFT OUTER JOIN \_TPJTProject AS D WITH(NOLOCK) ON A.CompanySeq= D.CompanySeq AND B.PJTSeq = D.PJTSeq

LEFT OUTER JOIN \_TPJTBOM AS M5 WITH(NOLOCK) ON B.CompanySeq= M5.CompanySEq

AND B.PJTSeq = M5.PJTSeq

AND B.BOMSerl = M5.BOMSerl

LEFT OUTER JOIN \_TPJTBOM AS M1 WITH(NOLOCK) ON B.CompanySeq= M1.CompanySeq

AND B.PJTSeq = M1.PJTSeq

AND M1.BOMSerl \<\> -1

AND M5.UpperBOMSerl = M1.BOMSerl

AND ISNULL(M1.BeforeBOMSerl,0) = 0 -- 상위 BOM

LEFT OUTER JOIN \_TDAItem AS M2 WITH(NOLOCK) ON B.CompanySEq = M2.CompanySeq

AND M1.ItemSeq = M2.ItemSeq

LEFT OUTER JOIN \_TPJTBOM AS M3 WITH(NOLOCK) ON B.CompanySeq = M3.CompanySeq

AND B.PJTSeq = M3.PJTSeq

AND M3.BOMSerl \<\> -1

AND ISNULL(M3.BeforeBOMSerl,0) = 0

AND SUBSTRING(M1.TreeCode,1,6) = M3.TreeCode -- 최상위

AND ISNUMERIC(REPLACE(M3.BOMLevel,'.','/')) = 1 -- 가장 최상위 조건 추가 :: 20.02.21 \#임희진

LEFT OUTER JOIN \_TDAItem AS M4 WITH(NOLOCK) ON B.CompanySeq = M4.CompanySeq

AND M3.ItemSeq = M4.ItemSeq

LEFT OUTER JOIN \_TDAItemClass AS E WITH(NOLOCK) ON B.CompanySeq = E.CompanySeq

AND E.UMajorItemClass IN (2001, 2004)

AND B.ItemSeq = E.ItemSeq

LEFT OUTER JOIN \_TPDMRPDailyitem AS F WITH(NOLOCK) ON B.CompanySeq = F.CompanySeq

AND B.UMPOReqType = 6501014

AND B.SourceSeq = F.MRPDailySeq

AND B.SourceSerl = F.Serl

LEFT OUTER JOIN \_TPDMPSDailyProdPlan AS G WITH(NOLOCK) ON F.CompanySeq = G.CompanySeq

AND F.ProdPlanSeq = G.ProdPlanSeq

LEFT OUTER JOIN \_TPDSFCWorkOrder AS H WITH(NOLOCK) ON F.CompanySeq = H.CompanySeq

AND F.ProdPlanSeq = H.ProdPlanSeq

WHERE A.CompanySeq = @CompanySeq

AND (@MRPNo = '' OR A.POReqNo like @MRPNo + '%')

AND (@PJTSeq = 0 OR B.PJTSeq = @PJTSeq)

AND (@WBSSeq = 0 OR B.WBSSeq = @WBSSeq)

AND (@ItemSeq = 0 OR B.ItemSeq = @ItemSeq)

AND (@FrDate = '' OR A.ReqDate \>= @FrDate)

AND (@ToDate = '' OR A.ReqDate \<= @ToDate)

AND (@CMOrPM = 0 OR A.UMPOReqType = @CMOrPM)

AND (@ItemName = '' OR C.ItemName LIKE @ItemName + '%')

AND (@ItemNo = '' OR C.ItemNo LIKE @ItemNo + '%')

AND (@PurchaseCustSeq = 0 OR B.CustSeq = @PurchaseCustSeq)

AND (@PJTName = '' OR D.PJTName LIKE @PJTName + '%')

AND (@PJTTypeSeq = 0 OR D.PJTTypeSeq = @PJTTypeSeq)

AND (@PJTNo = '' OR D.PJTNo LIKE @PJTNo + '%')

AND (@TopUnitName = '' OR M4.ItemName LIKE @TopUnitName + '%')

AND (@TopUnitNo = '' OR M4.ItemNo LIKE @TopUnitNo + '%')

AND (@MRPEmpSeq = 0 OR A.EmpSeq = @MRPEmpSeq)

AND (@MRPDeptSeq = 0 OR A.DeptSeq = @MRPDeptSeq)

AND (@ProdPlanNo = '' OR G.ProdPlanNo LIKE @ProdPlanNo + '%')

AND (@WorkOrderNo = '' OR H.WorkOrderNo LIKE @WorkOrderNo + '%')

AND ( (@POReqNoFr = '' AND @POReqNoTo = '') OR (@POReqNoFr = '' AND A.POReqNo \<= @POReqNoTo)

OR (@POReqNoTo = '' AND A.POReqNo \>= @POReqNoFr)

OR (A.POReqNo BETWEEN @POReqNoFr AND @POReqNoTo))

AND (@UMItemClassS = 0 OR E.UMItemClass = @UMItemClassS)

AND (@ReqIsStop = '0' OR B.IsStop \<\> @ReqIsStop)

EXEC \_SCOMProgressTracking @CompanySeq, '\_TPUORDPOReqItem', '#Temp_POReq', 'POReqSeq', 'POReqSerl', ''
