# TRY CATCH 예외처리

BEGIN TRANSACTION;

 

BEGIN TRY

-- 첫 번째 INSERT 구문

INSERT INTO \_TSIASServiceReq(CompanySeq, ReceiptSeq, ReceiptNo, ReceiptDate, EmpSeq,

ItemSeq, CustSeq, UMContactKind, UMConsultKind, ServiceKind,

UMPriority, SMChargeType, ServiceDeptSeq, ServiceEmpSeq, ProcExpTime,

ProcExpDate, ReceiptRemark, FileSeq, LastUserSeq, LastDateTime,

ReqDate, ReceiptTime, ExpProcTime, LOTNo, SerialNo,

ContactSeq,        AddressSeq,                AcceptSeq,                ClientSeq,                ReqCustName

)

SELECT 1, 1, '12312', '' AS ReceiptDate, 0 AS EmpSeq,

0 AS ItemSeq, 0 AS CustSeq, 0 AS UMContactKind, 0 AS UMConsultKind, 'ㄹㄴㅇ' AS ServiceKind,

0 AS UMPriority,0 AS SMChargeType,0 AS ServiceDeptSeq,0 AS ServiceEmpSeq,'' AS ProcExpTime,

'' AS ProcExpDate, '' AS ReceiptRemark, 0 AS FileSeq, 1, GETDATE(),

'' AS ReqDate, '' AS ReceiptTime, '' ASExpProcTime, '' AS LOTNo, '31231',

0 AS ContactSeq, 0 AS AddressSeq,                0,                                0,                                ''

-- 첫 번째 INSERT 후 오류 체크

IF @@ERROR \<\> 0

BEGIN

GOTO StockERROR_PROC

END

 

 

-- 모든 작업이 성공하면 커밋

COMMIT TRANSACTION;

RETURN;

 

StockERROR_PROC:

-- 오류 발생 시 롤백

ROLLBACK TRANSACTION;

 

-- 오류 메시지 출력

PRINT '오류 발생: ' + ERROR_MESSAGE();

RETURN;

 

END TRY

BEGIN CATCH

-- 오류 발생 시 롤백

ROLLBACK TRANSACTION;

 

-- 오류 메시지 출력

PRINT '오류 발생: ' + ERROR_MESSAGE();

RETURN;

END CATCH;
