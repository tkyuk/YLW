# 날짜 중첩 체크 로직

DECLARE @MessageType INT,

@Status INT,

@Results NVARCHAR(250)

 

EXEC dbo.\_SWCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

6 , -- 중복된 @1 @2가(이) 입력되었습니다.(SELECT \* FROM \_TCAMessageLanguage WHERE MessageSeq = 6)

@LanguageSeq

UPDATE C

SET Result = @Results,

MessageType = @MessageType,

Status = @Status

FROM \#BIZ_OUT_DataBlock2 AS C JOIN (SELECT A.UMPgSeq

FROM shilla_TPRRetAddWorkTerm AS A WITH(NOLOCK)

JOIN shilla_TPRRetAddWorkTerm AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

>                                                                           AND A.UMPgSeq = B.UMPgSeq

AND A.Serl \<\> B.Serl

AND ((A.StartYear \>= B.StartYear AND A.StartYear \< B.EndYear)

OR (A.EndYear \> B.StartYear AND A.EndYear \<= B.EndYear))

WHERE A.CompanySeq = @CompanySeq

GROUP BY A.UMPgSeq

HAVING COUNT(1) \> 1 ) AS Z ON C.UMPgSeq = Z.UMPgSeq
