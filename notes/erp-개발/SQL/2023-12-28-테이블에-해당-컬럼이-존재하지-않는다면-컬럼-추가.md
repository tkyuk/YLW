# 테이블에 해당 컬럼이 존재하지 않는다면 컬럼 추가

IF NOT EXISTS (SELECT \* FROM sys.columns WHERE object_id = OBJECT_ID(N'\[dbo\].\[sys_TSLCustItemPriceReqLog\]') AND name='ReqDate')

BEGIN

ALTER TABLE sys_TSLCustItemPriceReqLog ADD ReqDate NCHAR(8)

END
