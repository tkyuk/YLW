# 사업자 영문명

D2.TaxEngName AS CompanyName -- 사업자 영문명

 

LEFT OUTER JOIN \_TTAXBizTaxUnit AS D1 WITH(NOLOCK) ON D1.CompanySeq = @CompanySeq

AND D1.BizUnit = A.BizUnit

LEFT OUTER JOIN \_TDATaxUnit AS D2 WITH(NOLOCK) ON D2.CompanySeq = D1.CompanySeq

AND D2.TaxUnit = D1.TaxUnit
