# 연차발생 테이블 서브 SP

DROP TABLE \#Tmp_YyUseDaysEmp

-- 연차 임시 테이블

CREATE TABLE \#Tmp_YyUseDaysEmp

(

Yy NCHAR(4), -- 사용연도

EmpSeq INT, -- 사원코드

SMYyType INT, -- 연차구분

WkTypeDate NCHAR(8), -- 근무유형기준일자

OccurDays DECIMAL(19,5), -- 총발생일수(이월적치일수+발생일수+추가발생일수)

OccurDTCnt DECIMAL(19,5), -- 총발생시간(이월적치시간+발생시간+추가발생시간)

UseFrDate NCHAR(8),

UseToDate NCHAR(8),

YyRuleSeq INT, -- 연차기준코드

YyTypeCnt INT, -- 연차구분갯수 : 1이면 자체 사용시간계산, 2이면 우선순위적용

UseDispSeq INT, -- 연차사용기준에 따른 사용일수 적용 순서 (5044001 :개정연차먼저사용, 5044002: 기본연차먼저사용)

SMYyUseBase INT, -- 연차사용기준

WkTypeSeq INT, -- 근무유형코드

UnitTm DECIMAL(19,5), -- 단위시간

UseDTCnt DECIMAL(19,5), -- 사용시간

PayDTCnt DECIMAL(19,5), -- 잔여시간

IsOccurOver NCHAR(1) -- 발생시간을 넘어선경우

)

 

INSERT INTO \#Tmp_YyUseDaysEmp(Yy, EmpSeq, SMYyType, WkTypeDate, OccurDays, OccurDTCnt, UseFrDate, UseToDate,

YyRuleSeq, YyTypeCnt, UseDispSeq, SMYyUseBase, WkTypeSeq, UnitTm, UseDTCnt, PayDTCnt, IsOccurOver)

SELECT A.Yy, A.EmpSeq, SMYyType, '20241231', -- @WkToDate -\> OccurToDate로(2020.05.12) -\> @WkToDate 사용시점으로 무조건 유지하기로(2020.12.14)

ISNULL(PileDays, 0) + ISNULL(OccurDays, 0) + ISNULL(AddOccurDays, 0) - ISNULL(BasePileDays, 0) - ISNULL(AddPileDays, 0),

ISNULL(PileDTCnt, 0) + ISNULL(OccurDTCnt, 0) + ISNULL(AddOccurDTCnt, 0) - ISNULL(BasePileDTCnt, 0) - ISNULL(AddPileDTCnt, 0),

UseFrDate, UseToDate, 0,0,0,0,0,0, 0,0,'0'

FROM \_TXPRWkYyOccurDays AS A WITH(NOLOCK) -- 연차발생 테이블

WHERE A.CompanySeq = 1

AND EmpSeq = 315

AND UseFrDate \<= '20240101'

AND UseToDate \>= '20241231'

-- 연차 사용 내역 조회

EXEC \_SXPRWkYyUseDaysGet 1

SELECT ISNULL(UseDTCnt, 0) / CASE WHEN ISNULL(UnitTm, 0) = 0 THEN 1.0 ELSE ISNULL(UnitTm, 0) END

,ISNULL(OccurDTCnt, 0) / CASE WHEN ISNULL(UnitTm, 0) = 0 THEN 1.0 ELSE ISNULL(UnitTm, 0) END

FROM \#Tmp_YyUseDaysEmp
