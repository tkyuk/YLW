# 거래처별 구매 담당자 등록의 담당자

UPDATE A

SET EmpSeq = ISNULL(B.EmpSeq, 0)

FROM \#Fix AS A

LEFT OUTER JOIN \_TPUBaseBuyEmp AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.CustSeq = A.CustSeq

AND B.SMEmpRegKind = 6218002 -- 거래처별구매담당자
