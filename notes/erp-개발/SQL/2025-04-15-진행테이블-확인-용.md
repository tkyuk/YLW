# 진행테이블 확인 용

--\_TCOMSourceDailyTrace

SELECT \* FROM \_TCOMProgTable WHERE ProgTableName = '\_TSLSalesItem'

SELECT \* FROM \_TCOMProgTable WHERE ProgTableName = '\_TPJTSLBillItem'

SELECT \* FROM \_TCOMProgTable WHERE ProgTableName LIKE '\_TPJT%'

 

SELECT DISTINCT B.ProgTableKorName,C.ProgTableKorName,A.FromTableSeq,A.ToTableSeq

FROM \_TCOMSourceDailyTrace AS A

JOIN \_TCOMProgTable AS B ON B.ProgTableSeq = A.FromTableSeq

JOIN \_TCOMProgTable AS C ON C.ProgTableSeq = A.ToTableSeq

WHERE A.FromTableSeq = 55

 

 

SELECT \* FROM \_TPJTProjectItemBillCond -- 55

SELECT \* FROM \_TCOMSourceDailyTrace WHERE FromTableSeq = 55

 

SELECT \* FROM \_TCOMProgTable WHERE ProgTableSeq = 55

SELECT \* FROM \_TCOMProgTable WHERE ProgTableSeq = 54

 

SELECT \* FROM \_TPJTProjectItemBillCond WHERE PJTSeq = 527

SELECT \* FROM \_TPJTSLBillItem WHERE BillSeq = 1182

 

SELECT TOP 10 \* FROM \_TCOMSourceDailyTrace ORDER BY LastDateTime DESC
