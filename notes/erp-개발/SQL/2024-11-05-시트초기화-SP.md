# 시트초기화 SP

/\*

> DELETE OCASheetMasterSetting WHERE PgmSeq = @PgmSeq AND ControlID = @ControlID AND CompanySeq = @CompanySeq
>
> DELETE OCASheetSetting WHERE PgmSeq = @PgmSeq AND ControlID = @ControlID AND CompanySeq = @CompanySeq

\*/

BEGIN TRAN

DELETE A

FROM OCASheetMasterSetting AS A

WHERE A.ControlID = 'SS1'

AND A.PgmSeq IN (SELECT A.PgmSeq

FROM \_TCAPgm AS A

WHERE A.Caption IN('수주입력_gcos','수주품목조회','수주품목조회_gcos','출하의뢰품목조회_gcos','출하의뢰품목조회','출하의뢰입력','거래명세서품목조회_gcos','거래명세서품목조회','거래명세서입력'

,'수출Order입력_gcos','수출Order입력','수출Order품목조회_gcos','수출Order품목조회','수출출하의뢰품목조회_gcos','수출출하의뢰품목조회','수출출하의뢰입력'

,'수출Invoice입력','수출Invoice품목조회')

)

 

DELETE A

FROM OCASheetSetting AS A

WHERE A.ControlID = 'SS1'

AND A.PgmSeq IN (SELECT A.PgmSeq

FROM \_TCAPgm AS A

WHERE A.Caption IN('수주입력_gcos','수주품목조회','수주품목조회_gcos','출하의뢰품목조회_gcos','출하의뢰품목조회','출하의뢰입력','거래명세서품목조회_gcos','거래명세서품목조회','거래명세서입력'

,'수출Order입력_gcos','수출Order입력','수출Order품목조회_gcos','수출Order품목조회','수출출하의뢰품목조회_gcos','수출출하의뢰품목조회','수출출하의뢰입력'

,'수출Invoice입력','수출Invoice품목조회')

)

 

 

 

 

 

ROLLBACK TRAN
