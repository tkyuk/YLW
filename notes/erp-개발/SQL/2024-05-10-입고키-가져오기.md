# 입고키 가져오기

-- 입고키값 가져오기

UPDATE A

SET DelvInSeq = B.ToSeq,

DelvInSerl = B.ToSerl,

DelvInNo = CASE WHEN A.POType = 1 THEN C.DelvInNo WHEN A.POType = 2 THEN D.OSPDelvInNo END

FROM \#TDelvItem AS A

JOIN \_TCOMSourceDailyTrace AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.FromSeq = A.DelvSeq

AND B.FromSerl = A.DelvSerl

AND B.FromSubSerl = 0

LEFT OUTER JOIN \_TPUDelvIn AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq AND C.DelvInSeq = B.ToSeq

LEFT OUTER JOIN \_TPDOSPDelvIn AS D WITH(NOLOCK) ON D.CompanySeq = B.CompanySeq AND D.OSPDelvInSeq = B.ToSeq

WHERE A.WorkingTag IN ('U', 'D')

AND A.Status = 0

AND A.IsAutoIn = '1'

AND A.SMQcType = 6035001 -- 무검사

AND B.FromTableSeq = CASE WHEN A.POType = 1 THEN 10 WHEN A.POType = 2 THEN 3 END -- \_TCOMProgTable WHERE ProgTableName IN ('\_TPUDelvItem', '\_TPDOSPDelvItem' )

AND B.ToTableSeq = CASE WHEN A.POType = 1 THEN 9 WHEN A.POType = 2 THEN 2 END -- \_TCOMProgTable WHERE ProgTableName IN ('\_TPUDelvInItem', '\_TPDOSPDelvInItem')
