CREATE PROCEDURE dbo.tnt_SWSLOrderItemListStop

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

AS

 

CREATE TABLE \#TCOMStop

(

IDX_NO INT

, StopSeq INT

, StopSerl INT

, StopSubSerl INT

, IsStop NCHAR(1)

, UMStopReason INT

, StopReason NVARCHAR(500)

, Result NVARCHAR(255)

, MessageType INT

, Status INT

, DataSeq INT

)

INSERT INTO \#TCOMStop(IDX_NO, StopSeq, StopSerl, StopSubSerl, IsStop, UMStopReason, StopReason, Result, MessageType, Status, DataSeq)

SELECT IDX_NO, OrderSeq, OrderSerl, 0, IsStop, 0, '', NULL, NULL, 0, DataSeq

FROM \#BIZ_OUT_DATABLOCK1

WHERE Status = 0

EXEC \_SWCOMStop

@CompanySeq = @CompanySeq

, @LanguageSeq = @LanguageSeq

, @UserSeq = @UserSeq

, @PgmSeq = @PgmSeq

 

UPDATE \#BIZ_OUT_DATABLOCK1

SET Result = B.Result

, MessageType = B.MessageType

, Status = B.Status

FROM \#BIZ_OUT_DATABLOCK1 AS A

JOIN \#TCOMStop AS B ON A.IDX_NO = B.IDX_NO

WHERE B.Status \<\> 0

 

 

RETURN

CREATE PROCEDURE dbo.tnt_SWSLOrderItemListStopCheck

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

AS

 

DECLARE @IsStop INT

, @MessageType INT

, @Status INT

, @Results NVARCHAR(250)

 

SELECT @IsStop = IsStop FROM \#BIZ_OUT_DATABLOCK1

 

IF @IsStop = 1 -- 확정처리

BEGIN

EXEC dbo.\_SWCOMStopCheck

@CompanySeq = @CompanySeq

, @LanguageSeq = @LanguageSeq

, @PgmSeq = @PgmSeq

, @TempTableName = N'#BIZ_OUT_DATABLOCK1'

, @Seq = N'OrderSeq'

, @Serl = N'OrderSerl'

, @SubSerl = N''

, @IsStop = N'Status' -- 체크 컬럼

, @StopTableSeq = 19 -- 수주

 

EXEC dbo.\_SWCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

1152 ,

@LanguageSeq ,

1326 , N'중단'

 

END

ELSE -- 중단취소

BEGIN

UPDATE A

SET Status = 99999,

Result = '중단 되지 않은 건은 중단취소 할수 없습니다.'

FROM \#BIZ_OUT_DATABLOCK1 AS A LEFT OUTER JOIN \_TCOMStop AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.TableSeq = 19

AND A.OrderSeq = B.StopSeq

AND A.OrderSerl = B.StopSerl

WHERE A.Status = 0

AND ISNULL(B.IsStop,'') \<\> '1'

-- 1326 , N'중단',

END

 

UPDATE A

SET Result = @Results

, MessageType = @MessageType

, Status = @Status

FROM \#BIZ_OUT_DATABLOCK1 AS A

WHERE A.Status = 1

 

 

RETURN
