# 창고 구하기 (구매, 외주)

구매

외주

-- 창고 찾아오기.

UPDATE A

SET WHSeq = CASE WHEN ISNULL(A.WHSeq, 0) \<\> 0 THEN A.WHSeq

WHEN ISNULL(B.WHSeq, 0) \<\> 0 THEN B.WHSeq

WHEN ISNULL(C.WHSeq, 0) \<\> 0 THEN C.WHSeq

WHEN ISNULL(D.WHSeq, 0) \<\> 0 THEN D.WHSeq

WHEN ISNULL(E.WHSeq, 0) \<\> 0 THEN E.WHSeq

ELSE 0

END

FROM \#TDelvItem AS A

OUTER APPLY(SELECT TOP 1 Z.WHSeq

FROM \_TPUReqDeptBasicWH AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = @CompanySeq

AND Z.DeptSeq = A.DeptSeq

ORDER BY Z.Seq DESC

) AS B -- 1.부서별 납품창고

OUTER APPLY(SELECT TOP 1 Z.WHSeq

FROM \_TDAWHItem AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = @CompanySeq

AND Z.ItemSeq = A.ItemSeq

ORDER BY Z.WHSeq DESC

) AS C -- 2.창고별관리품목

OUTER APPLY(SELECT TOP 1 Z.WHSeq

FROM \_TPUDelvItem AS Z WITH(NOLOCK)

JOIN \_TPUDelv AS Y WITH(NOLOCK) ON Z.CompanySeq = Y.CompanySeq

AND Z.DelvSeq = Y.DelvSeq

WHERE Z.CompanySeq = @CompanySeq

AND ISNULL(Y.IsReturn, '0') \<\> '1' -- 반품건 제외

AND Z.ItemSeq = A.ItemSeq

AND Y.BizUnit = A.BizUnit

ORDER BY Y.DelvDate DESC, Y.DelvNo DESC, Z.DelvSeq DESC, Z.DelvSerl DESC

) AS D -- 3.이전 품목별 납품창고

OUTER APPLY(SELECT TOP 1 Z.WHSeq

FROM \_TDAWH AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = @CompanySeq

AND Z.SMWHKind = 8002026

AND Z.BizUnit = A.BizUnit

ORDER BY Z.WHSeq DESC

) AS E -- 4. 위의 창고도 모두 없을 경우, 사업단위 창고

WHERE A.WorkingTag IN ('A', 'U')

AND A.Status = 0

AND A.POType = 1 -- 구매

AND ISNULL(A.WHSeq, 0) = 0 -- 창고 코드가 없다면!

-- 창고 찾아오기.

UPDATE A

SET WHSeq = CASE WHEN ISNULL(A.WHSeq, 0) \<\> 0 THEN A.WHSeq

WHEN ISNULL(B.WHSeq, 0) \<\> 0 THEN B.WHSeq

WHEN ISNULL(C.WHSeq, 0) \<\> 0 THEN C.WHSeq

WHEN ISNULL(D.WHSeq, 0) \<\> 0 THEN D.WHSeq

WHEN ISNULL(E.WHSeq, 0) \<\> 0 THEN E.WHSeq

ELSE 0

END

FROM \#TDelvItem AS A

OUTER APPLY(SELECT TOP 1 Y.DelvWHSeq AS WHSeq

FROM \_TPDSFCWorkOrder AS Z WITH(NOLOCK)

JOIN \_TPDBaseWorkCenter AS Y WITH(NOLOCK) ON Y.CompanySeq = Z.CompanySeq

AND Y.WorkCenterSeq = Z.WorkCenterSeq

WHERE Z.CompanySeq = @CompanySeq

AND Z.WorkOrderSeq = A.WorkOrderSeq

AND Z.WorkOrderSerl = A.WorkOrderSerl

ORDER BY Z.WorkCenterSeq DESC

) AS B -- 1.작업지시 워크센터의 창고

OUTER APPLY(SELECT TOP 1 X.DelvWHSeq AS WHSeq

FROM \_TPDROUItemProc AS Z WITH(NOLOCK)

JOIN \_TPDROUItemProcWC AS Y WITH(NOLOCK) ON Y.CompanySeq = Z.CompanySeq

AND Y.BizUnit = Z.BizUnit

AND Y.ItemSeq = Z.ItemSeq

AND Y.ProcRev = Z.ProcRev

AND Y.ProcSeq = Z.ProcSeq

JOIN \_TPDBaseWorkCenter AS X WITH(NOLOCK) ON X.CompanySeq = Y.CompanySeq

AND X.WorkCenterSeq = Y.WorkCenterSeq

WHERE Z.CompanySeq = @CompanySeq

AND Z.BizUnit = A.BizUnit

AND Z.ItemSeq = A.ItemSeq

AND Z.ProcRev = A.ProcRev

ORDER BY Y.Ranking ASC, Y.ProcSeq DESC, Y.Serl DESC

) AS C -- 2.제품별 공정별 워크센터의 창고

OUTER APPLY(SELECT TOP 1 Z.WHSeq

FROM \_TDAWHItem AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = @CompanySeq

AND Z.ItemSeq = A.ItemSeq

ORDER BY Z.WHSeq DESC

) AS D -- 3.창고별관리품목

OUTER APPLY(SELECT TOP 1 Z.WHSeq

FROM \_TDAWH AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = @CompanySeq

AND Z.SMWHKind = 8002026

AND Z.BizUnit = A.BizUnit

ORDER BY Z.WHSeq DESC

) AS E -- 4. 위의 창고도 모두 없을 경우, 사업단위 창고

WHERE A.WorkingTag IN ('A', 'U')

AND A.Status = 0

AND A.POType = 2 -- 외주

AND ISNULL(A.WHSeq, 0) = 0 -- 창고 코드가 없다면!
