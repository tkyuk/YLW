# SQL 부가세 포함 계산식

djb_SWSLOrderCfmReqJumpQuery

 

ISNULL(A.IsInclusedVAT,'0') AS IsInclusedVAT, -- 부가세포함

CASE WHEN ISNULL(A.IsInclusedVAT,'0') = '1' THEN ROUND( (ISNULL(X.ToQty, 0) \* ISNULL(A.CurPrice, 0)) \* (100 / (100 + A.VatRate)), 0 )

ELSE (ISNULL(X.ToQty, 0) \* ISNULL(A.CurPrice, 0))

END AS CurAmt , -- 판매금액

CASE WHEN ISNULL(A.IsInclusedVAT,'0') = '1' THEN (ISNULL(X.ToQty, 0) \* ISNULL(A.CurPrice, 0)) - (ROUND( (ISNULL(X.ToQty, 0) \* ISNULL(A.CurPrice, 0)) \* (100 / (100 + A.VatRate)), 0 ))

ELSE (ISNULL(X.ToQty, 0) \* ISNULL(A.CurPrice, 0)) \* (A.VatRate / 100)

END AS CurVAT, -- 부가세

CASE WHEN ISNULL(A.IsInclusedVAT,'0') = '1' THEN (ISNULL(X.ToQty, 0) \* ISNULL(A.CurPrice, 0))

ELSE (ISNULL(X.ToQty, 0) \* ISNULL(A.CurPrice, 0)) + (ISNULL(X.ToQty, 0) \* ISNULL(A.CurPrice, 0)) \* (A.VatRate / 100)

END AS CurAmtTotal, -- 판매금액계

 

CASE WHEN ISNULL(A.IsInclusedVAT,'0') = '1' THEN ROUND( (ISNULL(X.ToQty, 0) \* ISNULL(A.CurPrice, 0)) \* (100 / (100 + A.VatRate)), 0 )

ELSE (ISNULL(X.ToQty, 0) \* ISNULL(A.CurPrice, 0))

END AS DomAmt , -- 원화판매금액

CASE WHEN ISNULL(A.IsInclusedVAT,'0') = '1' THEN (ISNULL(X.ToQty, 0) \* ISNULL(A.CurPrice, 0)) - (ROUND( (ISNULL(X.ToQty, 0) \* ISNULL(A.CurPrice, 0)) \* (100 / (100 + A.VatRate)), 0 ))

ELSE (ISNULL(X.ToQty, 0) \* ISNULL(A.CurPrice, 0)) \* (A.VatRate / 100)

END AS DomVAT , -- 원화부가세

 

CASE WHEN ISNULL(A.IsInclusedVAT,'0') = '1' THEN (ISNULL(X.ToQty, 0) \* ISNULL(A.CurPrice, 0))

ELSE (ISNULL(X.ToQty, 0) \* ISNULL(A.CurPrice, 0)) + (ISNULL(X.ToQty, 0) \* ISNULL(A.CurPrice, 0)) \* (A.VatRate / 100)

END AS DomAmtTotal, -- 원화판매금액계
