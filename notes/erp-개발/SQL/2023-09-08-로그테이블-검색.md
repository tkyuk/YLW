# 로그테이블 검색

-- 테이블 키값으로만 가능(빠름) : 테이블, CompanySeq, 테이블 키, 테이블 키 값

EXEC \_SWCOMGetTableLog '\_TDAItem', 1, 'ItemSeq', 998

 

 

 

-- 기간, 컬럼, 컬럼값으로 가능(느림) : CompanySeq, 테이블, 기간Fr, 기간To, 컬럼명, 컬럼값

EXEC \_SWCOMTableLogQuery

@CompanySeq = 1,

@TableName = '\_TDAItem',

@LogDateFr = '20240901',

@LogDateTo = '20241030',

@TXTCol1 = 'ItemSeq',

@TXTData1 = '998'
