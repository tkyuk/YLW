# 창고가져오기

-- 창고우선순위 변경

-- 담당자창고 \> 판매부서별창고 \> 품목분류별기본창고(예외품목) \> 품목분류별기본창고(소분류) \> 창고별품목(동일사업단위) \> 창고별품목(타사업단위) \> 사업단위창고

 

-- 담당자창고

UPDATE \#BIZ_OUT_DataBlock1

SET WHSeq = B.WHSeq,

WHName = ISNULL(B.WHName, '')

FROM \#BIZ_OUT_DataBlock1 AS A

JOIN \_TDAWH AS B WITH (NOLOCK) ON B.CompanySeq = @CompanySeq

AND A.EmpSeq = B.EmpSeq

WHERE ISNULL(A.EmpSeq,0) \<\> 0

AND B.IsNotUse \<\> '1'

 

-- 부서별창고

UPDATE \#BIZ_OUT_DataBlock1

SET UMItemClass = ISNULL(A.UMItemClass,0)

FROM \_TDAItemClass AS A WITH (NOLOCK)

JOIN \#BIZ_OUT_DataBlock1 AS B ON A.CompanySeq = @CompanySeq

AND A.ItemSeq = B.ItemSeq

WHERE A.UMajorItemClass IN (2001, 2004)

UPDATE \#BIZ_OUT_DataBlock1

SET WHSeq = B.WHSeq,

WHName = ISNULL(C.WHName, '')

FROM \#BIZ_OUT_DataBlock1 AS A

JOIN \_TSLExceptionDeptWH AS B WITH (NOLOCK) ON B.CompanySeq = @CompanySeq

AND A.DeptSeq = B.DeptSeq

AND A.UMItemClass = B.UMItemClass

JOIN \_TDAWH AS C WITH(NOLOCK) ON B.CompanySeq = C.CompanySeq

AND B.WHSeq = C.WHSeq

WHERE ISNULL(A.WHSeq,0) = 0

AND C.IsNotUse \<\> '1'

UPDATE \#BIZ_OUT_DataBlock1

SET WHSeq = B.WHSeq,

WHName = ISNULL(C.WHName, '')

FROM \#BIZ_OUT_DataBlock1 AS A

JOIN \_TSLSalesDeptWH AS B WITH (NOLOCK) ON B.CompanySeq = @CompanySeq

AND A.DeptSeq = B.DeptSeq

JOIN \_TDAWH AS C WITH(NOLOCK) ON B.CompanySeq = C.CompanySeq

AND B.WHSeq = C.WHSeq

WHERE ISNULL(A.WHSeq,0) = 0

AND C.IsNotUse \<\> '1'

-- 동일사업단위 내 창고별품목

UPDATE \#BIZ_OUT_DataBlock1

SET WHSeq = ISNULL(B.WHSeq, 0),

WHName = (SELECT WHName FROM \_TDAWH WITH (NOLOCK) WHERE CompanySeq = @CompanySeq AND WHSeq = B.WHSeq)

FROM \#BIZ_OUT_DataBlock1 AS A

JOIN (SELECT X.ItemSeq, MIN(ISNULL(Y.WHSeq,0)) AS WHSeq

FROM \#BIZ_OUT_DataBlock1 AS X

JOIN \_TDAWHItem AS Y WITH(NOLOCK) ON Y.CompanySeq = @CompanySeq

AND X.ItemSeq = Y.ItemSeq

JOIN \_TDAWH AS Z WITH(NOLOCK) ON Z.CompanySeq = @CompanySeq

AND Y.WHSeq = Z.WHSeq

AND X.BizUnit = Z.BizUnit

GROUP BY X.ItemSeq) AS B ON A.ItemSeq = B.ItemSeq

WHERE ISNULL(A.WHSeq,0) = 0

 

-- 타사업단위 창고별품목

UPDATE \#BIZ_OUT_DataBlock1

SET WHSeq = ISNULL(B.WHSeq, 0),

WHName = (SELECT WHName FROM \_TDAWH WITH (NOLOCK) WHERE CompanySeq = @CompanySeq AND WHSeq = B.WHSeq)

FROM \#BIZ_OUT_DataBlock1 AS A

JOIN (SELECT X.ItemSeq, MIN(ISNULL(Y.WHSeq,0)) AS WHSeq

FROM \#BIZ_OUT_DataBlock1 AS X

JOIN \_TDAWHItem AS Y WITH(NOLOCK) ON Y.CompanySeq = @CompanySeq

AND X.ItemSeq = Y.ItemSeq

GROUP BY X.ItemSeq) AS B ON A.ItemSeq = B.ItemSeq

WHERE ISNULL(A.WHSeq,0) = 0

 

-- 사업단위창고

UPDATE \#BIZ_OUT_DataBlock1

SET WHSeq = ISNULL(B.WHSeq, 0),

WHName = ISNULL(B.WHName,'')

FROM \#BIZ_OUT_DataBlock1 AS A

JOIN \_TDAWH AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND A.BizUnit = B.BizUnit

AND B.SMWHKind = 8002026 -- 사업단위창고

WHERE ISNULL(A.WHSeq,0) = 0

AND B.IsNotUse \<\> '1'
