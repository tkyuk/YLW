# 다이나믹 Title 2개 예시

CREATE PROC cymc_SPJTPurchaseProgListQuery

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

AS

DECLARE @EmpSeq INT,

@CustSeq INT,

@PJTName NVARCHAR(100)

SELECT @EmpSeq = ISNULL(EmpSeq , 0),

@CustSeq = ISNULL(CustSeq, 0),

@PJTName = ISNULL(PJTName, '')

FROM \#BIZ_IN_DataBlock1

--타이틀구하기(담당자별 협력사 = 담당자별협력사프로젝트입력_cymc)

CREATE TABLE \#Title

(

Title NVARCHAR(100),

TitleSeq INT,

Title2 NVARCHAR(100),

TitleSeq2 INT,

ColIDX INT

)

 

CREATE TABLE \#Title2

(

Title2 NVARCHAR(100),

TitleSeq2 INT

)

 

INSERT \#Title2(Title2, TitleSeq2)

SELECT '발주수량' , 1 UNION ALL

SELECT '발주중단수량', 2 UNION ALL

SELECT '입고수량' , 3 UNION ALL

SELECT '미입고수량' , 4

 

INSERT \#Title(Title, TitleSeq, Title2, TitleSeq2, ColIDX)

SELECT DISTINCT

C.CustName AS Title ,

A.CustSeq AS TitleSeq ,

Z.Title2 AS Title2 ,

Z.TitleSeq2 AS TitleSeq2,

DENSE_RANK() OVER(ORDER BY C.CustName, A.SortOrder) -1 AS ColIDX

FROM \#Title2 AS Z

JOIN cymc_TPJTEmpCust AS A WITH(NOLOCK) ON 1=1

JOIN \_TPJTProject AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.PJTSeq = A.PJTSeq

JOIN \_TDACust AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.CustSeq = A.CustSeq

WHERE A.CompanySeq = @CompanySeq

AND A.EmpSeq = @EmpSeq

AND (@CustSeq = 0 OR A.CustSeq = @CustSeq)

AND B.PJTName LIKE @PJTName + '%'

 

 

CREATE TABLE \#Fix

(

PJTNo NVARCHAR(100),

PJTName NVARCHAR(100),

PJTSeq INT,

RowIDX INT

)

 

INSERT \#Fix(PJTNo, PJTName, PJTSeq, RowIDX)

SELECT B.PJTNo ,

B.PJTName,

B.PJTSeq ,

ROW_NUMBER() OVER(ORDER BY B.PJTNo) - 1 AS RowIDX

FROM cymc_TPJTEmpCust AS A

JOIN \_TPJTProject AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.PJTSeq = A.PJTSeq

WHERE A.CompanySeq = @CompanySeq

AND A.EmpSeq = @EmpSeq

AND (@CustSeq = 0 OR A.CustSeq = @CustSeq)

AND B.PJTName LIKE @PJTName + '%'

 

CREATE TABLE \#TMP_PROGRESSTABLE

(

> IDOrder INT,
>
> TABLENAME NVARCHAR(100)
>
> )

 

CREATE TABLE \#TCOMProgressTracking

(

> IDX_NO INT,
>

>

>
> Serl INT,
>

>
> StdQty DECIMAL(19,5) ,
>
> Amt DECIMAL(19, 5),
>
> VAT DECIMAL(19,5)
>
> )

CREATE TABLE \#Temp_POProgress

(

> IDX_NO INT IDENTITY,
>
> POSeq INT,
>
> POSerl INT,

PJTSeq INT,

CustSeq INT,

> Qty DECIMAL(19, 5),
>
> IsDelv NCHAR(1),
>
> IsStop NCHAR(1),
>
> SMCurrStatus INT,
>
> CurAmt DECIMAL(19, 5),
>
> CurVAT DECIMAL(19, 5),
>
> DelvInSeq INT,
>
> DelvInSerl INT,
>
> DelvInAmt DECIMAL(19, 5),
>
> DelvInVAT DECIMAL(19, 5),
>
> DelvInQty DECIMAL(19, 5),
>
> DelvInDate NCHAR(8),
>
> DelvInExRate DECIMAL(19, 5),
>
> CompleteCHECK INT
>

INSERT \#Temp_POProgress(POSeq, POSerl, PJTSeq, CustSeq, Qty, IsDelv, IsStop, SMCurrStatus, CurAmt, CurVAT, DelvInSeq, DelvInSerl, DelvInAmt, DelvInVAT, DelvInQty, DelvInExRate, CompleteCHECK)

SELECT A.POSeq, B.POSerl, B.PJTSeq, A.CustSeq, B.Qty, '1', B.IsStop, 6036001, B.CurAmt, B.CurVAT, 0, 0, 0, 0, 0, 0, 0

FROM \_TPUORDPO AS A WITH(NOLOCK)

JOIN \_TPUORDPOItem AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.POSeq = A.POSeq

WHERE A.CompanySeq = @CompanySeq

AND A.CustSeq IN (SELECT DISTINCT TitleSeq FROM \#Title)

AND B.PJTSeq IN (SELECT DISTINCT PJTSeq FROM \#Fix)

 

> -- 진행상태 체크                 

EXEC \_SCOMProgStatus @CompanySeq, '\_TPUORDPOItem', 1036002, '#Temp_POProgress', 'POSeq', 'POSerl', '', '', '', '', '', '', 'CompleteCHECK', 1, 'Qty', 'StdUnitQty', 'CurAmt', '', 'POSeq', 'POSerl', '', '\_TPUORDPOItem', 1133

 

UPDATE \#Temp_POProgress

SET SMCurrStatus = (SELECT CASE WHEN A.CompleteCHECK = '5' OR B.MinorSeq = 1037009 THEN 6036007 -- 결재진행중 2014.07.03 김용현추가

WHEN A.IsStop = '1' OR B.MinorSeq = 1037005 THEN 6036005 -- 중단

WHEN A.CompleteCHECK = '40' OR B.MinorSeq = 1037008 THEN 6036004 -- 완료

WHEN A.CompleteCHECK = 1 OR B.MinorSeq = 1037003 THEN 6036002 -- 확정(승인)

WHEN A.CompleteCHECK = '20' OR B.MinorSeq = 1037006 THEN 6036003 -- 진행

ELSE 6036001 END) -- 작성(미확정)

FROM \#Temp_POProgress AS A

LEFT OUTER JOIN \_TDASMinor AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq AND B.MajorSeq = 1037 AND A.CompleteCHECK = B.MinorValue

>

>

SELECT 1, '\_TPUDelvInItem'                         -- 구매입고

EXEC \_SCOMProgressTracking @CompanySeq, '\_TPUORDPOItem', '#Temp_POProgress', 'POSeq', 'POSerl', ''

> AND B.Seq                = C.DelvInSeq

JOIN \#OrderTracking AS D ON A.IDX_No = D.IDX_No

> WHERE D.IDOrder = 1        -- 구매입고

 

CREATE TABLE \#Value_Temp

(

PJTSeq INT,

CustSeq INT,

POQty DECIMAL(19,5),

StopPOQty DECIMAL(19,5),

InQty DECIMAL(19,5),

NotInQty DECIMAL(19,5)

)

 

CREATE TABLE \#Value

(

PJTSeq INT,

CustSeq INT,

POQty DECIMAL(19,5),

StopPOQty DECIMAL(19,5),

InQty DECIMAL(19,5),

NotInQty DECIMAL(19,5)

)

 

-- POQty

INSERT \#Value_Temp(PJTSeq, CustSeq, POQty, StopPOQty, InQty, NotInQty)

SELECT PJTSeq, CustSeq, SUM(Qty), 0, 0, 0

FROM \#Temp_POProgress

WHERE IsStop \<\> '1'

GROUP BY PJTSeq, CustSeq

 

-- StopPOQty

INSERT \#Value_Temp(PJTSeq, CustSeq, POQty, StopPOQty, InQty, NotInQty)

SELECT PJTSeq, CustSeq, 0, SUM(Qty), 0, 0

FROM \#Temp_POProgress

WHERE IsStop = '1'

GROUP BY PJTSeq, CustSeq

 

-- InQty

INSERT \#Value_Temp(PJTSeq, CustSeq, POQty, StopPOQty, InQty, NotInQty)

SELECT PJTSeq, CustSeq, 0, 0, SUM(Qty), 0

FROM \#Temp_POProgress

WHERE (DelvInQty \<\> 0 AND Qty = DelvInQty)

GROUP BY PJTSeq, CustSeq

 

-- NotInQty

INSERT \#Value_Temp(PJTSeq, CustSeq, POQty, StopPOQty, InQty, NotInQty)

SELECT PJTSeq, CustSeq, 0, 0, 0, SUM(Qty)

FROM \#Temp_POProgress

WHERE (DelvInQty = 0 OR (DelvInQty \<\> 0 AND Qty \> DelvInQty))

GROUP BY PJTSeq, CustSeq

 

-- 최종 집계

INSERT \#Value(PJTSeq, CustSeq, POQty, StopPOQty, InQty, NotInQty)

SELECT PJTSeq, CustSeq, SUM(POQty), SUM(StopPOQty), SUM(InQty), SUM(NotInQty)

FROM \#Value_Temp

GROUP BY PJTSeq, CustSeq

 

-- 최종SELECT

SELECT \*

FROM \#Title

ORDER BY ColIDX, TitleSeq2

 

SELECT \*

FROM \#Fix

ORDER BY RowIDX

 

SELECT B.ColIDX ,

A.RowIDX ,

C.POQty ,

C.StopPOQty,

C.InQty ,

C.NotInQty

FROM \#Fix AS A

JOIN \#Title AS B ON 1=1

JOIN \#Value AS C WITH(NOLOCK) ON C.PJTSeq = A.PJTSeq

AND C.CustSeq = B.TitleSeq

ORDER BY B.ColIDX, A.RowIDX

 

RETURN

GO
