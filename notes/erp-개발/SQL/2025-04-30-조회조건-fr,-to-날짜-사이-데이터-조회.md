# 조회조건 fr, to 날짜 사이 데이터 조회

DROP PROC gcos_SWSCMWHInputLimitItemQuery

GO

CREATE PROC gcos_SWSCMWHInputLimitItemQuery

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

AS

DECLARE @DelvReqDateFr NCHAR(8)

,@DelvReqDateTo NCHAR(8)

,@WHSeq INT

,@UMLimitType INT

 

SELECT @DelvReqDateFr = ISNULL(A.DelvReqDateFr,'')

,@DelvReqDateTo = ISNULL(A.DelvReqDateTo,'')

,@WHSeq = ISNULL(A.WHSeq , 0)

,@UMLimitType = ISNULL(A.UMLimitType , 0)

FROM \#BIZ_IN_DataBlock2 AS A

IF @DelvReqDateTo = '' SELECT @DelvReqDateTo = '99991231'

 

-- 기준 테이블 생성

CREATE TABLE \#Tmp_DelvReqDate(

DelvReqDate NCHAR(8)

,UMDelvReqTimeName NVARCHAR(100)

,UMDelvReqTime INT

)

INSERT INTO \#Tmp_DelvReqDate(DelvReqDate, UMDelvReqTimeName, UMDelvReqTime)

SELECT CONVERT(CHAR(8), DATEADD(DAY, B.number, CONVERT(date, @DelvReqDateFr)), 112) AS DelvReqDate

,A.MinorName

,A.MinorSeq

FROM \_TDAUMinor AS A

CROSS JOIN master.dbo.spt_values AS B

WHERE A.CompanySeq = @CompanySeq

AND A.MajorSeq = 1110279

AND B.type = 'P' -- 숫자만 추출

AND B.number \<= DATEDIFF(DAY, CONVERT(date, @DelvReqDateFr), CONVERT(date, @DelvReqDateTo))

ORDER BY DelvReqDate, A.MinorSeq

 

SELECT A.DelvReqDate AS DelvReqDate

,A.UMDelvReqTimeName AS UMDelvReqTimeName

,A.UMDelvReqTime AS UMDelvReqTime

,ISNULL(B.LimitQty, 0) AS LimitQty

,@WHSeq AS WHSeq

,@UMLimitType AS UMLimitType

FROM \#Tmp_DelvReqDate AS A

LEFT OUTER JOIN gcos_TSCMWHInputLimit AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.WHSeq = @WHSeq

AND B.DelvReqDate = A.DelvReqDate

AND B.UMDelvReqTime = A.UMDelvReqTime

ORDER BY A.DelvReqDate, A.UMDelvReqTime

RETURN
