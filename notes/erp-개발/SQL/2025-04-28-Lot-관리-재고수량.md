# Lot 관리 재고수량

DROP PROC SMC_SMESLGGetInOutLotStock

GO

CREATE PROC SMC_SMESLGGetInOutLotStock

AS

 

DECLARE @CompanySeq INT = 1

,@BizUnit INT = 1

CREATE TABLE \#GetInOutItemLot

(

LotNo NVARCHAR(30),

ItemSeq INT,

ValiDate NCHAR(8),

CreateDate NCHAR(8)

)

 

 

CREATE TABLE \#GetInOutLotStock

(

BizUnit INT,

WHSeq INT,

MainWHSeq INT,

LotNo NVARCHAR(30),

ItemSeq INT,

UnitSeq INT,

PrevQty DECIMAL(19,5),

InQty DECIMAL(19,5),

OutQty DECIMAL(19,5),

StockQty DECIMAL(19,5),

STDPrevQty DECIMAL(19,5),

STDInQty DECIMAL(19,5),

STDOutQty DECIMAL(19,5),

STDStockQty DECIMAL(19,5),

StockAmt DECIMAL(19,5)

)

 

 

INSERT INTO \#GetInOutItemLot (LotNo, ItemSeq, ValiDate, CreateDate)

SELECT DISTINCT L.LotNo, L.ItemSeq, L.ValiDate, L.CreateDate

FROM \_TDAItem AS A

JOIN \_TLGLotMaster AS L WITH(NOLOCK) ON L.CompanySeq = A.CompanySeq

AND L.ItemSeq = A.ItemSeq

JOIN \_TDAItemAsset AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.AssetSeq = A.AssetSeq

JOIN \_TDAItemStock AS F WITH(NOLOCK) ON F.CompanySeq = A.CompanySeq

AND F.ItemSeq = A.ItemSeq

WHERE A.CompanySeq = @CompanySeq

AND C.IsQty \<\> '1'

AND F.IsLotMng = '1'

 

-- 현재일 기준

DECLARE @CurrDate NCHAR(8)

SELECT @CurrDate = CONVERT(NCHAR(8), GETDATE(),112)

 

 

EXEC \_SWLGGetInOutLotStock @CompanySeq = @CompanySeq, -- 법인코드

@BizUnit = @BizUnit, -- 사업부문

@DateFr = @CurrDate, -- 조회기간Fr

@DateTo = @CurrDate, -- 조회기간To

@WHSeq = 0, -- 창고지정

@SMWHKind = 0, -- 창고구분별 조회

@IsSubDisplay = '1', -- 기능창고포함('1' : 기능창고별 조회, '0' : 사업단위창고에 포함)

@IsOnlyStock = '1', -- 재고만 표시('1' : 재고, '0' : 이월,입고,출고,재고)

@IsStockAmt = '0', -- 재고금액 계산

@StockAmtType = 0, -- 전월재고단가, 표준재고단가

@UnitType = 1, -- 1: 기준단위, 2: 수불단위, 3: 환산단위(단위별 조회)

@ConvUnitSeq = 0, -- 환산단위코드

@QryType = 'S', -- 'S': 1007 실재고, 'B':1008 자산포함, 'A':1011 가용재고

@IsTrustCust = '' -- 수탁여부

-- Lot 관리 품목 시 프로젝트별 합산

SELECT A.ItemSeq, A.LotNo, SUM(A.StockQty) AS StockQty

FROM \#GetInOutLotStock AS A

WHERE StockQty \<\> 0

GROUP BY A.ItemSeq, A.LotNo

ORDER BY A.ItemSeq

GO

EXEC SMC_SMESLGGetInOutLotStock
