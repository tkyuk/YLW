# BOM 예시

\<\<WITH문 sample.sql\>\>

 

CREATE TABLE \#TDAGood

(

GoodSeq INT,

GoodVer NVARCHAR(2),

Qty DECIMAL(19,5)

)

INSERT INTO \#TDAGood

SELECT 799, '01', 2

UNION

SELECT 727, '00', 100

 

;WITH CTS_TPDBOM(GoodSeq, GoodVer, ItemSeq, ItemVer, SubItemSeq, SubItemVer, NeedQty, Lvl) AS -- 재귀함수....

(

SELECT A.GoodSeq, A.GoodVer, B.ItemSeq, CONVERT(NVARCHAR(2), B.ItemBomRev), B.SubItemSeq, CONVERT(NVARCHAR(2), B.SubItemBomRev),

CONVERT(DECIMAL(19, 5), CASE WHEN NeedQtyDenominator = 0 THEN 0 ELSE NeedQtyNumerator / NeedQtyDenominator END \* A.Qty),

CONVERT(INT, 1)

FROM \#TDAGood A

JOIN \_TPDBOM AS B ON B.ItemSeq = A.GoodSeq AND B.ItemBomRev = A.GoodVer

UNION ALL

SELECT A.GoodSeq, GoodVer, B.ItemSeq, CONVERT(NVARCHAR(2), B.ItemBomRev), B.SubItemSeq, CONVERT(NVARCHAR(2), B.SubItemBomRev),

CONVERT(DECIMAL(19, 5), CASE WHEN NeedQtyDenominator = 0 THEN 0 ELSE NeedQtyNumerator / NeedQtyDenominator END \* A.NeedQty),

CONVERT(INT, Lvl + 1)

FROM CTS_TPDBOM AS A

JOIN \_TPDBOM AS B ON B.ItemSeq = A.SubItemSeq AND B.ItemBomRev = A.SubItemVer

)

SELECT A.\*,

B.ItemName AS GoodName, B.AssetSeq AS GoodAsset, C.ItemName AS ItemName, C.AssetSeq AS ItemAsset, D.ItemName AS SubItemName, D.AssetSeq AS SubItemAsset

INTO \#CTS_TPDBOM

FROM CTS_TPDBOM AS A

JOIN \_TDAItem AS B ON B.ItemSeq = A.GoodSeq

JOIN \_TDAItem AS C ON C.ItemSeq = A.ItemSeq

JOIN \_TDAItem AS D ON D.ItemSeq = A.SubItemSeq

WHERE D.AssetSeq NOT IN (2,4)

ORDER BY Lvl

 

SELECT \* FROM \#CTS_TPDBOM

 

 

 

DROP TABLE \#CTS_TPDBOM

DROP TABLE \#TDAGood
