# 키값 생성

SELECT @Count = COUNT(1) FROM \#BIZ_OUT_DataBlock1 WHERE WorkingTag = 'A' --@Count값수정(AND Status = 0 제외)

IF @Count \> 0

BEGIN

-- 키값생성코드부분 시작

EXEC @Seq = dbo.\_SWCOMCreateSeq @CompanySeq, 'MCNK_TWACLoanFund', 'LoanFundSeq', @Count

-- Temp Table 에 생성된 키값 UPDATE

UPDATE \#BIZ_OUT_DataBlock1

SET LoanFundSeq = @Seq + DataSeq

WHERE WorkingTag = 'A'

AND Status = 0

END
