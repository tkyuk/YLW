# 숫자 테이블 0 ~ 2047 (master..spt_values)

<span class="mark">숫자 출력</span>

 

SELECT number

FROM master..spt_values

WHERE type = 'P' AND number BETWEEN 0 AND 2047

 

 

<span class="mark">날짜 사이 모든 날짜 출력</span>

 

DECLARE @StartDate DATE = '2024-01-01'

DECLARE @EndDate DATE = '2024-12-31'

 

SELECT DATEADD(DAY, number, @StartDate) AS GeneratedDate

FROM master..spt_values

WHERE type = 'P'

AND number BETWEEN 0 AND DATEDIFF(DAY, @StartDate, @EndDate)

 

 

<span class="mark">알파벳 생성</span>

 

SELECT CHAR(number + 65) AS Letter

INTO \#TMP

FROM master..spt_values

WHERE type = 'P'

AND number BETWEEN 0 AND 25

 

 

SELECT \* FROM \#TMP
