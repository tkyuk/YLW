# ACE 진행연결

BEGIN TRAN

 

 

IF NOT EXISTS (SELECT TOP 1 1

FROM \_TCOMSourceDaily

WHERE CompanySeq = 1

AND ToTableSeq = 20

AND ToSeq         = 14329

AND FromTableSeq        = 18

AND FromSeq         = 15336

)

 

 

 

DECLARE @CompanySeq INT = 1

,@UserSeq INT = 1

 

 

 

CREATE TABLE \#SComSourceDailyBatch

(

ToTableName NVARCHAR(100),

ToSeq INT,

ToSerl INT,

ToSubSerl INT,

FromTableName NVARCHAR(100),

FromSeq INT,

FromSerl INT,

FromSubSerl INT,

ToQty DECIMAL(19,5),

ToStdQty DECIMAL(19,5),

ToAmt DECIMAL(19,5),

ToVAT DECIMAL(19,5),

ToDOMAmt DECIMAL(19,5),

ToDOMVAT DECIMAL(19,5),

FromQty DECIMAL(19,5),

FromSTDQty DECIMAL(19,5),

FromAmt DECIMAL(19,5),

FromVAT DECIMAL(19,5),

FromDOMAmt DECIMAL(19,5),

FromDOMVAT DECIMAL(19,5)

)

TRUNCATE TABLE \#SComSourceDailyBatch

 

INSERT INTO \#SComSourceDailyBatch

SELECT '\_TSLSalesItem', A.SalesSeq, A.SalesSerl, 0,

'\_TSLInvoiceItem', B.InvoiceSeq, B.InvoiceSerl, 0,

A.Qty, A.STDQty, A.CurAmt, A.CurVAT, A.DOMAmt, A.DOMVAT,

B.Qty, B.STDQty, B.CurAmt, B.CurVAT, B.DOMAmt, B.DOMVAT

FROM \_TSLSalesItem AS A

JOIN \_TSLInvoiceItem AS B ON B.CompanySeq = A.CompanySeq

AND B.ItemSeq = A.ItemSeq

AND A.SalesSeq = 14329

AND B.InvoiceSeq = 15336

 

--SELECT \* FROM \#SComSourceDailyBatch

 

-- 진행연결

EXEC \_SWComSourceDailyBatch 'A', @CompanySeq, @UserSeq

SELECT \*

AND FromSeq         = 15336

 

 

ROLLBACK TRAN
