# 미진행, 진행중, 완료, 중단

DROP PROC djb_SWSLOrderCfmReqQuery

GO

CREATE PROC djb_SWSLOrderCfmReqQuery

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

AS

DECLARE @BizUnit INT , -- 사업단위

@ReqDateFr NCHAR(8) , -- 요청일자Fr

@ReqDateTo NCHAR(8) , -- 요청일자To

@DeptSeq INT , -- 부서

@EmpSeq INT , -- 담당자

@ItemName NVARCHAR(100), -- 품명

@ItemNo NVARCHAR(100), -- 품번

@CustItemName NVARCHAR(100), -- 거래처품명

@CustSeq INT , -- 거래처코드

@UpperCustSeq INT , -- 청구처코드

@ProgressSeq INT , -- 확인여부

@SMIsCfm INT , -- 수주여부

@IsPUCfm INT -- 생산요청여부 추가 23.04.05 SJPARK

 

 

SELECT @BizUnit = ISNULL(BizUnit , 0),

@ReqDateFr = ISNULL(ReqDateFr ,''),

@ReqDateTo = ISNULL(ReqDateTo ,''),

@DeptSeq = ISNULL(DeptSeq , 0),

@EmpSeq = ISNULL(EmpSeq , 0),

@ItemName = ISNULL(ItemName ,''),

@ItemNo = ISNULL(ItemNo ,''),

@CustItemName = ISNULL(CustItemName,''),

@UpperCustSeq = ISNULL(UpperCustSeq, 0),

@CustSeq = ISNULL(CustSeq , 0),

@ProgressSeq = ISNULL(ProgressSeq , 0),

@SMIsCfm = ISNULL(SMIsCfm , 0),

@IsPUCfm = ISNULL(IsPUCfm , 0)

FROM \#BIZ_IN_DataBlock1 AS A

 

IF @ReqDateTo = '' SELECT @ReqDateTo = '99991231'

 

-- 기생산여부 임시테이블 (생산실적테이블에 등록된 품목 찾아오기)

CREATE TABLE \#TMP_WorkReport

(

ItemSeq INT,

IsYN NCHAR(1)

)

INSERT INTO \#TMP_WorkReport

SELECT DISTINCT A.ItemSeq, '1'

FROM djb_TSLOrderCfmReq AS A WITH(NOLOCK)

JOIN \_TPDSFCWorkReport AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.GoodItemSeq = A.ItemSeq

 

 

SELECT ISNULL(B.BizUnitName ,'') AS BizUnitName ,-- 사업단위

ISNULL(A.BizUnit , 0) AS BizUnit ,-- 사업단위코드

ISNULL(A.ReqDate ,'') AS ReqDate ,-- 요청일자

ISNULL(A.DVDate ,'') AS DVDate ,-- 출고일자

ISNULL(P.CustName ,'') AS RequestCustName ,-- 청구처

ISNULL(O.UpperCustSeq , 0) AS RequestCustSeq ,-- 청구처코드

ISNULL(C.CustName ,'') AS CustName ,-- 거래처

ISNULL(C.CustNo ,'') AS CustNo ,-- 거래처번호

ISNULL(A.CustSeq , 0) AS CustSeq ,-- 거래처코드

ISNULL(E.DeptName ,'') AS DeptName ,-- 부서

ISNULL(A.DeptSeq , 0) AS DeptSeq ,-- 부서코드

ISNULL(D.EmpName ,'') AS EmpName ,-- 담당자

ISNULL(A.EmpSeq , 0) AS EmpSeq ,-- 담당자코드

ISNULL(F.MinorName ,'') AS SMExpKindName ,-- 수출구분

ISNULL(A.SMExpKind , 0) AS SMExpKind ,-- 수출구분코드

ISNULL(F1.ValueText , 0) AS IsSMExpKind ,-- 내수/수출구분

ISNULL(G.MinorName ,'') AS UMOrderKindName ,-- 수주구분

ISNULL(A.UMOrderKind , 0) AS UMOrderKind ,-- 수주구분코드

ISNULL(H.ItemNo ,'') AS ItemNo ,-- 품번

ISNULL(H.ItemName ,'') AS ItemName ,-- 품명

ISNULL(H.Spec ,'') AS Spec ,-- 규격

ISNULL(A.ItemSeq , 0) AS ItemSeq ,-- 품목내부코드

> CASE WHEN ISNULL(N.CustItemName, '') = '' THEN N2.CustItemName

ELSE N.CustItemName END AS CustItemName,

CASE WHEN ISNULL(N.CustItemNo, '') = '' THEN N2.CustItemNo

ELSE N.CustItemNo END AS CustItemNo,

ISNULL(I.UnitName ,'') AS UnitName ,-- 단위

ISNULL(A.UnitSeq , 0) AS UnitSeq ,-- 단위코드

ISNULL(A.Qty , 0) AS Qty ,-- 수량

ISNULL(J.CurrName ,'') AS CurrName ,-- 화폐단위

ISNULL(A.CurrSeq , 0) AS CurrSeq ,-- 화폐단위코드

ISNULL(A.CurPrice , 0) AS Price ,-- 단가

ISNULL(A.Remark ,'') AS Remark ,-- 비고

ISNULL(A.Dummy1 ,'') AS Dummy1 ,-- Packing비고

ISNULL(A.Dummy2 ,'') AS Dummy2 ,-- 배송비고

ISNULL(A.Dummy3 ,'') AS Dummy3 ,-- 비고3

ISNULL(A.Dummy4 ,'') AS Dummy4 ,-- 비고4

ISNULL(A.Dummy5 ,'') AS Dummy5 ,-- 비고5

ISNULL(A.Dummy6 ,'') AS Dummy6 ,-- 비고6

ISNULL(A.Dummy7 , 0) AS Dummy7 ,-- 비고7

ISNULL(A.Dummy8 , 0) AS Dummy8 ,-- 비고8

ISNULL(A.Dummy9 , 0) AS Dummy9 ,-- 비고9

ISNULL(A.Dummy10 , 0) AS Dummy10 ,-- 비고10

ISNULL(A.IsCfm ,'') AS IsCfm ,-- 확인

ISNULL(A.CfmRemark ,'') AS CfmRemark ,-- 확인비고

A.CfmDate AS CfmDate ,-- 확인일자

ISNULL(U1.UserName ,'') AS CfmUserName ,-- 확인자

ISNULL(A.CfmUserSeq , 0) AS CfmUserSeq ,-- 확인자코드

ISNULL(A.IsStop ,'') AS IsStop ,-- 중단여부

A.StopDate AS StopDate ,-- 중단일

ISNULL(U2.UserName ,'') AS StopUserName ,-- 중단자

ISNULL(A.StopUserSeq , 0) AS StopUserSeq ,-- 중단자코드

ISNULL(R.OrderQty , 0) AS OrderQty ,-- 수주수량

ISNULL(A.Qty,0) - ISNULL(R.OrderQty,0) AS NotQty ,-- 미진행수량

A.RegDateTime AS RegDateTime ,-- 등록일자

ISNULL(U3.UserName ,'') AS RegUserName ,-- 등록자

ISNULL(A.RegUserSeq , 0) AS RegUserSeq ,-- 등록자코드

A.LastDateTime AS LastDateTime ,-- 수정일자

ISNULL(U4.UserName ,'') AS LastUserName ,-- 수정자

ISNULL(A.LastUserSeq , 0) AS LastUserSeq ,-- 수정자코드

ISNULL(A.ReqSeq , 0) AS ReqSeq ,-- 내부코드

ISNULL(Q.UnitName ,'') AS STDUnitName ,-- 재고단위

ISNULL(A.STDUnitSeq , 0) AS STDUnitSeq ,-- 재고단위코드

ISNULL(A.STDQty , 0) AS STDQty ,-- 재고단위수량

ISNULL(S.IsYN ,'') AS IsWorkReport ,-- 기생산여부 ( 품목기준 생산실적등록여부 )

ISNULL(A.IsInclusedVAT ,'') AS IsInclusedVAT ,-- 부가세포함 -- 23.03.02 SJPARK

ISNULL(A.VATRate , 0) AS VATRate ,-- 부가세율 -- 23.03.02 SJPARK

ISNULL(T.DVReqQty , 0) AS DVReqQty ,-- 출하의뢰수량

ISNULL(U.ProdReqQty , 0) AS ProdReqQty -- 생산의뢰수량

FROM djb_TSLOrderCfmReq AS A

LEFT OUTER JOIN \_TDABizUnit AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

> AND B.BizUnit = A.BizUnit

LEFT OUTER JOIN \_TDACust AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

> AND C.CustSeq = A.CustSeq

LEFT OUTER JOIN \_TDAEmp AS D WITH(NOLOCK) ON D.CompanySeq = A.CompanySeq

> AND D.EmpSeq = A.EmpSeq

LEFT OUTER JOIN \_TDADept AS E WITH(NOLOCK) ON E.CompanySeq = A.CompanySeq

> AND E.DeptSeq = A.DeptSeq

LEFT OUTER JOIN \_TDASMinor AS F WITH(NOLOCK) ON F.CompanySeq = A.CompanySeq

> AND F.MinorSeq = A.SMExpKind

LEFT OUTER JOIN \_TDASMinorValue AS F1 WITH(NOLOCK) ON F1.CompanySeq = F.CompanySeq

AND F1.MinorSeq = F.MinorSeq

AND F1.Serl = 1001

AND F.MajorSeq = 8009

LEFT OUTER JOIN \_TDAUMinor AS G WITH(NOLOCK) ON G.CompanySeq = A.CompanySeq

> AND G.MinorSeq = A.UMOrderKind

LEFT OUTER JOIN \_TDAItem AS H WITH(NOLOCK) ON H.CompanySeq = A.CompanySeq

> AND H.ItemSeq = A.ItemSeq

LEFT OUTER JOIN \_TDAUnit AS I WITH(NOLOCK) ON I.CompanySeq = A.CompanySeq

> AND I.UnitSeq = A.UnitSeq

LEFT OUTER JOIN \_TDACurr AS J WITH(NOLOCK) ON J.CompanySeq = A.CompanySeq

> AND J.CurrSeq = A.CurrSeq

LEFT OUTER JOIN \_TDAEmp AS K WITH(NOLOCK) ON K.CompanySeq = A.CompanySeq

> AND K.EmpSeq = A.CfmUserSeq

LEFT OUTER JOIN \_TDAEmp AS L WITH(NOLOCK) ON L.CompanySeq = A.CompanySeq

> AND L.EmpSeq = A.StopUserSeq

LEFT OUTER JOIN \_TDAEmp AS M WITH(NOLOCK) ON M.CompanySeq = A.CompanySeq

> AND M.EmpSeq = A.RegUserSeq

LEFT OUTER JOIN \_TSLCustItem         AS N WITH(NOLOCK) ON N.CompanySeq = A.CompanySeq

>                                    AND N.ItemSeq         = A.ItemSeq
>
>                                    AND N.CustSeq         = A.CustSeq
>
>                                    AND N.UnitSeq         = A.UnitSeq

LEFT OUTER JOIN \_TSLCustItem AS N2 WITH(NOLOCK) ON A.CompanySeq = N2.CompanySeq

AND A.ItemSeq         = N2.ItemSeq

AND A.CustSeq         = N2.CustSeq

AND N2.UnitSeq = 0

LEFT OUTER JOIN \_TDACustGroup AS O WITH(NOLOCK) ON O.CompanySeq = A.CompanySeq

> AND O.CustSeq = A.CustSeq

AND O.UMCustGroup = 8014002

LEFT OUTER JOIN \_TDACust AS P WITH(NOLOCK) ON P.CompanySeq = O.CompanySeq

> AND P.CustSeq = O.UpperCustSeq

LEFT OUTER JOIN \_TDAUnit AS Q WITH(NOLOCK) ON Q.CompanySeq = A.CompanySeq

> AND Q.UnitSeq = A.STDUnitSeq

LEFT OUTER JOIN \_TCAUser AS U1 WITH(NOLOCK) ON U1.CompanySeq = A.CompanySeq

> AND U1.UserSeq = A.CfmUserSeq

LEFT OUTER JOIN \_TCAUser AS U2 WITH(NOLOCK) ON U2.CompanySeq = A.CompanySeq

> AND U2.UserSeq = A.StopUserSeq

LEFT OUTER JOIN \_TCAUser AS U3 WITH(NOLOCK) ON U3.CompanySeq = A.CompanySeq

> AND U3.UserSeq = A.RegUserSeq

LEFT OUTER JOIN \_TCAUser AS U4 WITH(NOLOCK) ON U4.CompanySeq = A.CompanySeq

> AND U4.UserSeq = A.LastUserSeq

LEFT OUTER JOIN \#TMP_WorkReport AS S WITH(NOLOCK) ON @CompanySeq = A.CompanySeq

> AND S.ItemSeq = A.ItemSeq

 

 

OUTER APPLY (SELECT SUM(Z.ToQty) AS OrderQty

FROM \_TCOMSourceDailyTrace AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = A.CompanySeq

AND Z.FromSeq = A.ReqSeq

--AND Z.FromSerl = A.

AND Z.FromTableSeq = 15000063 -- 출고가능확인요청_djb

AND Z.ToTableSeq = 19 -- 수주

) AS R

OUTER APPLY (SELECT SUM(Z.ToQty) AS DVReqQty

FROM \_TCOMSourceDailyTrace AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = A.CompanySeq

AND Z.FromSeq = A.ReqSeq

--AND Z.FromSerl = A.

AND Z.FromTableSeq = 15000063 -- 출고가능확인요청_djb

AND Z.ToTableSeq = 16 -- 출하의뢰

) AS T

OUTER APPLY (SELECT SUM(Z.ToQty) AS ProdReqQty

FROM \_TCOMSourceDailyTrace AS Z WITH(NOLOCK)

WHERE Z.CompanySeq = A.CompanySeq

AND Z.FromSeq = A.ReqSeq

--AND Z.FromSerl = A.

AND Z.FromTableSeq = 15000063 -- 출고가능확인요청_djb

AND Z.ToTableSeq = 1 -- 생산의뢰

) AS U

WHERE A.CompanySeq = @CompanySeq

AND ((@BizUnit = 0) OR (@BizUnit = A.BizUnit ))

AND ((@ReqDateFr = '') OR (A.ReqDate \>= @ReqDateFr ))

AND ((@ReqDateTo = '') OR (A.ReqDate \<= @ReqDateTo ))

AND ((@DeptSeq = 0) OR (@DeptSeq = A.DeptSeq ))

AND ((@EmpSeq = 0) OR (@EmpSeq = A.EmpSeq ))

AND ((@ItemName = '') OR (H.ItemName LIKE @ItemName + '%'))

AND ((@ItemNo = '') OR (H.ItemNo LIKE @ItemNo + '%'))

AND ((@CustItemName = '') OR (N.CustItemName LIKE @CustItemName + '%'))

AND ((@CustSeq = 0) OR (@CustSeq = A.CustSeq ))

AND ((@UpperCustSeq = 0) OR (@UpperCustSeq = O.UpperCustSeq))

AND (@ProgressSeq = 0 OR -- 공백 / 요청 / 확인

(@ProgressSeq = 1 AND A.IsCfm \<\> '1') OR

(@ProgressSeq = 2 AND A.IsCfm = '1'))

AND (@SMIsCfm = 0 OR -- 공백 / 미진행 / 진행중 / 완료/ 중단

(@SMIsCfm = 6052001 AND R.OrderQty IS NULL) OR

(@SMIsCfm = 6052002 AND ISNULL(A.Qty,0) - ISNULL(R.OrderQty,0) \> 0 AND R.OrderQty\>0) OR

(@SMIsCfm = 6052003 AND ISNULL(A.Qty,0) - ISNULL(R.OrderQty,0) = 0) OR

(@SMIsCfm = 6052004 AND A.IsStop = 1))

 

AND (@IsPUCfm = 0 OR -- 공백 / 미진행 / 진행중 / 완료

(@IsPUCfm = 1 AND U.ProdReqQty IS NULL) OR

(@IsPUCfm = 2 AND ISNULL(A.Qty,0) - ISNULL(U.ProdReqQty,0) \> 0 AND U.ProdReqQty\>0) OR

(@IsPUCfm = 3 AND ISNULL(A.Qty,0) - ISNULL(U.ProdReqQty,0) = 0) )

--SELECT \* FROM \_TDASMinor WHERE MajorSeq = 6052

ORDER BY A.ReqSeq

 

 

 

 

RETURN
