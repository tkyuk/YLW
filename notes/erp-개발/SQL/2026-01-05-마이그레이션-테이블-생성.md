# 마이그레이션 테이블 생성

IF EXISTS (SELECT \* FROM SYSOBJECTS WHERE Name = 'svs_TSIASServiceHist' AND xtype = 'U' )

DROP TABLE svs_TSIASServiceHist

 

CREATE TABLE svs_TSIASServiceHist

(

Seq INT NOT NULL,

ReceiptDeptName NVARCHAR(100) NOT NULL,

ReceiptDate NCHAR(8) NOT NULL,

ProcDate NCHAR(8) NOT NULL,

ReceiptEmpName NVARCHAR(100) NOT NULL,

ProcEmpName NVARCHAR(100) NOT NULL,

CustomerName NVARCHAR(100) NOT NULL,

ItemNo NVARCHAR(200) NOT NULL,

ItemName NVARCHAR(200) NOT NULL,

Spec NVARCHAR(200) NOT NULL,

BizUnitName NVARCHAR(100) NOT NULL,

SerialNo NVARCHAR(100) NOT NULL,

ServiceNo NVARCHAR(100) NOT NULL,

ReceiptDesc NVARCHAR(MAX) NOT NULL,

Reason NVARCHAR(MAX) NOT NULL,

ProcDesc NVARCHAR(MAX) NOT NULL,

ServiceStatus NVARCHAR(100) NOT NULL,

OutDate NCHAR(8) NOT NULL,

LaborChargeType DECIMAL(19,5) NOT NULL,

LaborAmt DECIMAL(19,5) NOT NULL,

PartChargeType NVARCHAR(100) NOT NULL,

PartAmt DECIMAL(19,5) NOT NULL,

Remark NVARCHAR(MAX) NOT NULL,

CONSTRAINT PKsvs_TSIASServiceHist PRIMARY KEY CLUSTERED (Seq ASC)

)

 

CREATE INDEX IX_svs_TSIASServiceHist_ReceiptDate ON svs_TSIASServiceHist (ReceiptDate)

CREATE INDEX IX_svs_TSIASServiceHist_SerialNo ON svs_TSIASServiceHist (SerialNo)

CREATE INDEX IX_svs_TSIASServiceHist_ServiceNo ON svs_TSIASServiceHist (ServiceNo)

CREATE INDEX IX_svs_TSIASServiceHist_CustomerName ON svs_TSIASServiceHist (CustomerName)
