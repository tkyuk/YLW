# 추가정보정의

SELECT \* FROM \_TDAUMinor WHERE MajorSeq = 2101110

 

-- 사용자정의코드로 찾기 -\> ValueSeq =\> 테이블 내부 키값

SELECT ValueSeq,\* FROM \_TDAUMinorValue WHERE MajorSeq = 2101110

 

-- DefineUnitSeq = MajorSeq / CodeHelpConst =\> 테이블 확인 가능

SELECT DefineUnitSeq, CodeHelpConst,\* FROM \_TCOMUserDefine WHERE DefineUnitSeq = 2101110

 

SELECT CodeHelpSeq,CodeHelpTitle, TableName,\* FROM \_TCACodeHelpData WHERE CodeHelpSeq = 10006

 

SELECT \* FROM \_TDAWH WHERE WHSeq IN(1027,6)
