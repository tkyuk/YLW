--=========================

 

-- (INPUT) 사용자정의코드 대분류

CREATE TABLE \#TDAUMajorLanuage(

MajorSeq INT

)

 

-- (OUTPUT) 사용자정의코드 소분류_사전

CREATE TABLE \#TDAUMinorLanguage(

CompanySeq INT ,

MajorSeq INT ,

MinorSeq INT ,

MinorName NVARCHAR(200)

)

 

INSERT INTO \#TDAUMajorLanuage(MajorSeq)

SELECT 4001 -- 비용구분

UNION ALL SELECT 4024 -- 연료종류

 

 

EXEC dbo.\_SWUTGetUMinorLanguage @CompanySeq = @CompanySeq

, @LanguageSeq = @LanguageSeq

)

CompanySeq INT
