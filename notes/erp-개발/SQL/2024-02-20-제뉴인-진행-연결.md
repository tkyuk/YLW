# 제뉴인 진행 연결

-- 품목 중복인지 체크

BEGIN TRAN

 

DECLARE @CompanySeq INT = 1

,@UserSeq INT = 1

CREATE TABLE \#SComSourceDailyBatch

(

ToTableName NVARCHAR(100),

ToSeq INT,

ToSerl INT,

ToSubSerl INT,

FromTableName NVARCHAR(100),

FromSeq INT,

FromSerl INT,

FromSubSerl INT,

ToQty DECIMAL(19,5),

ToStdQty DECIMAL(19,5),

ToAmt DECIMAL(19,5),

ToVAT DECIMAL(19,5),

FromQty DECIMAL(19,5),

FromSTDQty DECIMAL(19,5),

FromAmt DECIMAL(19,5),

FromVAT DECIMAL(19,5)

)

TRUNCATE TABLE \#SComSourceDailyBatch

-- 진행연결(구매발주 =\> 구매납품)

INSERT INTO \#SComSourceDailyBatch

SELECT '\_TSLSalesItem' AS ToTableName, B.SalesSeq AS ToSeq, B.SalesSerl AS ToSerl, 0,

'\_TSLInvoiceItem' AS FromTableName, A.InvoiceSeq AS FromSeq, A.InvoiceSerl AS FromSerl, 0,

B.Qty, B.Qty, B.CurAmt, B.CurVAT,

A.Qty, A.STDQty, A.CurAmt, A.CurVAT

FROM \_TSLInvoiceItem AS A WITH(NOLOCK)

JOIN \_TSLSalesItem AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.ItemSeq = A.ItemSeq

AND B.SalesSeq = 32414

AND A.InvoiceSeq = 32556

SELECT \* FROM \#SComSourceDailyBatch

 

-- 진행연결

EXEC \_SComSourceDailyBatch 'A', @CompanySeq, @UserSeq

 

SELECT \*

FROM \_TCOMSourceDaily AS A

JOIN \#SComSourceDailyBatch AS B ON B.FromSeq = A.FromSeq

AND B.FromSerl = A.FromSerl

AND B.ToSeq = A.ToSeq

AND B.ToSerl = A.ToSerl

 

ROLLBACK TRAN
