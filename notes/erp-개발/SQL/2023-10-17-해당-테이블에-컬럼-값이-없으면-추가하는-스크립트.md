# 해당 테이블에 컬럼 값이 없으면 추가하는 스크립트

DECLARE @pID INT

SELECT @pID = id FROM Sysobjects WHERE name = 'mcnk_TPDBOMECO'

 

IF NOT EXISTS(SELECT NAME FROM SYSCOLUMNS WHERE id = @pID AND name = 'ECOTitle_New')

BEGIN

SELECT 1

ALTER TABLE mcnk_TPDBOMECO ADD ECOTitle_New NVARCHAR(100)

END

 

IF NOT EXISTS (SELECT \* FROM sys.columns WHERE object_id = OBJECT_ID(N'\[dbo\].\[das_TARBizTripCostPJTAmt\]') AND name='ENDDateKind')

BEGIN

ALTER TABLE das_TARBizTripCostPJTAmt ADD ENDDateKind INT

END
