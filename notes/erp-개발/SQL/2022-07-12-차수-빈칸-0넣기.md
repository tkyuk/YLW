# 차수 / 빈칸 0넣기

RIGHT('00' + CAST(B.PlanYearAmd + 1 AS NVARCHAR), 2)

 

 

 

-- 계획연도에 해당하는 회계 계산월 가져오기

UPDATE \#BIZ_OUT_DataBlock1

SET EndYM = CASE WHEN @SttlMonth BETWEEN 1 AND 6 THEN CONVERT(NCHAR(4),CONVERT(INT,PlanYear) + 1) + RIGHT('00' + CAST(@SttlMonth AS NVARCHAR), 2)

ELSE PlanYear + RIGHT('00' + CAST(@SttlMonth AS NVARCHAR), 2) END

FROM \#BIZ_OUT_DataBlock1 AS A

WHERE WorkingTag IN ('A','U')

AND Status = 0
