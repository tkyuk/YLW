SELECT @XmlData1 = ''

SELECT @XmlData1 = @XmlData1 +

REPLACE(CONVERT(NVARCHAR(MAX), A.xmlData),

'\<A',

'\<A LogType="' + A.LogType+ '" ' +

'LogSeq="' + CONVERT(NVARCHAR(100), ISNULL(A.LogSeq, 0)) + '" ' +

'LogUserSeq="' + CONVERT(NVARCHAR(100), ISNULL(A.LogUserSeq, 0)) + '" ' +

'LogPgmSeq="' + CONVERT(NVARCHAR(100), ISNULL(A.LogPgmSeq , 0)) + '" ' +

'LogDateTime="' + CONVERT(NVARCHAR(100), A.LogDateTime, 120) + '" ')
