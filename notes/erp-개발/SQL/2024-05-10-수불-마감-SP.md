# 수불 마감 SP

-- 수불 마감

CREATE TABLE \#TLGClosingCheck

(

IDX_NO INT ,

InOutYM NCHAR(6) ,

ItemSeq INT ,

WHSeq INT ,

MessageType INT ,

Status INT ,

Result NVARCHAR(255)

)

 

INSERT INTO \#TLGClosingCheck

(

IDX_NO, InOutYM, ItemSeq, WHSeq,

MessageType, Status, Result

)

SELECT A.IDX_NO, LEFT(A.DelvDate,6), A.ItemSeq, A.WHSeq,

0 AS MessageType, 0 AS Status, '' AS Result

FROM \#TDelvItem AS A

WHERE A.Status = 0

 

EXEC \_SWLGClosingCheck @CompanySeq, @LanguageSeq

 

UPDATE A

SET Result = B.Result,

MessageType = B.MessageType,

Status = B.Status

FROM \#TDelvItem AS A

JOIN \#TLGClosingCheck AS B WITH(NOLOCK) ON A.IDX_NO = B.IDX_NO

WHERE A.Status = 0

AND B.Status \<\> 0

DROP TABLE \#TLGClosingCheck
