# 특정 테이블 쓰는 sp 찾기

-- 특정 테이블 쓰는 sp 찾기

SELECT

DISTINCT O.NAME

FROM

SYSDEPENDS AS D

INNER JOIN

SYSOBJECTS AS O

ON

D.ID = O.ID

WHERE

O.XTYPE = 'P'

AND D.DEPID =

(

SELECT

ID

FROM

SYSOBJECTS

WHERE

NAME = '\_TXPRWkItemBasicInfo'

)
