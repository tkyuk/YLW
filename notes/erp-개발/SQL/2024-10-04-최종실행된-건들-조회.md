# 최종실행된 건들 조회

SELECT B.UserName,A.CrDateTime,\*

FROM OTCANoticeMessage AS A

LEFT OUTER JOIN \_TCAUser AS B ON B.UserSeq = A.RecvUserSeq

JOIN (SELECT MAX(convert(varchar, CrDateTime, 120)) AS CrDateTime FROM OTCANoticeMessage) AS C ON C.CrDateTime=convert(varchar, A.CrDateTime, 120)

ORDER BY MessageID DESC
