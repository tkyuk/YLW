# 실제 테이블에 채번 테이블 키값보다 큰 값 존재하는 경우 MAX 키 값 업데이트

25.09.03

 

IF EXISTS (SELECT 1 FROM \_TDAItem WITH(NOLOCK) WHERE CompanySeq = @CompanySeq AND ItemSeq \> @Seq)

BEGIN

SELECT @Seq = (SELECT MAX(ItemSeq) FROM \_TDAItem WITH(NOLOCK) WHERE CompanySeq = @CompanySeq)

UPDATE \_TCOMCreateSeqMax

SET MaxSeq = @Seq + @Count,

WorkCount = @Count

WHERE CompanySeq = @CompanySeq AND TableName = '\_TDAItem'

END

 

IF EXISTS (SELECT 1 FROM \_TPDQCTestReport WITH(NOLOCK) WHERE CompanySeq = @CompanySeq AND QCSeq \>= @QcSeq)

BEGIN

SELECT @QcSeq = (SELECT MAX(QCSeq) + 1 FROM \_TPDQCTestReport WITH(NOLOCK) WHERE CompanySeq = @CompanySeq)

UPDATE \_TCOMCreateSeqMax

SET MaxSeq = @QcSeq + 1

WHERE CompanySeq = @CompanySeq

AND TableName = '\_TPDQCTestReport'

END
