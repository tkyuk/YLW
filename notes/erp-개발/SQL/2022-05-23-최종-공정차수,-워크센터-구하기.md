# 최종 공정차수, 워크센터 구하기

UPDATE \#TMESWorkReport

SET ProcRev = B.ProcRev

FROM \#TMESWorkReport AS A JOIN (SELECT A.ItemSeq, MAX(B.ProcRev) AS ProcRev

FROM \#TMESWorkReport AS A JOIN \_TPDROUItemProcRevFactUnit AS B ON B.CompanySeq = @CompanySeq

AND A.BizUnit = B.BizUnit

AND A.ItemSeq = B.ItemSeq

GROUP BY A.ItemSeq) AS B ON A.ItemSeq = B.ItemSeq

WHERE ISNULL(A.WorkOrderSeq,0) = 0

 

UPDATE \#TMESWorkReport

SET ProcSeq = B.ProcSeq,

WorkCenterSeq = C.WorkcenterSeq

FROM \#TMESWorkReport AS A JOIN (SELECT A.ItemSeq, B.ProcRev, MAX(B.ProcSeq) AS ProcSeq

FROM \#TMESWorkReport AS A JOIN \_TPDROUItemProc AS B ON B.CompanySeq = @CompanySeq

AND A.BizUnit = B.BizUnit

AND A.ItemSeq = B.ItemSeq

AND A.ProcRev = B.ProcRev

WHERE ISNULL(A.WorkOrderSeq,0) = 0

GROUP BY A.ItemSeq, B.ProcRev) AS B ON A.ItemSeq = B.ItemSeq

AND A.ProcRev = B.ProcRev

JOIN \_TPDROUItemProcWC AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

AND A.BizUnit = C.BizUnit

AND A.ItemSeq = C.ItemSeq

AND A.ProcRev = C.ProcRev

AND B.ProcSeq = C.ProcSeq

WHERE ISNULL(A.WorkOrderSeq,0) = 0
