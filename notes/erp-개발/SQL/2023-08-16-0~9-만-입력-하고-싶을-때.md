# 0~9 만 입력 하고 싶을 때

--SELECT TRY_PARSE('13423432N4234' AS INT)

--SELECT PARSE('13423432N4234' AS INT)

SELECT ISNUMERIC ('\\)

 

CREATE TABLE \#TMP(

A NVARCHAR(100)

)

 

INSERT INTO \#TMP VALUES

('0'),

('1'),

('100'),

('+'),

('++'),

('-'),

('\\)

 

SELECT CASE WHEN A IN('+','-') THEN 0 ELSE CASE WHEN TRY_CAST(A AS INT) IS NOT NULL THEN 1 ELSE 0 END END , A,TRY_CAST(A AS INT)

FROM \#TMP

 

DROP TABLE \#TMP

ISNUMERIC 은 +,-, \\\$ 도 숫자로 인식

 

=\> CASE WHEN TRY_CAST(A AS INT) IS NOT NULL THEN 1 ELSE 0 END 사용

=\> 이 경우에 +,-는 여전히 숫자로 인식

=\> 별도로 CASE 문을 더 써줘야함
