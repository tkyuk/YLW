# 생산 JOIN

SELECT A.WorkOrderSeq,A.WorkOrderSerl,B.ProdPlanSeq,C.ProdReqSeq,D.Serl

FROM \_TPDSFCWorkOrder AS A WITH(NOLOCK)

JOIN \_TPDMPSDailyProdPlan AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.ProdPlanSeq = A.ProdPlanSeq

JOIN \_TPDMPSProdReq AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.ProdReqSeq = B.ProgFromSeq

AND B.ProgFromTableSeq = 1

JOIN \_TPDMPSProdReqItem AS D WITH(NOLOCK) ON D.CompanySeq = B.CompanySeq

AND D.ProdReqSeq = B.ProgFromSeq

AND D.Serl = B.ProgFromSerl

 

 

WHERE C.ProdReqSeq =35
