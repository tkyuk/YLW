@ServiceSeq INT = 0,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

CREATE TABLE \#Tmp_Result

--DelvSeq INT ,

UPDATE \#BIZ_IN_DataBlock1

SET Dummy10 = 1

WHERE Dummy10 = 0

 

DECLARE @Start INT ,

@End INT ,

@Loop INT

 

SELECT @Start = MIN(A.DataSeq),

@End = MAX(A.DataSeq)

FROM \#BIZ_IN_DataBlock1 AS A

--반복처리

WHILE (@Start \<= @End)

BEGIN

SELECT @Loop = Dummy10

FROM \#BIZ_IN_DataBlock1 AS A

WHERE A.DataSeq = @Start

WHILE (@Loop \> 0)

BEGIN

INSERT INTO \#Tmp_Result

SELECT G.ItemName AS ItemName , -- 품명

G.ItemNo AS ItemNo , -- 품번

B.LotNo AS LotNo , -- LotNo

A.DelvDate AS DelvDate , -- 납품일

B.Dummy5 AS Dummy5 , -- 사용기한

C.CustName AS CustName , -- 납품처

B.Dummy6 AS Dummy6 -- 제조처

--B.DelvSeq AS DelvSeq , --

--B.DelvSerl AS DelvSerl , --

--B.Dummy10 AS Dummy10 -- 라벨 출력 횟수

FROM \#BIZ_IN_DataBlock1 AS Z

JOIN \_TPUDelv AS A WITH(NOLOCK) ON A.CompanySeq = @CompanySeq

AND A.DelvSeq = Z.DelvSeq

JOIN \_TPUDelvItem AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.DelvSeq = Z.DelvSeq

AND B.DelvSerl = Z.DelvSerl

LEFT OUTER JOIN \_TDACust AS C WITH(NOLOCK) ON A.CompanySeq = C.CompanySeq

AND A.CustSeq = C.CustSeq

LEFT OUTER JOIN \_TDAItem AS G WITH(NOLOCK) ON B.CompanySeq = G.CompanySeq

AND B.ItemSeq = G.ItemSeq

WHERE A.CompanySeq = @CompanySeq

AND Z.DataSeq = @Start

ORDER BY A.DelvDate, A.DelvNo, B.DelvSeq, B.DelvSerl, G.ItemName, G.ItemNo, B.LOTNo

SELECT @Loop = @Loop - 1

END

SELECT @Start = @Start +1

END

INSERT INTO \#Result(NUM, InvoiceSeq, ItemSeq)

SELECT CASE WHEN (InvNUM % 8) = 0 THEN (FLOOR(InvNUM/8) -1) ELSE FLOOR(InvNUM/8) end +1 , InvoiceSeq, ItemSeq

FROM \#TEMP

 

UPDATE \#Result

SET MaxNUM = A.minValue

FROM (SELECT MAX(NUM) AS minValue, InvoiceSeq FROM \#Result AS B GROUP BY InvoiceSeq) AS A

WHERE A.InvoiceSeq = \#Result.InvoiceSeq

 

CREATE TABLE \#ResultSUM

(

Serl INT IDENTITY(1,1),

InvoiceSeq INT ,

TotCurAmt DECIMAL(19,5),

TotVAT DECIMAL(19,5),

TotQty DECIMAL(19,5),

TotAmt DECIMAL(19,5),

TotDomAmt DECIMAL(19,5),

ABSSumTot DECIMAL(19,5),

HanAmt         NVARCHAR(MAX)

)

INSERT INTO \#ResultSUM

SELECT B.InvoiceSeq,

SUM(B.CurAmt) AS TotCurAmt,

SUM(B.CurVAT) AS TotVAT ,

SUM(B.Qty) AS TotQty ,

ISNULL(SUM(B.CurAmt), 0) + ISNULL(SUM(B.CurVAT), 0) AS TotAmt ,

ISNULL(SUM(B.DomAmt), 0) + ISNULL(SUM(B.DomVAT), 0) AS TotDomAmt,

ABS(ISNULL(SUM(B.CurAmt), 0) + ISNULL(SUM(B.CurVAT), 0)) AS ABSSumTot,

''

FROM \#BIZ_IN_DataBlock1 AS A JOIN \_TSLInvoiceItem AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND A.InvoiceSeq = B.InvoiceSeq

GROUP BY B.InvoiceSeq

 

DECLARE @Start INT ,

@End INT ,

@ABSSumTot DECIMAL(19,5)

SELECT @Start = MIN(A.Serl),

@End = MAX(A.Serl)

FROM \#ResultSUM AS A

 

--반복처리

WHILE (@Start \<= @End)

BEGIN

 

SELECT @ABSSumTot = ABSSumTot

FROM \#ResultSUM AS A

WHERE A.Serl = @Start

 

> EXEC \_SWDAGetAmtHan @ABSSumTot , @HanAmt OUTPUT

 

UPDATE \#ResultSUM

SET HanAmt = @HanAmt

FROM \#ResultSUM AS A

WHERE A.Serl = @Start

 

 

SELECT @Start = @Start +1,

@HanAmt = 0

END
