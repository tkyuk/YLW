# 트리코드 부여

-- 1. \_TPJTWBS 원본 데이터를 담을 임시 테이블 생성 (실제 SP에서는 메인 임시 테이블일 수 있음)

-- SortNum_New 컬럼을 정렬 가능한 형태(INT, BIGINT 혹은 NVARCHAR)로 추가합니다.

-- 여기서는 최대 9자리 이상의 값이 나올 수 있으므로 BIGINT를 권장합니다.

CREATE TABLE \#Temp_TPJTWBS

(

CompanySeq INT NOT NULL,

PJTSeq INT NOT NULL,

> WBSSeq INT NOT NULL,

SortNum NVARCHAR(100) NOT NULL, -- 원본 SortNum

SortNum_New BIGINT NULL -- 변환된 정렬용 SortNum

);

 

-- 예시 데이터 삽입 (테스트용)

INSERT INTO \#Temp_TPJTWBS (CompanySeq, PJTSeq,WBSSeq, SortNum)

SELECT PJTSeq FROM \_TPJTWBS WHERE SortNum LIKE '%.%'

 

SELECT CompanySeq, PJTSeq,WBSSeq, SortNum FROM \_TPJTWBS WHERE PJTSEQ IN (SELECT PJTSeq FROM \_TPJTWBS WHERE SortNum LIKE '%.%') ORDER BY 1,2,3,4

 

-- 2. SortNum 변환 로직 적용 및 SortNum_New 업데이트 (4단계 구조, 2자리 패딩)

UPDATE T1

SET

T1.SortNum_New =

-- 최종적으로 BIGINT로 변환합니다. (TRY_CAST로 변환 오류 방지)

TRY_CAST(

(

-- \[1단계 (X) 추출\]: 패딩 없이 원래 길이 그대로 사용

ISNULL(

CASE WHEN CHARINDEX('.', T1.SortNum) = 0 THEN T1.SortNum

ELSE LEFT(T1.SortNum, CHARINDEX('.', T1.SortNum) - 1)

END, N'0'

)

)

\+

(

-- \[2단계 (YY) 추출\]: 첫 마침표 이후. 정렬 유지를 위해 2자리 패딩 적용.

RIGHT('00' + -- 3자리 대신 2자리 패딩

CASE

WHEN CHARINDEX('.', T1.SortNum) = 0 THEN N''

ELSE

SUBSTRING(

T1.SortNum,

CHARINDEX('.', T1.SortNum) + 1,

CASE

WHEN CHARINDEX('.', T1.SortNum, CHARINDEX('.', T1.SortNum) + 1) = 0 THEN 1000

ELSE CHARINDEX('.', T1.SortNum, CHARINDEX('.', T1.SortNum) + 1) - CHARINDEX('.', T1.SortNum) - 1

END

)

END, 2) -- 3자리 대신 2자리 추출

)

\+

-- \[3단계 (ZZZ) 추출\]: 마침표가 2개 이상일 때만 결합. 2자리 패딩 적용.

CASE

WHEN LEN(T1.SortNum) - LEN(REPLACE(T1.SortNum, '.', N'')) \>= 2 THEN

RIGHT('00' + -- 3자리 대신 2자리 패딩

SUBSTRING(

T1.SortNum,

CHARINDEX('.', T1.SortNum, CHARINDEX('.', T1.SortNum) + 1) + 1,

CASE

WHEN CHARINDEX('.', T1.SortNum, CHARINDEX('.', T1.SortNum, CHARINDEX('.', T1.SortNum) + 1) + 1) = 0 THEN 1000

ELSE CHARINDEX('.', T1.SortNum, CHARINDEX('.', T1.SortNum, CHARINDEX('.', T1.SortNum) + 1) + 1) - CHARINDEX('.', T1.SortNum, CHARINDEX('.', T1.SortNum) + 1) - 1

END

)

, 2) -- 3자리 대신 2자리 추출

ELSE N''

END

\+

-- \[4단계 (AAAA) 추출\]: 마침표가 3개 이상일 때만 결합. 2자리 패딩 적용.

CASE

WHEN LEN(T1.SortNum) - LEN(REPLACE(T1.SortNum, '.', N'')) \>= 3 THEN

RIGHT('00' + -- 3자리 대신 2자리 패딩

SUBSTRING(

T1.SortNum,

CHARINDEX('.', T1.SortNum, CHARINDEX('.', T1.SortNum, CHARINDEX('.', T1.SortNum) + 1) + 1) + 1,

1000

)

, 2) -- 3자리 대신 2자리 추출

ELSE N''

END

AS BIGINT)

FROM \#Temp_TPJTWBS AS T1

WHERE

T1.SortNum IS NOT NULL;

-- 4. 임시 테이블 정리

DROP TABLE \#Temp_TPJTWBS;

<span class="mark">-- WBS 트리코드 업데이트 (참고 : \_SWPJTWBSSave) / 6,3,3,3 자리 (9개까지 가능)</span>

begin tran

DROP TABLE \#Tmp

select QrySort ,UpperWBSSeq,TreeCode,SortNum,\* from \_TPJTWBS

/\*트리 정렬\*/

-- 정렬

CREATE TABLE \#Tmp

( WBSSeq INT --

> , PJTSeq INT

, SortNum NVARCHAR(30) -- 작업순서

, QrySort INT NULL -- 리프의 순서

, UpperSortNum NVARCHAR(30) NULL -- 상위 순서

, UpperWBSSeq INT NULL -- 상위 WBSSeq

, NodeCode NVARCHAR(6) NULL --

, TreeCode NVARCHAR(100) NULL --

, TmpSortNum NCHAR(30) NULL --

, Pos INT NULL --

)

INSERT INTO \#Tmp(WBSSeq,PJTSeq, SortNum, TmpSortnum)

SELECT WBSSeq, PJTSeq,SortNum, SortNum+'.'

FROM \_TPJTWBS WITH(NOLOCK)

WHERE LEN(TreeCode) \< 6

 

-- 작업순서로 리프랑 상위 Order

UPDATE \#Tmp

SET QrySort = CONVERT(INT, RIGHT(SortNum, CHARINDEX('.', REVERSE('.'+SortNum))-1)),

UpperSortNum = SUBSTRING(SortNum, 1, LEN(SortNum) - CHARINDEX('.', REVERSE(SortNum)))

-- 상위 WBSSeq 찾기

UPDATE \#Tmp

SET UpperWBSSeq = CASE WHEN B.WBSSeq = A.WBSSeq THEN -1

ELSE B.WBSSeq END,

NodeCode = CASE WHEN B.WBSSeq = A.WBSSeq THEN REPLICATE('0',6- len(convert(varchar,A.QrySort)))+convert(varchar,A.QrySort)

ELSE REPLICATE('0',3- len(convert(varchar,A.QrySort)))+convert(varchar,A.QrySort) END

FROM \#Tmp AS A

LEFT OUTER JOIN \_TPJTWBS AS B WITH(NOLOCK) ON B.CompanySeq = 1

AND B.PJTSeq = A.PJTSeq

> AND B.WBSSeq = A.WBSSeq

AND B.SortNum = A.UpperSortNum

--

UPDATE \#Tmp

SET TreeCode = NodeCode,

TmpSortNum = SortNum

WHERE UpperWBSSeq = -1

-- 하위를 9세대까지 연결할 수 있다. ex)1.2.3.4.5.6.7.8.9

DECLARE @Cnt INT

SELECT @Cnt = 0

WHILE(@Cnt \< 9)

BEGIN

UPDATE B

SET TreeCode = A.TreeCode + LTRIM(RTRIM(B.NodeCode))

FROM \#Tmp AS A

JOIN \#Tmp AS B ON A.WBSSeq = B.UpperWBSSeq

WHERE A.TreeCode is not null

SELECT @Cnt = @Cnt+1

END

 

UPDATE \_TPJTWBS

SET TreeCode = ISNULL(B.TreeCode,'')

FROM \_TPJTWBS AS A WITH(NOLOCK)

JOIN \#Tmp AS B on A.WBSSeq = B.WBSSeq

WHERE A.CompanySeq = 1

AND B.PJTSeq = A.PJTSeq

AND B.WBSSeq = A.WBSSeq
