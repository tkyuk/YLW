# 저장SP 마지막에 체크

-- 겹치는 날짜 체크

DECLARE @MessageType INT,

@Status INT,

@Results NVARCHAR(250)

 

IF EXISTS(SELECT TOP 1 1

FROM shilla_TPRRetAddWorkTermExcept AS A

JOIN shilla_TPRRetAddWorkTermExcept AS B ON B.EmpSeq = A.EmpSeq

AND (B.FromYM \<\> A.FromYM OR B.ToYM \<\> A.ToYM)

WHERE A.FromYM BETWEEN B.FromYM AND B.ToYM

OR A.ToYM BETWEEN B.FromYM AND B.ToYM

)

BEGIN

UPDATE \#BIZ_OUT_DataBlock1

SET Result = '적용시작월, 적용종료월에 대해 중첩된 데이터가 존재합니다.',

MessageType = 17009 ,

Status = 9999

FROM \#BIZ_OUT_DataBlock1 AS A

WHERE A.Status = 0

AND A.WorkingTag IN ('A','U')

END
