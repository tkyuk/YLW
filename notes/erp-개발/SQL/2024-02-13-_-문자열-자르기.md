# \_/ 문자열 자르기

DECLARE @TmpItemSeq NVARCHAR(MAX)

SELECT @TmpItemSeq = @ItemSeq

 

IF (ISNULL(@TmpItemSeq, '') \<\> '')

BEGIN

IF SUBSTRING(@TmpItemSeq, LEN(@TmpItemSeq)-1, LEN(@TmpItemSeq)) \<\> '\_/'

BEGIN

SELECT @TmpItemSeq = @TmpItemSeq + '\_/'

END

 

CREATE TABLE \#TempItemTable

(

IDX_NO INT,

ItemSeq INT

)

INSERT INTO \#TempItemTable

SELECT A.IDX_NO, A.Code

FROM \_FCOMSplit(@TmpItemSeq,'\_/') AS A

 

END

<span class="mark">-- 앞에 공백 포함</span>

 

INSERT INTO \#MESData(WorkReportSeq, ItemSeq, InputQty, WHSeq, UnitSeq, LotNo)

SELECT @WorkReportSeq, A.Code, C.Code, B.Code, D.Code, E.value

FROM \_FCOMSplit(@ItemSeq, '\_/') AS A

JOIN \_FCOMSplit(@WHSeq , '\_/') AS B ON A.IDX_No = B.IDX_No

JOIN \_FCOMSplit(@InputQty, '\_/') AS C ON A.IDX_No = C.IDX_No

JOIN \_FCOMSplit(@UnitSeq , '\_/') AS D ON A.IDX_No = D.IDX_No

JOIN \_FCOMStrSplitOne(@LotNo , '\_/') AS E ON A.IDX_No = E.idx
