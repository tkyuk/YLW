# 복사저장 sp

CREATE PROCEDURE ovcos_SWSLPriceCopySave

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

AS

 

DECLARE @TargetDateFr NCHAR(8),

@TargetDateTo NCHAR(8),

@RegDate NCHAR(8)

 

 

SELECT @TargetDateFr        = ISNULL(TargetDateFr, ''),

@TargetDateTo        = ISNULL(TargetDateTo, ''),

@RegDate = ISNULL(RegDate,'')

FROM \#BIZ_OUT_DataBlock1

IF NOT EXISTS(SELECT 1 FROM ovcos_TSLPriceList AS A WHERE A.CompanySeq = @CompanySeq

AND A.RegDate = @RegDate)

BEGIN

UPDATE \#BIZ_OUT_DataBlock1

SET Status = 9999,

Result = '해당 작성일자에 데이터가 존재하지 않습니다.'

WHERE Status = 0

 

END

-- JOIN ovcos_TWEBPriceListOrder AS C WITH(NOLOCK) ON B.CompanySeq = C.CompanySeq

-- AND B.PLSeq = C.PLSeq

-- WHERE A.Status = 0

 

IF EXISTS(SELECT 1 FROM \#BIZ_OUT_DataBlock1 WHERE Status \<\> 0)

BEGIN

RETURN

END

 

-- 기존데이터 삭제.

DELETE

FROM ovcos_TSLPriceList

WHERE CompanySeq = @CompanySeq

AND RegDate BETWEEN @TargetDateFr AND @TargetDateTo

 

IF @@ERROR \<\> 0

BEGIN

RETURN

END

-- 넣기

-- 대상건찾기.

DECLARE @Date NCHAR(8)

 

SELECT @Date = @TargetDateFr

CREATE TABLE \#ovcos_TSLPriceList (

DataSeq INT IDENTITY,

PLSeq INT,

RegDate NCHAR(8),

UMPLTarget INT,

ItemSeq INT,

UnitSeq INT,

UMPLCategory INT,

ItemChnName NVARCHAR(1000),

FOBPrice DECIMAL(19,5),

WholesalePrice DECIMAL(19,5),

RetailPrice DECIMAL(19,5),

UMPLFuction INT,

UMPLNMPA INT,

IsCPNP NCHAR(1),

IsBPOM NCHAR(1),

IntroURL NVARCHAR(1000),

DetailURL NVARCHAR(1000),

VideoURL NVARCHAR(1000),

UMPLPromotion INT

)

 

WHILE 1=1

BEGIN

INSERT INTO \#ovcos_TSLPriceList

SELECT 0,

@Date,

UMPLTarget,

ItemSeq,

UnitSeq,

UMPLCategory,

ItemChnName,

FOBPrice,

WholesalePrice,

RetailPrice,

UMPLFuction,

UMPLNMPA,

IsCPNP,

IsBPOM,

IntroURL,

DetailURL,

VideoURL,

UMPLPromotion

FROM ovcos_TSLPriceList

WHERE CompanySeq = @CompanySeq

AND RegDate = @RegDate

 

SELECT @Date = CONVERT(NCHAR(8), DATEADD(DD, 1, @Date), 112)

 

IF @Date \> @TargetDateTo BREAK

END

 

DECLARE @Count INT ,

@Seq INT

 

SELECT @Count = COUNT(1) FROM \#ovcos_TSLPriceList

IF @Count \> 0

BEGIN

-- 키값생성코드부분 시작

EXEC @Seq = dbo.\_SWCOMCreateSeq @CompanySeq, 'ovcos_TSLPriceList', 'PLSeq', @Count

-- Temp Table 에 생성된 키값 UPDATE

UPDATE \#ovcos_TSLPriceList

SET PLSeq = @Seq + DataSeq

END

-- INSERT

BEGIN

INSERT INTO ovcos_TSLPriceList

(

RegDate ,

UMPLFuction ,

UMPLNMPA ,

IsBPOM ,

VideoURL ,

UMPLPromotion ,

)

A.PLSeq ,

A.RegDate,

ISNULL(A.UMPLTarget , 0),

ISNULL(A.ItemSeq , 0),

ISNULL(A.UnitSeq , 0),

ISNULL(A.UMPLCategory , 0),

ISNULL(A.ItemChnName ,''),

ISNULL(A.FOBPrice , 0),

ISNULL(A.WholesalePrice, 0),

ISNULL(A.RetailPrice , 0),

ISNULL(A.UMPLFuction , 0),

ISNULL(A.UMPLNMPA , 0),

ISNULL(A.IsCPNP ,''),

ISNULL(A.IsBPOM ,''),

ISNULL(A.IntroURL ,''),

ISNULL(A.DetailURL ,''),

ISNULL(A.VideoURL ,''),

ISNULL(A.UMPLPromotion , 0),

@UserSeq

FROM \#ovcos_TSLPriceList A

IF @@ERROR \<\> 0

BEGIN

RETURN

END

END
