SET ProcRev = B.ProcRev

FROM \#GoodItem AS A JOIN (SELECT A.ItemSeq, MAX(B.ProcRev) AS ProcRev

FROM \#GoodItem AS A JOIN \_TPDROUItemProcMat AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

;WITH CTE_MatItem (GoodSeq, ItemSeq, ProcRev, MatItemSeq, SubItemProcRev, Lev, NeedQty) AS

(

SELECT A.ItemSeq, B.ItemSeq, B.ProcRev, B.MatItemSeq, B.SubItemProcRev, 1,

CONVERT(DECIMAL(19,5), A.Qty \* (CASE WHEN ISNULL(B.NeedQtyDenominator, 0) = 0 THEN 0 ELSE B.NeedQtyNumerator / B.NeedQtyDenominator END))

FROM \#GoodItem AS A

JOIN \_TPDROUItemProcMat AS B ON B.CompanySeq = @CompanySeq AND A.ItemSeq = B.ItemSeq AND A.ProcRev = B.ProcRev AND B.BizUnit = @BizUnit

UNION ALL

SELECT A.GoodSeq, B.ItemSeq, B.ProcRev, B.MatItemSeq, B.SubItemProcRev, Lev+1,

CONVERT(DECIMAL(19,5), A.NeedQty \* (CASE WHEN ISNULL(B.NeedQtyDenominator, 0) = 0 THEN 0 ELSE B.NeedQtyNumerator / B.NeedQtyDenominator END))

FROM CTE_MatItem AS A

JOIN \_TPDROUItemProcMat AS B ON B.CompanySeq = @CompanySeq AND A.MatItemSeq = B.ItemSeq AND A.SubItemProcRev = B.ProcRev AND B.BizUnit = @BizUnit

)

B.Spec AS Spec,

D.UnitName AS UnitName,

SUM(A.NeedQty) AS Qty,

A.MatItemSeq AS SubItemSeq

FROM CTE_MatItem A

JOIN \_TDAItem AS B WITH (NOLOCK) ON B.CompanySeq = @CompanySeq AND A.MatItemSeq = B.ItemSeq

A.MatItemSeq

ORDER BY B.AssetSeq ,

H.ItemClassLName,

H.ItemClassMName,

H.ItemClassSName,

B.ItemName ,

B.ItemNo ,

B.Spec

-- 차수 중복 제거 후 (같은 제품의 차수가 여러 개면 여러 행이 나오므로)

SELECT A.ItemSeq,

A.MatItemSeq

INTO \#Tmp_Only_Rev

FROM \_TPDROUItemProcMat AS A

JOIN \#Result AS B ON A.MatItemSeq = B.SubItemSeq

WHERE A.CompanySeq = @CompanySeq

GROUP BY A.ItemSeq, A.MatItemSeq

 

 

-- 제품별 소요자재 계산

SELECT A.MatItemSeq,

'1' AS IsOnly

INTO \#Tmp_Only

FROM \#Tmp_Only_Rev AS A JOIN \_TDAItem AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND A.MatItemSeq = B.ItemSeq

AND B.SMStatus = 2001002

GROUP BY A.MatItemSeq

HAVING COUNT(\*) = 1

 

 

SELECT A.\*,

B.StkQty,

CASE WHEN ISNULL(A.Qty,0) - ISNULL(B.StkQty,0) \> 0 THEN ISNULL(A.Qty,0) - ISNULL(B.StkQty,0) ELSE 0 END AS Excess,

ISNULL(C.IsOnly, 'O') AS IsOnly

FROM \#Result AS A

LEFT OUTER JOIN \#Stk AS B ON A.SubItemSeq = B.ItemSeq

LEFT OUTER JOIN \#Tmp_Only AS C ON A.SubItemSeq = C.MatItemSeq
