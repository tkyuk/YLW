# 금액 계산 (부가세포함 / 수량 \* 단가)

-- 금액 계산

UPDATE A

SET CurAmt = CASE WHEN ISNULL(A.IsVat, '') \<\> '1' THEN (A.Qty \* A.Price)

ELSE (A.Qty \* A.Price) - ((A.Qty \* A.Price) \* (ISNULL(A.VatRate, 10) / (100 + ISNULL(A.VatRate, 10))))

END,

CurVAT = CASE WHEN ISNULL(A.IsVat, '') \<\> '1' THEN (A.Qty \* A.Price) \* (ISNULL(A.VatRate, 10) / 100)

ELSE ((A.Qty \* A.Price) \* (ISNULL(A.VatRate, 10) / (100 + ISNULL(A.VatRate, 10))))

END,

DomAmt = CASE WHEN ISNULL(A.IsVat, '') \<\> '1' THEN (A.Qty \* A.DomPrice)

ELSE (A.Qty \* A.DomPrice) - ((A.Qty \* A.DomPrice) \* (ISNULL(A.VatRate, 10) / (100 + ISNULL(A.VatRate, 10))))

END,

DomVAT = CASE WHEN ISNULL(A.IsVat, '') \<\> '1' THEN (A.Qty \* A.DomPrice) \* (ISNULL(A.VatRate, 10) / 100)

ELSE ((A.Qty \* A.DomPrice) \* (ISNULL(A.VatRate, 10) / (100 + ISNULL(A.VatRate, 10))))

END

FROM \#TDelvItem AS A

WHERE A.WorkingTag IN ('A', 'U')

AND A.Status = 0
