CREATE TABLE \#TMP(

A NVARCHAR(100)

)

INSERT INTO \#TMP

SELECT 'ABC%DEF'

UNION SELECT'ABC\[DEF'

UNION SELECT'ABC/DEF'

UNION SELECT'ABC\DEF'

UNION SELECT'ABC_DEF'

 

SELECT \* FROM \#TMP WHERE A LIKE '%/\[%' ESCAPE '/' -- \[ 검색
