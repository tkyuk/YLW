# 거래명세서 / 세금계산서에 대한 수금, 매출, 미수금

CREATE PROC emk_SWSLPlanSalesCollectSumListQuery

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

AS

 

-- 조회조건

DECLARE @DateMMFr NCHAR(6),

@SalesYM NCHAR(6), -- 기준월

@EnvValue INT, -- 환경설정 소수점 자리수

@SalesKind INT, -- 매출기준

@EmpSeq INT, -- 사원명

@PreDateMMFr NCHAR(6), -- 전년도 1월

@PostDateMMFr NCHAR(6) -- 전년도 12월

 

-- 조회조건 변수에 넣기

SELECT @SalesYM = ISNULL(SalesYM ,''), -- 기준년도

@SalesKind = ISNULL(SalesKind, 0), -- 매출기준

@EmpSeq = ISNULL(EmpSeq , 0) -- 사원명

FROM \#BIZ_IN_DataBlock1

 

-- 기준월 없으면 현재일

IF @SalesYM = '' SELECT @SalesYM = LEFT(CONVERT(NCHAR(6),GETDATE(),112),4)

 

SELECT @DateMMFr = LEFT(@SalesYM,4) + '01', -- 당해년도 1월

@PreDateMMFr = CONVERT(NCHAR(4),DATEADD(Y , -1, LEFT(@SalesYM,4)+'0101'),112) + '01', -- 전년도 1월

@PostDateMMFr = CONVERT(NCHAR(8),DATEADD(DAY, -1, LEFT(@SalesYM,4)+'0101'),112) -- 전년도 12월

 

-- 최종결과 Temp테이블

CREATE TABLE \#ResultReceipt

(

EmpSeq INT, -- 담당자코드

CustSeq INT, -- 매출업체코드

ReceiptAmt01 DECIMAL(19,5), -- 수금 1월

ReceiptAmt02 DECIMAL(19,5), -- 수금 2월

ReceiptAmt03 DECIMAL(19,5), -- 수금 3월

ReceiptAmt04 DECIMAL(19,5), -- 수금 4월

ReceiptAmt05 DECIMAL(19,5), -- 수금 5월

ReceiptAmt06 DECIMAL(19,5), -- 수금 6월

ReceiptAmt07 DECIMAL(19,5), -- 수금 7월

ReceiptAmt08 DECIMAL(19,5), -- 수금 8월

ReceiptAmt09 DECIMAL(19,5), -- 수금 9월

ReceiptAmt10 DECIMAL(19,5), -- 수금 10월

ReceiptAmt11 DECIMAL(19,5), -- 수금 11월

ReceiptAmt12 DECIMAL(19,5), -- 수금 12월

PreMiAmt DECIMAL(19,5), -- 이월 미수금

SalesAmt DECIMAL(19,5), -- 매출액

ReceiptAmt DECIMAL(19,5), -- 수금액

MiAmt DECIMAL(19,5) -- 미수금

)

 

-- 전년도 이월미수금 구하기 위한 임시테이블 생성 (이월미수금 = 매출액 - 수금액)

CREATE TABLE \#Tmp_PreMiAmt

(

EmpSeq INT, -- 담당자코드

CustSeq INT, -- 매출업체코드

SalesAmt DECIMAL(19,5), -- 매출액

ReceiptAmt DECIMAL(19,5) -- 수금액

)

 

 

-- @SalesKind = 2 거래명세서 / @SalesKind = 3 세금계산서

IF @SalesKind = 2 -- 거래명세서

BEGIN

--========================================================

-- 이월 미수금 구하기 시작(전년도 매출액 합계 - 수금액 합계)

--========================================================

-- 1. 전년도 매출액

INSERT INTO \#Tmp_PreMiAmt

(EmpSeq,CustSeq, SalesAmt)

SELECT A.EmpSeq, A.CustSeq, ISNULL(SUM(ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0)),0)

FROM \_TSLCreditSum AS A WITH(NOLOCK)

WHERE A.CompanySeq = @CompanySeq

AND A.SumYM BETWEEN @PreDateMMFr AND @PostDateMMFr

AND (@EmpSeq = 0 OR @EmpSeq = A.EmpSeq)

AND A.SumType = 1 -- 매출액

GROUP BY A.EmpSeq, A.CustSeq

 

-- 2. 전년도 수금액

INSERT INTO \#Tmp_PreMiAmt

(EmpSeq,CustSeq, ReceiptAmt)

SELECT A.EmpSeq, A.CustSeq, ISNULL(SUM(ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0)),0)

FROM \_TSLCreditSum AS A WITH(NOLOCK)

WHERE A.CompanySeq = @CompanySeq

AND A.SumYM BETWEEN @PreDateMMFr AND @PostDateMMFr

AND (@EmpSeq = 0 OR @EmpSeq = A.EmpSeq)

AND A.SumType = 2 -- 수금액

GROUP BY A.EmpSeq, A.CustSeq

--========================================================

-- 이월 미수금 종료

--========================================================

 

-- 최종 결과 테이블 수금액 INSERT

INSERT INTO \#ResultReceipt

(EmpSeq,CustSeq,

ReceiptAmt01,ReceiptAmt02,ReceiptAmt03,ReceiptAmt04,ReceiptAmt05,ReceiptAmt06,

ReceiptAmt07,ReceiptAmt08,ReceiptAmt09,ReceiptAmt10,ReceiptAmt11,ReceiptAmt12)

SELECT A.EmpSeq, A.CustSeq,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '01' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt01,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '02' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt02,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '03' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt03,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '04' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt04,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '05' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt05,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '06' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt06,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '07' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt07,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '08' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt08,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '09' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt09,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '10' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt10,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '11' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt11,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '12' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt12

FROM \_TSLCreditSum AS A WITH(NOLOCK)

WHERE A.CompanySeq = @CompanySeq

AND A.SumYM BETWEEN LEFT(@DateMMFr, 6) AND LEFT(@SalesYM,6)

AND (@EmpSeq = 0 OR @EmpSeq = A.EmpSeq)

AND A.SumType = 2

GROUP BY A.EmpSeq, A.CustSeq

 

-- 최종 결과 테이블 매출액 INSERT

INSERT INTO \#ResultReceipt

(EmpSeq,CustSeq,SalesAmt)

SELECT A.EmpSeq, A.CustSeq, ISNULL(SUM(ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0)),0)

FROM \_TSLCreditSum AS A WITH(NOLOCK)

WHERE A.CompanySeq = @CompanySeq

AND A.SumYM BETWEEN LEFT(@DateMMFr, 6) AND LEFT(@SalesYM,6)

AND (@EmpSeq = 0 OR @EmpSeq = A.EmpSeq)

AND A.SumType = 1

GROUP BY A.EmpSeq, A.CustSeq

 

 

END

ELSE -- 세금계산서 (거래명세서와 동일한 로직)

BEGIN

-- 이월 미수금 구하기 전년도 매출액 합계 - 수금액 합계

INSERT INTO \#Tmp_PreMiAmt

(EmpSeq,CustSeq, SalesAmt)

SELECT A.EmpSeq, A.CustSeq, ISNULL(SUM(ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0)),0)

FROM \_TSLBillCreditSum AS A WITH(NOLOCK)

WHERE A.CompanySeq = @CompanySeq

AND A.SumYM BETWEEN @PreDateMMFr AND @PostDateMMFr

AND (@EmpSeq = 0 OR @EmpSeq = A.EmpSeq)

AND A.SumType = 1 -- 매출액

GROUP BY A.EmpSeq, A.CustSeq

 

INSERT INTO \#Tmp_PreMiAmt

(EmpSeq,CustSeq, ReceiptAmt)

SELECT A.EmpSeq, A.CustSeq, ISNULL(SUM(ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0)),0)

FROM \_TSLBillCreditSum AS A WITH(NOLOCK)

WHERE A.CompanySeq = @CompanySeq

AND A.SumYM BETWEEN @PreDateMMFr AND @PostDateMMFr

AND (@EmpSeq = 0 OR @EmpSeq = A.EmpSeq)

AND A.SumType = 2 -- 수금액

GROUP BY A.EmpSeq, A.CustSeq

 

 

 

INSERT INTO \#ResultReceipt

(EmpSeq,CustSeq,

ReceiptAmt01,ReceiptAmt02,ReceiptAmt03,ReceiptAmt04,ReceiptAmt05,ReceiptAmt06,

ReceiptAmt07,ReceiptAmt08,ReceiptAmt09,ReceiptAmt10,ReceiptAmt11,ReceiptAmt12)

SELECT A.EmpSeq, A.CustSeq,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '01' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt01,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '02' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt02,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '03' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt03,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '04' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt04,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '05' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt05,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '06' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt06,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '07' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt07,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '08' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt08,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '09' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt09,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '10' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt10,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '11' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt11,

ISNULL(SUM(CASE WHEN RIGHT(A.SumYM,2) = '12' THEN ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0) END),0) AS ReceiptAmt12

FROM \_TSLBillCreditSum AS A WITH(NOLOCK)

WHERE A.CompanySeq = @CompanySeq

AND A.SumYM BETWEEN LEFT(@DateMMFr, 6) AND LEFT(@SalesYM,6)

AND (@EmpSeq = 0 OR @EmpSeq = A.EmpSeq)

AND A.SumType = 2

GROUP BY A.EmpSeq, A.CustSeq

 

INSERT INTO \#ResultReceipt

(EmpSeq,CustSeq,SalesAmt)

SELECT A.EmpSeq, A.CustSeq, ISNULL(SUM(ISNULL(A.CurAmt,0) + ISNULL(A.CurVat,0)),0)

FROM \_TSLBillCreditSum AS A WITH(NOLOCK)

WHERE A.CompanySeq = @CompanySeq

AND A.SumYM BETWEEN LEFT(@DateMMFr, 6) AND LEFT(@SalesYM,6)

AND (@EmpSeq = 0 OR @EmpSeq = A.EmpSeq)

AND A.SumType = 1

GROUP BY A.EmpSeq, A.CustSeq

 

 

END

 

 

 

-- 작년 이월 미수금 최종 임시테이블 생성

CREATE TABLE \#PreMiAmt

(

EmpSeq INT, -- 담당자코드

CustSeq INT, -- 매출업체코드

MiAmt DECIMAL(19,5) -- 미수금 (매출액-수금액)

--SalesAmt DECIMAL(19,5), -- 매출액

--ReceiptAmt DECIMAL(19,5) -- 수금액

)

INSERT INTO \#PreMiAmt

SELECT EmpSeq ,

CustSeq,

ISNULL(ISNULL(SUM(SalesAmt),0) - ISNULL(SUM(ReceiptAmt),0),0) AS MiAmt

FROM \#Tmp_PreMiAmt

GROUP BY EmpSeq, CustSeq

 

 

-- 최종 테이블 '이월 미수금' 구하기 : 전년도의 미수금

UPDATE \#ResultReceipt

SET PreMiAmt = ISNULL(B.MiAmt,0)

FROM \#ResultReceipt AS A

LEFT OUTER JOIN \#PreMiAmt AS B ON B.EmpSeq = A.EmpSeq

AND B.CustSeq = A.CustSeq

 

-- 최종 테이블 '수금액' 구하기 : 당해년도 1~12월 수금 합계

UPDATE \#ResultReceipt

SET ReceiptAmt = ISNULL(ReceiptAmt01,0) + ISNULL(ReceiptAmt02,0) + ISNULL(ReceiptAmt03,0) + ISNULL(ReceiptAmt04,0) + ISNULL(ReceiptAmt05,0) + ISNULL(ReceiptAmt06,0) +

ISNULL(ReceiptAmt07,0) + ISNULL(ReceiptAmt08,0) + ISNULL(ReceiptAmt09,0) + ISNULL(ReceiptAmt10,0) + ISNULL(ReceiptAmt11,0) + ISNULL(ReceiptAmt12,0)

 

 

-- 최종 테이블 '미수금' 구하기 : 이월 미수금 + 매출액 - 수금액

UPDATE \#ResultReceipt

SET MiAmt = ISNULL(PreMiAmt,0) + ISNULL(SalesAmt,0) - ISNULL(ReceiptAmt,0)

 

SELECT B.Empid AS Empid , -- 사원번호

B.EmpName AS EmpName , -- 담당

A.EmpSeq AS EmpSeq , -- 담당자코드

A.CustSeq AS CustSeq , -- 매출업체코드

C.CustName AS CustName , -- 매출업체명

C.BizNo AS BizNo , -- 사업자번호

SUM(ReceiptAmt01) AS ReceiptAmt01, -- 수금 1월

SUM(ReceiptAmt02) AS ReceiptAmt02, -- 수금 2월

SUM(ReceiptAmt03) AS ReceiptAmt03, -- 수금 3월

SUM(ReceiptAmt04) AS ReceiptAmt04, -- 수금 4월

SUM(ReceiptAmt05) AS ReceiptAmt05, -- 수금 5월

SUM(ReceiptAmt06) AS ReceiptAmt06, -- 수금 6월

SUM(ReceiptAmt07) AS ReceiptAmt07, -- 수금 7월

SUM(ReceiptAmt08) AS ReceiptAmt08, -- 수금 8월

SUM(ReceiptAmt09) AS ReceiptAmt09, -- 수금 9월

SUM(ReceiptAmt10) AS ReceiptAmt10, -- 수금 10월

SUM(ReceiptAmt11) AS ReceiptAmt11, -- 수금 11월

SUM(ReceiptAmt12) AS ReceiptAmt12, -- 수금 12월

SUM(PreMiAmt ) AS PreMiAmt , -- 이월 미수금

SUM(SalesAmt ) AS SalesAmt , -- 매출액

SUM(ReceiptAmt ) AS ReceiptAmt , -- 수금액

SUM(MiAmt ) AS MiAmt -- 미수금

FROM \#ResultReceipt AS A

LEFT OUTER JOIN \_TDAEmp AS B ON B.CompanySeq = @CompanySeq

AND B.EmpSeq = A.EmpSeq

LEFT OUTER JOIN \_TDACust AS C ON C.CompanySeq = @CompanySeq

AND C.CustSeq = A.CustSeq

WHERE ISNULL(A.EmpSeq ,0) \<\> 0

AND ISNULL(A.CustSeq,0) \<\> 0

GROUP BY B.Empid, B.EmpName, A.EmpSeq, A.CustSeq, C.CustName, C.BizNo

 

RETURN
