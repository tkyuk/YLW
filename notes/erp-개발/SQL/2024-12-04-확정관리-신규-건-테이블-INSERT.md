# 확정관리 신규 건 테이블 INSERT

--SELECT \* FROM \_TCOMConfirm WHERE ConfirmSeq = 15000125

BEGIN TRAN

INSERT INTO \_TCOMConfirm(

CompanySeq

,ConfirmSeq

,CfmSeq

,CfmSerl

,CfmSubSerl

,IsAuto

,CfmCode

,CfmDate

,CfmEmpSeq

,UMCfmReason

,CfmReason

,LastDateTime

)

SELECT 1 AS CompanySeq

,15000125 AS ConfirmSeq

,A.CustBankAccSerl AS CfmSeq

,0 AS CfmSerl

,0 AS CfmSubSerl

,0 AS IsAuto

,'1' AS CfmCode

,'' AS CfmDate

,0 AS CfmEmpSeq

,0 AS UMCfmReason

,0 AS CfmReason

,GETDATE() AS LastDateTime

FROM \_TDACustBankAcc AS A

LEFT OUTER JOIN \_TCOMConfirm AS B ON B.CompanySeq = A.CompanySeq

AND B.CfmSeq = A.CustBankAccSerl

AND B.ConfirmSeq = 15000125

WHERE B.CompanySeq IS NULL

 

ROLLBACK TRAN
