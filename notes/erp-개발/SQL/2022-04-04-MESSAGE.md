# MESSAGE

BEGIN

EXEC dbo.\_SWCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

1213 , -- @1일경우 @2, @3은 필수입력 항목입니다.(SELECT \* FROM \_TCAMessageLanguage WHERE MessageSeq = 1213)

@LanguageSeq ,

15000064, N'단가소급적용', 15000065, N'시작일', 15000066, N'종료일' -- SELECT \* FROM \_TCADictionary WHERE WordSeq = 15000064

-- '단가소급 적용시 시작일, 종료일은 필수 입니다.'

-- 소급적용시작일 : 15000065

-- 소급적용종료일 : 15000066

 

UPDATE \#BIZ_OUT_DataBlock1

SET Result = @Results,

MessageType = @MessageType,

Status = @Status

WHERE WorkingTag IN ('A','U')

AND ISNULL(IsRetroac,'') = '1'

AND (ISNULL(RetroacSDate,'') = '' OR ISNULL(RetroacEDate,'') = '')

END

- 메시지 등록 후 사용할 경우

- 이 경우 테이블에 추가하는 작업이 필요 (하단 참조)

 

BEGIN

EXEC dbo.\_SWCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

<span class="mark">15000001</span> , -- MCO품목은 최대 2개만 등록가능합니다. (SELECT \* FROM \_TCAMessageLanguage WHERE MessageSeq = 15000001)

<span class="mark">@LanguageSeq ,</span>

<span class="mark">0, ''</span>

UPDATE \#BIZ_OUT_DataBlock1

SET Result = @Results,

MessageType = @MessageType,

Status = @Status

FROM \#BIZ_OUT_DataBlock1 AS A

INNER JOIN ( SELECT S.MCOItemSeq

FROM (

SELECT A1.MCOItemSeq

FROM \#BIZ_OUT_DataBlock1 AS A1

WHERE A1.WorkingTag IN ('A','U')

AND A1.Status = 0

UNION ALL

SELECT A1.MCOItemSeq

FROM mco_TDAItem AS A1

WHERE A1.CompanySeq = @CompanySeq

AND NOT EXISTS(SELECT 1

FROM \#BIZ_OUT_DataBlock1

 

WHERE WorkingTag IN('D','U')

AND MCOCompanySeq = A1.MCOCompanySeq

AND ItemSeq = A1.ItemSeq

AND MCOItemSeq = A1.MCOItemSeq

AND Status = 0

)

) AS S

GROUP BY S.MCOItemSeq

HAVING COUNT(1) \> 2

) AS B

ON A.MCOItemSeq = B.MCOItemSeq

END
