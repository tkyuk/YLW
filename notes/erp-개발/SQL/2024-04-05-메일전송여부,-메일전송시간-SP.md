# 메일전송여부, 메일전송시간 SP

SELECT A.POSeq,

MAX(LEFT(K.SndDtm,4) + '-' + SUBSTRING(K.SndDtm, 5, 2) + '-' + SUBSTRING(K.SndDtm, 7, 2) + ' ' + SUBSTRING(K.SndDtm, 9, 2) + ':' + SUBSTRING(K.SndDtm, 11, 2) + ':' + SUBSTRING(K.SndDtm, 13, 2)) AS MailDateTime

INTO \#SendMail

FROM \#TEMP_TPUORDPOItem AS A

JOIN \_TPUORDPO AS B WITH(NOLOCK) ON A.CompanySeq = B.CompanySeq

AND A.POSeq = B.POSeq

LEFT OUTER JOIN OTCAKumsTransLog AS K WITH(NOLOCK) ON B.CompanySeq = K.CompanySeq

AND B.PONo = SUBSTRING(K.Subject, CHARINDEX('PO', K.Subject), 11)

GROUP BY A.POSeq

 

 

 

SELECT

, CASE WHEN ISNULL(K.MailDateTime,'') = '' THEN '0' ELSE '1' END AS IsMail -- 메일전송여부 추가

, K.MailDateTime AS MailDateTime -- 메일전송시간 추가

 

LEFT OUTER JOIN \#SendMail AS K ON A.POSeq = K.POSeq
