# 입력값 유형이 숫자인 경우 숫자만 입력되도록 체크

DROP PROC dbo.djb_SWPDQAItemExtQCSubCheck

GO

CREATE PROC dbo.djb_SWPDQAItemExtQCSubCheck

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

 

AS

 

DECLARE @Count INT,

@MessageType INT,

@Status INT,

@Results NVARCHAR(250)

 

-------------------------------------------

-- 중복여부체크

-------------------------------------------

EXEC dbo.\_SWCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

6 , -- 중복된 @1 @2가(이) 입력되었습니다.(SELECT \* FROM \_TCAMessageLanguage WHERE MessageSeq = 6)

@LanguageSeq ,

476, N'검사항목등록' -- SELECT \* FROM \_TCADictionary WHERE Word like '%검사항목%'

UPDATE \#BIZ_OUT_DataBlock2

SET Result = REPLACE(@Results,'@2', ''),

MessageType = @MessageType,

Status = @Status

FROM \#BIZ_OUT_DataBlock2 AS A JOIN \_TPDQAItemQcTitle AS B WITH(NOLOCK) ON A.UMQCTitleSeq = B.UMQCTitleSeq

AND A.ItemSeq = B.ItemSeq

AND A.SMQCKind = B.SMQcKind

WHERE B.CompanySeq = @CompanySeq

AND A.WorkingTag IN ('A')

AND A.Status = 0

 

UPDATE \#BIZ_OUT_DataBlock2

SET Result = REPLACE(@Results,'@2', ''),

MessageType = @MessageType,

Status = @Status

FROM \#BIZ_OUT_DataBlock2 AS A JOIN \_TPDQAItemQcTitle AS B WITH(NOLOCK) ON 1=1 -- A.UMQCTitleSeq \<\> B.UMQCTitleSeq

AND A.ItemSeq = B.ItemSeq

AND A.UMQCTitleSeq = B.UMQcTitleSeq

AND A.SMQCKind = B.SMQcKind

AND B.CompanySeq = @CompanySeq

WHERE A.WorkingTag IN ('U')

AND A.Status = 0

AND A.UMQCTitleSeq \<\> A.UMQCTitleOldSeq

UPDATE A

SET Result = REPLACE(@Results,'@2', ''),

MessageType = @MessageType,

Status = @Status

FROM \#BIZ_OUT_DataBlock2 AS A JOIN \#BIZ_OUT_DataBlock2 AS B ON A.UMQCTitleSeq = B.UMQCTitleSeq

AND A.ItemSeq = B.ItemSeq

AND A.IDX_NO \<\> B.IDX_NO

AND A.SMQCKind = B.SMQcKind

WHERE A.WorkingTag IN ('U','A')

AND A.Status = 0

 

 

-------------------------------------------

-- 품목 입력 여부

-------------------------------------------

EXEC dbo.\_SWCOMMessage @MessageType OUTPUT,

@Status OUTPUT,

@Results OUTPUT,

1008 , -- @1을(를) 입력하지 않았습니다. (SELECT \* FROM \_TCAMessageLanguage WHERE MessageSeq = 1008)

@LanguageSeq ,

20899, N'품목명' -- SELECT \* FROM \_TCADictionary WHERE Word like '%품목명%'

 

UPDATE \#BIZ_OUT_DataBlock2

SET Result = @Results,

MessageType = @MessageType,

Status = @Status

FROM \#BIZ_OUT_DataBlock2 AS A

WHERE A.WorkingTag IN ('A')

AND A.Status = 0

AND A.ItemSeq = 0

 

 

-- 입력방법이 숫자인경우 상한치,하한치 체크 23.08.11 SJPARK

CREATE TABLE \#TempNumResultCheck(

IDX_NO INT,

WorkingTag NCHAR(1),

Status INT,

Result NVARCHAR(255),

MessageType INT,

SMInputType INT,

LowerLimit NVARCHAR(100),

UpperLimit NVARCHAR(100)

)

 

INSERT INTO \#TempNumResultCheck(

IDX_NO ,

WorkingTag ,

Status ,

Result ,

MessageType ,

SMInputType ,

LowerLimit ,

UpperLimit

)

SELECT IDX_NO ,

WorkingTag ,

Status ,

'' ,

0 ,

SMInputType ,

ISNULL(LowerLimit , ''),

ISNULL(UpperLimit , '')

FROM \#BIZ_OUT_DataBlock2 AS A

WHERE A.Status = 0

AND ISNULL(A.ItemSeq,0) \<\> 0

 

-- 입력구분이 숫자인데 문자가 들어온경우

UPDATE \#TempNumResultCheck

SET Result = CASE WHEN ISNUMERIC(CASE WHEN ISNULL(A.LowerLimit ,'') = '' THEN '0' ELSE A.LowerLimit END) \<\> '1' OR (CHARINDEX(',',A.LowerLimit) \<\> 0) THEN '합격하한치에 '

WHEN ISNUMERIC(CASE WHEN ISNULL(A.UpperLimit ,'') = '' THEN '0' ELSE A.UpperLimit END) \<\> '1' OR (CHARINDEX(',',A.UpperLimit) \<\> 0) THEN '합격상한치에 '

END + '숫자를 입력하세요',

MessageType = 99999,

Status = 99999

FROM \#TempNumResultCheck AS A

WHERE A.WorkingTag IN ('A','U')

AND A.Status = 0

AND A.SMInputType = 1018001 -- 입력구분 - 숫자

AND (ISNUMERIC(CASE WHEN ISNULL(A.LowerLimit ,'') = '' THEN '0' ELSE A.LowerLimit END) \<\> '1'

OR ISNUMERIC(CASE WHEN ISNULL(A.UpperLimit ,'') = '' THEN '0' ELSE A.UpperLimit END) \<\> '1'

OR CHARINDEX(',',A.LowerLimit) \<\> 0

OR CHARINDEX(',',A.UpperLimit) \<\> 0

)

 

IF EXISTS (SELECT TOP 1 1 FROM \#TempNumResultCheck WHERE Status \<\> 0)

BEGIN

UPDATE B

SET Result = A.Result ,

MessageType = A.MessageType,

Status = A.Status

FROM \#TempNumResultCheck AS A

JOIN \#BIZ_OUT_DataBlock2 AS B ON A.IDX_NO = B.IDX_NO

WHERE B.Status = 0

AND A.Status \<\> 0

END

 

 

 

RETURN
