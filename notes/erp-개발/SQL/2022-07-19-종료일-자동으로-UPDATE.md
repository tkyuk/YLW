# 종료일 자동으로 UPDATE

-- 신규 최종사용자 =\> 종료일 = 99991231

-- 기존 최종사용자 =\> 종료일 = 신규사용자의 시작일 - 1

DECLARE @BeforeStartDate NCHAR(8),

@BeforeUserSeq INT

-- 기존에 99991231 이었던 데이터 찾기

SELECT @BeforeStartDate = StartDate,

@BeforeUserSeq = UserSeq

FROM hsg_TDAHWUser AS A WITH(NOLOCK)

WHERE A.EndDate = '99991231'

AND A.HwSeq = @HwSeq

-- 시작일이 제일 높은 데이터의 종료일 99991231 로 업데이트

UPDATE hsg_TDAHWUser

SET EndDate = '99991231'

WHERE StartDate = (SELECT MAX(A.StartDate)

FROM hsg_TDAHWUser AS A WITH(NOLOCK)

WHERE A.HwSeq = @HwSeq)

DECLARE @AfterStartDate NCHAR(8)

SELECT @AfterStartDate = StartDate

FROM hsg_TDAHWUser AS A WITH(NOLOCK)

WHERE StartDate = (SELECT max(StartDate) FROM hsg_TDAHWUser WHERE HwSeq = @HwSeq)

 

-- 바뀌었는지 확인 (1에서 찾은 시작일과 현재 max시작일 비교)

IF(@BeforeStartDate \<\> @AfterStartDate)

BEGIN

UPDATE hsg_TDAHWUser

SET EndDate = CONVERT(NCHAR(8),DATEADD(DAY,-1, CONVERT(date,@AfterStartDate)),112)

WHERE StartDate = @BeforeStartDate

END
