# 구매단가

-- 구매단가 : 구매발주

 

CREATE TABLE \#TPUBaseGetPrice (WorkingTag NCHAR(1) NULL)

EXEC dbo.\_SCAOpenXmlToTemp '\<ROOT\>\</ROOT\>', 2, @CompanySeq, 2640, 'DataBlock1', '#TPUBaseGetPrice'

-- 단가 정보 구해옴.

INSERT INTO \#TPUBaseGetPrice

(

IDX_NO, ItemSeq, PUType, CustSeq -- 거래처 추가 23.10.18 SJPARK

)

SELECT A.IDX_NO, B.ItemSeq, 'PUORDPO' AS PUType, A.CustSeq

FROM \#MESPO AS A

JOIN \_TDAItem AS B ON B.CompanySeq = @CompanySeq AND A.ItemSeq = B.ItemSeq

IF EXISTS(SELECT TOP 1 1 FROM \#TPUBaseGetPrice)

BEGIN

EXEC \_SPUBaseGetPrice @xmlDocument= '',@xmlFlags=2,@ServiceSeq=2640,@WorkingTag=N'AUTO',@CompanySeq=@CompanySeq,@LanguageSeq=1,@UserSeq=@UserSeq,@PgmSeq=@PgmSeq

END

 

--구매

SELECT LEFT(M.DelvInDate,6) AS DelvYM,

M.CustSeq AS CustSeq,

A.ItemSeq AS ItemSeq,

A.UnitSeq AS UnitSeq,

SUM(A.Qty) AS Qty,

ISNULL(B.Price,0) AS AdjPrice -- 해당월 조정가.

FROM \_TPUDelvIn AS M WITH(NOLOCK)

JOIN \_TPUDelvInItem AS A WITH(NOLOCK) ON A.DelvInSeq = M.DelvInSeq

AND A.CompanySeq = M.CompanySeq

LEFT OUTER JOIN \_TPUBASEBuyPriceItem AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND A.ItemSeq = B.ItemSeq

AND M.CustSeq = B.CustSeq

AND B.CurrSeq = @CurrSeq

AND A.UnitSeq = B.UnitSeq

AND CONVERT(NCHAR(8), DATEADD(DD, -1, DATEADD(MM, 1, LEFT(M.DelvInDate,6) + '01')), 112) BETWEEN B.StartDate AND B.EndDate

 

 

 

 

LEFT OUTER JOIN \_TPUBASEBuyPriceItem AS I WITH(NOLOCK) ON A.CompanySeq = I.CompanySeq

AND A.ItemSeq = I.ItemSeq

AND ISNULL(I.IsPrice,'0') = '1'

AND @CurrDate BETWEEN I.StartDate AND I.EndDate
