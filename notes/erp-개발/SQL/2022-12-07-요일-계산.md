# 요일 계산

SP

 

DECLARE @TempDate NCHAR(8),

@WeeKDay NCHAR(8),

@WeeKDayTitle NCHAR(10)

SELECT @TempDate = '20221207'

SELECT @WeeKDay = DATEPART(WEEKDAY, @TempDate)

 

SELECT @WeeKDayTitle = CASE @WeeKDay WHEN '1' THEN '(일)'

WHEN '2' THEN '(월)'

WHEN '3' THEN '(화)'

WHEN '4' THEN '(수)'

WHEN '5' THEN '(목)'

WHEN '6' THEN '(금)'

WHEN '7' THEN '(토)'

END

SELECT @WeeKDay,@WeeKDayTitle

DROP PROCEDURE dbo.jis_SXPRWkCalcWeekday

GO

CREATE PROCEDURE dbo.jis_SXPRWkCalcWeekday

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10) = '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

 

AS

SELECT CASE DATEPART(WEEKDAY, A.WkDate) WHEN '1' THEN '(일)'

WHEN '2' THEN '(월)'

WHEN '3' THEN '(화)'

WHEN '4' THEN '(수)'

WHEN '5' THEN '(목)'

WHEN '6' THEN '(금)'

WHEN '7' THEN '(토)'

END AS DateName

FROM \#BIZ_IN_DataBlock1 AS A

RETURN
