SELECT R.BizUnit ,

W.WorkOrderNo ,

I.InputDate ,

D.DeptName AS DeptName , -- 입고부서

D.DeptSeq,

C.WorkCenterName AS WorkCenterName ,

C.WorkCenterSeq,

G.ItemName AS GoodItemName ,

G.ItemNo AS GoodItemNo ,

G.ItemSeq,

M.ItemName AS MatItemName ,

M.ItemNo AS MatItemNo ,

M.Spec AS MatItemSpec ,

U.UnitName AS MatUnitName , -- 입고단위

U.UnitSeq,

ISNULL(I.Qty, 0) AS Qty ,

ISNULL(I.StdUnitQty, 0)AS StdUnitQty ,

I.RealLotNo AS RealLotNo , -- 투입자재 RealLotNo

R.RealLotNo AS RRealLotNo , -- (생산실적 RealLotNo)

I.SerialNoFrom ,

I.InputType ,

I.WorkReportSeq ,

I.ItemSerl ,

I.MatItemSeq ,

'' ,

0, --(SELECT TOP 1 AMT FROM \#Temp_Amt WHERE InOutSeq = I.WorkReportSeq AND InOutSubSerl = I.ItemSerl) AS Amt,

0, --CASE WHEN I.Qty = 0 THEN 0 ELSE ISNULL((SELECT TOP 1 AMT FROM \#Temp_Amt WHERE InOutSeq = I.WorkReportSeq AND InOutSubSerl = I.ItemSerl), 0) / I.Qty END AS Price,

R.WorkType ,

B.MinorName AS WorkTypeName ,

ISNULL(R.ProdQty, 0) AS ProdQty ,

--ISNULL(R.OKQty,0) AS ProdQty,

R.ProcRev AS ProcRev ,

M.AssetSeq AS AssetSeq , -- 재고자산분류코드

M1.AssetName AS AssetName , -- 재고자산분류

I.ProcSeq AS MatProcSeq ,

C.CCtrSeq ,

E.CCtrName ,

I.Remark AS Remark ,

Z.CustName AS OrderCustName ,

R.WorkDate AS WorkDate ,

VC.ItemClassLName AS ItemClassLName ,

VC.ItemClassMName AS ItemClassMName ,

VC.ItemClassSName AS ItemClassSName ,

VC.ItemClassLSeq AS ItemClassLSeq ,

VC.ItemClassMSeq AS ItemClassMSeq ,

VC.ItemClassSSeq AS ItemClassSSeq

FROM \_TPDSFCMatinput AS I WITH(NOLOCK) -- 자재투입 테이블

JOIN \_TPDSFCWorkReport AS R WITH(NOLOCK) ON I.WorkReportSeq = R.WorkReportSeq -- 생산실적 테이블

AND I.CompanySeq = R.CompanySeq

--JOIN \#GoodItem AS KA ON R.GoodItemSeq = KA.ItemSeq

JOIN \_TPDBaseWorkCenter AS C WITH(NOLOCK) ON R.WorkCenterSeq = C.WorkCenterSeq

AND I.CompanySeq = C.CompanySeq

JOIN \_TDADept AS D WITH(NOLOCK) ON R.DeptSeq = D.DeptSeq

AND I.CompanySeq = D.CompanySeq

JOIN \_TDAItem AS G WITH(NOLOCK) ON R.GoodItemSeq = G.ItemSeq

AND I.CompanySeq = G.CompanySeq

JOIN \_TDAItem AS M WITH(NOLOCK) ON I.MatItemSeq = M.ItemSeq

AND I.CompanySeq = M.CompanySeq

JOIN \_TDAItemAsset AS M1 WITH(NOLOCK) ON M.AssetSeq = M1.AssetSeq

AND I.CompanySeq = M1.CompanySeq

LEFT OUTER JOIN \_TDAUnit AS U WITH(NOLOCK) ON I.MatUnitSeq = U.UnitSeq

AND I.CompanySeq = U.CompanySeq

JOIN \_TPDBaseProcess AS P WITH(NOLOCK) ON R.ProcSeq = P.ProcSeq

AND I.CompanySeq = P.CompanySeq

LEFT OUTER JOIN \_TPDSFCWorkOrder AS W WITH(NOLOCK) ON R.WorkOrderSeq = W.WorkOrderSeq

AND R.WorkOrderSerl = W.WorkOrderSerl

AND I.CompanySeq = W.CompanySeq

LEFT OUTER JOIN \_TPDMPSDailyProdPlan AS V WITH(NOLOCK) ON V.ProdPlanSeq = W.ProdPlanSeq

AND I.CompanySeq = V.CompanySeq

LEFT OUTER JOIN \_TDASMinor AS B WITH(NOLOCK) ON B.MinorSeq = R.WorkType

AND I.CompanySeq = B.CompanySeq

LEFT OUTER JOIN \_TDACCtr AS E WITH(NOLOCK) ON C.CCtrSeq = E.CCtrSeq

AND I.CompanySeq = E.CompanySeq

LEFT OUTER JOIN \_TDACust AS Z WITH(NOLOCK) ON R.CustSeq = Z.CustSeq

AND R.CompanySeq = Z.CompanySeq

LEFT OUTER JOIN \_TDAItemClass AS TC WITH(NOLOCK) ON TC.CompanySeq = M.CompanySeq --2022.08.30 우승현 추가

AND TC.ItemSeq = M.ItemSeq

AND UMajorItemClass IN (2001,2004)

LEFT OUTER JOIN \_VDAItemClass AS VC WITH(NOLOCK) ON VC.CompanySeq = TC.CompanySeq --2022.08.30 우승현 추가

AND VC.ItemClassSSeq = TC.UMItemClass

WHERE I.CompanySeq = @CompanySeq

AND (@BizUnit = 0 OR R.BizUnit = @BizUnit)

AND (I.InputDate LIKE @StdMonth + '%')
