# 기간에 해당되는 거래처 가져오기

---------------------------------------------------------------------------------------------------------------------------------

-- 거래처 : 구매단가등록, 외주계약등록의 기간에 해당되는 구매거래처, 외주거래처.(구매거래처는 대표단가여부 체크된건 적용)

---------------------------------------------------------------------------------------------------------------------------------

-- 구매단가등록

SELECT A.ItemSeq, MAX(B.Serl) AS Serl

INTO \#TPUBASEBuyPriceItem_MAX_Serl

FROM \#Fix AS A

JOIN \_TPUBASEBuyPriceItem AS B WITH(NOLOCK) ON A.ItemSeq = b.ItemSeq

WHERE B.CompanySeq = @CompanySeq

AND @FoProdDate BETWEEN B.StartDate AND B.EndDate

AND B.IsPrice = '1' -- 대표단가

GROUP BY A.ItemSeq

 

-- 외주계약등록

SELECT A.ItemSeq, MAX(B.Serl) AS Serl

INTO \#TPDOSPPriceItem_MAX_Serl

FROM \#Fix AS A

JOIN \_TPDOSPPriceItem AS B WITH(NOLOCK) ON A.ItemSeq = b.ItemSeq

WHERE B.CompanySeq = @CompanySeq

AND @FoProdDate BETWEEN B.StartDate AND B.EndDate

GROUP BY A.ItemSeq

 

-- 구매거래처

UPDATE A

SET CustSeq = ISNULL(C.CustSeq, 0)

FROM \#Fix AS A

JOIN \#TPUBASEBuyPriceItem_MAX_Serl AS B ON B.ItemSeq = A.ItemSeq

JOIN \_TPUBASEBuyPriceItem AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

AND C.ItemSeq = B.ItemSeq

AND C.Serl = B.Serl

 

-- 외주거래처

UPDATE A

SET CustSeq = ISNULL(C.CustSeq, 0)

FROM \#Fix AS A

JOIN \#TPDOSPPriceItem_MAX_Serl AS B ON B.ItemSeq = A.ItemSeq

JOIN \_TPDOSPPriceItem AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq

AND C.ItemSeq = B.ItemSeq

AND C.Serl = B.Serl
