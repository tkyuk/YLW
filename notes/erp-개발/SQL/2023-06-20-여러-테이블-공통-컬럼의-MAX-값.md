# 여러 테이블 공통 컬럼의 MAX 값

CREATE PROC emk_SWACMRProcRevQuery

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10) = '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

AS

SET NOCOUNT ON                        

 

DECLARE @AccYM NCHAR(6) -- 회계연월

SELECT @AccYM = ISNULL(AccYM, '')

FROM \#BIZ_IN_DataBlock1

 

SELECT ISNULL(MAX(RIGHT('00' + RTRIM(A.MRRev), 2)), '00') AS MRRev

FROM (SELECT MR1Rev AS MRRev FROM emk_TACMR1Mng WHERE CompanySeq = @CompanySeq AND AccYM = @AccYM UNION ALL

SELECT MR2Rev AS MRRev FROM emk_TACMR2Mng_BS WHERE CompanySeq = @CompanySeq AND AccYM = @AccYM UNION ALL

SELECT MR2Rev AS MRRev FROM emk_TACMR2Mng_CF WHERE CompanySeq = @CompanySeq AND AccYM = @AccYM UNION ALL

SELECT MR2Rev AS MRRev FROM emk_TACMR2Mng_PL WHERE CompanySeq = @CompanySeq AND AccYM = @AccYM UNION ALL

SELECT MR3Rev AS MRRev FROM emk_TACMR3Mng WHERE CompanySeq = @CompanySeq AND AccYM = @AccYM UNION ALL

SELECT MR4Rev AS MRRev FROM emk_TACMR4Mng WHERE CompanySeq = @CompanySeq AND AccYM = @AccYM UNION ALL

SELECT MR5Rev AS MRRev FROM emk_TACMR5Mng WHERE CompanySeq = @CompanySeq AND AccYM = @AccYM UNION ALL

SELECT MR6Rev AS MRRev FROM emk_TACMR6Mng WHERE CompanySeq = @CompanySeq AND AccYM = @AccYM UNION ALL

SELECT MR7Rev AS MRRev FROM emk_TACMR7Mng WHERE CompanySeq = @CompanySeq AND AccYM = @AccYM UNION ALL

SELECT MR8Rev AS MRRev FROM emk_TACMR8Mng WHERE CompanySeq = @CompanySeq AND AccYM = @AccYM) AS A

RETURN
