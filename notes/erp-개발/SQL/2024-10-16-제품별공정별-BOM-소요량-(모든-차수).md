CREATE TABLE \#DiffQty(GoodItemSeq INT, DiffQty DECIMAL(19,5), DiffStdQty DECIMAL(19,5))

DECLARE @MatItemSeq INT,

@TotQty DECIMAL(19,5),

@TotStdQty DECIMAL(19,5)

 

SELECT @MatItemSeq = 0,

@TotQty = 0,

@TotStdQty = 0

 

WHILE 1=1

BEGIN

SELECT TOP 1

@MatItemSeq = MatItemSeq,

@TotQty = TotQty,

@TotStdQty = TotStdQty

FROM \#MatItemRealQty

WHERE MatItemSeq \> @MatItemSeq

ORDER BY MatItemSeq

IF @@ROWCOUNT = 0 BREAK

 

TRUNCATE TABLE \#DiffQty

INSERT INTO \#DiffQty(GoodItemSeq, DiffQty, DiffStdQty)

SELECT TOP 1 GoodItemSeq, (SELECT @TotQty - SUM(RealQty) FROM \#MatItemRealQty AS A WHERE MatItemSeq = @MatItemSeq HAVING SUM(RealQty) \<\> @TotQty),

(SELECT @TotStdQty - SUM(RealStdQty) FROM \#MatItemRealQty AS A WHERE MatItemSeq = @MatItemSeq HAVING SUM(RealStdQty) \<\> @TotStdQty)

FROM \#MatItemRealQty

WHERE RealQty = (SELECT MAX(RealQty) FROM \#MatItemRealQty WHERE MatItemSeq = @MatItemSeq)

AND MatItemSeq = @MatItemSeq

 

IF @@ROWCOUNT \<\> 0

BEGIN

UPDATE A

SET RealQty = A.RealQty + ISNULL(B.DiffQty , 0) ,

RealStdQty = A.RealStdQty + ISNULL(B.DiffStdQty , 0)

FROM \#MatItemRealQty AS A

LEFT OUTER JOIN \#DiffQty AS B ON B.GoodItemSeq = A.GoodItemSeq

WHERE A.MatItemSeq = @MatItemSeq

END

 

END

 

-- 보정된 수량Update

UPDATE \#Result

SET Qty = B.RealQty,

StdQty = B.RealStdQty,

MatNeedBalance = B.RealQty - A.MatNeedQty,

MatAmt = A.MatPrice \* B.RealStdQty

FROM \#Result AS A JOIN \#MatItemRealQty AS B ON A.GoodItemSeq = B.GoodItemSeq

AND A.ProcRev = B.ProcRev

AND A.MatItemSeq = B.MatItemSeq

 

 

UPDATE \#Result

SET LossAmt = MatAmt - MatNeedAmt

 

UPDATE \#Result

SET LossRate = CASE WHEN ISNULL(MatNeedAmt,0) = 0 THEN 0 ELSE (LossAmt / MatNeedAmt) \* 100 END
