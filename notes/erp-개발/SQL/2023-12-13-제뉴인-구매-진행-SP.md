# 제뉴인 구매 진행 SP

CREATE TABLE \#TEMP_DelvInItem

(

IDX_NO                                        INT IDENTITY(1,1) ,

DelvSeq                                        INT NULL,

DelvNo                                        NVARCHAR(20) NULL,

DelvInSeq                                INT,

DelvInSerl                                INT,

CompanySeq                                INT

)

 

INSERT INTO \#TEMP_DelvInItem

(DelvInSeq, DelvInSerl, CompanySeq)

SELECT D.DelvInSeq, D.DelvInSerl, M.CompanySeq

FROM \_TPUBuyingAcc AS A

JOIN \_TPUDelvInItem AS D ON A.CompanySeq = D.CompanySeq

AND A.SourceSeq = D.DelvInSeq

AND A.SourceSerl = D.DelvInSerl

AND A.SourceType = '1'

JOIN \_TPUDelvIn AS M WITH(NOLOCK) ON D.CompanySeq = M.CompanySeq

AND D.DelvInSeq = M.DelvInSeq

WHERE D.CompanySeq = @CompanySeq
