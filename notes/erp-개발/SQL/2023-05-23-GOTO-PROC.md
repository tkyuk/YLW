# GOTO / PROC

DROP PROC emk_SWACForeSCReportResultDiv

GO

CREATE PROC emk_SWACForeSCReportResultDiv

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

AS

 

DECLARE @ForeVersion NCHAR(6),

@YYMM NCHAR(6)

 

 

SELECT @ForeVersion = ISNULL(A.ForeVersion,''),

@YYMM = ISNULL(A.YYMM ,'')

FROM \#BIZ_IN_DataBlock1 AS A

 

 

CREATE TABLE \#TempSlipCost

( YYMM NCHAR(6) ,

DeptCd NCHAR(5) ,

AccCd NCHAR(7) ,

SumAmt DECIMAL(19, 5)

)

 

 

CREATE TABLE \#TempToDivCost

( YYMM NCHAR(6) ,

DeptCd NCHAR(5) ,

AccCd NCHAR(7) ,

DivStdCd NCHAR(8) ,

Amt DECIMAL(19, 5)

)

 

CREATE TABLE \#TempToDivResultCost

( YYMM NCHAR(6) ,

DivDeptCd NCHAR(8) ,

DeptCd NCHAR(5) ,

AccCd NCHAR(7) ,

DivStdCd NCHAR(8) ,

DirectAmt DECIMAL(19, 5) ,

DivAmt DECIMAL(19, 5) ,

DivRate DECIMAL(19, 5)

)

 

CREATE TABLE \#TempDiff

( YYMM NCHAR(6) ,

DivDeptCd NCHAR(8) ,

DeptCd NCHAR(5) ,

AccCd NCHAR(7) ,

DivStdCd NCHAR(8) ,

DivAmt DECIMAL(19, 5)

)

 

 

DECLARE @cSysDate NCHAR(12)

 

SELECT @cSysDate = FORMAT(CAST(GETDATE() AS DATETIME), 'yyyyMMddHHmm')

 

 

IF @WorkingTag = 'D1'

GOTO Div1_Proc

ELSE IF @WorkingTag = 'D2'

GOTO Div2_Proc

ELSE IF @WorkingTag ='C'

GOTO Cancel_Proc

RETURN

 

--=============================================================================================================

Div1_Proc:

DELETE emk_TACForeSCReportResult

WHERE YYMM = @YYMM

AND ForeVersion = @ForeVersion

 

Insert Into \#TempSlipCost

SELECT s.YYMM , s.DeptCd, s.AccCd, Sum(s.ConfirmAmt)

FROM emk_TACForecastingAmt s Join TAForeSCAccReg_ESK c

on s.AccCd = c.AccCd

AND s.YYMM = c.YYMM

AND (c.DeptCd = '' or s.DeptCd = c.DeptCd )

AND s.YYVersion = c.ForeVersion

WHERE 1=1

AND s.YYMM= @YYMM

AND s.YYVersion = @ForeVersion

Group By s.YYMM , s.DeptCd, s.AccCd

/\*

 

Insert Into \#TempSlipCost

SELECT s.YYMM , s.DeptCd, s.AccCd, Sum(s.ConfirmAmt)

FROM emk_TACForecastingAmt s Join TAForeSCAccReg_ESK c

on s.AccCd = c.AccCd

AND s.YYMM = c.YYMM

AND (c.DeptCd = '' or s.DeptCd = c.DeptCd )

AND s.YYVersion = c.ForeVersion

WHERE 1=1

AND s.YYMM= '200601'

AND s.YYVersion = '200501'

Group By s.YYMM , s.DeptCd, s.AccCd

 

SELECT \* FROM \#TempSlipCost WHERE deptcd ='00014' AND acccd ='4110102'

SELECT i.YYMM, d.DivDeptCd, i.DeptCd, i.AccCd, '', i.SumAmt ,0 , 100

FROM \#TempSlipCost i Join TAForeSCDirDept_ESK d

on i.YYMM = d.YYMM

AND i.DeptCd = d.DeptCd

 

SELECT \* FROM TAForeSCDirDept_ESK WHERE yymm ='200601'

SELECT \* FROM \#TempSlipCost WHERE deptcd ='00014' AND acccd ='4110102'

 

\*/

 

 

Insert into \#TempToDivResultCost(YYMM, DivDeptCd, DeptCd, AccCd , DivStdCd, DirectAmt, DivAmt , DivRate )

SELECT i.YYMM, d.DivDeptCd, i.DeptCd, i.AccCd, '', i.SumAmt ,0 , 100

FROM \#TempSlipCost i Join emk_TACDirDept d

on i.YYMM = d.YYMM

AND i.DeptCd = d.DeptCd

 

---간접부서의 것을 배부하는 루틴.

Insert into \#TempToDivCost(YYMM, DeptCd, AccCd, DivStdCd, Amt)

SELECT i.YYMM, i.DeptCd, i.AccCd, r.DivStdCd, i.SumAmt

FROM \#TempSlipCost i Left Outer Join emk_TACDirDept d

on i.YYMM = d.YYMM

AND i.DeptCd = d.DeptCd

Join TAForeSCAccDeptDivReg_ESK r

on i.YYMM = r.YYMM

AND i.AccCd = r.AccCd

AND i.DeptCd = r.InDirDeptCd

AND r.ForeVersion = @ForeVersion

WHERE 1=1

AND ISNULL(d.YYMM , '') = ''

Insert into \#TempToDivResultCost(YYMM, DivDeptCd, DeptCd, AccCd , DivStdCd, DivAmt , DirectAmt , DivRate )

SELECT c.YYMM, v.DivDeptCd, c.DeptCd, c.AccCd, c.DivStdCd, Round(c.Amt\*v.DivRate/100 , 0) , 0 , v.DivRate

FROM \#TempToDivCost c Join emk_TACForeSCDivValueReg v

on c.YYMM = v.YYMM

AND c.DivStdCd = v.DivStdCd

AND v.ForeVersion = @ForeVersion

WHERE Div2StdYn \<\> '1'

 

Insert into \#TempToDivResultCost(YYMM, DivDeptCd, DeptCd, AccCd , DivStdCd, DivAmt , DirectAmt , DivRate )

SELECT k.YYMM, Max(k.DivDeptCd) , k.DeptCd, k.AccCd , k.DivStdCd, d.DiffAmt , 0 , 0

FROM \#TempToDivResultCost k Join ( SELECT YYMM, DeptCd, AccCd , DivStdCd, Max(DivAmt) As MaxAmt FROM \#TempToDivResultCost

Group By YYMM, DeptCd, AccCd , DivStdCd ) c

on k.YYMM = c.YYMM

AND k.DeptCd = c.DeptCd

AND k.AccCd = c.AccCd

AND k.DivStdCd = c.DivStdCd

AND k.DivAmt = c.MaxAmt

 

join ( SELECT k.YYMM, k.DeptCd, k.AccCd, k.DivStdCd, c.Amt - k.DivAmt as DiffAmt

FROM ( SELECT YYMM as YYMM , DeptCd as DeptCd, AccCd as AccCd, DivStdCd as DivStdCd, Sum(DivAmt) as DivAmt

FROM \#TempToDivResultCost Group by YYMM , DeptCd , AccCd , DivStdCd ) k

Join \#TempToDivCost c

on k.YYMM = c.YYMM

AND k.DeptCd = c.DeptCd

AND k.AccCd = c.AccCd

AND k.DivStdCd = c.DivStdCd ) d

on k.YYMM = d.YYMM

AND k.DeptCd = d.DeptCd

AND k.AccCd = d.AccCd

AND k.DivStdCd = d.DivStdCd

Group by k.YYMM , k.DeptCd , k.AccCd , k.DivStdCd, d.DiffAmt

-- exec SASForeSCReportResultDiv_ESK N'D1', N'200601', N'19890021', N'200501'

Insert into emk_TACForeSCReportResult(YYMM, DivDeptCd, AccCd, DeptCd, DivStdCD, DivideAmt, RegEmpid , RegDate, Version , DirectAmt , DivRate , ForeVersion )

SELECT YYMM, DivDeptCd, AccCd , DeptCd, DivStdCd, Sum(DivAmt) , @pEmpId, @cSysDate, '1' , Sum( DirectAmt) , Sum(DivRate ) , @ForeVersion

FROM \#TempToDivResultCost

Group by YYMM , DivDeptCd, DeptCd , AccCd , DivStdCd

--SELECT 'OK', '1st Allocation Work is Completed'

RETURN

--=============================================================================================================

Div2_Proc:

 

DECLARE @cStdDivDept NCHAR(8) , @nDivAmt DECIMAL(19, 5) , @cDivStdCd NCHAR(8) , @cRate DECIMAL(19, 5)

 

 

SELECT @cStdDivDept = DivDeptCd , @cDivStdCd= DivStdCd , @cRate = DivRate \* (-1)

FROM emk_TACForeSCDivValueReg

WHERE Div2StdYn ='1'

AND YYMM = @YYMM

AND ForeVersion = @ForeVersion

 

DELETE emk_TACForeSCReportResult

WHERE 1=1

AND YYMM = @YYMM

AND Version ='2'

AND ForeVersion = @ForeVersion

 

Insert into \#TempToDivCost(YYMM, AccCd, DivStdCd, Amt )

SELECT i.YYMM, '9000501', @cDivStdCd , Round(Sum(i.DivideAmt + i.DirectAmt) \* @cRate/100, 0)

FROM emk_TACForeSCReportResult i Join emk_TACForeSC2thAccReg e

on i.YYMM = e.YYMM

AND i.AccCd = e.AccCd

AND i.ForeVersion = e.ForeVersion

WHERE 1=1

AND i.YYMM = @YYMM

AND i.DivDeptCd = @cStdDivDept

AND i.ForeVersion = @ForeVersion

Group By i.YYMM

 

Insert into \#TempToDivResultCost(YYMM, DivDeptCd, AccCd , DivStdCd, DivAmt , DivRate )

SELECT c.YYMM, v.DivDeptCd, c.AccCd, c.DivStdCd, Round(c.Amt\*v.DivRate/100 , 0 ), v.DivRate

FROM \#TempToDivCost c Join emk_TACForeSCDivValueReg v

on c.YYMM = v.YYMM

AND c.DivStdCd = v.DivStdCd

AND v.ForeVersion = @ForeVersion

WHERE 1= 1

AND v.DivDeptCd \<\> @cStdDivDept

DECLARE @cDivSum DECIMAL(19, 5) , @cIniSum DECIMAL(19, 5)

 

SELECT @cDivSum = Sum(DivAmt) FROM \#TempToDivResultCost

SELECT @cIniSum = Sum(Amt) FROM \#TempToDivCost

IF @cDivSum \<\> @cIniSum

BEGIN

Insert into \#TempToDivResultCost(YYMM, DivDeptCd, AccCd , DivStdCd, DivAmt , DivRate)

SELECT top 1 YYMM, DivDeptCd, AccCd, DivStdCd, @cIniSum- @cDivSum , 0

FROM \#TempToDivResultCost

WHERE DivAmt = (SELECT Max(DivAmt) FROM \#TempToDivResultCost )

 

END

 

Insert into \#TempToDivResultCost(YYMM, DivDeptCd, AccCd , DivStdCd, DivAmt , DivRate )

SELECT c.YYMM, @cStdDivDept, c.AccCd, c.DivStdCd, c.Amt\* (-1) , @cRate \* (-1)

FROM \#TempToDivCost c

 

 

Insert into emk_TACForeSCReportResult(YYMM, DivDeptCd, AccCd, DeptCd, DivStdCD, DivideAmt, RegEmpid , RegDate, Version , DirectAmt , DivRate, ForeVersion )

SELECT YYMM, DivDeptCd, AccCd, @cStdDivDept, DivStdCd, Sum(DivAmt) , @pEmpId, @cSysDate, '2' , 0 , Sum(DivRate ) , @ForeVersion

FROM \#TempToDivResultCost

Group by YYMM , DivDeptCd, AccCd , DivStdCd

 

 

 

--SELECT 'OK', '2nd Allocation Work is Completed'

 

RETURN

--==================================================================================================

Cancel_Proc:

 

DELETE emk_TACForeSCReportResult

WHERE 1=1

AND YYMM = @YYMM

AND ForeVersion = @ForeVersion

 

 

 

--SELECT 'OK', 'Allocation is Conceled'

RETURN
