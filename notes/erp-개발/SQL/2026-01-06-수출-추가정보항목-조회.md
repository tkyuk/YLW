# 수출 추가정보항목 조회

SELECT \*

FROM \_TCOMUserDefine AS A WITH(NOLOCK)

LEFT OUTER JOIN \_TCADictionary AS Dic WITH(NOLOCK) ON A.CompanySeq = 1

AND Dic.LanguageSeq = 1

AND A.WordSeq = Dic.WordSeq

WHERE A.CompanySeq = 1

AND A.TableName = '\_TSLExpImpInfo'

AND A.DefineUnitSeq = 8212001

ORDER BY A.QrySort
