# 

OUTER APPLY (SELECT DISTINCT N.CompanySeq, N.ItemSeq, N.LOTNo

FROM \_TLGInOutLotStock AS N

WHERE N.InOut = -1

AND N.CompanySeq = @CompanySeq

AND N.LotNo = N.LotNo

AND N.ItemSeq = N.ItemSeq

AND N.WHSeq = @AnydoWhSeq) AS N

WHERE (@InStockSeq = 1000707001 -- 재고유무가 전체

OR (@InStockSeq = 1000707002 AND (A.PreQty \<\> 0 OR A.InQty \<\> 0 OR A.OutQty \<\> 0 OR A.StockQty \<\> 0))) -- 재고유무가 재고有

ORDER BY LOTNo

 

SELECT \* FROM \_TSLInvoiceItem WHERE WHSeq = 45 AND LOTNo \<\> ''

SELECT \* FROM \_TLGInOutDailyItem WHERE OutWHSeq = 45 AND LOTNo \<\> ''

SELECT \* FROM \_TLGInOutLotStock WHERE WHSeq = 45 AND InOut = -1
