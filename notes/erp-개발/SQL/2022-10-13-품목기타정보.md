# 품목기타정보

SELECT \* FROM \_TDAUMajor WHERE MajorSeq = 2140170

SELECT \* FROM \_TDAUMINOR WHERE MajorSeq = 2140170

SELECT \* FROM \_TDAUMinorValue WHERE MajorSeq = 2140170 AND Serl = 15000001

SELECT \* FROM \_TDAUMinorValue WHERE MajorSeq = 2140170 AND Serl = 15000002

 

SELECT \* FROM \_TCOMUserDefine WHERE DefineUnitSeq = 2140170 and TableName = '\_TDAUMajor'

 

select \* from \_TDAItem where ItemSeq = 1013

SELECT \* FROM \_TCOMUserDefine WHERE TableName = '\_TDAItem'

 

 

select \* from \_TDADefineItemClass AS A WITH(NOLOCK)

LEFT OUTER JOIN \_TDAItemClass B With(Nolock) ON A.CompanySeq = B.CompanySeq

AND A.UMajorItemClass = B.UMajorItemClass

AND B.ItemSeq = 1013

where A.UMajorItemClass =2140170

SELECT \* FROM \_TDAItemClass AS B WHERE B.UMajorItemClass = 2140170

 

SELECT C.ValueSeq,\* FROM \_TDAItem AS A

LEFT OUTER JOIN \_TDAItemClass AS B ON A.ItemSeq = B.ItemSeq AND B.UMajorItemClass = 2140170

LEFT OUTER JOIN \_TDAUMinorValue AS C ON C.MinorSeq= B.UMItemClass AND C.SERL IN (15000001,15000002)

WHERE C.CompanySeq IS NOT NULL

 

ISNULL(H.UMItemClass , 0) AS UmUse , -- 사용처코드

ISNULL(I.MinorName ,'') AS UmUseName -- 사용처

 

 

LEFT OUTER JOIN \_TDAItemClass AS H WITH(NOLOCK) ON H.CompanySeq = B.CompanySeq

AND H.ItemSeq = B.ItemSeq

AND H.UMajorItemClass = 2140257

LEFT OUTER JOIN \_TDAUMinor AS I WITH(NOLOCK) ON I.CompanySeq = H.CompanySeq

AND I.MinorSeq = H.UMItemClass
