# 구매 진행 수량

-- 진행체크할 테이블값 테이블

CREATE TABLE \#TMP_PROGRESSTABLE

(

IDOrder INT,

TABLENAME NVARCHAR(100)

)

-- 진행체크할 데이터 테이블

CREATE TABLE \#Temp_PJTBOM

(

IDX_NO INT IDENTITY,

PJTSeq INT,

BOMSerl INT,

CompleteCHECK INT,

SMProgressType INT NULL,

ReqQty DECIMAL(19,5),

ItemSeq INT

)

 

CREATE TABLE \#TCOMProgressTracking

(

IDX_NO INT,

IDOrder INT,

Seq INT,

Serl INT,

SubSerl INT,

Qty DECIMAL(19, 5),

StdQty DECIMAL(19, 5),

Amt DECIMAL(19, 5),

VAT DECIMAL(19, 5),

DOMAmt DECIMAL(19, 5),

DOMVAT DECIMAL(19, 5)

)

CREATE TABLE \#PJTBOM_Tracking

(

IDX_NO INT

,POReqQty DECIMAL(19,5)

,POQty DECIMAL(19,5)

,DelvQty DECIMAL(19,5)

,DelvInQty DECIMAL(19,5)

)

--===================================================================

-- 구매 진행 수량 가져오기

--===================================================================

INSERT \#TMP_PROGRESSTABLE(IDOrder,TABLENAME)

SELECT 1, '\_TPUORDPOItem' -- 구매발주

UNION

SELECT 2, '\_TPUDelvItem' -- 구매납품

UNION

SELECT 3, '\_TPUDelvInItem' -- 구매입고

UNION

SELECT 4, '\_TUIImpBLItem' -- 수입BL

UNION

SELECT 5, '\_TUIImpDelvItem' -- 수입입고

UNION

SELECT 6, '\_TPUORDPOReqItem' -- 구매요청

INSERT INTO \#Temp_PJTBOM(PJTSeq, BOMSerl, CompleteCHECK,ReqQty, ItemSeq)

SELECT A.PJTSeq,

A.BOMSerl,

-1,

A.TotQty,

A.ItemSeq

FROM \_TPJTBOM AS A WITH(NOLOCK)

JOIN das_TPJTBOM AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.PJTSeq = A.PJTSeq

AND B.BOMSerl = A.BOMSerl

LEFT OUTER JOIN \_TDAUMinorValue AS S2 WITH(NOLOCK) ON S2.CompanySeq = A.CompanySeq

AND S2.MinorSeq = A.UMSupplyType-- 구매여부

AND S2.Serl = 1000001

LEFT OUTER JOIN \_TDAUMinorValue AS S4 WITH(NOLOCK) ON S4.CompanySeq = A.CompanySeq

AND S4.MinorSeq = A.UMSupplyType -- 생산여부

AND S4.Serl = 1000004

WHERE A.BOMSerl \<\> -1

AND A.PJTSeq IN(SELECT PJTSeq FROM \#PJTMST)

AND (

(@SupplyType = 0 )

OR (@SupplyType = 1 AND S2.ValueText = '1') -- 구매

OR (@SupplyType = 2 AND S4.ValueText = '1') -- 제조 (생산)

)

EXEC \_SWCOMProgressTracking @CompanySeq, '\_TPJTBOM', '#Temp_PJTBOM', 'PJTSeq', 'BOMSerl', ''

 

INSERT INTO \#PJTBOM_Tracking

(

IDX_NO

,POReqQty

,POQty

,DelvQty

,DelvInQty

> )        

SELECT A.IDX_NO,

SUM(CASE IDOrder WHEN 6 THEN A.Qty                         ELSE 0 END),

SUM(CASE IDOrder WHEN 1 THEN A.Qty                         ELSE 0 END),

SUM(CASE WHEN IDOrder IN (2, 4) THEN A.Qty ELSE 0 END),

SUM(CASE WHEN IDOrder IN (3, 5) THEN A.Qty ELSE 0 END)

FROM \#TCOMProgressTracking AS A

GROUP BY A.IDX_No

TRUNCATE TABLE \#TMP_PROGRESSTABLE

TRUNCATE TABLE \#TCOMProgressTracking
