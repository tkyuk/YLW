UPDATE \#Temp_InOutItemProg

SET ReqSeq = ISNULL(C.ReqSeq, 0)

, ReqNo = ISNULL(C.ReqNo, '')

> , ReqDeptSeq = ISNULL(C.DeptSeq, 0)        
>
> , ReqEmpSeq = ISNULL(C.EmpSeq, 0)        
>
> , Qty         = ISNULL(D.Qty, 0)        

FROM \#Temp_InOutItemProg                        AS A

JOIN \#TCOMSourceTracking                        AS B ON A.IDX_NO = B.IDX_NO

AND B.IDOrder = 2

JOIN \_TLGInOutReq                                        AS C WITH(NOLOCK) ON C.CompanySeq = @CompanySeq AND B.Seq = C.ReqSeq

> LEFT OUTER JOIN \_TLGInOutReqItem        AS D WITH(NOLOCK) ON D.CompanySeq = @CompanySeq AND D.ReqSeq = C.ReqSeq

 

--2022.08.10 InvoiceNo 찾기 :: 우승현

UPDATE \#Temp_InOutItemProg

SET InvoiceNo = ISNULL(C.InvoiceNo, '')

FROM \#Temp_InOutItemProg AS A

JOIN \#TCOMSourceTracking AS B ON B.IDX_NO = A.IDX_NO

AND B.IDOrder = 3

JOIN \_TSLInvoice AS C ON C.InvoiceSeq = B.Seq

 

--\[기타출고품목조회\] 화면 금액 컬럼 수정 :: 22.04.07 우승현

--\[기타입고품목조회\] InOutType조건 삭제 :: 22.06.17 방혁

UPDATE \#Temp_InOutItemProg

SET Amt = ISNULL(C.Amt, 0)

FROM \#Temp_InOutItemProg AS A

JOIN (

SELECT A.InOutType, A.InOutSeq, A.InOutSerl, SUM(B.Amt) AS AMT

FROM \#Temp_InOutItemProg AS A

JOIN \_TLGInOutStock AS B WITH(NOLOCK) ON B.CompanySeq = @CompanySeq

AND B.InOutType = A.InOutType

AND B.InOutSeq = A.InOutSeq

AND B.InOutSerl = A.InOutSerl

WHERE B.CompanySeq = @CompanySeq

--AND B.InOutType = 30

GROUP BY A.InOutType, A.InOutSeq, A.InOutSerl

) AS C ON C.InOutType = A.InOutType

AND C.InOutSeq = A.InOutSeq

AND C.InOutSerl = A.InOutSerl

--WHERE A.InOutType = 30
