# 컬럼 중간에 추가하기 (동적쿼리)

/\*

테이블 중간에 컬럼 추가 시 기존 데이터 백업 후 처리 (컬럼 명칭 변경은 처리할 수 없습니다.)

 

1\. @TableName 명칭 지정

2\. CREATE문 넣기

\*/

 

 

DECLARE @TableName NVARCHAR(128) = '테이블명';

DECLARE @BackupName NVARCHAR(128) = @TableName + '\_BAK';

DECLARE @Columns NVARCHAR(MAX) = '';

DECLARE @Sql NVARCHAR(MAX);

 

-- 1. 백업 테이블 생성 (기존 데이터 보관)

SET @Sql = 'IF OBJECT_ID(''' + @BackupName + ''', ''U'') IS NOT NULL DROP TABLE ' + QUOTENAME(@BackupName);

EXEC sp_executesql @Sql;

 

SET @Sql = 'SELECT \* INTO ' + QUOTENAME(@BackupName) + ' FROM ' + QUOTENAME(@TableName);

EXEC sp_executesql @Sql;

 

-- 2. 원본 테이블 삭제

SET @Sql = 'IF OBJECT_ID(''' + @TableName + ''', ''U'') IS NOT NULL DROP TABLE ' + QUOTENAME(@TableName);

EXEC sp_executesql @Sql;

-- 4. \[핵심\] 백업 테이블에 존재하는 컬럼 리스트 추출 (STRING_AGG 사용 - SQL 2017 이상)

-- 만약 하위 버전이라면 STUFF/FOR XML 방식을 써야 합니다.

SELECT @Columns = STRING_AGG(QUOTENAME(name), ', ')

FROM sys.columns

WHERE object_id = OBJECT_ID(@BackupName);

 

-- 5. 동적 INSERT 실행 (백업에 있던 컬럼만 골라서 복구)

IF @Columns IS NOT NULL AND @Columns \<\> ''

BEGIN

SET @Sql = 'INSERT INTO ' + QUOTENAME(@TableName) + ' (' + @Columns + ') ' +

'SELECT ' + @Columns + ' FROM ' + QUOTENAME(@BackupName);

EXEC sp_executesql @Sql;

END

 

-- 6. 백업 테이블 삭제

SET @Sql = 'DROP TABLE ' + QUOTENAME(@BackupName);

EXEC sp_executesql @Sql;
