# @SPName 없는 경우 sp 지정

IF NOT EXISTS(SELECT 1 FROM SYSOBJECTS WHERE id = object_id(@SPName) AND sysstat & 0xf = 4)

BEGIN

SELECT @SPName = 'dbo.\_SWPRAdjTotAbrIncomePrint'

END

DROP PROCEDURE cymc_SWPRAdjTotAbrIncomePrint

GO

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@PgmSeq INT = 0,

,@SPName NVARCHAR(200)

FROM \#BIZ_IN_DataBlock1

> SELECT @SPName = '\_SWPRAdjTotAbrIncomePrint\_' + @YY
