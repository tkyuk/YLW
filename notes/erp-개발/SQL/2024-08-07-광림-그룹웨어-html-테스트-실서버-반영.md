# 광림 그룹웨어 html 테스트 -\> 실서버 반영

UPDATE A

SET Body = B.Body

FROM KL.DBO.\_TCOMEnvGroupWare AS A

JOIN KLTest.DBO.\_TCOMEnvGroupWare AS B ON B.CompanySeq = A.CompanySeq

AND B.WorkKind = A.WorkKind

WHERE A.WorkKind IN ('Estimate1','Estimate2')
