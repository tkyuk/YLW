--=============================================================

-- 환경설정 가져오기

--=============================================================

DECLARE @DomVatPoint INT -- 6506 원화부가세소수점처리(외주)

, @DomAmtPoint INT -- 6507 원화금액소수점처리(외주)

, @KorCurrSeq INT -- 13 자국통화

, @CurAmtDec INT -- 14 거래금액 소수점 자리수(외화)

, @DomAmtDec INT -- 15 자국통화금액 소수점 자리수(원화)

, @DomPriceDec INT -- 6 구매/자재 원화단가 소수점 자릿수

, @CurPriceDec INT -- 7 구매/자재 거래통화단가 소수점 자릿수

, @MatDec INT -- 20230303.추가

 

 

-- 원화부가세 소수점 처리방식 환경설정 가져오기(외주)

EXEC dbo.\_SWCOMEnv @CompanySeq, 6506, @UserSeq, @@PROCID, @DomVatPoint OUTPUT

-- 원화금액 소수점 처리방식 환경설정 가져오기(외주)

EXEC dbo.\_SWCOMEnv @CompanySeq, 6507, @UserSeq, @@PROCID, @DomAmtPoint OUTPUT

-- 구매/자재 원화단가 소수점 자릿수 환경설정 가져오기

EXEC dbo.\_SWCOMEnv @CompanySeq, 6 , @UserSeq, @@PROCID, @DomPriceDec OUTPUT

-- 구매/자재 거래통화단가 소수점 자릿수 환경설정 가져오기

EXEC dbo.\_SWCOMEnv @CompanySeq, 7 , @UserSeq, @@PROCID, @CurPriceDec OUTPUT

-- 자국통화

EXEC dbo.\_SWCOMEnv @CompanySeq, 13 , @UserSeq, @@PROCID, @KorCurrSeq OUTPUT

-- 거래금액 소수점 자릿수 환경설정 가져오기

EXEC dbo.\_SWCOMEnv @CompanySeq, 14 , @UserSeq, @@PROCID, @CurAmtDec OUTPUT

-- 자국통화금액 소수점 자릿수 환경설정 가져오기

EXEC dbo.\_SWCOMEnv @CompanySeq, 15 , @UserSeq, @@PROCID, @DomAmtDec OUTPUT

EXEC dbo.\_SWCOMEnv @CompanySeq, 5, 0, @@PROCID, @MatDec OUTPUT -- 20230303.구매/자재 수량 소숫점 자릿수 추가
