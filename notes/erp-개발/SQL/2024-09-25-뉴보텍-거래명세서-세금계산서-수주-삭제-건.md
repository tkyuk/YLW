@xmlDocument NVARCHAR(MAX),

@xmlFlags INT = 0,

@ServiceSeq INT = 0,-- 서비스등록한것 Seq가 넘어온다.

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0

AS

DECLARE @docHandle INT,

@InvoiceSeq INT,

@TotCurAmt DECIMAL(19,5),

@TotVAT DECIMAL(19,5),

@TotAmt DECIMAL(19,5),

@LineKind NCHAR(1),

@IsUsedCustEmp NCHAR(1),

@OrderSerl INT

-- 서비스 마스타 등록 생성

CREATE TABLE \#TSLInvoice (WorkingTag NCHAR(1) NULL)

ExEC \_SCAOpenXmlToTemp @xmlDocument, @xmlFlags, @CompanySeq, @ServiceSeq, 'DataBlock1', '#TSLInvoice'

/\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*/

-- 마스터 정보를 조회(이부분은 거래명세서 기준으로 마스터를 조회함)

/\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*/

SELECT @InvoiceSeq = MIN(B.InvoiceSeq)

FROM \#TSLInvoice AS B

/\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*/

-- 거래명세서비고진행 환경설정 가져오기 : 체크시 거래명세서에서 세금계산서진행시 비고를 넘기지 않는다.

EXEC dbo.\_SCOMEnv @CompanySeq, 8046, @UserSeq, @@PROCID, @LineKind OUTPUT

/\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*/

-- 영업담당 사용여부 환경설정 가져오기 : 거래명세서 -\> 세금계산서 점프 시, 영업담당 사용여부를 체크하여,

-- 사용일 경우, 청구처의 영업담당자로 설정해준다.

-- 수정자 : 이성덕 2011년 11월 8일

EXEC dbo.\_SCOMEnv @CompanySeq, 8001, @UserSeq, @@PROCID, @IsUsedCustEmp OUTPUT

SELECT @IsUsedCustEmp = '0' -- 청구처의 영업담당자가 아닌 거래처의 영업담당자를 지정함.(뉴보텍 요청 2015.12.21)

SELECT @TotCurAmt = SUM(CurAmt),

@TotVAT = SUM(CurVAT),

@TotAmt = ISNULL(SUM(CurAmt), 0) + ISNULL(SUM(CurVAT), 0)

FROM \_TSLInvoiceItem

WHERE CompanySeq = @CompanySeq

AND InvoiceSeq = @InvoiceSeq

 

SELECT A.InvoiceSeq AS InvoiceSeq, --내부번호

B.BizUnitName AS BizUnitName, --사업부문

A.BizUnit AS BizUnit, --사업부문코드

A.InvoiceNo AS InvoiceNo, --거래명세서번호

A.InvoiceDate AS InvoiceDate, --거래명세서일

G.MinorName AS SMExpKindName, --수출구분

A.SMExpKind AS SMExpKind, --수출구분코드

H.MinorName AS UMOutKindName, --출고구분

--CASE ISNULL(A1.ValueSeq, 0) WHEN 0 THEN 8020001 ELSE A1.ValueSeq END AS UMOutKind, --출고구분코드

A.UMOutKind AS UMOutKind, --출고구분코드

-- C.DeptName AS DeptName, --부서

-- A.DeptSeq AS DeptSeq, --부서코드

-- D.EmpName AS EmpName, --담당자

-- A.EmpSeq AS EmpSeq, --담당자코드

CASE @WorkingTag WHEN 'R' THEN C.DeptName

ELSE CASE WHEN @IsUsedCustEmp = '1'

THEN CASE WHEN ISNULL(R.UpperCustSeq,0) = 0

THEN ISNULL(C.DeptName,'')

ELSE ISNULL(T.DeptName,'') END

ELSE ISNULL(C.DeptName,'') END END AS DeptName, --부서

CASE @WorkingTag WHEN 'R' THEN A.DeptSeq

ELSE CASE WHEN @IsUsedCustEmp = '1'

THEN CASE WHEN ISNULL(R.UpperCustSeq,0) = 0

THEN ISNULL(A.DeptSeq,0)

ELSE ISNULL(F.DeptSeq,0) END

ELSE ISNULL(A.DeptSeq,0) END END AS DeptSeq, --부서코드

ELSE CASE WHEN @IsUsedCustEmp = '1'

THEN CASE WHEN ISNULL(R.UpperCustSeq,0) = 0

THEN ISNULL(D.EmpName,'')

ELSE ISNULL(S.EmpName,'') END

ELSE ISNULL(D.EmpName,'') END END AS EmpName, --담당자

CREATE TABLE \#TInvoiceBillFIFO

(

IDX_NO INT,

IDX_Serl INT,

Seq INT,

Serl INT,

BillProgQty DECIMAL(19,5),

BillProgSTDQty DECIMAL(19,5),

BillProgAmt DECIMAL(19,5),

BillProgVAT DECIMAL(19,5)

)

CREATE INDEX IX\_#TInvoiceBillFIFO ON \#TInvoiceBillFIFO (IDX_NO, IDX_Serl)

DECLARE @IDX_NO INT,

@IDX_Serl INT

SELECT @IDX_NO = 0, @IDX_Serl = 0 -- 초기화

WHILE 1=1

BEGIN

SELECT TOP 1

@IDX_NO = A.IDX_NO,

@IDX_Serl = 0

FROM \#TOrderBill AS A

WHERE A.IDX_NO \> @IDX_NO

ORDER BY A.IDX_NO

IF @@ROWCOUNT = 0 BREAK

WHILE 1=1

BEGIN

SELECT TOP 1

@IDX_Serl = A.IDX_Serl

FROM \#TOrderInvoicePre AS A

WHERE A.IDX_NO = @IDX_NO

AND A.IDX_Serl \> @IDX_Serl

ORDER BY A.IDX_Serl

IF @@ROWCOUNT = 0 BREAK

INSERT INTO \#TInvoiceBillFIFO

(

IDX_NO, IDX_Serl, Seq, Serl, BillProgQty, BillProgSTDQty, BillProgAmt, BillProgVAT

)

SELECT A.IDX_NO, A.IDX_Serl, A.Seq, A.Serl,

CASE WHEN A.PlusYnQty + B.PlusYnQty = 0 THEN 0 ELSE CASE WHEN ABS(A.Qty ) \< ABS(B.Qty ) THEN A.Qty ELSE B.Qty END END AS Qty ,

CASE WHEN A.PlusYnSTDQty + B.PlusYnSTDQty = 0 THEN 0 ELSE CASE WHEN ABS(A.STDQty) \< ABS(B.STDQty) THEN A.STDQty ELSE B.STDQty END END AS STDQty,

CASE WHEN A.PlusYnAmt + B.PlusYnAmt = 0 THEN 0 ELSE CASE WHEN ABS(A.Amt ) \< ABS(B.Amt ) THEN A.Amt ELSE B.Amt END END AS Amt ,

CASE WHEN A.PlusYnVAT + B.PlusYnVAT = 0 THEN 0 ELSE CASE WHEN ABS(A.VAT ) \< ABS(B.VAT ) THEN A.VAT ELSE B.VAT END END AS VAT

FROM \#TOrderInvoicePre AS A

JOIN \#TOrderBill AS B ON A.IDX_NO = B.IDX_NO

WHERE A.IDX_NO = @IDX_NO

AND A.IDX_Serl = @IDX_Serl

UPDATE B

SET Qty = B.Qty - A.Qty ,

STDQty = B.STDQty - A.STDQty,

Amt = B.Amt - A.Amt ,

VAT = B.VAT - A.VAT

FROM \#TOrderInvoicePre AS A

JOIN \#TOrderBill AS B ON A.IDX_NO = B.IDX_NO

WHERE A.IDX_NO = @IDX_NO

AND A.IDX_Serl = @IDX_Serl

IF EXISTS(SELECT TOP 1 1 FROM \#TOrderBill WHERE IDX_NO = @IDX_NO AND PlusYnQty \<\> CASE WHEN Qty \<= 0 THEN -1 ELSE 1 END)

BEGIN

BREAK

END

END

END
