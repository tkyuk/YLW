# 전표 그룹웨어 건

SELECT \* FROM \_TSLBILL WHERE BILLSeq = 15

SELECT \* FROM \_TSLSalesBILLRelation WHERE BILLSeq = 15

SELECT \* FROM \_TSLSales WHERE SalesSeq = 52

 

 

SELECT \* FROM \_TCOMConfirm WHERE ConfirmSeq = 6363 AND CfmSeq = 65

 

 

SELECT \* FROM \_TCOMGroupWare WHERE WorkKind = 'Slip' ORDER BY LastDateTime DESC

 

SELECT \* FROM \_TCOMGroupWare WHERE WorkKind = 'Slip' AND TblKey = 196

 

 

SELECT \* FROM \_TCOMGroupWareLog WHERE WorkKind = 'Slip' AND TblKey = 196

SELECT \* FROM \_TCOMGroupWareStatusLog WHERE WorkKind = 'Slip' AND TblKey = 196

 

SELECT \* FROM \_TACSlip WHERE SlipMstSeq = 196

SELECT \* FROM \_TACSlipRow WHERE SlipSeq = 556
