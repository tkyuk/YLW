SELECT B.WHSeq, B.ItemSeq, B.SerialNo, B.InOut, A.\*

FROM \_TLGInOutSerialSub AS A

LEFT OUTER JOIN \_TLGInOutSerialStock AS B ON B.CompanySeq = A.CompanySeq

AND B.InOutType = A.InOutType

AND B.InOutSeq = A.InOutSeq

AND B.InOutSerl = A.InOutSerl

AND B.InOutSerialSerl = A.InOutSerialSerl

WHERE A.SerialNo in ('svs_SerialNo2','AA003BB')
