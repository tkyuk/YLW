# 옵션 기준 변경 (소분류 -\> 대분류)

LEFT OUTER JOIN \_VDAItemClass AS D0 WITH(NOLOCK) ON D0.CompanySeq = C1.CompanySeq

AND D0.ItemClassSSeq = C1.UMItemClass

LEFT OUTER JOIN svs_TPDSerialBase AS D1 WITH(NOLOCK) ON D1.CompanySeq = D0.CompanySeq

AND D1.UMItemClassL = D0.ItemClassLSeq

AND ISNULL(D1.IsNotUse, '') \<\> '1'

LEFT OUTER JOIN svs_TPDSerialExcp AS E1 WITH(NOLOCK) ON E1.CompanySeq = I.CompanySeq

AND E1.ItemSeq = I.ItemSeq

AND ISNULL(E1.IsNotUse, '') \<\> '1'
