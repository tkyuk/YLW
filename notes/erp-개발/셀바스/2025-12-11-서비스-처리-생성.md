# 서비스 처리 생성

SELECT InvoiceSeq,InOutSeq,\* FROM svs_TSIASServiceMaster where ReceiptSeq = 57

SELECT InvoiceSeq,InOutSeq,InvoiceSerl,InOutSerl,\* FROM svs_TSIASServiceRepair where ReceiptSeq = 57

 

 

 

SELECT A.LastDateTime, A.LastUserSeq, A.InvoiceSeq, B.\*

FROM svs_TSIASServiceRepair AS A

JOIN \_TSLInvoiceItem AS B ON B.InvoiceSeq = A.InvoiceSeq

AND B.InvoiceSerl = A.InvoiceSerl

JOIN \_TLGInOutStock AS C ON C.InOutSeq = B.InvoiceSeq

AND C.InOutSerl = B.InvoiceSerl

AND C.InOutType = 10

WHERE A.ReceiptSeq = 57

 

 

SELECT A.LastDateTime, A.LastUserSeq, A.InOutSeq, B.\*

FROM svs_TSIASServiceRepair AS A

JOIN \_TLGInOutDailyItem AS B ON B.InOutSeq = A.InOutSeq

AND B.InOutSerl = A.InOutSerl

JOIN \_TLGInOutStock AS C ON C.InOutSeq = B.InOutSeq

AND C.InOutSerl = B.InOutSerl

AND C.InOutType = B.InOutType

 

WHERE A.ReceiptSeq = 57

 

 

SELECT \* FROM \_TLGInOutStock WHERE ItemSeq IN(7463,7462) AND InOutType IN(10)

SELECT \* FROM \_TLGWHStock         WHERE ItemSeq IN(7463,7462)

 

SELECT \* FROM \_TLGInOutStock WHERE ItemSeq IN(5301,2420) AND InOutType IN(30)

SELECT \* FROM \_TLGWHStock         WHERE ItemSeq IN(5301,2420)

SELECT \* FROM \_TSLORDER WHERE OrderSeq IN(2417,2420)

SELECT \* FROM \_TSLORDERITEM WHERE OrderSeq IN(2417,2420)

 

SELECT \* FROM svs_TSLORDER WHERE OrderSeq IN(2417,2420)

SELECT \* FROM svs_TSLORDERITEM WHERE OrderSeq IN(2417,2420)
