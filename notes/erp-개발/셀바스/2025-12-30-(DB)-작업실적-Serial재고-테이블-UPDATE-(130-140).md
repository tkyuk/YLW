# (DB) 작업실적 Serial재고 테이블 UPDATE (130 -\> 140)

BEGIN TRAN

 

SELECT C.CompanySeq

,C.InOutType

,B.GoodInSeq

,C.InOutSerl

,C.DataKind

,C.InOutDataSerl

,C.InOutSubSerl

,C.InOutSerialSerl

,C.InOut

,C.InOutDate

,C.SerialNo

,C.WHSeq

,C.FunctionWHSeq

,C.ItemSeq

,C.InOutKind

,C.InOutDataKind

,C.PgmSeq

,C.LastUserSeq

,C.LastDateTime

INTO \#Tmp

FROM \_TPDSFCWorkReport AS A

JOIN \_TPDSFCGoodIn AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.WorkReportSeq = A.WorkReportSeq

JOIN \_TLGInOutSerialStock AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.InOutSeq = A.WorkReportSeq

AND C.InOutType = 140

 

 

DELETE A

FROM \_TLGInOutSerialStock AS A

WHERE InOutType = 140

/\*

UPDATE C

SET InOutSeq = B.GoodInSeq

FROM \_TPDSFCWorkReport AS A

JOIN \_TPDSFCGoodIn AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.WorkReportSeq = A.WorkReportSeq

JOIN \_TLGInOutSerialStock AS C WITH(NOLOCK) ON C.CompanySeq = A.CompanySeq

AND C.InOutSeq = A.WorkReportSeq

AND C.InOutType = 140

\*/

INSERT INTO \_TLGInOutSerialStock

SELECT \*

FROM \#Tmp

 

COMMIT TRAN
