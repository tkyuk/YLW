function fnSLGetItemInfo_djb(ObjectSS, IntSRow, IntERow, StrColCustSeq, StrColDeptSeq, StrColCurrSeq, DatSalesDate, StrColItemSeq, StrColUnitSeq, StrColSTDQty,

StrColItemStdPrice, StrColCustStdPrice, StrColPrice, StrColQty, StrColVatRate, StrColInclusedVAT, StrColCurAmt, StrColCurVat, StrColCurAmtTotal,

StrColDomAmt, StrColDomVat, StrColDomTotAmt, StrColOptionSeq, StrColExRate, StrExpKind, StrColDomPrice, StrColUnitName,

StrColSTDUnitName, StrColSTDUnitSeq, StrColIsQtyChange, IntInOut, StrColWHName, StrColWHSeq, StrColBizUnit, StrColEmpSeq)

local ds, Qty, STDQty

local columnSchema

local addData

 

StrCol = ObjectSS.ActiveColumnName

 

if IntInOut == '' or IntInOut == nil then

IntInOut = 1

end

columnSchema = '{WorkingTag, string}{IDX_NO, int}{Status, int}{DataSeq, int}{Selected, int}{TABLE_NAME, string}'

..'{DeptSeq, int}{WHSeq, int}{BizUnit, int}{ItemSeq, int}{InOrOut, int}{SalesDate, string}{CurrSeq, int}{CustSeq, int}'

..'{UnitSeq, int}{UnitName, string}{OptionSeq, int}{Qty, double}{IsPurchase, string}{SMExpKind, int}{ItemStdPrice, double}{CustStdPrice, double}'

..'{ItemMinPrice, double}{ItemCustPrice, double}{PriceStep, int}{PriceStepName, string}{IsAppQty, int}{CurrName, string}{CustName, string}{OptionName, string}'

..'{ItemClassName, string}{UMItemClass, int}{IsInclusedVAT, string}{VATRate, double}{SMBillType, int}{StdUnitSeq, int}{StdUnitName, string}'

..'{StdQty, double}{IsQtyChange, string}{WHName, string}{ColName, string}{EmpSeq, int}'

ds = CreateDataSet(ds,'DataBlock1', columnSchema)

for iRow = IntSRow, IntERow, 1 do

if StrColQty == '' then

Qty = 0

else

Qty = CellText(ObjectSS, iRow, StrColQty)

end

if StrColSTDQty == '' then

STDQty = 0

else

STDQty = CellText(ObjectSS, iRow, StrColSTDQty)

end

if StrColWHSeq == '' then

WHSeq = 0

else

WHSeq = CellText(ObjectSS, iRow, StrColWHSeq)

end

if StrColOptionSeq == '' then

OptionSeq = 0

else

OptionSeq = CellText(ObjectSS, iRow, StrColOptionSeq)

end

 

addData = 'A'..CNST_SP..tostring(iRow)..CNST_SP..'0'..CNST_SP..tostring(iRow)..CNST_SP..'1'..CNST_SP..'DataBlock1'..CNST_SP

..CellText(ObjectSS, iRow, StrColDeptSeq)..CNST_SP

..tostring(WHSeq)..CNST_SP

..CellText(ObjectSS, iRow, StrColBizUnit)..CNST_SP

..CellText(ObjectSS, iRow, StrColItemSeq)..CNST_SP

..tostring(IntInOut)..CNST_SP

..CellText(ObjectSS, iRow, DatSalesDate)..CNST_SP

..CellText(ObjectSS, iRow, StrColCurrSeq)..CNST_SP

..CellText(ObjectSS, iRow, StrColCustSeq)..CNST_SP

..CellText(ObjectSS, iRow, StrColUnitSeq)..CNST_SP

..CellText(ObjectSS, iRow, StrColUnitName)..CNST_SP

..tostring(OptionSeq)..CNST_SP

..tostring(Qty)..CNST_SP

..'0'..CNST_SP

..CellText(ObjectSS, iRow, StrExpKind)..CNST_SP

..'0'..CNST_SP

..'0'..CNST_SP

 

..'0'..CNST_SP

..'0'..CNST_SP

..'0'..CNST_SP

..''..CNST_SP

..'0'..CNST_SP

..''..CNST_SP

..''..CNST_SP

..''..CNST_SP

..''..CNST_SP

..'0'..CNST_SP

..''..CNST_SP

..'0'..CNST_SP

..'0'..CNST_SP

..'0'..CNST_SP

..''..CNST_SP

..'0'..CNST_SP

..''..CNST_SP

..CellText(ObjectSS, iRow, StrColWHName)..CNST_SP

..tostring(StrCol)..CNST_SP

..CellText(ObjectSS, iRow, StrColEmpSeq)

AddDataRow(ds, 'DataBlock1', addData)

end

 

rtnDs = ExecuteService(501144, 1, ds, rtnDs, 'SQLWF')

if (rtnDs == nil) then

else

for row=0, GetDataBlock(rtnDs, 'DataBlock1').Rows.Count-1, 1 do

SetText(ObjectSS, IntSRow + row, StrColItemStdPrice, GetDataSetValue(rtnDs, 'DataBlock1', 'ItemStdPrice', row))

SetText(ObjectSS, IntSRow + row, StrColCustStdPrice, GetDataSetValue(rtnDs, 'DataBlock1', 'CustStdPrice', row))

SetText(ObjectSS, IntSRow + row, StrColPrice, GetDataSetValue(rtnDs, 'DataBlock1', 'CustStdPrice', row))

SetText(ObjectSS, IntSRow + row, StrColSTDUnitName, GetDataSetValue(rtnDs, 'DataBlock1', 'StdUnitName', row))

SetText(ObjectSS, IntSRow + row, StrColSTDUnitSeq, GetDataSetValue(rtnDs, 'DataBlock1', 'StdUnitSeq', row))

SetText(ObjectSS, IntSRow + row, StrColSTDQty, GetDataSetValue(rtnDs, 'DataBlock1', 'StdQty', row))

SetText(ObjectSS, IntSRow + row, StrColWHName, GetDataSetValue(rtnDs, 'DataBlock1', 'WHName', row))

SetText(ObjectSS, IntSRow + row, StrColWHSeq, GetDataSetValue(rtnDs, 'DataBlock1', 'WHSeq', row))

SetText(ObjectSS, IntSRow + row, StrColInclusedVAT, GetDataSetValue(rtnDs, 'DataBlock1', 'IsInclusedVAT', row))

SetText(ObjectSS, IntSRow + row, StrColVatRate, GetDataSetValue(rtnDs, 'DataBlock1', 'VATRate', row))

if StrColIsQtyChange ~= '' then

SetText(ObjectSS, IntSRow + row, StrColIsQtyChange, GetDataSetValue(rtnDs, 'DataBlock1', 'IsQtyChange', row))

end

if CellText(ObjectSS, IntSRow + row, StrColUnitSeq) == '' or CellText(ObjectSS, IntSRow + row, StrColUnitSeq) == '0' then

SetText(ObjectSS, IntSRow + row, StrColUnitSeq, GetDataSetValue(rtnDs, 'DataBlock1', 'UnitSeq', row))

SetText(ObjectSS, IntSRow + row, StrColUnitName, GetDataSetValue(rtnDs, 'DataBlock1', 'UnitName', row))

end

 

CustItemName = GetDataSetValue(rtnDs, 'DataBlock1', 'CustItemName', row)

 

--if CustItemName == nil or CustItemName == '' then

-- test

--else

SetText(ObjectSS, IntSRow + row, 'CustItemName', GetDataSetValue(rtnDs, 'DataBlock1', 'CustItemName', row))

SetText(ObjectSS, IntSRow + row, 'CustItemNo', GetDataSetValue(rtnDs, 'DataBlock1', 'CustItemNo', row))

SetText(ObjectSS, IntSRow + row, 'CustItemSpec', GetDataSetValue(rtnDs, 'DataBlock1', 'CustItemSpec', row))

SetText(ObjectSS, IntSRow + row, 'CustItemUnit', GetDataSetValue(rtnDs, 'DataBlock1', 'CustItemUnit', row))

--end

 

end

nVatDec = GetDataSetValue(rtnDs, 'DataBlock1', 'VATDecEnv', 0)

nAmtDec = GetDataSetValue(rtnDs, 'DataBlock1', 'AmtDecEnv', 0)

nCurr = GetDataSetValue(rtnDs, 'DataBlock1', 'CurrEnv', 0)

nCurrName = GetDataSetValue(rtnDs, 'DataBlock1', 'CurrEnvName', 0)

nCurAmt = GetDataSetValue(rtnDs, 'DataBlock1', 'CurAmtEnv', 0)

nKorAmt = GetDataSetValue(rtnDs, 'DataBlock1', 'DomAmtEnv', 0)

BasicAmt = GetDataSetValue(rtnDs, 'DataBlock1', 'BasicAmt', 0)

 

end

ObjectSS.ActiveColumnName = StrCol

end
