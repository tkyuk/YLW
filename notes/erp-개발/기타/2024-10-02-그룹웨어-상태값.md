# 그룹웨어 상태값

-- 1-5. 처리 후 상태값 UPDATE

-- S : 전자결재 시작(잔자결재문서 저장되는 시점에 호출되어야 함)

-- E : 전자결재를 모두 마치고, ERP에서 다음단계로 진행되어도 되는 시점에 호출

-- C : 전자결재 취소(전자결재 문서가 삭제되는 시점에 호출되어야 함)

-- R : 전자결재가 모두 마쳤졌다가 취소되어 Resart되는 경우(ERP다음단계 진행여부를 체크해야함)

-- T : 전자결재가 모두 마치고 수신완료까지 한상태

-- B : 잔자결재가 반려 처리가 되었을 때 ERP상에서 반송된 문서의 상태확인도 하고 재상신도 해야 하는경우. 2011.06.09. nhsong.

-- L : 최종수신자가 취소한 경우 (환경설정 체크값 읽어와서 체크된 경우에는 R-S 호출, 체크되지 않았으면 작업 없음)

-- J : 선결, 중간결재자가 결재시 데이터를 전달받기 위함, 사이트용 후처리 예정. 서연탑메탈 요청 202205100249

CREATE PROC dbo.apro_SWPJTSLPORegProc

@pWorkingTag NCHAR(1),

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@WorkKind NVARCHAR(40) = '',

@TblKey NVARCHAR(40) = '', -- 넘겨준 키를 그대로 보낸다.

@TopUserName NVARCHAR(50) = '',

@TopUserPosition NVARCHAR(50) = '',

@RevUserName NVARCHAR(50) = '',

@RevUserPosition NVARCHAR(50) = '',

@DocID NVARCHAR(MAX) = '',

@SiteInit NVARCHAR(40) = '' ,

@MessageType INT OUTPUT,

@Status INT OUTPUT,

@Results NVARCHAR(250) OUTPUT,

@GWDate NVARCHAR(8) = '',

@EmpID NVARCHAR(40) = ''

AS

SET NOCOUNT ON;

 

-- 1-5. 처리 후 상태값 UPDATE

-- S : 전자결재 시작(잔자결재문서 저장되는 시점에 호출되어야 함)

-- E : 전자결재를 모두 마치고, ERP에서 다음단계로 진행되어도 되는 시점에 호출

-- C : 전자결재 취소(전자결재 문서가 삭제되는 시점에 호출되어야 함)

-- R : 전자결재가 모두 마쳤졌다가 취소되어 Resart되는 경우(ERP다음단계 진행여부를 체크해야함)

-- T : 전자결재가 모두 마치고 수신완료까지 한상태

-- B : 잔자결재가 반려 처리가 되었을 때 ERP상에서 반송된 문서의 상태확인도 하고 재상신도 해야 하는경우. 2011.06.09. nhsong.

-- L : 최종수신자가 취소한 경우 (환경설정 체크값 읽어와서 체크된 경우에는 R-S 호출, 체크되지 않았으면 작업 없음)

-- J : 선결, 중간결재자가 결재시 데이터를 전달받기 위함, 사이트용 후처리 예정. 서연탑메탈 요청 202205100249

 

DECLARE @POSeq INT

SELECT @POSeq = CONVERT(INT, @TblKey)

 

CREATE TABLE \#TCOMConfirm

(

IDX_NO INT IDENTITY(1,1)

, CfmSeq INT

, CfmSerl INT

, CfmSubSerl INT

, CfmCode INT

, UMCfmReason INT

, CfmReason NVARCHAR(500)

, Result NVARCHAR(255)

, MessageType INT

, Status INT

, DataSeq INT

)

 

IF @pWorkingTag = 'E'

BEGIN

INSERT INTO \#TCOMConfirm

(

CfmSeq, CfmSerl, CfmSubSerl, CfmCode, UMCfmReason, CfmReason, Result, MessageType, Status, DataSeq

)

SELECT A.PJTSeq, B.PJTRevSeq, 0, 1, N'0', N'', NULL, NULL, 0, A.POSerl

FROM apro_TPJTSLPORegItem AS A WITH(NOLOCK)

JOIN \_TPJTProject AS B WITH(NOLOCK) ON B.CompanySeq = A.CompanySeq

AND B.PJTSeq = A.PjtSeq

LEFT OUTER JOIN \_TCOMConfirm AS C WITH(NOLOCK) ON C.CompanySeq = B.CompanySeq

AND C.ConfirmSeq = 6307

AND C.CfmSeq = B.PJTSeq

AND C.CfmSerl = B.PJTRevSeq

AND C.CfmCode = 1

WHERE A.CompanySeq = @CompanySeq

AND A.POSeq = @POSeq

AND C.CompanySeq IS NULL -- 기 확정건 제외

 

EXEC \_SWCOMConfirm

@CompanySeq = @CompanySeq

, @LanguageSeq = @LanguageSeq

, @UserSeq = @UserSeq

, @PgmSeq = 114320043 -- 프로젝트확정처리_apro

END

 

RETURN
