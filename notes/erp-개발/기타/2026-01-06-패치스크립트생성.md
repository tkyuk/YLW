@ServiceSeq INT

, @WorkingTag NVARCHAR(10)

, @CompanySeq INT

, @LanguageSeq INT

, @UserSeq INT

, @PgmSeq INT

, @IsTransaction BIT

AS

 

SELECT TblName, GrpName, IsCommon

FROM sysdevwk_TDATableScript AS A

ORDER BY GrpName, TblName

SELECT TblName AS TblNameOld, TblName, GrpName,

KeyCol01, KeyCol02, KeyCol03, KeyCol04, KeyCol05, KeyCol06, KeyCol07, KeyCol08,

KeyCol09, KeyCol10, KeyCol11, KeyCol12, KeyCol13, KeyCol14, KeyCol15, IsCommon

ORDER BY GrpName, TblName
