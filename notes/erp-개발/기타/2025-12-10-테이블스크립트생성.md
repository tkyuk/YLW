# 테이블스크립트생성

CREATE PROCEDURE sys_SWCACreateTableScriptQuery

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

AS

SET @CompanySeq = 0

SET NOCOUNT ON

 

DECLARE @TableName NVARCHAR(200),

@DropTable NCHAR(1)='0',

@IsText NCHAR(1)='0',

@IsTextLog NCHAR(1)='0',

@IsPrimary NCHAR(1)='0',

@IsLogTableAdd NCHAR(1)='0'

 

SELECT @TableName = ISNULL(TableName , '0'),

@DropTable = ISNULL(DropTable , '0'),

@IsText = ISNULL(IsText , '0'),

@IsTextLog = ISNULL(IsTextLog , '0'),

@IsPrimary = ISNULL(IsPrimary , '0'),

@IsLogTableAdd = ISNULL(IsLogTableAdd, '0')

FROM \#BIZ_IN_DataBlock1

DECLARE @TableSeq INT,

@fieldName VARCHAR(200),

@Type VARCHAR(200),

@Type2 VARCHAR(200),

@length INT,

@DecLength INT,

@NullType INT,

@KeyYN INT,

@ColumnCaption NVARCHAR(200),

@sqlText VARCHAR(MAX),

@sqlTextLog VARCHAR(MAX),

@sql4 VARCHAR(MAX),

@sqlkey4 VARCHAR(MAX),

@sqlPrikey4 VARCHAR(MAX),

@sqlIndex VARCHAR(MAX),

@sqlIndexLog VARCHAR(MAX),

@sqlPrimaryKey VARCHAR(MAX),

@sqlLogTableAdd VARCHAR(MAX),

@MaxLength INT

 

SELECT @sql4 = '', @sqlText = '', @sqlTextLog = '', @sqlIndex = '', @sqlIndexLog = '', @sqlkey4 = '', @sqlPrikey4 = ''

SELECT @TableSeq = TableSeq, @TableName = TableName FROM \_TCATables WHERE TableName = @TableName

 

SELECT @MaxLength = ((CEILING(MAX(LEN(ISNULL(b.ColumnName,'')))) / 4) + 1) \* 4

FROM \_TCATables a

INNER JOIN \_TCAColumns b ON (b.CompanySeq = @CompanySeq AND a.TableSeq = b.TableSeq)

WHERE A.CompanySeq = @CompanySeq

AND A.TableSeq = @TableSeq

DECLARE Tables_Cursor CURSOR FOR

SELECT ISNULL(b.ColumnName,'') as fieldName,

ISNULL(C.MinorName,'') as Type,

ISNULL(B.length,0), ISNULL(B.DecLength,0), ISNULL(B.NullType,0),

B.KeyYN AS KeyYN,

B.Remark AS ColumnCaption

FROM \_TCATables a

INNER JOIN \_TCAColumns b ON (b.CompanySeq = @CompanySeq AND a.TableSeq = b.TableSeq)

LEFT OUTER JOIN \_TCAMinor C ON(C.CompanySeq = @CompanySeq AND B.DataType = C.MinorSeq AND C.LanguageSeq=1)

WHERE A.CompanySeq = @CompanySeq

AND A.TableSeq = @TableSeq

ORDER BY KeyYN DESC, b.ColumnSort

 

IF @DropTable = '1' -- DropTable문장넣기

BEGIN

SET @sqlText = 'IF EXISTS (SELECT \* FROM SYSOBJECTS WHERE Name = ''' + @TableName + ''' AND xtype = ''U'' )' + char(13) + char(10)

SET @sqlText = @sqlText + ' DROP TABLE ' + @TableName + char(13) + char(10) + char(13) + char(10)

SET @sqlTextLog = 'IF EXISTS (SELECT \* FROM SYSOBJECTS WHERE Name = ''' + @TableName + 'Log'' AND xtype = ''U'' )' + char(13) + char(10)

SET @sqlTextLog = @sqlTextLog + ' DROP TABLE ' + @TableName + 'Log' + char(13) + char(10) + char(13) + char(10)

END

ELSE

BEGIN

SET @sqlText = @sqlText + 'IF NOT EXISTS (SELECT \* FROM SYSOBJECTS WHERE Name = ''' + @TableName + ''' AND xtype = ''U'' )' + char(13) + char(10)

SET @sqlText = @sqlText + 'BEGIN' + char(13) + char(10)

SET @sqlTextLog = @sqlTextLog + 'IF NOT EXISTS (SELECT \* FROM SYSOBJECTS WHERE Name = ''' + @TableName + 'Log' + ''' AND xtype = ''U'' )' + char(13) + char(10)

SET @sqlTextLog = @sqlTextLog + 'BEGIN' + char(13) + char(10)

END

SET @sqlText = @sqlText + 'CREATE TABLE ' + @TableName + char(13) + char(10) + '(' + char(13) + char(10)

SET @sqlTextLog = @sqlTextLog + 'CREATE TABLE ' + @TableName + 'Log' + char(13) + char(10) + '(' + char(13) + char(10)

SET @sqlTextLog = @sqlTextLog + SUBSTRING(' LogSeq' + SPACE(@MaxLength), 1, @MaxLength + 4) + 'INT IDENTITY(1,1) NOT NULL, ' + char(13) + char(10)

SET @sqlTextLog = @sqlTextLog + SUBSTRING(' LogUserSeq' + SPACE(@MaxLength), 1, @MaxLength + 4) + 'INT NOT NULL, ' + char(13) + char(10)

SET @sqlTextLog = @sqlTextLog + SUBSTRING(' LogDateTime' + SPACE(@MaxLength), 1, @MaxLength + 4) + 'DATETIME NOT NULL, ' + char(13) + char(10)

SET @sqlTextLog = @sqlTextLog + SUBSTRING(' LogType' + SPACE(@MaxLength), 1, @MaxLength + 4) + 'NCHAR(1) NOT NULL, ' + char(13) + char(10)

SET @sqlTextLog = @sqlTextLog + SUBSTRING(' LogPgmSeq' + SPACE(@MaxLength), 1, @MaxLength + 4) + 'INT NULL, ' + char(13) + char(10)

IF @IsPrimary = '1'

SET @sqlPrimaryKey = 'CONSTRAINT PK' + @TableName + ' PRIMARY KEY CLUSTERED ('

ELSE

SET @sqlIndex = 'CREATE UNIQUE CLUSTERED INDEX IDXTemp' + @TableName + ' ON ' + @TableName + '('

SET @sqlIndexLog = 'CREATE UNIQUE CLUSTERED INDEX IDXTemp' + @TableName + 'Log' + ' ON ' + @TableName + 'Log ' + '(LogSeq)' + char(13) + char(10)

OPEN Tables_Cursor

FETCH NEXT FROM Tables_Cursor INTO @fieldName, @Type, @length, @DecLength , @NullType, @KeyYN, @ColumnCaption

WHILE @@FETCH_STATUS = 0

BEGIN

IF @Type in ('char', 'varchar', 'nchar', 'nvarchar')

BEGIN

IF (@length = -1)

SELECT @sql4 = @sql4 + SPACE(4) + SUBSTRING(@fieldName + SPACE(@MaxLength), 1, @MaxLength) + UPPER(@Type) + '(MAX) '

ELSE

SELECT @sql4 = @sql4 + SPACE(4) + SUBSTRING(@fieldName + SPACE(@MaxLength), 1, @MaxLength) + UPPER(@Type) + '(' + CONVERT(VARCHAR(30), @length) + ') '

END

ELSE -- 숫자타입인경우

BEGIN

IF @Type in ('decimal', 'numeric' )

SELECT @sql4 = @sql4 + SPACE(4) + SUBSTRING(@fieldName + SPACE(@MaxLength), 1, @MaxLength) + UPPER(@Type) + '(' + CONVERT(varchar(30), @length) + ',' + CONVERT(varchar(30), @DecLength) + ') '

ELSE

SELECT @sql4 = @sql4 + SPACE(4) + SUBSTRING(@fieldName + SPACE(@MaxLength), 1, @MaxLength) + UPPER(@Type) + ' '

END

SELECT @sql4 = @sql4 + CASE WHEN @Type = 'NVARCHAR' THEN SPACE(5 - (CASE WHEN @length = -1 THEN 3 ELSE LEN(CONVERT(VARCHAR(5), @length)) END))

WHEN @Type = 'INT' THEN SPACE(12)

WHEN @Type = 'NCHAR' THEN SPACE(7)

WHEN @Type = 'BIT' THEN SPACE(8)

WHEN @Type = 'DATETIME' THEN SPACE(7)

WHEN @Type = 'DECIMAL' THEN SPACE(2)

WHEN @Type = 'IMAGE' THEN SPACE(8)

WHEN @Type = 'BIGINT' THEN SPACE(8)

ELSE SPACE(1)

END

IF (@KeyYN = 1 )

BEGIN

SELECT @sqlkey4 = @sqlkey4 + @fieldName + ', '

SELECT @sqlPrikey4 = @sqlPrikey4 + @fieldName + ' ASC, '

END

IF (@NullType = 1)

SELECT @sql4 = @sql4 + ' NULL, '

ELSE

SELECT @sql4 = @sql4 + 'NOT NULL, '

 

SELECT @sql4 = @sql4 + char(13) + char(10)

FETCH NEXT FROM Tables_Cursor INTO @fieldName, @Type, @length , @DecLength, @NullType, @KeyYN, @ColumnCaption

END

CLOSE Tables_Cursor

DEALLOCATE Tables_Cursor

 

SET @sqlPrimaryKey = @sqlPrimaryKey + @sqlPrikey4

SET @sqlPrimaryKey = substring(@sqlPrimaryKey, 1, datalength(@sqlPrimaryKey)-2)

SET @sqlPrimaryKey = @sqlPrimaryKey + ')' + char(13) + char(10)

SET @sqlText = @sqlText + @sql4

SET @sqlText = substring(@sqlText, 1, datalength(@sqlText)-4)

IF @IsPrimary = '1'

SET @sqlText = @sqlText + ', ' + char(13) + char(10) + @sqlPrimaryKey + ')'

ELSE

BEGIN

SET @sqlText = @sqlText + char(13) + char(10) + ')' + char(13) + char(10)

SET @sqlIndex = @sqlIndex + @sqlkey4

SET @sqlIndex = substring(@sqlIndex, 1, datalength(@sqlIndex)-2)

SET @sqlIndex = @sqlIndex + ')' + char(13) + char(10)

END

IF @DropTable != '1' -- DropTable 아닌 경우에만 테이블 추가에 END 구문 넣어준다.

BEGIN

SET @sqlText = @sqlText + 'END' + char(13) + char(10)

END

 

-- WBS용 로그를 남기기 위해서 로그테이블 넣어주는 스크립트

IF @IsLogTableAdd = '1'

BEGIN

SET @sqlLogTableAdd = 'IF NOT EXISTS (SELECT 1 FROM \_TCOMTableLogInfo WHERE TableSeq = ' + RTRIM(CONVERT(NCHAR(10), @TableSeq)) + ')' + NCHAR(13)

SET @sqlLogTableAdd = @sqlLogTableAdd + 'BEGIN' + NCHAR(13)

SET @sqlLogTableAdd = @sqlLogTableAdd + SPACE(4) + 'INSERT \_TCOMTableLogInfo (TableName, CompanySeq, TableSeq, UseLog, LastUserSeq, LastDateTime)' + NCHAR(13)

SET @sqlLogTableAdd = @sqlLogTableAdd + SPACE(4) + 'SELECT ''' + @TableName + ''', 1, ' + RTRIM(CONVERT(NCHAR(10), @TableSeq)) + ', ''1'', 1, GETDATE()' + NCHAR(13)

SET @sqlLogTableAdd = @sqlLogTableAdd + + 'END' + NCHAR(13)

END

ELSE

BEGIN

SET @sqlLogTableAdd = ''

END

 

SET @sqlTextLog = @sqlTextLog + @sql4

SET @sqlTextLog = substring(@sqlTextLog, 1, datalength(@sqlTextLog)-4)

SET @sqlTextLog = @sqlTextLog + char(13) + char(10) + ')' + char(13) + char(10)

IF @DropTable != '1' -- DropTable 아닌 경우에만 테이블 추가에 END 구문 넣어준다.

BEGIN

SET @sqlTextLog = @sqlTextLog + 'END' + char(13) + char(10)

END

 

IF @IsText = 1 AND @IsTextLog = 1

SELECT @sqlText + char(13) + char(10) + @sqlIndex + char(13) + char(10) + @sqlTextLog + char(13) + char(10) + @sqlIndexLog + NCHAR(13) + @sqlLogTableAdd + ' ' as TableSql

ELSE IF @IsText = 1 AND @IsTextLog = 0

SELECT @sqlText + char(13) + char(10) + @sqlIndex + NCHAR(13) + @sqlLogTableAdd + ' ' as TableSql

ELSE IF @IsText = 0 AND @IsTextLog = 1

SELECT @sqlTextLog + char(13) + char(10) + @sqlIndexLog + NCHAR(13) + @sqlLogTableAdd + ' ' as TableSql

ELSE

SELECT '' as TableSql

 

RETURN

GO

DECLARE @CONST\_#BIZ_IN_DataBlock1 INT

, @CONST\_#BIZ_OUT_DataBlock1 INT

SELECT @CONST\_#BIZ_IN_DataBlock1 = 0

, @CONST\_#BIZ_OUT_DataBlock1 = 0

 

 

IF @CONST\_#BIZ_IN_DataBlock1 = 0

BEGIN

CREATE TABLE \#BIZ_IN_DataBlock1

(

WorkingTag NCHAR(1)

, IDX_NO INT

, DataSeq INT

, Selected INT

, MessageType INT

, Status INT

, Result NVARCHAR(255)

, ROW_IDX INT

, IsChangedMst NCHAR(1)

, TABLE_NAME NVARCHAR(255)

 

, TableName NVARCHAR(200), IsText NCHAR(1), IsTextLog NCHAR(1), DropTable NCHAR(1), IsPrimary NCHAR(1), IsLogTableAdd NCHAR(1)

)

SET @CONST\_#BIZ_IN_DataBlock1 = 1

 

END

 

IF @CONST\_#BIZ_OUT_DataBlock1 = 0

BEGIN

CREATE TABLE \#BIZ_OUT_DataBlock1

(

WorkingTag NCHAR(1)

, IDX_NO INT

, DataSeq INT

, Selected INT

, MessageType INT

, Status INT

, Result NVARCHAR(255)

, ROW_IDX INT

, IsChangedMst NCHAR(1)

, TABLE_NAME NVARCHAR(255)

 

, TableName NVARCHAR(200), IsText NCHAR(1), IsTextLog NCHAR(1), DropTable NCHAR(1), IsPrimary NCHAR(1), IsLogTableAdd NCHAR(1), TableSql NVARCHAR(4000)

)

SET @CONST\_#BIZ_OUT_DataBlock1 = 1

 

END

 

DECLARE @INPUT_ERROR_MESSAGE NVARCHAR(4000)

, @INPUT_ERROR_SEVERITY INT

, @INPUT_ERROR_STATE INT

, @INPUT_ERROR_PROCEDURE NVARCHAR(128)

INSERT INTO \#BIZ_IN_DataBlock1 (WorkingTag, IDX_NO, DataSeq, Selected, Status, Result, ROW_IDX, IsChangedMst, TABLE_NAME, TableName, IsText, IsTextLog, DropTable, IsPrimary, IsLogTableAdd)

SELECT N'A', 1, 1, 1, 0, NULL, NULL, N'0', N'', N'svs_TSIASServiceRepairClient', N'1', N'0', N'1', N'1', N'1'

IF @@ERROR \<\> 0 RETURN

 

 

DECLARE @HasError NCHAR(1)

, @UseTransaction NCHAR(1)

-- 내부 SP용 파라메터

, @ServiceSeq INT

, @MethodSeq INT

, @WorkingTag NVARCHAR(10)

, @CompanySeq INT

, @LanguageSeq INT

, @UserSeq INT

, @PgmSeq INT

, @IsTransaction BIT

 

SET @HasError = N'0'

SET @UseTransaction = N'0'

 

BEGIN TRY

 

SET @ServiceSeq = 18820580

--SET @MethodSeq = 1

SET @WorkingTag = N''

SET @CompanySeq = 1

SET @LanguageSeq = 1

SET @UserSeq = 27292

SET @PgmSeq = 18820687

SET @IsTransaction = 0

 

-- ExecuteOrder : 1 : Start

EXEC sys_SWCACreateTableScriptQuery

@ServiceSeq

, @WorkingTag

, @CompanySeq

, @LanguageSeq

, @UserSeq

, @PgmSeq

, @IsTransaction

IF @@ERROR \<\> 0

BEGIN

SET @HasError = N'1'

GOTO GOTO_END

END

 

-- ExecuteOrder : 1 : End

 

GOTO_END:

END TRY

BEGIN CATCH

-- SQL 오류인 경우는 여기서 처리가 된다

IF @UseTransaction = N'1'

ROLLBACK TRAN

DECLARE @ERROR_MESSAGE NVARCHAR(4000)

, @ERROR_SEVERITY INT

, @ERROR_STATE INT

, @ERROR_PROCEDURE NVARCHAR(128)

 

SELECT @ERROR_MESSAGE = ERROR_MESSAGE()

, @ERROR_SEVERITY = ERROR_SEVERITY()

, @ERROR_STATE = ERROR_STATE()

, @ERROR_PROCEDURE = ERROR_PROCEDURE()

RAISERROR (@ERROR_MESSAGE, @ERROR_SEVERITY, @ERROR_STATE, @ERROR_PROCEDURE)

 

RETURN

END CATCH

 

-- SQL 오류를 제외한 체크로직으로 발생된 오류는 여기서 처리

IF @HasError = N'1' AND @UseTransaction = N'1'

ROLLBACK TRAN

 

 

DROP TABLE \#BIZ_IN_DataBlock1

DROP TABLE \#BIZ_OUT_DataBlock1
