function fnCalcSheetWBS_djb(ObjectSS, StrColCurrSeq, ObjectExRate, StrColCustSeq, StrColDeptSeq, StrColReqDate, IntSRow, IntERow, StrColItemSeq, StrColUnitSeq, StrColSTDQty, StrColItemPrice, StrColCustPrice,

StrColPrice, StrColQty, StrColVatRate, StrColInclusedVAT, StrColCurAmt, StrColCurVat, StrColCurAmtTotal, StrColDomAmt, StrColDomVat, StrColDomTotAmt, StrColOptionSeq,

StrColExRate, StrExpKind, StrColDomPrice, StrColUnitName, StrColSTDUnitName, StrColSTDUnitSeq, StrColIsQtyChange, IntInOut, StrColWHName, StrColWHSeq, StrColBizUnit, StrColEmpSeq)

 

StrCol = ObjectSS.ActiveColumnName

if StrCol == '' or StrCol == nil then

StrCol = '-1'

end

 

if StrCol == 'ItemName' or StrCol == 'ItemNo' or StrCol == 'Spec' or StrCol == 'UnitName' or StrCol == 'CustName' or

StrCol == 'CustItemName' or StrCol == 'CustItemNo' or StrCol == 'CustItemSpec' or StrCol == StrColUnitName then

fnSLGetItemInfo_djb(ObjectSS, IntSRow, IntERow, StrColCustSeq, StrColDeptSeq, StrColCurrSeq, StrColReqDate, StrColItemSeq, StrColUnitSeq, StrColSTDQty,

StrColItemPrice, StrColCustPrice, StrColPrice, StrColQty, StrColVatRate, StrColInclusedVAT, StrColCurAmt, StrColCurVat, StrColCurAmtTotal,

StrColDomAmt, StrColDomVat, StrColDomTotAmt, StrColOptionSeq, StrColExRate, StrExpKind, StrColDomPrice, StrColUnitName,

StrColSTDUnitName, StrColSTDUnitSeq, StrColIsQtyChange, IntInOut, StrColWHName, StrColWHSeq, StrColBizUnit, StrColEmpSeq)

elseif StrCol == 'Qty' or StrCol == StrColQty then

fnSLGetStdQty_djb(ObjectSS, IntSRow, IntERow, StrColItemSeq, StrColUnitSeq, StrColQty, StrColSTDUnitName, StrColSTDUnitSeq, StrColSTDQty, StrColIsQtyChange, StrColUnitName)

end

 

ObjectSS.ActiveColumnName = StrCol

end
