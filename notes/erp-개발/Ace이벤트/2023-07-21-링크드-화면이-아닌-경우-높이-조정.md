# 링크드 화면이 아닌 경우 높이 조정

LinkPgm = \_base.LinkCreateID

 

if LinkPgm == '' then

> JumpOut('SalesExpense')        

else

> JumpOut('SalesExpenseLink')        

end

 

 

LinkPgm = \_base.LinkCreateID

 

-- 링크드 화면이 아닌 \[구매품의조회\] 원본일 경우 조회조건 높이 조정

if LinkPgm == '' then

        SetGridRowDefinition('GridItem',0,160,false)

end
