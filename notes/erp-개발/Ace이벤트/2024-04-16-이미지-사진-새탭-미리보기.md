# 이미지 사진 새탭 미리보기

local AttachFileSeq, AttachFileNo, msg, FileIdx, AppVersion

 

AttachFileSeq= CellText(SS1, SS1.ActiveRow, 'FileSeq')

AttachFileNo = CellText(SS1, SS1.ActiveRow, 'FileNo')

URL = CellText(SS1, SS1.ActiveRow, 'file_url')

FileIdx = tonumber(AttachFileNo) - 1                        

AppVersion = fltAppVersion.Value                 -- 웹공통버전

 

akFileUpload2.Value = AttachFileSeq                -- FileSeq, FileNo 값 파일컨트롤에 넣어주기

 

if HtmlScript('') ~= '' then -- 웹인 경우에만

> if AttachFileSeq ~= 0 and AttachFileSeq ~= '' and AppVersion \>= 6.1005 then                         -- 웹공통버전이 6.1.6.1005 버전 이상이 경우, 새탭에서 이미지 열기

        WPageOpenFileUploadImage(AttachFileSeq, AttachFileNo)                         

> elseif AttachFileSeq ~= 0 and AttachFileSeq ~= '' and AppVersion \< 6.1005 then                 -- 웹공통버전이 6.1.6.1005 버전보다 낮을 경우, 파일 다운로드
>
> akFileUpload2.ListFileDownloadRun = FileIdx                                                                 
>
> elseif URL ~= '' then
>
> WebOpen(URL, true, false)
>
> else
>
> msg = GetMessage('1001', GetDictionary(7818),'', '', '', '', '', '', '', '', '') -- @1이(가) 없습니다.
>
> MessageBox(msg, '', 'MsgBoxTypeOK')
>
> end

else

> if URL ~= '' then

        WebOpen(URL, true, false)

> end

end
