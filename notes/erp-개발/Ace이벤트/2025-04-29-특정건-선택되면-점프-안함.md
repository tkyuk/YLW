# 특정건 선택되면 점프 안함

local checkrow, PUPDSeq

\_base.IsJumpEventCancel = false

 

for row = 0, SS1.DataRowCnt - 1 do

> if CellText(SS1, row, 'Select') == '1' then

                checkrow = '1'

        if CellText(SS1, row, 'PUPDSeq') == '2' then

> PUPDSeq = '1'

        end

        end

end

 

if checkrow ~= '1' then

> msg = GetMessage('1001', GetDictionary(1572), '', '', '', '', '', '', '', '', '')

        MessageBox(msg, '', 'MsgBoxTypeOK')

        \_base.IsJumpEventCancel = true

        GoTo('END')

> return

end

 

if PUPDSeq == '1' then

msg = '생산 데이터는 구매요청으로 점프할 수 없습니다'

> MessageBoxShow(msg, '')

\_base.IsJumpEventCancel = true

> GoTo('END')

return

end
