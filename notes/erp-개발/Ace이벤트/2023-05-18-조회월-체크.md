# 조회월 체크

--==============================================================

-- 조회조건 날짜 Fr,To 숫자 입력

--==============================================================

if tonumber(fltSetDateMM.Value) \< 1 or tonumber(fltSetDateMM.Value) \> 12 then

> MessageBoxShow('조회월은 1~12월 사이의 숫자로 입력해야합니다.','')
>
> GoTo('END')
>
> return

end

 

if tonumber(fltSetDateMMTo.Value) \< 1 or tonumber(fltSetDateMMTo.Value) \> 12 then

> MessageBoxShow('조회월은 1~12월 사이의 숫자로 입력해야합니다.','')
>
> GoTo('END')
>
> return

end

 

if tonumber(fltSetDateMM.Value) \> tonumber(fltSetDateMMTo.Value) then

> MessageBoxShow('조회월이 잘못 입력 되었습니다.','')
>
> GoTo('END')
>
> return
>
>  

end
