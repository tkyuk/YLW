# 다이나믹 가변부 소수점 자리수 세팅

-- 원화 소수점 자리수

local MonthList, MonthListLength, i, QuarterList, QuarterListLength, HalfList, HalfListLength

 

MonthList = {'01', '02', '03', '04', '05', '06', '07', '08', '09', '10', '11', '12'} -- 월별

QuarterList = {'01', '02', '03', '04'} -- 분기별

HalfList = {'01', '02'} -- 반기별

 

-- \#List변수 : 해당 List의 총 길이를 가져올 수 있습니다.

MonthListLength = \#MonthList

QuarterListLength = \#QuarterList

HalfListLength = \#HalfList

 

-- 월별 원화 소수점자리수

for i = 1, MonthListLength do

SS1.ActiveColumnName = 'AmtQty'..MonthList\[i\]

if tonumber(cmbQryTypeName.Value) == 1 then -- 1 : 금액

SS1.ColumnDeclength = \_base.ENV_DecLenKOR

else -- 2 : 수량

SS1.ColumnDeclength = \_base.ENV_DecLenGoodsQty

end

end

 

-- 분기별 원화 소수점자리수

for i = 1, QuarterListLength do

SS1.ActiveColumnName = 'AmtQty'..QuarterList\[i\]..'Tot'

if tonumber(cmbQryTypeName.Value) == 1 then -- 1 : 금액

SS1.ColumnDeclength = \_base.ENV_DecLenKOR

else -- 2 : 수량

SS1.ColumnDeclength = \_base.ENV_DecLenGoodsQty

end

end

 

-- 반기별 원화 소수점자리수

for i = 1, HalfListLength do

SS1.ActiveColumnName = 'AmtQty'..HalfList\[i\]..'Half'

if tonumber(cmbQryTypeName.Value) == 1 then -- 1 : 금액

SS1.ColumnDeclength = \_base.ENV_DecLenKOR

else -- 2 : 수량

SS1.ColumnDeclength = \_base.ENV_DecLenGoodsQty

end

end

 

SS1.ActiveColumnName = 'Tot'

if tonumber(cmbQryTypeName.Value) == 1 then -- 1 : 금액

SS1.ColumnDeclength = \_base.ENV_DecLenKOR

else -- 2 : 수량

SS1.ColumnDeclength = \_base.ENV_DecLenGoodsQty

end
