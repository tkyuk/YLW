CREATE PROCEDURE rot_SPJTProjectWarrantyDueDateCheck

@ServiceSeq INT = 0,

@WorkingTag NVARCHAR(10)= '',

@CompanySeq INT = 1,

@LanguageSeq INT = 1,

@UserSeq INT = 0,

@PgmSeq INT = 0,

@IsTransaction BIT = 0

AS

DECLARE @MessageType INT,

@Status INT,

@Results NVARCHAR(250)

 

 

/\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*

프로젝트 완료여부 체크

\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*/

--EXEC dbo.\_SCOMMessage @MessageType OUTPUT,

-- @Status OUTPUT,

-- @Results OUTPUT,

-- 18 , -- @1는(은) 수정/삭제 할수 없습니다..(SELECT \* FROM \_TCAMessageLanguage WHERE messagedefault like '%건은%')

-- @LanguageSeq ,

-- 11351,'' -- SELECT \* FROM \_TCADictionary WHERE Word like '%완료%건%'

UPDATE \#BIZ_OUT_DataBlock1

SET Result = '완료된 프로젝트는 저장 할 수 없습니다.',

MessageType = 17009,

Status = 99999

FROM \#BIZ_OUT_DataBlock1 AS A

JOIN \_TPJTProject AS B ON A.PJTSeq = B.PJTSeq

WHERE A.WorkingTag IN ('U')

AND A.Status = 0

AND B.CompanySeq = @CompanySeq

AND B.ISComplete = '1'

 

 

/\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*

프로젝트 중단여부 체크

-- @LanguageSeq ,

-- 48413,'' -- SELECT \* FROM \_TCADictionary WHERE Word like '%중단%'

UPDATE \#BIZ_OUT_DataBlock1

SET Result = '중단된 프로젝트는 저장 할 수 없습니다.',

MessageType = 17009,

Status = 99999

FROM \#BIZ_OUT_DataBlock1 AS A

JOIN \_TPJTProject AS B ON A.PJTSeq = B.PJTSeq

WHERE A.WorkingTag IN ('U')

AND A.Status = 0

AND B.CompanySeq = @CompanySeq

AND B.ISSTOP = '1'

 

RETURN
