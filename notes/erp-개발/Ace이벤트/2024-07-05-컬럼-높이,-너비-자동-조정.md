# 컬럼 높이, 너비 자동 조정

local row

 

for row = 0, SS1.DataRowCnt -1 do

> if CellText(SS1, row, 'OrderText') ~= '' then
>
> SS1.ActiveRow = row
>
> SS1.SetAutoWidthColumn(ColXName(SS1, 'OrderText')); -- 너비
>
> SS1.SetAutoHeightRow(row); -- 높이
>
> end

end
