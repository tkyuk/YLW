# 상위 시트에 데이터 반영

ExpanderC32.IsExpanded = false

 

-- Serial 저장 후 시트1 Serial수량에 반영

if SS3.DataRowCnt-1 ~= '-1' then

> SetText(SS1, SS1.DblClickRow, 'SerialNoQty' ,SS3.DataRowCnt)        
>
> SS1.ActiveRow = SS1.DblClickRow
>
> SS1.ActiveColumnName = 'SerialNoQty'
>
> if CellText(SS1, SS1.DblClickRow, 'IsSerial') == '1' and CellText(SS1, SS1.DblClickRow, 'Qty') ~= CellText(SS1, SS1.DblClickRow, 'SerialNoQty') then
>
> SS1.ActiveCellBackColor = '#FF8282'
>
> else
>
> SS1.ActiveCellBackColor = -1118482
>
> end                                
>
>  

end
