# 마스터 변경 시 시트 일괄 적용 (ex. 환율)

local ExRate, TransportPrice, TreatmentPrice, Price, KorPrice

 

ExRate = fltExRate.Value

if ExRate == 0 then

> ExRate = 1

end        

 

> for row = 0, SS1.DataRowCnt - 1 do
>
> SS1.ActiveRow = row        
>
> if SS1.ActiveColumnName == 'TransportPrice' or SS1.ActiveColumnName == 'TreatmentPrice' then

         TransportPrice = tonumber(CellText(SS1, row, 'TransportPrice'))

         TreatmentPrice = tonumber(CellText(SS1, row, 'TreatmentPrice'))

        

         Price = TransportPrice + TreatmentPrice

         KorPrice = (TransportPrice + TreatmentPrice) \* ExRate

        

         SetText(SS1, row, 'Price', Price)

         SetText(SS1, row, 'KorPrice', KorPrice)

> end        

end
