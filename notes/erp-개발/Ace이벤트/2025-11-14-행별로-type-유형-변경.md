# 행별로 type 유형 변경

local row, col, rowType

 

for row = 0, SS1.DataRowCnt - 1 do

SS1.ActiveRow = row

rowType = CellText(SS1, row, 'RowCellType') -- ActiveRow 대신 row 사용

 

if rowType == 'FLOAT' then

SS1.ActiveColumnName = 'TotAmt'

SS1.ActiveCellType = 'enFloat'

SS1.ActiveCellDecimalPlaces = 2

SS1.ActiveCellShowSeparator = true

elseif rowType == 'CHECK' then

for col = SS1.SheetFixCols, SS1.MaxCols - 1 do

SS1.ActiveCol = col

SS1.ActiveCellType = 'enCheck'

end

elseif rowType == 'DATE' then

for col = SS1.SheetFixCols, SS1.MaxCols - 1 do

SS1.ActiveCol = col

SS1.ActiveCellType = 'enDate'

end

else

for col = SS1.SheetFixCols, SS1.MaxCols - 1 do

SS1.ActiveCol = col

SS1.ActiveCellType = 'enText'

end

end

end
