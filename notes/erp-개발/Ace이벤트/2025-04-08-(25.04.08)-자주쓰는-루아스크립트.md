SELECT TOP 500 A.PgmSeq, (SELECT caption FROM \_TCAPgm where PgmSeq = A.PgmSeq), A.PgmMethodScript, '============ \[ROWCOUNT = '

\+ B.PgmMethodName

\+ '\] \[ExecuteSeq = '

\+ CAST(A.ExecuteSeq AS NVARCHAR)

\+ '\] ===================================================================== ' + CHAR(13) +

A.PgmMethodScript

ON A.CompanySeq = B.CompanySeq
