# 배열 생성 / 선택된 행 기준 반복

-- \### 변수 초기화 (수정된 부분) \###

-- \# 연산자 사용을 피하기 위해 배열의 크기를 직접 관리할 카운터 변수를 추가합니다.

local seenItems, duplicateItemsForCheck, duplicateListArray, hasDuplicates, duplicateListCounter

seenItems = {}

duplicateItemsForCheck = {}

duplicateListArray = {}

hasDuplicates = false

duplicateListCounter = 0 -- 중복 목록 배열의 크기를 저장할 변수

 

-- \### 시트(SS)의 모든 행을 순회 \###

for row = 0, SS.DataRowCnt - 1 do

if CellText(SS, row, 'SelPrice') == '1' then

local itemSeq = CellText(SS, row, 'ItemSeq')

local itemNo = CellText(SS, row, 'ItemNo')

 

if itemSeq ~= nil and itemSeq ~= '' then

if seenItems\[itemSeq\] then

hasDuplicates = true

if duplicateItemsForCheck\[itemNo\] == nil then

-- \### 수정된 부분 1: 카운터를 사용하여 배열에 항목 추가 \###

duplicateListCounter = duplicateListCounter + 1

duplicateListArray\[duplicateListCounter\] = itemNo

duplicateItemsForCheck\[itemNo\] = true

end

else

seenItems\[itemSeq\] = true

end

end

end

end

 

-- \### 결과 처리 \###

if hasDuplicates then

local duplicateListString = ""

-- \### 수정된 부분 2: \# 연산자 대신 카운터 변수 사용 \###

for i = 1, duplicateListCounter do

if i \> 1 then

duplicateListString = duplicateListString .. ", "

end

duplicateListString = duplicateListString .. duplicateListArray\[i\]

end

 

local message = "동일한 자재번호 " .. duplicateListString .. " 가 선택되었습니다."

 

MessageBoxShow(message, "중복 오류")

 

GoTo('END')

return

end

-- 1. 체크된 행의 정보를 저장할 테이블(배열)을 생성합니다.

local sourceData = {}

local sourceCounter = 0

for row = 0, SS.DataRowCnt - 1 do

if CellText(SS, row, 'SelPrice') == '1' then

sourceCounter = sourceCounter + 1

-- 각 행의 정보를 하나의 작은 테이블로 묶어서 저장합니다.

sourceData\[sourceCounter\] = {

itemSeq = CellText(SS, row, 'ItemSeq'),

price = CellText(SS, row, 'Price'),

domPrice = CellText(SS, row, 'DOMPrice'),

dummy9 = CellText(SS, row, 'Dummy9'),

dummy10 = CellText(SS, row, 'Dummy10')

}

end

end

 

-- 2. 시트 전체를 다시 순회하며, 저장된 원본 데이터를 기준으로 값을 업데이트합니다.

for row = 0, SS.DataRowCnt - 1 do

local targetItemSeq = CellText(SS, row, 'ItemSeq')

 

-- 현재 행의 ItemSeq가 유효한 경우에만 비교를 수행합니다.

if targetItemSeq ~= nil and targetItemSeq ~= '' and CellText(SS, row, 'SelPrice') ~= '1' then

-- 3. 원본 데이터 배열을 순회하며 일치하는 ItemSeq를 찾습니다.

for i = 1, sourceCounter do

local setdata = sourceData\[i\]

if setdata.itemSeq == targetItemSeq then

-- ItemSeq가 일치하면 값을 SetText 합니다.

SetText(SS, row, 'Price', setdata.price)

SetText(SS, row, 'DOMPrice', setdata.domPrice)

SetText(SS, row, 'Dummy9', setdata.dummy9)

SetText(SS, row, 'Dummy10', setdata.dummy10)

if SS.WorkingTag ~= 'A' then

SS.WorkingTag = 'U'

end

SS.ActiveRow = row

SS.ActiveColumnName = 'Price'

> SS.CodeHelpSelectedRows = 0

RunPgmMethod('NewCalcAmt')

-- 일치하는 데이터를 찾았으므로, 이 행에 대한 원본 데이터 검색을 중단합니다.

break

end

end

end

end
