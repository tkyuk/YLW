# 자바스크립트 코드도움 실행

1\. 코드도움 실행을 위한 데이터 담기

var args = {

    CodeHelpConst: 108720024,   //실행하려는 코드도움 Seq 값

    CodeHelpParam1: "",  //실행하려는 코드도움 Param1 값

    CodeHelpParam2: "",  //실행하려는 코드도움 Param2 값

    CodeHelpParam3: "",  //실행하려는 코드도움 Param3 값

    CodeHelpParam4: "",  //실행하려는 코드도움 Param4 값

    FieldName: "WorkOrderNo",            //실행하려는 코드도움 DataFieldName 값

    FieldCd: "WorkOrderSeq",              //실행하려는 코드도움 DataFieldCd 값

    editingText: "",               //실행하려는 코드도움의 Text 값

    col: 0,                          //Sheet ActiveCol 

    row: 0,                          //Sheet ActiveRow

    isPasting: true,

    Control: txtWorkOrderNo, //Active Sheet

    Page: \_Page //현재 화면 객체 정보

};

﻿

2\. 코드도움 실행

txtWorkOrderNo.kevent_codehelp(args);
