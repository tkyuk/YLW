# 선택된 행에 대해 필수값, 데이터 없으면 에러 체크

if SS1.DataRowCnt == 0 then

GoTo('END')

return

end

 

for row = 0, SS1.DataRowCnt do

if CellText(SS1, row, 'Select') == '1' then

-- 24.04.23 SJPARK

WeightDate = CellText(SS1, row, 'WeightDate')

CarNo = CellText(SS1, row, 'CarNo')

Driver = CellText(SS1, row, 'Driver')

HP = CellText(SS1, row, 'HP')

 

if WeightDate == '' or WeightDate == '0' or WeightDate == nil or

CarNo == '' or CarNo == '0' or CarNo == nil or

Driver == '' or Driver == '0' or Driver == nil or

HP == '' or HP == '0' or HP == nil then

        MessageBoxShow('필수값이 입력되지 않았습니다.','')

> GoTo('END')

        return

end

 

end

end
