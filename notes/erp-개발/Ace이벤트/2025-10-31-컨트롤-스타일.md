SELECT A.Style, A.PgmSeq, B.Caption

FROM \_TCAPgmControls AS A

JOIN \_TCAPgm AS B ON B.PgmSeq = A.PgmSeq

WHERE A.ControlType = 4 -- 체크박스

AND ISNULL(A.Style,'') \<\> ''

AND A.Style LIKE 'Style%'

 

SELECT DISTINCT A.Style

FROM \_TCAPgmControls AS A

JOIN \_TCAPgm AS B ON B.PgmSeq = A.PgmSeq

WHERE A.ControlType = 4 -- 체크박스

AND ISNULL(A.Style,'') \<\> ''

AND A.Style LIKE 'Style%'
