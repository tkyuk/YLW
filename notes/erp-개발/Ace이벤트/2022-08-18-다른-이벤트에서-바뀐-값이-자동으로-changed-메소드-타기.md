# 다른 이벤트에서 바뀐 값이 자동으로 changed 메소드 타기

local W_UnitQty, B_UnitQty, ConvQty, UnitSeq, W_UnitSeq, Qty

 

A_UnitQty = tonumber(CellText(SS1, SS1.ActiveRow, 'A_UnitQty'))

W_UnitQty = tonumber(CellText(SS1, SS1.ActiveRow, 'W_UnitQty'))

B_UnitQty = tonumber(CellText(SS1, SS1.ActiveRow, 'B_UnitQty'))

ConvQty = tonumber(CellText(SS1, SS1.ActiveRow, 'ConvQty'))

UnitSeq = tonumber(CellText(SS1, SS1.ActiveRow, 'UnitSeq'))

W_UnitSeq = tonumber(CellText(SS1, SS1.ActiveRow, 'W_UnitSeq'))

Qty = tonumber(CellText(SS1, SS1.ActiveRow, 'Qty'))

 

if UnitSeq == W_UnitSeq then                                                                                                         

Qty = W_UnitQty                                                                                                

else                                                                                                        

Qty = A_UnitQty \* B_UnitQty                                                                                                

end                                

 

SetText(SS1, SS1.ActiveRow, 'W_UnitQty', W_UnitQty)        

SetText(SS1, SS1.ActiveRow, 'Qty', Qty)        

수량, 중량 변경 =\> 판매수량 계산

 

 

 

판매수량 changed =\> TestCnt

 

 

단순히 RunPgmMethod 만 태우는게 아니라

ActiveColumnName으로 판매수량을 태워야함

 

같은 이벤트를 실행해도 ActiveColumnName에 따라 달라짐

 

ActiveRowOld = SS1.ActiveRow

for row = SS1.ActiveRow, EndRow, 1 do

SS1.ActiveRow = row

RunPgmMethod('RollCalc')

end

SS1.ActiveRow = ActiveRowOld

<span class="mark">SS1.ActiveColumnName = 'Qty'</span>

RunPgmMethod('TestCnt')
