# 엔터 개수, 개행문자 개수에 따라 높이 재설정

-- 개행문자 갯수에 따라 Row 높이 재설정 (개행 처리된 데이터를 한눈에 보기 위함)

for row = 0, SS1.DataRowCnt - 1 do

> Enter = CellText(SS1, row, 'Enter')
>
> Enter = tonumber(Enter)
>
>  
>
> if Enter \< 1 then
>
> Enter = 30
>
> elseif Enter == 1 then        
>
> Enter = 45
>
> else
>
> Enter = (Enter \* 18) + 23
>
> end
>
>  
>
> SetRowHeight(SS1, row, Enter)

end
