function round(num, idp)-- 반올림

num = tonumber(num)

idp = tonumber(idp)

if idp == nil then

mult = 10^0

else

mult = 10^idp

end

if num \>= 0 then

return math.floor(num \* mult + 0.50001) / mult

else

return math.ceil(num \* mult - 0.50001) / mult

end

end
