# 시트 비활성화 처리 (최종 차수 아닌 경우)

if CellText(SSRev, SSRev.DblClickRow, 'EndDate') ~= '99991231' then

> GetToolbar('Save').EventArgs.Visible = true
>
> GetToolbar('Cut').EventArgs.Visible = true
>
> SS1.ControlKey = ''

SS1.IsmenuRowInsertVisible = true

SS1.AddRowsCount = 10

SS2.IsmenuRowInsertVisible = true

SS2.AddRowsCount = 5

SS3.IsmenuRowInsertVisible = true

SS3.AddRowsCount = 3

else

> GetToolbar('Save').EventArgs.Visible = false
>
> GetToolbar('Cut').EventArgs.Visible = false

SS1.IsmenuRowInsertVisible = false

SS1.AddRowsCount = 0

SS2.IsmenuRowInsertVisible = false

SS2.AddRowsCount = 0

SS3.IsmenuRowInsertVisible = false

SS3.AddRowsCount = 0

end
