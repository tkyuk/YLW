# DIS처리 및 다시 NON처리

명칭.ControlKey = 'NON;' ==\> 신규, 삭제 시 사용

명칭.ControlKey = 'DIS;NON;' =\>\> 조회, 저장 시 사용

 

 

txtBizUnitName1.ControlKey = 'NON;'

datStdYM.ControlKey = 'NON;'

txtWHName.ControlKey = 'NON;'

-- 링크드화면 (체계사업)인 경우 콤보 3건 저장 가능

LinkPgm = \_base.LinkCreateID

 

if LinkPgm ~= '' then

> SS1.ActiveColumnName = 'UMBizHeadQuarterName'         
>
> SS1.ColumnControlKey = ''
>
> SS1.ActiveColumnName = 'UMHeadQuarterName'         
>
> SS1.ColumnControlKey = ''
>
> SS1.ActiveColumnName = 'UMBizSectorName'         
>
> SS1.ColumnControlKey = ''

else

> SS1.ActiveColumnName = 'UMBizHeadQuarterName'         
>
> SS1.ColumnControlKey = 'DIS;'
>
> SS1.ActiveColumnName = 'UMHeadQuarterName'         
>
> SS1.ColumnControlKey = 'DIS;'
>
> SS1.ActiveColumnName = 'UMBizSectorName'         
>
> SS1.ColumnControlKey = 'DIS;'

end
