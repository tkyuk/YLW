# 스플리터 숨기기

-- 단암 추가 링크드 화면 세팅

LinkPgm = \_base.LinkCreateID

 

if LinkPgm ~= '' then

-- 구매입고 툴바 숨기기

> GetToolbar('JumpOutDelvIn').EventArgs.Disable = true
>
> GetToolbar('JumpOutDelvIn').EventArgs.Visible = false        
>
> GetToolbar('JumpOutPJTDelvReturnQuery').EventArgs.Disable = true
>
> GetToolbar('JumpOutPJTDelvReturnQuery').EventArgs.Visible = false        

else

-- 링크드 화면 아닌 경우 첨부파일 높이 조정

> SetGridRowDefinition('GridItem',3,0,false)        
>
> <span class="mark">if GridItem.GetRowSplitter(2, 0) ~= nil then</span>

        <span class="mark">GridItem.GetRowSplitter(2, 0).ControlKey = 'HID';</span>

> <span class="mark">end</span>        

end
