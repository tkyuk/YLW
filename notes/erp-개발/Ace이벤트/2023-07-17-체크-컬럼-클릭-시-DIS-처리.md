# 체크 컬럼 클릭 시 DIS 처리

if CellText(SS1, SS1.ActiveRow, 'ISStd') == '1' and SS1.WorkingTag == 'A' then        

> SS1.ActiveColumnName = 'AssyName'         
>
> SS1.ActiveCellReadOnly = false
>
> SS1.ActiveColumnName = 'AfterTx'         
>
> SS1.ActiveCellReadOnly = false
>
> SS1.ActiveColumnName = 'HeatTx'         
>
> SS1.ActiveCellReadOnly = false
>
> SS1.ActiveColumnName = 'QCYN'         
>
> SS1.ActiveCellReadOnly = false

else

> SS1.ActiveColumnName = 'AssyName'         
>
> SS1.ActiveCellReadOnly = true
>
> SS1.ActiveColumnName = 'AfterTx'         
>
> SS1.ActiveCellReadOnly = true
>
> SS1.ActiveColumnName = 'HeatTx'         
>
> SS1.ActiveCellReadOnly = true
>
> SS1.ActiveColumnName = 'QCYN'         
>
> SS1.ActiveCellReadOnly = true

end
