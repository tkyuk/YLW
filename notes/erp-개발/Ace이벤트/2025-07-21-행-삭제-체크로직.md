# 행 삭제 체크로직

local row, IsNotCut

IsNotCut = '0'

for row = SS2.SelectedRowStart, SS2.SelectedRowEnd do

> if tonumber(CellText(SS2, row, 'IsAuto')) == '1' then
>
> IsNotCut = '1'
>
> end

end

if IsNotCut == '1' then

> MessageBox('자동 생성 건은 삭제할 수 없습니다.')
>
> GoTo('END')
>
> return

end
