# (루아) 음수인 경우 / 0보다 작을 경우 빨간색 (1개 컬럼)

local row, EndItemQty

 

for row = 0, SS1.DataRowCnt - 1 do

SS1.ActiveRow = row

> SS1.ActiveColumnName = 'EndItemQty'
>
> EndItemQty = CellText(SS1, row, 'EndItemQty')
>
> if tonumber(EndItemQty) \< 0 then
>
> SS1.ActiveCellForeColor = -65536 -- 빨강
>
> end

end
