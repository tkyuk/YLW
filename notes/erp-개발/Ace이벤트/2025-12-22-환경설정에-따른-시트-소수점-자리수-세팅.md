# 환경설정에 따른 시트 소수점 자리수 세팅

DomDec = tonumber(GetPgmEnv('COMEnv', 3549))                -- 자국통화 소수점 자리수

CurDec         = tonumber(GetPgmEnv('COMEnv', 3550))                -- 외화통화 소수점 자리수

 

if GetPgmEnv('COMEnv', 3548) == '1' then                                        -- 외화통화 사용

> txtCurrName.ControlKey = 'NOS;'
>
> txtCurrName.DefaultValue='ENV_BasicKorCurr' -- 202404150200
>
>  
>
> SS1.ActiveColumnName = 'Amt'

SS1.ColumnDeclength = CurDec        

> SS1.ActiveColumnName = 'CurrName'
>
> SS1.ColumnControlKey =''
>
>  

else

> txtCurrName.ControlKey = 'HID;DIS;'
>
>  
>
> SS1.ActiveColumnName = 'Amt'

SS1.ColumnDeclength = DomDec         

> SS1.ActiveColumnName = 'CurrName'

SS1.ColumnControlKey = 'HID;DIS;'

end
