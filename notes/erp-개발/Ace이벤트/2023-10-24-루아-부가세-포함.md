# 루아 부가세 포함

local Price, Amt, VATAmt, DomAmt, DomVAT, DomPrice, Qty, VATRate, CurrRate, IsVAT

 

if SS1.ActiveColumnName == 'Qty' or SS1.ActiveColumnName == 'Price' or SS1.ActiveColumnName == 'ExRate' then

Price = CellText(SS1, SS1.ActiveRow, 'Price')

VATRate = CellText(SS1, SS1.ActiveRow, 'VATRate')

Qty = CellText(SS1, SS1.ActiveRow, 'Qty')

CurrRate = CellText(SS1, SS1.ActiveRow, 'ExRate')

 

SetText(SS1, SS1.ActiveRow, 'CurAmt', Qty\*Price)

SetText(SS1, SS1.ActiveRow, 'CurVAT', Qty\*Price\*VATRate/100)

SetText(SS1, SS1.ActiveRow, 'TotCurAmt', Qty\*Price\*(100 + VATRate)/100)

 

SetText(SS1, SS1.ActiveRow, 'DomPrice', Price\*CurrRate)

SetText(SS1, SS1.ActiveRow, 'DomAmt', Qty\*Price\*CurrRate)

SetText(SS1, SS1.ActiveRow, 'DomVAT', Qty\*Price\*VATRate/100\*CurrRate)

SetText(SS1, SS1.ActiveRow, 'TotDomAmt', Qty\*Price\*(100 + VATRate)/100\*CurrRate)

 

end

 

if SS1.ActiveColumnName == 'CurAmt' then

Amt = CellText(SS1, SS1.ActiveRow, 'CurAmt')

VATRate = CellText(SS1, SS1.ActiveRow, 'VATRate')

CurrRate = CellText(SS1, SS1.ActiveRow, 'ExRate')

 

SetText(SS1, SS1.ActiveRow, 'CurVAT', Amt\*VATRate/100)

SetText(SS1, SS1.ActiveRow, 'TotCurAmt', Amt\*(100 + VATRate)/100)

 

SetText(SS1, SS1.ActiveRow, 'DomAmt', Amt\*CurrRate)

SetText(SS1, SS1.ActiveRow, 'DomVAT', Amt\*VATRate/100\*CurrRate)

SetText(SS1, SS1.ActiveRow, 'TotDomAmt', Amt\*(100 + VATRate)/100\*CurrRate)

end

 

if SS1.ActiveColumnName == 'CurVAT' then

VATAmt = CellText(SS1, SS1.ActiveRow, 'CurVAT')

Amt = CellText(SS1, SS1.ActiveRow, 'CurAmt')

 

CurrRate = CellText(SS1, SS1.ActiveRow, 'ExRate')

 

SetText(SS1, SS1.ActiveRow, 'TotCurAmt', Amt+VATAmt)

 

DomAmt = CellText(SS1, SS1.ActiveRow, 'DomAmt')

 

SetText(SS1, SS1.ActiveRow, 'DomVAT', VATAmt \* CurrRate )

SetText(SS1, SS1.ActiveRow, 'TotDomAmt', DomAmt + (VATAmt \* CurrRate))

end

if SS1.ActiveColumnName == 'DomPrice' then

DomPrice = CellText(SS1, SS1.ActiveRow, 'DomPrice')

VATRate = CellText(SS1, SS1.ActiveRow, 'VATRate')

Qty = CellText(SS1, SS1.ActiveRow, 'Qty')

IsVAT = CellText(SS1, SS1.ActiveRow, 'IsVAT')

 

if IsVAT == '1' then

SetText(SS1, SS1.ActiveRow, 'DomAmt', math.floor( (Qty \* DomPrice) \* (100 / (100 + VATRate)), 0.500001 ))

SetText(SS1, SS1.ActiveRow, 'DomVAT', (Qty \* DomPrice) - (math.floor( (Qty \* DomPrice) \* (100 / (100 + VATRate)), 0.500001 )))

SetText(SS1, SS1.ActiveRow, 'TotDomAmt', Qty \* DomPrice)

else

SetText(SS1, SS1.ActiveRow, 'DomAmt', DomPrice\*Qty)

SetText(SS1, SS1.ActiveRow, 'DomVAT', DomPrice\*Qty\*VATRate/100)

SetText(SS1, SS1.ActiveRow, 'TotDomAmt', DomPrice\*Qty\*(100+VATRate)/100)

end

end

 

 

 

if SS1.ActiveColumnName == 'DomAmt' then

DomAmt = CellText(SS1, SS1.ActiveRow, 'DomAmt')

VATRate = CellText(SS1, SS1.ActiveRow, 'VATRate')

 

SetText(SS1, SS1.ActiveRow, 'DomVAT', DomAmt\*VATRate/100)

SetText(SS1, SS1.ActiveRow, 'TotDomAmt', DomAmt\*(100+VATRate)/100)

end

 

if SS1.ActiveColumnName == 'DomVAT' then

DomAmt = CellText(SS1, SS1.ActiveRow, 'DomAmt')

DomVAT = CellText(SS1, SS1.ActiveRow, 'DomVAT')

 

SetText(SS1, SS1.ActiveRow, 'TotDomAmt', DomAmt + DomVAT)
