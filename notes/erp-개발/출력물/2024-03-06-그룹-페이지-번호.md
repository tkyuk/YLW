# 그룹 페이지 번호

-- 페이지풋터 Onbind

var GroupIndex = GetGlobal("GroupIndex");

var GroupPageInfo = GetGlobal("GroupPageInfo");

var GroupPageArray = GroupPageInfo.split(",");

 

if(isNaN(GroupPageArray\[GroupIndex\]) == true) {

GroupPageArray\[GroupIndex\] = 0;

}

 

GroupPageArray\[GroupIndex\] = parseInt(GroupPageArray\[GroupIndex\]) + 1;\_TraceLn("GroupPageArray\[:"+GroupIndex+"\]:"+GroupPageArray\[GroupIndex\]);

GroupPageInfo = GroupPageArray.join(",");

SetGlobal("GroupPageInfo", GroupPageInfo);

This.SetUserValue(GroupIndex);

// Report Onbind

// 총 6개 페이지 중 3장, 1장, 2장이 서로 그룹일 경우에 대한 예시를 같이 표기하였습니다.

SetGlobal("GroupPageInfo", ""); // 그룹별 전체페이지정보(3,3,3,1,2,2)

SetGlobal("GroupCurPageInfo", ""); // 그룹별 현재페이지정보(1,2,3,1,1,2)

SetGlobal("GroupIndex", 0); // 현재 순번(1,2,3,4,5,6)

SetGlobal("GroupCurPageIndex", 0); // 현재 페이지번호(1,2,3,1,1,2)

SetGlobal("GroupStartIndex", 1); // 1개 그룹에 대한 배열의 시작점(1,1,1,4,5,5)

SetGlobal("GroupEndIndex", 0); // 1개 그룹에 대한 배열의 마지막점(1,2,3,1,1,2)(각각 최종적으로 각 그룹의 마지막 값인 3, 1, 2를 채택하게됨)

SetGlobal("SlipMstSeqOld", 0); // SlipMstSeq 변화 감지용

 

SetGlobal("PageCursor", 1); // 현재 페이지번호

-- 페이지풋터 OnEndbind

var GroupIndex = This.GetUserValue();

var GroupPageInfo = GetGlobal("GroupPageInfo");

var GroupPageArray = GroupPageInfo.split(",");

if(isNaN(GroupIndex) == false && GroupPageArray != null) {

var currPage = GetGlobal("PageCursor");

var totalPage = GroupPageArray\[GroupIndex\];

This.SetText(currPage + "/" + totalPage);

 

if (totalPage == currPage) {

SetGlobal("PageCursor", 1);

} else {

SetGlobal("PageCursor", GetGlobal("PageCursor")+1);

}

}
