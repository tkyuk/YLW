# JS - 더미풋터

var idx = ReportTemplate.GetGlobal("idx");

if(idx ==0){

var empty = This.GetEmptySpace()-<span class="mark">200</span>;

var thisHeight = This.GetHeight();

var repeatN = Math.floor(empty /thisHeight);

ReportTemplate.SetGlobal("repeatN", repeatN);

}

This.SetRepeatNumber(GetGlobal("repeatN"));

ReportTemplate.SetGlobal("idx", idx+1);
