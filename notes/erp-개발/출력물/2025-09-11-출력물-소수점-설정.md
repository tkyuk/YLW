# 출력물 소수점 설정

<span class="mark">// 수량</span>

This.SetFormat(GetGlobal("DecLenGoodQty"));

 

<span class="mark">// 단가</span>

var CurrName = This.GetDataSetValue("DataBlock1.CurrName");

 

if ( This.GetDataSetValue("OZFormParam.BasicKorCurr") == CurrName ) {

> This.SetFormat(GetGlobal("DecLenDomPrice"));

 

 

} else {

> This.SetFormat(GetGlobal("DecLenPrice"));

 

\_TraceLn("\t\>\>CurrName = " + GetGlobal("DecLenPrice"));

\_TraceLn("\t\>\>BasicKorCurr = " + GetGlobal("DecLenDomPrice"));

}

This.SetFormat(GetGlobal("DecLenPrice"));

 

<span class="mark">// 금액</span>

var CurrName = This.GetDataSetValue("DataBlock1.CurrName");

 

if ( This.GetDataSetValue("OZFormParam.BasicKorCurr") == CurrName ) {

> This.SetFormat(GetGlobal("DecLenDomAmt"));

 

 

} else {

> This.SetFormat(GetGlobal("DecLenCurAmt"));

 

\_TraceLn("\t\>\>CurrName = " + GetGlobal("DecLenCurAmt"));

\_TraceLn("\t\>\>BasicKorCurr = " + GetGlobal("DecLenDomAmt"));

}
