# 더미풋터 페이지별로 설정

-- 자바스크립트 더미풋터설정

// 20220210 by ghpark 데이터 개수에 따라 더미풋터 개수 설정

var count = This.GetComponentByName<span class="mark">("FixedTableLabel67")</span>.GetDataSetRowCount(); // DB에서 가져온 데이터 개수

var firstCnt = 14;// 첫 페이지에서 최대로 표시가능한 데이터 개수 -- 합계나올것 고려

var lastCnt = 14;// 마지막 페이지에서 최대로 표시가능한 데이터 개수 -- 합계나올것 고려

var DummyCnt = 14;// 중간 페이지에서 최대로 표시가능한 데이터 개수 -- 합계없는 전체개수

var repeatN = 0; // 더미풋터 반복 횟수

 

 

if(count \< firstCnt){ // 데이터가 첫 페이지에 모두 보여줄 수 있을 정도의 개수라면, 첫 페이지에 모두 보여준다.

repeatN = firstCnt - count; // 데이터 개수에 따라 변함(만약 첫 페이지 최대 개수가 11개, 데이터 1개면, 더미풋터는 11-1, 즉 10번 반복해야함)

} else if((count - firstCnt) % DummyCnt == 0){ // 데이터 개수가 11, 11+36, 11+36\*2 즉 36n + 11개일 경우, 더미풋터 개수는 0이 된다.

repeatN = 0

} else {

repeatN = lastCnt - (count - firstCnt) % DummyCnt; // 데이터 개수가 11을 초과할 경우의 데이터 개수와 더미풋터 반복횟수 조율

}

//\_TraceLn("\t\t\>\>repeatN = " + repeatN);

//\_TraceLn("\t\t\>\>test = " + count);

This.SetRepeatNumber(repeatN);
