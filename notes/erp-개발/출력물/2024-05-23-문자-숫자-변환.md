# 문자 -\> 숫자 변환

var date = This.GetDataSetValue("DataBlock1.YM"); //라벨에 원래 찍히는 날짜 가져오기 This.GetText();로 대체해도 됨

var year = date.substring(0,4);

var month = date.substring(4,6);

 

 

//var set1 = Number(GetInputValue(year));

var monthNm = Number(month);

 

 

This.SetText(year+"년 "+monthNm+"월 말");
