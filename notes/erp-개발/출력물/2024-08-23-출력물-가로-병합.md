# 출력물 가로 병합

\<\<HorizontalMerge.ozr\>\>

1.  고정 테이블

    1.  오즈 스크립트 사용

> string gubun;
>
> string type;
>
> gubun = \#GUBUN#;
>
> type = \#TYPE#;
>
>  
>
> if(gubun == "A" \|\| type == "" \|\| type == null) {
>
> setattr("leftlabelmergeto","FixedTableLabel1");
>
> }
>
>  

2.  테이블

    1.  자바 스크립트 (OnBind)

        1.  왼쪽

> var gubun = This.GetDataSetValue("GUBUN");
>
> var type = This.GetDataSetValue("TYPE");
>
>  
>
> if(gubun == "A" \|\| type == "" \|\| type == null) {
>
> This.SetRightGroupingCount(1);
>
> }

2.  오른쪽

> var gubun = This.GetDataSetValue("GUBUN");
>
> var type = This.GetDataSetValue("TYPE");
>
>  
>
> if(gubun == "A" \|\| type == "" \|\| type == null) {
>
> This.SetLeftGroupingCount(1);
>
> This.SetText(gubun);
>
> }
