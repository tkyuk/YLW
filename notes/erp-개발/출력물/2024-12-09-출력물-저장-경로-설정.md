# 출력물 저장 경로 설정

-- PC명 함수지정1

%USERPROFILE%\Desktop\임가공 계약서\\

 

-- 경로지정 (경로만)

C:\Users\PC명\Desktop\임가공계약서\\

 

-- 경로지정 (경로 + 파일명 =\> 자동저장)

C:\Users\PC명\Desktop\임가공계약서\파일명

함수 사용하여 PC경로 지정 시 파일명까지 입력하는 자동 저장은 불가능

 

경로 지정 시 폴더 없는 경우 자동 생성되지만, 함수는 자동 생성되지 않는다

 

IF NOT EXISTS (SELECT TOP 1 1 FROM \_TCAOZReport WHERE CompanySeq= 1 AND ReportSeq = 59020223 AND OZPSeq = 28 AND OZPSerl = 9)

INSERT INTO \_TCAOZReport

SELECT 1,        59020223,        28,        9,        0,        '%USERPROFILE%\Desktop\임가공 계약서\\,        '0',        '0',        0,        1, GETDATE()
